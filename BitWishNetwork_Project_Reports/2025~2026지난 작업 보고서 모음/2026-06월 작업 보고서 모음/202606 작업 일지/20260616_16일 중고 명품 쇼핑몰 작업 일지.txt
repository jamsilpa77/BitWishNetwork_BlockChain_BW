    const res = await fetch(backendUrl, {
      headers: {
        'x-tenant-slug': tenantSlug,
      },
      cache: 'no-store', // 실시간 데이터 대조를 위해 캐시 무효화
    });

    if (res.ok) {
      rawProducts = await res.json();
      isLiveDatabase = true;
    }
  } catch (error) {
    // 로컬 백엔드가 꺼져 있거나 버셀 독립 구동 중인 경우 안전하게 가상 모드로 백업 전환
    isLiveDatabase = false;
  }

  // 4. 만약 데이터베이스가 오프라인이면, 가상 샌드박스 데이터로 동적 정렬 수행
  if (!isLiveDatabase) {
    const allFallback = fallbackProducts[tenantSlug] || [];
    if (selectedBrandId) {
      rawProducts = allFallback.filter(p => p.brand_id === selectedBrandId);
    } else {
      rawProducts = allFallback;
    }
  }

  return (
    <div className="min-h-screen bg-white py-12 px-6">
      <div className="max-w-7xl mx-auto">
        
        {/* [상단 타이틀 헤더 영역] */}
        <div className="border-b border-gray-100 pb-8 mb-12 flex flex-col md:flex-row justify-between items-baseline">
          <div>
            <h1 className="text-3xl font-black tracking-widest uppercase mb-2" style={{ color: 'var(--tenant-primary)' }}>
              PRODUCT COLLECTION
            </h1>
            <p className="text-xs text-gray-400 font-medium uppercase tracking-wider">
              엄격한 가치 감정과 물리 보안을 거친 명품 오리지널 전시관
            </p>
          </div>
          
          {/* [서버 실시간 연동 인디케이터] - 가동 상태를 시각화하여 표현 */}
          <div className="mt-4 md:mt-0 flex items-center space-x-2 text-2xs font-extrabold uppercase tracking-widest">
            {isLiveDatabase ? (
              <>
                <span className="h-2 w-2 rounded-full bg-green-500 animate-pulse"></span>
                <span className="text-green-600">● 실시간 DB 연동 완료 (Live Active)</span>
              </>
            ) : (
              <>
                <span className="h-2 w-2 rounded-full bg-amber-500"></span>
                <span className="text-amber-600">● 가상 데모 샌드박스 가동 상태 (Sandbox)</span>
              </>
            )}
          </div>
        </div>

        {/* [브랜드 동적 필터 탭 바 영역] - 클릭 시 주소 쿼리(?brandId=)를 변형하여 화면을 재정렬합니다. */}
        <div className="mb-12">
          <span className="text-3xs font-black text-gray-400 uppercase tracking-widest block mb-4">BRAND FILTERING</span>
          <div className="flex flex-wrap gap-2.5">
            {/* 전체보기 버튼 */}
            <a
              href={`/${tenantSlug}/products`}
              className={`text-2xs font-black uppercase tracking-widest px-5 py-3 border transition-all ${
                !selectedBrandId
                  ? 'bg-black text-white border-black'
                  : 'bg-white text-gray-600 border-gray-200 hover:border-black'
              }`}
            >
              ALL (전체보기)
            </a>

            {/* 개별 브랜드 선택 버튼 동적 생성 */}
            {brands.map((brand) => (
              <a
                key={brand.id}
                href={`/${tenantSlug}/products?brandId=${brand.id}`}
                className={`text-2xs font-black uppercase tracking-widest px-5 py-3 border transition-all ${
                  selectedBrandId === brand.id
                    ? 'bg-black text-white border-black'
                    : 'bg-white text-gray-600 border-gray-200 hover:border-black'
                }`}
              >
                {brand.name}
              </a>
            ))}
          </div>
        </div>

        {/* [상품 그리드 메인 영역] */}
        {rawProducts.length > 0 ? (
          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-8">
            {rawProducts.map((product) => (
              <div 
                key={product.id} 
                className="group cursor-pointer flex flex-col border border-gray-100 p-4 hover:shadow-lg transition-all duration-300 bg-white"
              >
                {/* 이미지 및 상태 등급 배지 */}
                <div className="relative overflow-hidden aspect-square bg-gray-50 mb-5">
                  <img
                    src={product.main_image || 'https://images.unsplash.com/photo-1584917865442-de89df76afd3?w=500'}
                    alt={product.name}
                    className="object-cover w-full h-full group-hover:scale-105 transition-transform duration-500"
                  />
                  <span 
                    className="absolute top-3 left-3 text-white text-3xs font-black py-1.5 px-3 tracking-wider"
                    style={{ backgroundColor: 'var(--tenant-primary)' }}
                  >
                    등급 {product.condition_grade}
                  </span>
                </div>

                {/* 상세 제품 메타 설명 */}
                <div className="flex flex-col">
                  <span className="text-3xs font-black text-gray-400 tracking-wider uppercase mb-1">
                    {product.brand_english_name || product.brand_name}
                  </span>
                  <h3 className="text-xs font-bold text-gray-800 line-clamp-1 group-hover:text-gray-900 transition-colors mb-3">
                    {product.name}
                  </h3>
                  <div className="flex justify-between items-baseline mt-auto border-t border-gray-50 pt-3">
                    <span className="text-sm font-black text-gray-900">
                      {typeof product.selling_price === 'number' 
                        ? product.selling_price.toLocaleString('ko-KR') + '원' 
                        : product.selling_price}
                    </span>
                    <span className="text-3xs text-gray-400 font-bold tracking-wide">AUTHENTIC 100%</span>
                  </div>
                </div>
              </div>
            ))}
          </div>
        ) : (
          /* 상품이 없는 경우의 예외 뷰 */
          <div className="border border-dashed border-gray-200 py-24 text-center rounded-sm">
            <span className="text-3xs font-black text-gray-300 uppercase tracking-widest block mb-2">NO PRODUCTS FOUND</span>
            <p className="text-xs text-gray-400">선택하신 브랜드 분류의 명품 원장 매물이 현재 전시 대기 중입니다.</p>
          </div>
        )}
      </div>
    </div>
  );
}
