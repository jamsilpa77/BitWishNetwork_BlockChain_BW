[1단계] 컴퓨터에서 '메모장' 실행하기

1.  모니터 왼쪽 아래에 있는 **Windows 시작 버튼(돋보기 모양 또는 검색창)**을 마우스로 클릭합니다.
2.  한글로 메모장이라고 타이핑한 뒤, 검색 결과에 나타난 메모장 앱을 클릭하여 실행합니다.

아래 적힌 검은색 박스 안의 내용을 마우스로 드래그하여 전부 복사(단축키: Ctrl + C)한 뒤, 방금 열어둔 메모장 화면에 그대로

매입/위탁/감정/마진정산 테이블을 유기적으로 연결하는 핵심 설계도입니다.

-- =========================================================================
-- LUXURY RESALE SaaS PLATFORM - CORE DATABASE SCHEMA (POSTGRESQL)
-- =========================================================================

-- UUID 생성 및 암호화 확장이 필요한 경우 활성화
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";
CREATE EXTENSION IF NOT EXISTS "pgcrypto";

-- 1. 테넌트(고객사 쇼핑몰) 메타 정보 테이블
CREATE TABLE tenants (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    name VARCHAR(100) NOT NULL,                        -- 쇼핑몰명 (예: 그랑파리, 시계천국)
    slug VARCHAR(50) UNIQUE NOT NULL,                  -- 고유 도메인 주소 식별자 (예: grandparis)
    status VARCHAR(20) NOT NULL DEFAULT 'ACTIVE',      -- 상태 (ACTIVE, SUSPENDED, PENDING)
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 2. 테넌트 도메인 매핑 테이블 (독립 도메인 연결용)
CREATE TABLE tenant_domains (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    tenant_id UUID REFERENCES tenants(id) ON DELETE CASCADE,
    domain VARCHAR(255) UNIQUE NOT NULL,               -- 연결된 독립 도메인 (예: www.grandparis.kr)
    is_primary BOOLEAN DEFAULT FALSE,                  -- 대표 도메인 여부
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 3. 테넌트별 테마 및 브랜딩 설정 테이블 (화이트라벨의 핵심)
CREATE TABLE tenant_themes (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    tenant_id UUID UNIQUE REFERENCES tenants(id) ON DELETE CASCADE,
    template_key VARCHAR(50) NOT NULL DEFAULT 'LUXURY_MARKET', -- 5대 데모 템플릿 코드
    primary_color VARCHAR(7) NOT NULL DEFAULT '#000000',      -- 메인 테마 색상 (Hex Code)
    secondary_color VARCHAR(7) NOT NULL DEFAULT '#FFFFFF',    -- 서브 테마 색상
    logo_url TEXT,                                            -- 로고 이미지 주소
    favicon_url TEXT,                                         -- 파비콘 이미지 주소
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 4. 회원 정보 테이블 (모든 쇼핑몰의 회원이 이 테이블에 담기되, tenant_id로 분리됨)
CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    tenant_id UUID NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
    email VARCHAR(150) NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    name VARCHAR(100) NOT NULL,
    phone VARCHAR(30),
    role VARCHAR(30) NOT NULL DEFAULT 'USER',          -- 권한 (USER, SUPER_ADMIN, TENANT_ADMIN, MD)
    status VARCHAR(20) NOT NULL DEFAULT 'ACTIVE',      -- 상태 (ACTIVE, BANNED, WITHDRAWN)
    points INT DEFAULT 0,                              -- 적립금
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    CONSTRAINT uniq_tenant_email UNIQUE (tenant_id, email) -- 테넌트 내부에서는 이메일이 유니크해야 함
);

-- 5. 명품 브랜드 메타 정보 테이블
CREATE TABLE brands (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    tenant_id UUID NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
    name VARCHAR(100) NOT NULL,                        -- 브랜드 한글명 (예: 에르메스)
    english_name VARCHAR(100) NOT NULL,                -- 브랜드 영문명 (예: Hermes)
    slug VARCHAR(100) NOT NULL,                        -- URL용 식별자
    is_popular BOOLEAN DEFAULT FALSE,                  -- 인기 브랜드 지정 노출 여부
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 6. 계층형 카테고리 테이블 (3단계 대/중/소 분류 지원)
CREATE TABLE categories (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    tenant_id UUID NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
    parent_id UUID REFERENCES categories(id) ON DELETE SET NULL, -- 부모 카테고리 ID
    name VARCHAR(100) NOT NULL,                        -- 카테고리명 (예: BAG, WATCH)
    depth INT NOT NULL DEFAULT 1,                      -- 대분류(1), 중분류(2), 소분류(3)
    sort_order INT DEFAULT 0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 7. 상품 테이블 (중고 명품의 정교한 속성 포함)
CREATE TABLE products (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    tenant_id UUID NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
    brand_id UUID REFERENCES brands(id) ON DELETE SET NULL,
    category_id UUID REFERENCES categories(id) ON DELETE SET NULL,
    name VARCHAR(255) NOT NULL,                        -- 상품명
    product_code VARCHAR(100) UNIQUE NOT NULL,         -- 상품 자체 관리코드
    serial_number VARCHAR(100),                        -- 명품 자체 고유 시리얼 번호
    condition_grade VARCHAR(5) NOT NULL,               -- 상태등급 (N, S, A, B, C)
    cost_price DECIMAL(12, 2) NOT NULL DEFAULT 0.00,   -- 매입 원가
    selling_price DECIMAL(12, 2) NOT NULL DEFAULT 0.00,-- 정가 판매가
    discount_price DECIMAL(12, 2),                     -- 할인가 판매가 (옵션)
    stock_quantity INT NOT NULL DEFAULT 1,             -- 중고 특성상 대부분 1개이나 다수 재고도 대응 가능
    view_count INT DEFAULT 0,                          -- 조회수
    wish_count INT DEFAULT 0,                          -- 찜수
    status VARCHAR(30) NOT NULL DEFAULT 'FOR_SALE',    -- 상태 (FOR_SALE, SOLD_OUT, HOLD, ON_APPRAISAL)
    description TEXT,                                  -- HTML 에디터 내용
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 8. 상품 이미지 테이블 (최대 10개 및 정렬 순서 보장)
CREATE TABLE product_images (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    product_id UUID NOT NULL REFERENCES products(id) ON DELETE CASCADE,
    image_url TEXT NOT NULL,                           -- S3 / CloudFront 웹 보안 주소
    is_main BOOLEAN DEFAULT FALSE,                     -- 대표 이미지 여부
    sort_order INT DEFAULT 0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 9. 명품 감정서 테이블 (1:1 구조)
CREATE TABLE appraisal_certificates (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    product_id UUID UNIQUE REFERENCES products(id) ON DELETE CASCADE,
    certificate_image_url TEXT,                       -- 감정서 이미지 파일 주소
    serial_number VARCHAR(100),                        -- OCR 추출 및 대조 완료된 시리얼넘버
    appraiser_name VARCHAR(100),                       -- 감정사명
    appraisal_date DATE,                               -- 감정일자
    is_authentic BOOLEAN DEFAULT TRUE,                 -- 진품 여부 (기본 진품, 가품일 시 회수/경보)
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 10. 매입 신청 테이블 (C2B 매입 프로세스 관리)
CREATE TABLE purchase_requests (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    tenant_id UUID NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
    user_id UUID REFERENCES users(id) ON DELETE SET NULL,
    product_name VARCHAR(255) NOT NULL,
    brand_name VARCHAR(100),
    condition_description TEXT,                        -- 고객이 적은 상태 설명
    estimated_price DECIMAL(12, 2),                    -- 가검수 책정가
    final_price DECIMAL(12, 2),                        -- 실물 입고 후 최종 확정 매입가
    status VARCHAR(30) NOT NULL DEFAULT 'REQUESTED',   -- 상태 (REQUESTED, RECEIVED, APPRAISED, PAID, REJECTED)
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

rem -- 11. 위탁 신청 테이블 (C2B 위탁 프로세스 관리)

@Injectable()
export class TenantContextMiddleware implements NestMiddleware {
  async use(req: Request, res: Response, next: NextFunction) {
    // 1. 요청이 들어온 도메인 호스트 주소 가져오기 (예: watch-exchange.localhost:4000)
    const host = req.headers.host || '';
    
    // API 개발 및 테스트 도구(Postman 등)를 위해 헤더로 수동 전달된 슬러그 값도 체크

    // 예: 'watch-exchange.localhost:4000' -> 'watch-exchange'만 파싱
    if (!tenantSlug && host) {

      const parts = cleanHost.split('.');
      if (parts.length > 1) {
        tenantSlug = parts[0]; 
      } else {
        tenantSlug = cleanHost;
      }
    }

    // 3. [가상 테넌트 주소 매핑 딕셔너리]

    // 지금은 데이터베이스 연결 이전에 핵심 시스템이 무결하게 동작하도록 데모용 매핑 회로를 구축합니다.

      'watch-exchange.localhost': 'b2c2d2e2-f2a2-11ee-8f19-325096b39f02',

      'localhost:4000': 'a1b1c1d1-e1f1-11ee-8f19-325096b39f01', // 기본 로컬 테스트용은 종합몰로 매핑

    let tenantId = demoDomainMap[cleanHostKey];

    // 호스트 주소 기반 매핑에 실패한 경우 슬러그 이름으로 한 번 더 확인
    if (!tenantId && tenantSlug) {
      const slugMap: Record<string, string> = {
        'luxury-market': 'a1b1c1d1-e1f1-11ee-8f19-325096b39f01',
        'watch-exchange': 'b2c2d2e2-f2a2-11ee-8f19-325096b39f02',
        'bag-gallery': 'c3d3e3f3-a3b3-11ee-8f19-325096b39f03',
        'premium-closet': 'd4e4f4a4-b4c4-11ee-8f19-325096b39f04',
        'luxury-auction': 'e5f5a5b5-c5d5-11ee-8f19-325096b39f05',
      };
      tenantId = slugMap[tenantSlug];

    if (!tenantId) {

    // 이 꼬리표는 하위 컨트롤러, 데이터베이스 쿼리 레이어까지 안전하게 이동합니다.
    req['tenantId'] = tenantId;
    req['tenantSlug'] = tenantSlug || host;

import { Pool, PoolClient, QueryResult } from 'pg';

@Injectable()
export class TenantDatabaseConnection implements OnModuleInit, OnModuleDestroy {
  // PostgreSQL 연결 객체 풀(Pool) 선언
  private pool: Pool;

  // 1. 서버가 부팅될 때 자동으로 가동되어 DB와 미리 악수(연결 배관 20개 유지)를 나눕니다.
  async onModuleInit() {
    // 앞서 저장해둔 .env 파일의 실제 설정 정보를 읽어옵니다.
    this.pool = new Pool({
      host: process.env.DB_HOST || 'localhost',
      port: parseInt(process.env.DB_PORT || '5432', 10),
      user: process.env.DB_USERNAME || 'postgres',
      password: process.env.DB_PASSWORD,
      database: process.env.DB_DATABASE || 'luxury_resale_db',
      max: 20,                          // 동시에 대기시킬 최대 연결선 개수 (기본 20개)
      idleTimeoutMillis: 30000,         // 미사용 시 연결선을 차단하고 반납하는 시간

  async onModuleDestroy() {
    if (this.pool) {
      await this.pool.end();
    }

  // 백엔드 내의 모든 API방(상품방, 회원방 등)에서 데이터 조회가 필요할 때 이 공통 쿼리 함수를 호출합니다.

  async query<T = any>(
    sql: string,
    params: any[] = [],
    tenantId?: string
  ): Promise<QueryResult<T>> {
    try {

      if (tenantId) {

        if (!lowerSql.includes('tenant_id') && !lowerSql.includes('tenants')) {

            `[SECURITY WARNING] Tenant ID is provided (${tenantId}), but tenant_id scope is missing in SQL: ${sql}`
          );
        }
      }

      // 안전하게 연결 풀에서 커넥션을 꺼내 쿼리 실행
      return await this.pool.query(sql, params);
    } catch (error: any) {

      );
    }
  }

  // "결제 완료"와 동시에 "재고 차감"이 한 번에 완벽히 이루어져야 하는 중요 금융 업무 시,

import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';

export function middleware(request: NextRequest) {
  const url = request.nextUrl.clone();
  const hostname = request.headers.get('host') || '';

  // 1. [정적 리소스 필터링]
  // 폰트, 아이콘, 넥스트 빌드 파일, static 정적 리소스 파일 등은 
  // 도메인 변환 감시 대상에서 제외하고 아무런 변형 없이 그대로 통과시킵니다.
  if (
    url.pathname.startsWith('/_next') ||
    url.pathname.startsWith('/api') ||
    url.pathname.startsWith('/static') ||
    url.pathname.includes('.')
  ) {
    return NextResponse.next();
  }

  // 2. [도메인 호스트명 정밀 분석]
  // 예: 'watch-exchange.localhost:3000' -> 'watch-exchange' 슬러그 분석
  const cleanHost = hostname.split(':')[0].trim(); // 포트번호 제거
  const parts = cleanHost.split('.');
  
  let tenant = 'luxury-market'; // 기본 매핑 템플릿 지정 (종합몰)

  if (parts.length > 1) {
    // 서브도메인이 있는 경우 (예: watch-exchange.luxury-saas.com)
    tenant = parts[0]; 
  } else {
    // 서브도메인이 없는 경우 로컬 환경 예외 매핑
    if (cleanHost === 'localhost' || cleanHost === '127.0.0.1') {
      tenant = 'luxury-market';
    } else {
      // 실제 독립 커스텀 도메인(예: grandparis.kr) 처리용 바인딩
      tenant = cleanHost; 
    }
  }

  // 3. [가상 호스팅 패스 Rewrite 처리]
  // 사용자의 눈에는 브라우저 주소창에 'http://watch-exchange.localhost:3000/products'로 깔끔하게 노출되지만,
  // 넥스트 서버 내부적으로는 멀티테넌트 렌더링 전용 가상 폴더인 '/[tenant]/products' 즉, 
  // '/watch-exchange/products' 폴더로 은밀하게 경로를 매핑하여 해당 템플릿 스킨을 호출합니다.
  url.pathname = `/${tenant}${url.pathname}`;
  
  return NextResponse.rewrite(url);
}

// 4. [감시망 대상 필터 매처 정의]
// 시스템 오동작을 줄이기 위해 이미지 최적화 주소나 API 주소는 해당 감시 매커니즘에서 물리 제외 처리

과정을 끈기 있게 완수해 내셨습니다. 이제 사장님의 컴퓨터는 이 고도화된 SaaS 플랫폼 엔진을 무사히 구동시키고 있습니다.

모든 엔진이 정상 가동되었으니, 이제 인터넷 창(크롬 등)을 열고 두 개의 주소에 각각 접속하여 실물을 감상해 보시기 바랍니다.

1. 첫 번째 확인: 실제 쇼핑몰 프론트엔드 (고객 화면)

  - 접속 주소: http://localhost:3001
  - 내용: 우리가 데이터베이스 시드로 심어놓은 에르메스 벌킨 백, 샤넬 클래식 백, 롤렉스 서브마리너 시계 카드 등이 예쁜 그리드 형태로
    화면에 뿌려지는 쇼핑몰 메인 화면입니다.
  - 테스트: 상단의 메뉴(SHOP, 명품 매입신청 등)가 공통 레이아웃 프레임에 맞춰 유기적으로 정합되어 있습니다.

2. 두 번째 확인: 비즈니스 솔루션용 자동 기술 설명서 (Swagger)

  - 접속 주소: http://localhost:4000/docs
  - 내용: 사장님이 이 솔루션을 통째로 다른 명품 매장 사장님들께 판매하실 때, **"우리 솔루션은 이러한 표준 백엔드 API
    명세서(Swagger)를 100% 자동 지원합니다"**라고 기술력을 증명할 수 있는 최고급 명세서 사이트입니다.
  - 의미: 백엔드가 정상적으로 가동되어야만 이 문서 페이지가 정상 노출됩니다.

훌륭한 첫 마일스톤을 달성하셨습니다!

이제 비즈니스의 가장 단단한 바닥(인프라 엔진)을 완전히 확보하셨습니다. 언제 어디서든 바탕화면에 만들어둔 쇼핑몰원클릭시작.bat 파일을
실행하시는 것만으로 두 서버가 동시에 안전하게 연동되어 시작될 것입니다.

정상적으로 화면이 뜨는지 인터넷 브라우저에서 차근차근 구경해 보시고, 다음 단계로 **개별 디테일한 기능 방들(명품 실물 감정서 팝업창 렌더링,
매입/위탁 신청서 연동 기능 등)**의 코드 작성을 이어서 진행하고자 하실 때 언제든 저를 호출해 주세요. 사장님의 비즈니스를 끝까지 든든하게
보좌해 드리겠습니다!

스웨거(Swagger) 문서 사이트(http://localhost:4000/docs)가 아주 깔끔하게 뜨는 모습을 보니 백엔드 서버는 무결하게
정말 잘 가동되고 있습니다!

화면을 그리지 못한 것입니다. 불편을 드려 대단히 죄송합니다.

아래의 완전히 수정된 코드로 덮어쓰기 해주시면 오류가 즉시 해결되고 화면이 나타날 것입니다.
