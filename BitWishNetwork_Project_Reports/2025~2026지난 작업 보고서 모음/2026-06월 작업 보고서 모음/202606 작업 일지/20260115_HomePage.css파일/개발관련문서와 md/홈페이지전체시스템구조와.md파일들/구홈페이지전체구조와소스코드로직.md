  const hashPassword = (password: string): string => {
    // 🎯 간단한 해싱 로직 (실제 운영에서는 bcrypt 사용)
    let hash = 0;
    for (let i = 0; i < password.length; i++) {
      const char = password.charCodeAt(i);
      hash = ((hash << 5) - hash) + char;
      hash = hash & hash; // 32bit 정수로 변환
    }
    return hash.toString(16);
  };

    const inputHash = hashPassword(inputPassword);
    return inputHash === hashedPassword;

    // React 상태 업데이트
    const newMap = new Map(walletPasswords).set(walletAddress, hashedPassword);
    setWalletPasswords(newMap);
    
    // localStorage에 해싱된 비밀번호만 저장 (원본 비밀번호는 절대 저장 안함)
    try {
      const currentStored = JSON.parse(localStorage.getItem('walletPasswords') || '{}');
      currentStored[walletAddress] = hashedPassword;
      localStorage.setItem('walletPasswords', JSON.stringify(currentStored));
      console.log(`🔐 ${walletAddress}의 2차 비밀번호 설정 완료 (해싱된 값만 localStorage 저장)`);

                        </div>
                      </div>

                  {!isMiningActive && showMiningStopStatus && (

                      <h3 className="text-lg font-bold text-purple-700 text-center mb-4">⏹️ 마이닝 정지 상태 표시 창</h3>
                      
                      <div className="grid grid-cols-2 gap-6">
                        {/* 좌측 열: 시간당 보상 및 보너스 */}
                        <div className="space-y-3">
                          <div className="text-center">
                            <div className="text-2xl font-bold text-purple-600">0.000 BW</div>
                            <p className="text-sm text-gray-600">시간당 보상</p>
                          </div>
                          <div className="text-center">
                            <div className="text-lg font-semibold text-purple-600">0%</div>
                            <p className="text-sm text-gray-600">출석 보너스</p>
                          </div>
                          <div className="text-center">
                            <div className="text-lg font-semibold text-purple-600">0%</div>
                            <p className="text-sm text-gray-600">락업 보너스</p>
                          </div>
                        </div>
                        
                        {/* 우측 열: 일일 보상 및 추천 보너스 */}
                        <div className="space-y-3">
                          <div className="text-center">
                            <div className="text-2xl font-bold text-purple-600">0.000 BW</div>
                            <p className="text-sm text-gray-600">일일 보상</p>
                          </div>
                          <div className="text-center">
                            <div className="text-lg font-semibold text-purple-600">0%</div>
                            <p className="text-sm text-gray-600">추천 보너스</p>
                          </div>
                          <div className="text-center">
                            <div className="text-lg font-semibold text-purple-600">0 BW</div>
                            <p className="text-sm text-gray-600">추천 보관 BW</p>
                          </div>
                        </div>
                      </div>
                      
                      {/* 마이닝 정지 상태 메시지 */}
                      <div className="text-center mt-4">
                        <p className="text-sm text-gray-600">
                          마이닝이 정지되었습니다
                        </p>
                      </div>

                      {/* 🎯 마이닝 정지 상태 표시만 - 하단 버튼들로 제어 */}
                      <div className="text-center mt-4">
                        <p className="text-xs text-gray-500">
                          💡 하단의 "마이닝 시작" 버튼을 클릭하여 마이닝을 다시 시작할 수 있습니다
                        </p>
                      </div>
                    </div>
                  )}

                  {/* 🧪 실시간 테스트 마이닝 보상 (녹색 배경) - 완벽 복원 */}
                  {testDataState.isTestActive && (
                    <div className="bg-green-100 border border-green-300 rounded-lg p-4 mt-4">
                      <h3 className="text-lg font-bold text-green-700 text-center mb-3">실시간 테스트 마이닝 보상</h3>
                      
                      <div className="grid grid-cols-2 gap-4">
                        <div className="text-center">
                          <div className="text-2xl font-bold text-green-600 mb-2">
                            {testDataState.testHourlyReward} BW/시간
                          </div>
                          <p className="text-sm text-green-700">시간당 기본 보상</p>
                        </div>
                        
                        <div className="text-center">
                          <div className="text-2xl font-bold text-green-600 mb-2">
                            {realTimeBlockchainData.miningRewards.toFixed(3)} BW

                          <p className="text-sm text-green-700">누적 보상</p>

                    let totalElapsedTime = pausedElapsedTime;

                    const hours = Math.floor(totalElapsedTime / (1000 * 60 * 60));
                    const minutes = Math.floor((totalElapsedTime % (1000 * 60 * 60)) / (1000 * 60));
                    const seconds = Math.floor((totalElapsedTime % (1000 * 60)) / 1000);

                    // 총 누적 보상을 계산합니다. (보너스가 포함된 강화된 채굴률 적용)

                    const lockupBonus = 0; // 락업 보너스는 0으로 설정

                    // 🎯 정확한 강화된 채굴률 계산
                    const totalEnhancedRate = baseRate * (1 + attendanceBonus + referralBonus + lockupBonus + merchantBonus);
                    
                    // 🎯 정밀한 시간 계산 (밀리초 단위로 정확하게)
                    const elapsedHours = totalElapsedTime / (1000 * 60 * 60);
                    const totalRewards = elapsedHours * totalEnhancedRate;
                    
                    // 🎯 디버깅용 로그
                    console.log('🔍 누적 보상 계산 상세:', {
                      baseRate,
                      attendanceBonus: attendanceBonus * 100 + '%',
                      referralBonus: referralBonus * 100 + '%',
                      lockupBonus: lockupBonus * 100 + '%',
                      merchantBonus: merchantBonus * 100 + '%',
                      totalEnhancedRate,
                      elapsedHours: elapsedHours.toFixed(6),
                      totalRewards: totalRewards.toFixed(6)
                    });

                    // tick 변수를 사용하여 강제 리렌더링 (실시간 업데이트)
                    console.log('실시간 업데이트 tick:', tick);

                    return (
                      <div className="bg-green-100 border border-green-300 rounded-lg p-3 mt-4 relative">
                        {/* 🎯 관리자 전용 마이닝 초기화 버튼 (오른쪽 위 모서리) */}
                        {authenticatedWalletAddress === 'GDG3OPHDK5WGTAG2UVEMDUUGMGIBSZMPF4RAUPVRQMG7PRNAZT5YU256' && (
                          <button
                            onClick={async () => {
                              if (confirm('🔒 관리자 전용: 마이닝을 완전히 초기화하시겠습니까?\n\n⚠️ 이 작업은 되돌릴 수 없습니다.')) {
                                try {
                                  console.log('🔒 관리자 마이닝 초기화 시작...');
                                  
                                  // 🎯 가상 마이닝 완전 초기화
                                  await resetVirtualMining();
                                  
                                  // 💾 localStorage에서 마이닝 상태 완전 삭제
                                  localStorage.removeItem('miningState');
                                  
                                  // 🎯 모든 마이닝 관련 상태 초기화
                                  setIsMiningActive(false);
                                  setMiningStartTime(null);
                                  setPausedElapsedTime(0);
                                  setTick(0);
                                  
                                  // 🎯 블록체인 상태 초기화
                                  setRealTimeBlockchainData(prev => ({
                                    ...prev,
                                    attendanceBonus: 0,
                                    referralBonus: 0,
                                    lockupBonus: 0,
                                    miningRewards: 0,
                                    lastUpdate: new Date()
                                  }));
                                  
                                  console.log('🔒 관리자 마이닝 초기화 완료');
                                  alert('✅ 관리자 마이닝 초기화가 완료되었습니다.');
                                } catch (error) {
                                  console.error('❌ 관리자 마이닝 초기화 오류:', error);
                                  alert('❌ 마이닝 초기화 중 오류가 발생했습니다.');
                                }
                              }
                            }}
                            className="absolute top-2 right-2 px-3 py-1 bg-red-600 text-white text-xs font-bold rounded-lg hover:bg-red-700 transition-colors shadow-lg z-10"
                            title="🔒 관리자 전용: 마이닝 완전 초기화"
                          >
                            🔒 초기화
                          </button>
                        )}
                        
                        <h3 className="text-lg font-bold text-green-700 text-center mb-3">실시간 마이닝 보상</h3>
                        <div className="grid grid-cols-2 gap-4">
                          {/* 누적 보상 */}
                          <div className="text-center">
                            <div className="text-2xl font-bold text-green-600 mb-2">
                              {totalRewards.toFixed(6)} BW
                            </div>
                            <p className="text-sm text-green-700">누적 보상</p>
                          </div>
                          
                          {/* 진행 시간 */}
                          <div className="text-center">
                            <div className="text-2xl font-bold text-green-600 mb-2 tabular-nums">
                              {/* 시:분:초 형태로 표시 */}
                              {String(hours).padStart(2, '0')}:
                              {String(minutes).padStart(2, '0')}:
                              {String(seconds).padStart(2, '0')}
                            </div>
                            <p className="text-sm text-green-700">진행 시간</p>
                          </div>
                        </div>
                        <div className="text-center mt-3">
                          <p className="text-sm text-green-700">
                            {isMiningActive ? '현재 마이닝 진행 중' : '마이닝이 정지됨'}
                          </p>
                        </div>
                      </div>
                    );
                  })()}

                  {/* 🎯 보너스 설정 섹션 (완벽한 설정) */}


                  {/* 🎯 오리지널 마이닝 시스템 액션 버튼들 (하단) - 테스트 데이터 완전 제거 */}
                  <div className="grid grid-cols-4 gap-4 mt-6">
                    <button 
                      onClick={async () => {
                        if (!authenticatedWalletAddress) {
                          alert('지갑 주소가 인증되지 않았습니다. 먼저 지갑 주소를 인증해주세요.');
                          return;
                        }
                        
                        // 🎯 마이닝 시작 시에는 보너스 유효성 검사하지 않음

                        try {
                          // 🎯 백서 정확한 보너스 공식과 함께 마이닝 시작
                          await startVirtualMining();
                          
                          setShowMiningStopStatus(false);
                          
                          console.log('✅ 백서 정확한 보너스 공식과 함께 마이닝 시작됨');
                        } catch (error) {
                          console.error('❌ 마이닝 시작 오류:', error);
                        }
                      }}
                      className="px-4 py-3 bg-purple-600 text-white font-semibold rounded-lg hover:bg-purple-700 transition-colors"
                    >
                      ⚡ 마이닝 시작 (보너스 적용)
                    </button>
                    
                    <button 
                      onClick={async () => {
                        if (!authenticatedWalletAddress) {
                          alert('지갑 주소가 인증되지 않았습니다. 먼저 지갑 주소를 인증해주세요.');
                          return;
                        }
                        try {
                          // 🎯 stopMining 함수 사용 (가상 더미 마이닝 시스템 연결)
                          await stopMining();
                          
                          // ✅ 마이닝 상태만 변경 (UI 데이터는 그대로 유지)
                          setIsMiningActive(false);
                          
                          // ✅ 마이닝 정지 시간 기록 (현재 진행 시간 고정용)
                          setMiningStopTime(new Date());
                          
                          // ✅ 마이닝 상태 저장 (localStorage에 현재 상태 보존)
                          saveMiningState();
                          
                          console.log('🛑 마이닝 정지됨 - UI 데이터는 그대로 유지');
                        } catch (error) {
                          console.error('❌ 가상 더미 마이닝 정지 오류:', error);
                          alert('가상 더미 마이닝 시스템에 연결할 수 없습니다.');
                        }
                      }}
                      className="px-4 py-3 bg-red-600 text-white font-semibold rounded-lg hover:bg-red-700 transition-colors"
                    >
                      ⏹️ 마이닝 정지
                    </button>
                    
                    <button 
                      onClick={() => {
                        setShowMiningBonusModal(false);
                        setSeedPhrase(''); // 🎯 시드문구 초기화
                        setShowMyWallet(true);
                      }}
                      className="px-4 py-3 bg-gray-600 text-white font-semibold rounded-lg hover:bg-gray-700 transition-colors"
                    >
                      🔑 나의 지갑
                    </button>
                    
                    <button 
                      onClick={() => {
                        // 🎯 마이닝 및 보너스 설정 창 닫기 (마이닝 상태 유지)
                        setShowMiningBonusModal(false);
                        // ❌ 마이닝 상태 초기화 제거 - setIsMiningActive(false), setMiningStartTime(null) 삭제
                        setShowMiningStopStatus(false);
                      }}
                      className="px-4 py-3 bg-gray-600 text-white font-semibold rounded-lg hover:bg-gray-700 transition-colors"
                    >
                      ❌ 닫기
                    </button>
                  </div>
            </div>
          </div>
        </div>
      )}

      {/* 푸터 */}
      <footer className="bg-gray-800 text-white py-12 mt-20">
        <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8">
          <div className="grid md:grid-cols-4 gap-8">
            <div>
              <h3 className="text-lg font-semibold mb-4">{t('footer.company')}</h3>
              <p className="text-gray-300 text-sm">
                {t('footer.description')}
              </p>
            </div>
            
            <div>
              <h4 className="text-md font-semibold mb-4">{t('footer.products')}</h4>
              <ul className="space-y-2 text-sm text-gray-300">
                <li><a href="#" className="hover:text-white transition-colors">{t('footer.bwMainnet')}</a></li>
                <li><a href="#" className="hover:text-white transition-colors">{t('footer.bwNode')}</a></li>
                <li><a href="#" className="hover:text-white transition-colors">{t('footer.bwWallet')}</a></li>
              </ul>
            </div>
            
            <div>
              <h4 className="text-md font-semibold mb-4">{t('footer.community')}</h4>
              <ul className="space-y-2 text-sm text-gray-300">
                <li><a href="#" className="hover:text-white transition-colors">{t('footer.bwCommunity')}</a></li>
                <li><a href="#" className="hover:text-white transition-colors">{t('footer.developerDocs')}</a></li>
                <li><a href="#" className="hover:text-white transition-colors">{t('footer.supportCenter')}</a></li>
              </ul>
            </div>
            
            <div>
              <h4 className="text-md font-semibold mb-4">{t('footer.companyInfo')}</h4>
              <ul className="space-y-2 text-sm text-gray-300">
                <li><a href="#" className="hover:text-white transition-colors">{t('footer.about')}</a></li>
                <li><a href="#" className="hover:text-white transition-colors">{t('footer.careers')}</a></li>
                <li><a href="#" className="hover:text-white transition-colors">{t('footer.contact')}</a></li>
              </ul>
            </div>
          </div>
          
          <div className="border-t border-gray-700 mt-8 pt-8 text-center">
            <p className="text-gray-300 text-sm">
              {t('footer.copyright')}
            </p>
          </div>
        </div>
      </footer>

    </div>
    );
}
