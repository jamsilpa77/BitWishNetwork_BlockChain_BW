app.post('/api/bitwish/wallet/authenticate', async (req, res) => {
  try {
    const { walletAddress, password } = req.body;
    
    // 1. 지갑 존재 확인
    const wallet = await walletsCollection.findOne({ walletAddress });
    if (!wallet) {
      return res.status(404).json({
        success: false,
        error: '지갑을 찾을 수 없습니다.'
      });
    }
    
    // 2. 인증 정보 조회
    const authInfo = await authCollection.findOne({ walletAddress });
    if (!authInfo) {
      return res.status(400).json({
        success: false,
        error: '비밀번호가 설정되지 않았습니다.'
      });
    }
    
    // 3. 계정 잠금 확인
    if (authInfo.lockedUntil && authInfo.lockedUntil > new Date()) {
      return res.status(423).json({
        success: false,
        error: '계정이 일시적으로 잠겼습니다. 나중에 다시 시도해주세요.'
      });
    }
    
    // 4. 비밀번호 검증
    const isPasswordValid = await verifyPassword(password, authInfo.hashedPassword);
    
    if (!isPasswordValid) {
      // 5. 실패 시도 횟수 증가
      const newFailedAttempts = authInfo.failedAttempts + 1;
      const lockedUntil = newFailedAttempts >= 5 ? 
        new Date(Date.now() + 30 * 60 * 1000) : null; // 5회 실패 시 30분 잠금
      
      await authCollection.updateOne(
        { walletAddress },
        { 
          $set: { 
            failedAttempts: newFailedAttempts,
            lockedUntil 
          }
        }
      );
      
      return res.status(401).json({
        success: false,
        error: '비밀번호가 일치하지 않습니다.'
      });
    }
    
    // 6. 성공 시 실패 시도 횟수 초기화
    await authCollection.updateOne(
      { walletAddress },
      { 
        $set: { 
          failedAttempts: 0,
          lockedUntil: null,
          lastLoginAt: new Date()
        }
      }
    );
    
    // 7. 사용자 정보 업데이트
    await usersCollection.updateOne(
      { walletAddress },
      { $set: { lastLoginAt: new Date() } }
    );
    
    // 8. 세션 생성
    const sessionId = generateSessionId();
    const sessionData = {
      sessionId,
      walletAddress,
      userId: wallet.userId,
      userType: 'USER',
      createdAt: new Date(),
      expiresAt: new Date(Date.now() + 24 * 60 * 60 * 1000) // 24시간
    };
    
    activeSessions.set(sessionId, sessionData);
    
    res.json({
      success: true,
      message: '지갑 인증이 성공했습니다.',
      sessionId,
      wallet: {
        address: wallet.address,
        createdAt: wallet.createdAt,
        lastLoginAt: new Date(),
        hasPassword: true
      }
    });
    
  } catch (error) {
    console.error('지갑 인증 오류:', error);
    res.status(500).json({
      success: false,
      error: '지갑 인증 중 오류가 발생했습니다.'
    });
  }
});
