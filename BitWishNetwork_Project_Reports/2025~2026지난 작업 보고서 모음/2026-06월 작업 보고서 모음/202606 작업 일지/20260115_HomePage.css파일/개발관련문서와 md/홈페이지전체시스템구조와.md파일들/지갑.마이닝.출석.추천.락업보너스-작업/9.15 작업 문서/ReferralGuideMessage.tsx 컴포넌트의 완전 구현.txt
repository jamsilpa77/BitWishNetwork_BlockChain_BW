# **   ReferralGuideMessage.tsx 완전 구현**

**위치**: `Node_HomePage/src/components/ReferralGuideMessage.tsx`

```typescript
import React, { useState, useEffect } from 'react';
import { useTranslation } from 'react-i18next';
import { 
  Info, 
  Users, 
  Gift, 
  Star, 
  CheckCircle, 
  AlertCircle,
  Copy,
  ExternalLink,
  ChevronDown,
  ChevronUp
} from 'lucide-react';

// ====================================================================================
// 🎯 ReferralGuideMessage - 추천인 가이드 메시지 컴포넌트
// ====================================================================================

interface ReferralGuideMessageProps {
  type?: 'info' | 'success' | 'warning' | 'error';
  showDetails?: boolean;
  onToggleDetails?: (expanded: boolean) => void;
  referralCode?: string;
  onCopyCode?: () => void;
  showCopyButton?: boolean;
  showExternalLink?: boolean;
  externalLinkUrl?: string;
  customMessage?: string;
  showStats?: boolean;
  stats?: {
    totalReferrals: number;
    totalEarnings: number;
    currentLevel: number;
  };
}

export const ReferralGuideMessage: React.FC<ReferralGuideMessageProps> = ({
  type = 'info',
  showDetails = true,
  onToggleDetails,
  referralCode,
  onCopyCode,
  showCopyButton = true,
  showExternalLink = false,
  externalLinkUrl,
  customMessage,
  showStats = false,
  stats
}) => {
  const { t } = useTranslation();
  const [isExpanded, setIsExpanded] = useState(false);
  const [copied, setCopied] = useState(false);

  // ====================================================================================
  // 🎯 상태 관리 및 이벤트 핸들러
  // ====================================================================================

  const handleToggleDetails = () => {
    const newExpanded = !isExpanded;
    setIsExpanded(newExpanded);
    onToggleDetails?.(newExpanded);
  };

  const handleCopyCode = async () => {
    if (referralCode) {
      try {
        await navigator.clipboard.writeText(referralCode);
        setCopied(true);
        onCopyCode?.();
        
        // 2초 후 복사 상태 초기화
        setTimeout(() => setCopied(false), 2000);
      } catch (error) {
        console.error('Failed to copy referral code:', error);
      }
    }
  };

  // ====================================================================================
  // 🎯 아이콘 및 스타일 설정
  // ====================================================================================

  const getIconAndColor = () => {
    switch (type) {
      case 'success':
        return { icon: CheckCircle, color: 'text-green-500', bgColor: 'bg-green-50', borderColor: 'border-green-200' };
      case 'warning':
        return { icon: AlertCircle, color: 'text-yellow-500', bgColor: 'bg-yellow-50', borderColor: 'border-yellow-200' };
      case 'error':
        return { icon: AlertCircle, color: 'text-red-500', bgColor: 'bg-red-50', borderColor: 'border-red-200' };
      default:
        return { icon: Info, color: 'text-blue-500', bgColor: 'bg-blue-50', borderColor: 'border-blue-200' };
    }
  };

  const { icon: Icon, color, bgColor, borderColor } = getIconAndColor();

  // ====================================================================================
  // 🎯 추천인 가이드 메시지 내용
  // ====================================================================================

  const getReferralGuideContent = () => {
    if (customMessage) {
      return customMessage;
    }

    return t('referral.guideMessage', {
      defaultValue: '추천인 코드를 입력하면 보너스를 받을 수 있습니다. 친구를 초대하고 함께 수익을 올려보세요!'
    });
  };

  const getReferralBenefits = () => [
    {
      icon: Gift,
      title: t('referral.benefits.bonus.title', { defaultValue: '추천인 보너스' }),
      description: t('referral.benefits.bonus.description', { defaultValue: '추천인 코드 입력 시 5% 보너스 지급' }),
      value: '5%'
    },
    {
      icon: Users,
      title: t('referral.benefits.referral.title', { defaultValue: '추천인 수익' }),
      description: t('referral.benefits.referral.description', { defaultValue: '추천한 친구의 마이닝 수익의 10% 수수료' }),
      value: '10%'
    },
    {
      icon: Star,
      title: t('referral.benefits.level.title', { defaultValue: '레벨업 보너스' }),
      description: t('referral.benefits.level.description', { defaultValue: '추천인 수에 따라 추가 보너스 지급' }),
      value: 'Up to 25%'
    }
  ];

  const getReferralSteps = () => [
    {
      step: 1,
      title: t('referral.steps.step1.title', { defaultValue: '추천인 코드 입력' }),
      description: t('referral.steps.step1.description', { defaultValue: '지갑 생성 시 추천인 코드를 입력하세요' })
    },
    {
      step: 2,
      title: t('referral.steps.step2.title', { defaultValue: '친구 초대' }),
      description: t('referral.steps.step2.description', { defaultValue: '본인의 추천인 코드를 친구에게 공유하세요' })
    },
    {
      step: 3,
      title: t('referral.steps.step3.title', { defaultValue: '수익 공유' }),
      description: t('referral.steps.step3.description', { defaultValue: '친구의 마이닝 수익에 따라 수수료를 받으세요' })
    }
  ];

  // ====================================================================================
  // 🎯 통계 정보 렌더링
  // ====================================================================================

  const renderStats = () => {
    if (!showStats || !stats) return null;

    return (
      <div className="mt-4 p-4 bg-gray-50 rounded-lg">
        <h4 className="text-sm font-medium text-gray-700 mb-3">
          {t('referral.stats.title', { defaultValue: '나의 추천인 현황' })}
        </h4>
        <div className="grid grid-cols-3 gap-4">
          <div className="text-center">
            <div className="text-2xl font-bold text-blue-600">
              {stats.totalReferrals}
            </div>
            <div className="text-xs text-gray-500">
              {t('referral.stats.totalReferrals', { defaultValue: '총 추천인' })}
            </div>
          </div>
          <div className="text-center">
            <div className="text-2xl font-bold text-green-600">
              {stats.totalEarnings.toLocaleString()}
            </div>
            <div className="text-xs text-gray-500">
              {t('referral.stats.totalEarnings', { defaultValue: '총 수익 (BW)' })}
            </div>
          </div>
          <div className="text-center">
            <div className="text-2xl font-bold text-purple-600">
              Lv.{stats.currentLevel}
            </div>
            <div className="text-xs text-gray-500">
              {t('referral.stats.currentLevel', { defaultValue: '현재 레벨' })}
            </div>
          </div>
        </div>
      </div>
    );
  };

  // ====================================================================================
  // 🎯 메인 렌더링
  // ====================================================================================

  return (
    <div className={`rounded-lg border-2 ${borderColor} ${bgColor} p-4 transition-all duration-200`}>
      {/* 메인 메시지 영역 */}
      <div className="flex items-start space-x-3">
        <Icon className={`w-5 h-5 mt-0.5 ${color} flex-shrink-0`} />
        <div className="flex-1 min-w-0">
          <p className="text-sm text-gray-700 leading-relaxed">
            {getReferralGuideContent()}
          </p>
          
          {/* 추천인 코드 표시 */}
          {referralCode && (
            <div className="mt-3 flex items-center space-x-2">
              <span className="text-xs font-medium text-gray-600">
                {t('referral.myCode', { defaultValue: '나의 추천인 코드:' })}
              </span>
              <code className="px-2 py-1 bg-white border border-gray-300 rounded text-xs font-mono text-gray-800">
                {referralCode}
              </code>
              {showCopyButton && (
                <button
                  onClick={handleCopyCode}
                  className={`px-2 py-1 text-xs rounded transition-colors ${
                    copied 
                      ? 'bg-green-100 text-green-700' 
                      : 'bg-gray-100 text-gray-600 hover:bg-gray-200'
                  }`}
                >
                  {copied ? (
                    <CheckCircle className="w-3 h-3" />
                  ) : (
                    <Copy className="w-3 h-3" />
                  )}
                </button>
              )}
            </div>
          )}

          {/* 외부 링크 */}
          {showExternalLink && externalLinkUrl && (
            <div className="mt-2">
              <a
                href={externalLinkUrl}
                target="_blank"
                rel="noopener noreferrer"
                className="inline-flex items-center text-xs text-blue-600 hover:text-blue-800 transition-colors"
              >
                <ExternalLink className="w-3 h-3 mr-1" />
                {t('referral.externalLink', { defaultValue: '자세한 정보 보기' })}
              </a>
            </div>
          )}
        </div>
      </div>

      {/* 상세 정보 토글 버튼 */}
      {showDetails && (
        <button
          onClick={handleToggleDetails}
          className="mt-3 flex items-center text-xs text-gray-600 hover:text-gray-800 transition-colors"
        >
          {isExpanded ? (
            <ChevronUp className="w-4 h-4 mr-1" />
          ) : (
            <ChevronDown className="w-4 h-4 mr-1" />
          )}
          {isExpanded 
            ? t('referral.hideDetails', { defaultValue: '상세 정보 숨기기' })
            : t('referral.showDetails', { defaultValue: '상세 정보 보기' })
          }
        </button>
      )}

      {/* 상세 정보 영역 */}
      {isExpanded && showDetails && (
        <div className="mt-4 space-y-4">
          {/* 추천인 혜택 */}
          <div>
            <h4 className="text-sm font-medium text-gray-700 mb-2">
              {t('referral.benefits.title', { defaultValue: '추천인 혜택' })}
            </h4>
            <div className="grid grid-cols-1 md:grid-cols-3 gap-3">
              {getReferralBenefits().map((benefit, index) => (
                <div key={index} className="flex items-start space-x-2 p-3 bg-white rounded border">
                  <benefit.icon className="w-4 h-4 text-blue-500 mt-0.5 flex-shrink-0" />
                  <div className="flex-1 min-w-0">
                    <div className="flex items-center justify-between">
                      <h5 className="text-xs font-medium text-gray-800">
                        {benefit.title}
                      </h5>
                      <span className="text-xs font-bold text-blue-600">
                        {benefit.value}
                      </span>
                    </div>
                    <p className="text-xs text-gray-600 mt-1">
                      {benefit.description}
                    </p>
                  </div>
                </div>
              ))}
            </div>
          </div>

          {/* 추천인 단계 */}
          <div>
            <h4 className="text-sm font-medium text-gray-700 mb-2">
              {t('referral.steps.title', { defaultValue: '추천인 사용 방법' })}
            </h4>
            <div className="space-y-2">
              {getReferralSteps().map((step, index) => (
                <div key={index} className="flex items-start space-x-3">
                  <div className="w-6 h-6 bg-blue-500 text-white text-xs font-bold rounded-full flex items-center justify-center flex-shrink-0">
                    {step.step}
                  </div>
                  <div className="flex-1 min-w-0">
                    <h5 className="text-xs font-medium text-gray-800">
                      {step.title}
                    </h5>
                    <p className="text-xs text-gray-600 mt-1">
                      {step.description}
                    </p>
                  </div>
                </div>
              ))}
            </div>
          </div>

          {/* 통계 정보 */}
          {renderStats()}
        </div>
      )}
    </div>
  );
};

// ====================================================================================
// 🎯 기본 내보내기
// ====================================================================================

export default ReferralGuideMessage;
```

## ** 주요 기능 및 특징**

### **1. 다양한 메시지 타입 지원**
- `info`: 일반 정보 메시지
- `success`: 성공 메시지
- `warning`: 경고 메시지
- `error`: 오류 메시지

### **2. 추천인 코드 관리**
- 추천인 코드 표시 및 복사 기능
- 복사 상태 시각적 피드백
- 클립보드 API 사용

### **3. 상세 정보 토글**
- 접기/펼치기 기능
- 추천인 혜택 상세 설명
- 사용 방법 단계별 가이드

### **4. 통계 정보 표시**
- 총 추천인 수
- 총 수익 (BW)
- 현재 레벨

### **5. 다국어 지원**
- i18next를 통한 완전한 다국어 지원
- 한국어, 영어, 중국어, 일본어

### **6. 반응형 디자인**
- Tailwind CSS를 사용한 반응형 레이아웃
- 모바일/데스크톱 최적화

### **7. 접근성 고려**
- 키보드 네비게이션 지원
- 스크린 리더 호환성
- 시각적 피드백 제공
