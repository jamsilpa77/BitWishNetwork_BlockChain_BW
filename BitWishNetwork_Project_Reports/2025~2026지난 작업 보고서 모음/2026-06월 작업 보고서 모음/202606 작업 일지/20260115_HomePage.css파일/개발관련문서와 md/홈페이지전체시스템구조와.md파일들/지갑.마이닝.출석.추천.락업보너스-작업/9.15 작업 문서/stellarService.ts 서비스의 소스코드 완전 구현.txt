export class StellarService {
  private server: Server;
  private config: StellarConfig;
  private network: Networks;

  constructor(config: StellarConfig) {
    this.config = config;
    this.network = config.network === 'mainnet' ? Networks.PUBLIC : Networks.TESTNET;
    this.server = new Server(config.horizonUrl);
  }

  // ====================================================================================
  // 🎯 연결 및 설정
  // ====================================================================================

  async connect(): Promise<boolean> {
    try {
      const response = await this.server.ledgers().order('desc').limit(1).call();
      return response.records.length > 0;
    } catch (error) {
      console.error('Failed to connect to Stellar network:', error);
      return false;
    }
  }

  getNetwork(): string {
    return this.config.network;
  }

  getHorizonUrl(): string {
    return this.config.horizonUrl;
  }

  // ====================================================================================
  // 🎯 계정 관리
  // ====================================================================================

  async createAccount(): Promise<{ accountId: string; secretKey: string; publicKey: string }> {
    const keypair = Keypair.random();
    return {
      accountId: keypair.publicKey(),
      secretKey: keypair.secret(),
      publicKey: keypair.publicKey()
    };
  }

  async getAccount(accountId: string): Promise<StellarAccount> {
    try {
      const account = await this.server.loadAccount(accountId);
      
      return {
        accountId: account.accountId(),
        secretKey: '', // 보안상 반환하지 않음
        publicKey: account.accountId(),
        balances: account.balances.map(balance => ({
          assetType: balance.asset_type,
          assetCode: balance.asset_code,
          assetIssuer: balance.asset_issuer,
          balance: balance.balance,
          limit: balance.limit,
          buyingLiabilities: balance.buying_liabilities,
          sellingLiabilities: balance.selling_liabilities,
          isAuthorized: balance.is_authorized,
          isAuthorizedToMaintainLiabilities: balance.is_authorized_to_maintain_liabilities,
          lastModifiedLedger: balance.last_modified_ledger,
          sponsor: balance.sponsor
        })),
        sequence: account.sequenceNumber(),
        subentryCount: account.subentry_count,
        flags: {
          authRequired: account.flags.auth_required,
          authRevocable: account.flags.auth_revocable,
          authImmutable: account.flags.auth_immutable
        },
        thresholds: {
          lowThreshold: account.thresholds.low_threshold,
          medThreshold: account.thresholds.med_threshold,
          highThreshold: account.thresholds.high_threshold
        },
        signers: account.signers.map(signer => ({
          key: signer.key,
          weight: signer.weight,
          type: signer.type,
          sponsor: signer.sponsor
        })),
        data: account.data
      };
    } catch (error) {
      throw new Error(`Failed to load account: ${error instanceof Error ? error.message : 'Unknown error'}`);
    }
  }

  async getAccountBalances(accountId: string): Promise<StellarBalance[]> {
    try {
      const account = await this.server.loadAccount(accountId);
      return account.balances.map(balance => ({
        assetType: balance.asset_type,
        assetCode: balance.asset_code,
        assetIssuer: balance.asset_issuer,
        balance: balance.balance,

  // ====================================================================================
  // 🎯 트랜잭션 관리
  // ====================================================================================

  async createTransaction(
    sourceAccount: string,
    operations: Operation[],
    memo?: string,
    memoType?: MemoType
  ): Promise<Transaction> {
    try {
      const account = await this.server.loadAccount(sourceAccount);
      const transaction = new TransactionBuilder(account, {
        fee: BASE_FEE,
        networkPassphrase: this.network
      });

      operations.forEach(op => transaction.addOperation(op));

      if (memo) {
        if (memoType) {
          transaction.addMemo(Memo[memoType](memo));
        } else {
          transaction.addMemo(Memo.text(memo));
        }
      }

      transaction.setTimeout(this.config.timeout);
      return transaction.build();
    } catch (error) {
      throw new Error(`Failed to create transaction: ${error instanceof Error ? error.message : 'Unknown error'}`);
    }
  }

  async submitTransaction(transaction: Transaction): Promise<StellarTransaction> {
    try {
      const response = await this.server.submitTransaction(transaction);
      
      return {
        id: response.id,
        hash: response.hash,
        ledger: response.ledger,
        createdAt: response.created_at,
        sourceAccount: response.source_account,
        sourceAccountSequence: response.source_account_sequence,
        feeAccount: response.fee_account,
        feeCharged: response.fee_charged,
        operationCount: response.operation_count,
        envelopeXdr: response.envelope_xdr,
        resultXdr: response.result_xdr,
        resultMetaXdr: response.result_meta_xdr,
        feeMetaXdr: response.fee_meta_xdr,
        memo: response.memo,
        memoType: response.memo_type,
        signatures: response.signatures,
        validAfter: response.valid_after,
        validBefore: response.valid_before
      };
    } catch (error) {
      throw new Error(`Failed to submit transaction: ${error instanceof Error ? error.message : 'Unknown error'}`);
    }
  }

  async getTransaction(transactionHash: string): Promise<StellarTransaction> {
    try {
      const response = await this.server.transactions().transaction(transactionHash).call();
      
      return {
        id: response.id,
        hash: response.hash,
        ledger: response.ledger,
        createdAt: response.created_at,
        sourceAccount: response.source_account,
        sourceAccountSequence: response.source_account_sequence,
        feeAccount: response.fee_account,
        feeCharged: response.fee_charged,
        operationCount: response.operation_count,
        envelopeXdr: response.envelope_xdr,
        resultXdr: response.result_xdr,
        resultMetaXdr: response.result_meta_xdr,

        memo: response.memo,
        memoType: response.memo_type,
        signatures: response.signatures,
        validAfter: response.valid_after,
        validBefore: response.valid_before
      };
    } catch (error) {
      throw new Error(`Failed to get transaction: ${error instanceof Error ? error.message : 'Unknown error'}`);
    }
  }

  async getAccountTransactions(
    accountId: string,
    limit: number = 10,

    try {
      const response = await this.server

        feeMetaXdr: record.fee_meta_xdr,
        memo: record.memo,
        memoType: record.memo_type,
        signatures: record.signatures,
        validAfter: record.valid_after,

  async sendPayment(
    fromSecretKey: string,
    to: string,
    amount: string,
    asset: string = 'XLM',
    memo?: string
  ): Promise<StellarTransaction> {
    try {
      const fromKeypair = Keypair.fromSecret(fromSecretKey);
      const fromAccount = await this.server.loadAccount(fromKeypair.publicKey());

      const transaction = new TransactionBuilder(fromAccount, {
        fee: BASE_FEE,
        networkPassphrase: this.network
      });

      const paymentAsset = asset === 'XLM' ? Asset.native() : Asset.fromCode(asset);
      
      transaction.addOperation(
        Operation.payment({
          destination: to,
          asset: paymentAsset,
          amount: amount
        })
      );

      if (memo) {
        transaction.addMemo(Memo.text(memo));
      }

      transaction.setTimeout(this.config.timeout);
      const builtTransaction = transaction.build();
      builtTransaction.sign(fromKeypair);

      return await this.submitTransaction(builtTransaction);
    } catch (error) {
      throw new Error(`Failed to send payment: ${error instanceof Error ? error.message : 'Unknown error'}`);
    }
  }

  async createTrustline(
    accountSecretKey: string,
    assetCode: string,
    assetIssuer: string,
    limit: string = '922337203685.4775807'
  ): Promise<StellarTransaction> {
    try {
      const accountKeypair = Keypair.fromSecret(accountSecretKey);
      const account = await this.server.loadAccount(accountKeypair.publicKey());

      const transaction = new TransactionBuilder(account, {
        fee: BASE_FEE,
        networkPassphrase: this.network
      });

      transaction.addOperation(
        Operation.changeTrust({
          asset: Asset.fromCode(assetCode, assetIssuer),
          limit: limit

      transaction.setTimeout(this.config.timeout);
      const builtTransaction = transaction.build();
      builtTransaction.sign(accountKeypair);

      return await this.submitTransaction(builtTransaction);
    } catch (error) {

  async createOffer(
    accountSecretKey: string,
    selling: { assetCode: string; assetIssuer: string },
    buying: { assetCode: string; assetIssuer: string },
    amount: string,
    price: string
  ): Promise<StellarTransaction> {
    try {
      const accountKeypair = Keypair.fromSecret(accountSecretKey);
      const account = await this.server.loadAccount(accountKeypair.publicKey());

      const transaction = new TransactionBuilder(account, {
        fee: BASE_FEE,
        networkPassphrase: this.network
      });

      const sellingAsset = selling.assetCode === 'XLM' 
        ? Asset.native() 
        : Asset.fromCode(selling.assetCode, selling.assetIssuer);
      
      const buyingAsset = buying.assetCode === 'XLM' 
        ? Asset.native() 
        : Asset.fromCode(buying.assetCode, buying.assetIssuer);

      transaction.addOperation(
        Operation.manageOffer({
          selling: sellingAsset,
          buying: buyingAsset,
          amount: amount,
          price: price,
          offerId: '0' // 새 오퍼 생성
        })
      );

      transaction.setTimeout(this.config.timeout);
      const builtTransaction = transaction.build();
      builtTransaction.sign(accountKeypair);

      return await this.submitTransaction(builtTransaction);
    } catch (error) {
      throw new Error(`Failed to create offer: ${error instanceof Error ? error.message : 'Unknown error'}`);
    }
  }

  }

  async getAccountThresholds(accountId: string): Promise<{ lowThreshold: number; medThreshold: number; highThreshold: number }> {
    try {
      const account = await this.server.loadAccount(accountId);
      return {
        lowThreshold: account.thresholds.low_threshold,
        medThreshold: account.thresholds.med_threshold,
        highThreshold: account.thresholds.high_threshold
      };
    } catch (error) {
      throw new Error(`Failed to get account thresholds: ${error instanceof Error ? error.message : 'Unknown error'}`);
    }
  }

  // ====================================================================================
  // 🎯 에러 처리
  // ====================================================================================

  private handleStellarError(error: any): StellarError {
    if (error.response) {
      const response = error.response;
      return {
        type: response.data?.type || 'Unknown',
        title: response.data?.title || 'Error',
        status: response.status,
        detail: response.data?.detail || 'Unknown error occurred',
        instance: response.data?.instance || '',
        extras: response.data?.extras
      };
    }
    
    return {
      type: 'NetworkError',
      title: 'Network Error',
      status: 0,
      detail: error.message || 'Network error occurred',
      instance: '',
      extras: { originalError: error }
    };
  }

  // ====================================================================================
  // 🎯 설정 업데이트
  // ====================================================================================

  updateConfig(newConfig: Partial<StellarConfig>): void {
    this.config = { ...this.config, ...newConfig };
    this.network = this.config.network === 'mainnet' ? Networks.PUBLIC : Networks.TESTNET;
    this.server = new Server(this.config.horizonUrl);
  }

  getConfig(): StellarConfig {
    return { ...this.config };
  }
}
