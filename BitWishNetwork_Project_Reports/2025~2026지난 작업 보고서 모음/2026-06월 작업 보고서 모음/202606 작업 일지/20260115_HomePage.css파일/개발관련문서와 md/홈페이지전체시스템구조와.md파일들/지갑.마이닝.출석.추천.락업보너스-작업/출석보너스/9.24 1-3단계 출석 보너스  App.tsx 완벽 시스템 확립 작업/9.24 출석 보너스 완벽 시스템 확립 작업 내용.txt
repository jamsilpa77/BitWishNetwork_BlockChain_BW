##    **오류 분석 및 해결 완료**

사진에서 보이는 **"Uncaught ReferenceError: userIsMiningActive is not defined"** 오류를 분석하고 해결했습니다.

###    **문제 원인**

1. **변수 미선언**: `userIsMiningActive` 변수가 App.tsx에서 선언되지 않았음
2. **개발 가이드와 코드 불일치**: 개발 가이드 문서에는 해당 변수가 사용되고 있지만, 실제 코드에서는 선언되지 않음
3. **useEffect 의존성 배열 오류**: 2924번째 줄에서 `userIsMiningActive`를 의존성으로 사용하고 있었지만 변수가 존재하지 않음

### ✅ **해결 방법**

App.tsx의 마이닝 상태 관리 섹션에 다음 변수들을 추가했습니다:

```typescript
// 유저용 마이닝 상태 관리
const [userIsMiningActive, setUserIsMiningActive] = useState(false);
const [userMiningStartTime, setUserMiningStartTime] = useState<Date | null>(null);
const [userMiningStopTime, setUserMiningStopTime] = useState<Date | null>(null);
const [userMiningRewards, setUserMiningRewards] = useState(0);
```

### 🎯 **개인별 마이닝 시스템 이해**

개발 가이드에 따르면, 이 시스템은:
- **개인별 독립적 마이닝 환경**: 각 사용자가 자신만의 마이닝 상태를 가짐
- **상태 관리**: 마이닝 활성화/비활성화, 시작/정지 시간, 보상 관리
- **실시간 계산**: Decimal.js 기반 50자리 정밀도로 보상 계산
- **상태 저장**: localStorage를 통한 마이닝 상태 자동 저장/복원

이제 브라우저를 새로고침하면 `userIsMiningActive is not defined` 오류가 해결되어 정상적으로 작동할 것입니다.

if (referralCount >= 91) {
  // 구간 4: 91명 이상 (0.75% × N명)

}

### **🚨 1.5 출석 보너스 시스템 문제점**
1. **AttendanceService.ts**: 6단계 보너스 시스템이 **5% 고정**으로 완전히 단순화됨

4. **완전한 불일치**: 백서의 정교한 6단계 시스템이 전혀 구현되지 않음

### **3.2 LockupService.ts 구현 - 완벽함**
```typescript
// 🎯 LockupService.ts의 락업 보너스 계산 - 백서와 완벽히 일치
static calculateBasicLockupBonus(ratio: number, months: number): {
  apy: number;
  totalAPY: number;
  monthlyRate: number;
  enhancedRate: number;
  description: string;
  lockedAmount: number;
} {
  // 🎯 백서 기준 APY 계산
  let baseAPY = 0;
  if (months >= 36) {
    baseAPY = 0.072; // 7.2% ✅ 백서와 일치
  } else if (months >= 12) {
    baseAPY = 0.057; // 5.7% ✅ 백서와 일치
  } else if (months >= 6) {
    baseAPY = 0.042; // 4.2% ✅ 백서와 일치
  } else {
    baseAPY = 0.027; // 2.7% ✅ 백서와 일치
  }
  
  const ratioBonus = ratio * 0.01; // 1% per 1% ratio
  const totalAPY = baseAPY + ratioBonus;
  const monthlyRate = totalAPY / 12;
  const enhancedRate = 0.25 * (1 + totalAPY * (ratio / 100));
  
  return {
    apy: baseAPY,
    totalAPY: totalAPY,
    monthlyRate: monthlyRate,
    enhancedRate: enhancedRate,
    description: `${months}개월 락업 (${(totalAPY * 100).toFixed(1)}% APY)`,
    lockedAmount: lockedAmount
  };
}
```

```typescript
//    가맹점 보너스 계산 함수 (BigDecimal 기반 완전 재구현)
const calculateMerchantBonus = (isMerchant: boolean): {
  bonusRate: number;
  enhancedRate: number;
  description: string;
  decimalBonusRate: Decimal;
  decimalEnhancedRate: Decimal;
} => {
  if (isMerchant) {
    const decimalBonusRate = new Decimal(1.25); // 125% (백서 정책) ✅
    const decimalEnhancedRate = new Decimal(0.25).times(new Decimal(1).plus(decimalBonusRate));
    return {
      bonusRate: decimalBonusRate.toDecimalPlaces(PRECISION_CONFIG.DISPLAY_PRECISION).toNumber(),
      enhancedRate: decimalEnhancedRate.toDecimalPlaces(PRECISION_CONFIG.DISPLAY_PRECISION).toNumber(),
      description: '🏪 가맹점 보너스 활성화 (125%)',
      decimalBonusRate,
      decimalEnhancedRate
    };
  } else {
    const decimalBonusRate = new Decimal(0); // 0%
    const decimalEnhancedRate = new Decimal(0.25); // 기본값
    return {
      bonusRate: decimalBonusRate.toDecimalPlaces(PRECISION_CONFIG.DISPLAY_PRECISION).toNumber(),
      enhancedRate: decimalEnhancedRate.toDecimalPlaces(PRECISION_CONFIG.DISPLAY_PRECISION).toNumber(),
      description: '❌ 가맹점 보너스 미활성화',
      decimalBonusRate,
      decimalEnhancedRate
    };
  }
};
```

```typescript
// 🎯 개발 가이드에서 예상하는 로직
const calculateUserAttendanceBonus = (consecutiveDays: number): { 
  bonusRate: number; 
  enhancedRate: number; 
  description: string;
  decimalBonusRate: Decimal;
  decimalEnhancedRate: Decimal;
} => {
  let bonusRate = 0;
  let description = '';
  
  if (consecutiveDays >= 30) {
    bonusRate = 0.17; // 17%
    description = '30일 이상 연속 출석';
  } else if (consecutiveDays >= 29) {
    bonusRate = 0.12; // 12%
    description = '29일 연속 출석';
  } else if (consecutiveDays >= 22) {
    bonusRate = 0.068; // 6.8%
    description = '22-28일 연속 출석';
  } else if (consecutiveDays >= 15) {
    bonusRate = 0.05; // 5%
    description = '15-21일 연속 출석';
  } else if (consecutiveDays >= 8) {
    bonusRate = 0.035; // 3.5%
    description = '8-14일 연속 출석';
  } else if (consecutiveDays >= 1) {
    bonusRate = 0.02; // 2%
    description = '1-7일 연속 출석';
  } else {
    bonusRate = 0; // 0%
    description = '출석 보너스 없음';
  }
  
  // Decimal.js 기반 정밀 계산
  const decimalBonusRate = new Decimal(bonusRate);
  const decimalBaseRate = new Decimal(0.25);
  const decimalEnhancedRate = decimalBaseRate.times(new Decimal(1).plus(decimalBonusRate));
  
  return {
    bonusRate,
    enhancedRate: decimalEnhancedRate.toDecimalPlaces(50).toNumber(),
    description,
    decimalBonusRate,
    decimalEnhancedRate
  };
};

// 라인 6434-6450: 추천 보관 BW 계산 함수
const calculateReferralStorageBW = (referralCount: number): number => {
  if (referralCount <= 0) return 0;
  
  // 기본 채굴률: 0.25 BW/시간
  const baseRate = 0.25;
  
  // 구간별 보너스율 계산 (백서 기준)
  let totalBonusRate = 0;
  
  if (referralCount <= 30) {
    totalBonusRate = referralCount * 0.01875; // 1.875% × N명
  } else if (referralCount <= 60) {
    totalBonusRate = 30 * 0.01875 + (referralCount - 30) * 0.015; // 30명(56.25%) + (N-30)명(1.5%)
  } else if (referralCount <= 90) {
    totalBonusRate = 30 * 0.01875 + 30 * 0.015 + (referralCount - 60) * 0.01125; // 30명(56.25%) + 30명(45%) + (N-60)명(1.125%)
  } else {
    totalBonusRate = 30 * 0.01875 + 30 * 0.015 + 30 * 0.01125 + (referralCount - 90) * 0.0075; // 30명(56.25%) + 30명(45%) + 30명(33.75%) + (N-90)명(0.75%)
  }
  
  return baseRate * totalBonusRate;
};
```
