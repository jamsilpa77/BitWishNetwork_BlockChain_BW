# **📄 ReferralCodeInput.tsx 완전 구현**

**위치**: `Node_HomePage/src/components/ReferralCodeInput.tsx`

```typescript
import React, { useState, useEffect, useRef } from 'react';
import { useTranslation } from 'react-i18next';

// ====================================================================================
// 🎯 ReferralCodeInput - 추천인 코드 입력 컴포넌트
// ====================================================================================

interface ReferralCodeInputProps {
  onReferralCodeChange: (code: string) => void;
  onValidationResult: (isValid: boolean, message?: string) => void;
  placeholder?: string;
  showHelpText?: boolean;
  showValidationIcon?: boolean;
  size?: 'small' | 'medium' | 'large';
  variant?: 'outlined' | 'filled' | 'underlined';
  disabled?: boolean;
  initialValue?: string;
  maxLength?: number;
  autoFocus?: boolean;
  className?: string;
}

export const ReferralCodeInput: React.FC<ReferralCodeInputProps> = ({
  onReferralCodeChange,
  onValidationResult,
  placeholder = "추천인 코드를 입력하세요",
  showHelpText = true,
  showValidationIcon = true,
  size = 'medium',
  variant = 'outlined',
  disabled = false,
  initialValue = '',
  maxLength = 20,
  autoFocus = false,
  className = ''
}) => {
  const { t } = useTranslation();
  const [inputValue, setInputValue] = useState(initialValue);
  const [isValidating, setIsValidating] = useState(false);
  const [validationState, setValidationState] = useState<'idle' | 'validating' | 'valid' | 'invalid'>('idle');
  const [validationMessage, setValidationMessage] = useState('');
  const [showSuggestions, setShowSuggestions] = useState(false);
  const inputRef = useRef<HTMLInputElement>(null);
  const validationTimeoutRef = useRef<NodeJS.Timeout | null>(null);

  // 크기별 스타일 설정
  const sizeStyles = {
    small: {
      input: 'px-3 py-2 text-sm',
      icon: 'w-4 h-4',
      helpText: 'text-xs',
      suggestion: 'text-sm py-1 px-2'
    },
    medium: {
      input: 'px-4 py-3 text-base',
      icon: 'w-5 h-5',
      helpText: 'text-sm',
      suggestion: 'text-sm py-2 px-3'
    },
    large: {
      input: 'px-5 py-4 text-lg',
      icon: 'w-6 h-6',
      helpText: 'text-base',
      suggestion: 'text-base py-3 px-4'
    }
  };

  // 변형별 스타일 설정
  const variantStyles = {
    outlined: 'border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent',
    filled: 'bg-gray-100 border-0 rounded-lg focus:ring-2 focus:ring-blue-500 focus:bg-white',
    underlined: 'border-0 border-b-2 border-gray-300 rounded-none focus:ring-0 focus:border-blue-500'
  };

  // 검증 상태별 아이콘
  const getValidationIcon = () => {
    if (!showValidationIcon) return null;
    
    switch (validationState) {
      case 'validating':
        return (
          <div className="animate-spin">
            <svg className={`${sizeStyles[size].icon} text-blue-500`} fill="none" viewBox="0 0 24 24">
              <circle className="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" strokeWidth="4"></circle>
              <path className="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z"></path>
            </svg>
          </div>
        );
      case 'valid':
        return (
          <svg className={`${sizeStyles[size].icon} text-green-500`} fill="none" viewBox="0 0 24 24" stroke="currentColor">
            <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M5 13l4 4L19 7" />
          </svg>
        );
      case 'invalid':
        return (
          <svg className={`${sizeStyles[size].icon} text-red-500`} fill="none" viewBox="0 0 24 24" stroke="currentColor">
            <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M6 18L18 6M6 6l12 12" />
          </svg>
        );
      default:
        return (
          <svg className={`${sizeStyles[size].icon} text-gray-400`} fill="none" viewBox="0 0 24 24" stroke="currentColor">
            <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M13 16h-1v-4h-1m1-4h.01M21 12a9 9 0 11-18 0 9 9 0 0118 0z" />
          </svg>
        );
    }
  };

  // 추천인 코드 검증 함수
  const validateReferralCode = async (code: string): Promise<{ isValid: boolean; message: string }> => {
    if (!code || code.trim().length === 0) {
      return { isValid: false, message: '추천인 코드를 입력해주세요' };
    }

    if (code.length < 3) {
      return { isValid: false, message: '추천인 코드는 최소 3자 이상이어야 합니다' };
    }

    if (code.length > maxLength) {
      return { isValid: false, message: `추천인 코드는 최대 ${maxLength}자까지 입력 가능합니다` };
    }

    // 영문자, 숫자, 하이픈, 언더스코어만 허용
    const validPattern = /^[A-Za-z0-9_-]+$/;
    if (!validPattern.test(code)) {
      return { isValid: false, message: '추천인 코드는 영문자, 숫자, 하이픈, 언더스코어만 사용 가능합니다' };
    }

    try {
      setIsValidating(true);
      setValidationState('validating');

      // 백엔드에서 추천인 코드 검증
      const response = await fetch('http://localhost:4001/referral/validate', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({ code: code.trim().toUpperCase() }),
      });

      const data = await response.json();
      
      if (data.success && data.isValid) {
        return { isValid: true, message: '유효한 추천인 코드입니다' };
      } else {
        return { isValid: false, message: data.message || '존재하지 않는 추천인 코드입니다' };
      }
    } catch (error) {
      console.error('추천인 코드 검증 오류:', error);
      return { isValid: false, message: '추천인 코드 검증 중 오류가 발생했습니다' };
    } finally {
      setIsValidating(false);
    }
  };

  // 입력값 변경 처리
  const handleInputChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    const value = e.target.value;
    
    // 최대 길이 제한
    if (value.length > maxLength) {
      return;
    }

    setInputValue(value);
    onReferralCodeChange(value);

    // 이전 검증 타이머 취소
    if (validationTimeoutRef.current) {
      clearTimeout(validationTimeoutRef.current);
    }

    // 입력이 비어있으면 검증 상태 초기화
    if (!value.trim()) {
      setValidationState('idle');
      setValidationMessage('');
      onValidationResult(false);
      return;
    }

    // 500ms 후 검증 실행 (디바운싱)
    validationTimeoutRef.current = setTimeout(async () => {
      const result = await validateReferralCode(value);
      setValidationState(result.isValid ? 'valid' : 'invalid');
      setValidationMessage(result.message);
      onValidationResult(result.isValid, result.message);
    }, 500);
  };

  // 포커스 처리
  const handleFocus = () => {
    setShowSuggestions(true);
  };

  // 포커스 아웃 처리
  const handleBlur = () => {
    // 약간의 지연을 두어 클릭 이벤트가 처리되도록 함
    setTimeout(() => {
      setShowSuggestions(false);
    }, 200);
  };

  // 추천인 코드 제안 목록
  const suggestions = [
    { code: 'REF001', description: '추천인 코드 1' },
    { code: 'REF002', description: '추천인 코드 2' },
    { code: 'REF003', description: '추천인 코드 3' },
    { code: 'WELCOME', description: '신규 가입자용' },
    { code: 'VIP', description: 'VIP 회원용' }
  ];

  // 제안 클릭 처리
  const handleSuggestionClick = (suggestion: string) => {
    setInputValue(suggestion);
    onReferralCodeChange(suggestion);
    setShowSuggestions(false);
    
    // 즉시 검증 실행
    validateReferralCode(suggestion).then(result => {
      setValidationState(result.isValid ? 'valid' : 'invalid');
      setValidationMessage(result.message);
      onValidationResult(result.isValid, result.message);
    });
  };

  // 컴포넌트 마운트 시 초기값 설정
  useEffect(() => {
    if (initialValue) {
      setInputValue(initialValue);
      onReferralCodeChange(initialValue);
    }
  }, [initialValue, onReferralCodeChange]);

  // 자동 포커스
  useEffect(() => {
    if (autoFocus && inputRef.current) {
      inputRef.current.focus();
    }
  }, [autoFocus]);

  // 컴포넌트 언마운트 시 타이머 정리
  useEffect(() => {
    return () => {
      if (validationTimeoutRef.current) {
        clearTimeout(validationTimeoutRef.current);
      }
    };
  }, []);

  return (
    <div className={`relative ${className}`}>
      {/* 입력 필드 */}
      <div className="relative">
        <input
          ref={inputRef}
          type="text"
          value={inputValue}
          onChange={handleInputChange}
          onFocus={handleFocus}
          onBlur={handleBlur}
          placeholder={placeholder}
          disabled={disabled}
          maxLength={maxLength}
          className={`
            w-full
            ${sizeStyles[size].input}
            ${variantStyles[variant]}
            ${disabled ? 'bg-gray-100 cursor-not-allowed' : 'bg-white'}
            ${validationState === 'valid' ? 'border-green-500' : ''}
            ${validationState === 'invalid' ? 'border-red-500' : ''}
            focus:outline-none
            transition-colors
            duration-200
          `}
        />
        
        {/* 검증 아이콘 */}
        <div className="absolute right-3 top-1/2 transform -translate-y-1/2">
          {getValidationIcon()}
        </div>
      </div>

      {/* 제안 목록 */}
      {showSuggestions && !disabled && (
        <div className="absolute top-full left-0 right-0 mt-1 bg-white border border-gray-200 rounded-lg shadow-lg z-50 max-h-48 overflow-y-auto">
          {suggestions
            .filter(suggestion => 
              suggestion.code.toLowerCase().includes(inputValue.toLowerCase()) ||
              suggestion.description.toLowerCase().includes(inputValue.toLowerCase())
            )
            .map((suggestion, index) => (
              <button
                key={index}
                onClick={() => handleSuggestionClick(suggestion.code)}
                className={`
                  w-full text-left
                  ${sizeStyles[size].suggestion}
                  hover:bg-gray-100
                  transition-colors
                  duration-150
                  flex items-center justify-between
                `}
              >
                <div>
                  <div className="font-medium text-gray-900">{suggestion.code}</div>
                  <div className="text-gray-500 text-xs">{suggestion.description}</div>
                </div>
                <svg className="w-4 h-4 text-gray-400" fill="none" viewBox="0 0 24 24" stroke="currentColor">
                  <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M9 5l7 7-7 7" />
                </svg>
              </button>
            ))}
        </div>
      )}

      {/* 도움말 텍스트 */}
      {showHelpText && (
        <div className="mt-2">
          {/* 검증 메시지 */}
          {validationMessage && (
            <div className={`
              ${sizeStyles[size].helpText}
              ${validationState === 'valid' ? 'text-green-600' : 'text-red-600'}
              flex items-center space-x-1
            `}>
              <span>{validationMessage}</span>
            </div>
          )}
          
          {/* 기본 도움말 */}
          {!validationMessage && (
            <div className={`${sizeStyles[size].helpText} text-gray-500`}>
              💡 추천인 코드를 입력하면 1.2% 추가 보너스를 받을 수 있습니다
            </div>
          )}
          
          {/* 입력 길이 표시 */}
          <div className={`${sizeStyles[size].helpText} text-gray-400 mt-1`}>
            {inputValue.length}/{maxLength}자
          </div>
        </div>
      )}

      {/* 검증 중 표시 */}
      {isValidating && (
        <div className="absolute top-full left-0 right-0 mt-1 bg-blue-50 border border-blue-200 rounded-lg p-2 z-50">
          <div className="flex items-center space-x-2">
            <div className="animate-spin">
              <svg className="w-4 h-4 text-blue-500" fill="none" viewBox="0 0 24 24">
                <circle className="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" strokeWidth="4"></circle>
                <path className="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z"></path>
              </svg>
            </div>
            <span className="text-sm text-blue-700">추천인 코드를 검증하는 중...</span>
          </div>
        </div>
      )}
    </div>
  );
};

// ====================================================================================
// 🎯 ReferralCodeInputWithLabel - 라벨이 포함된 추천인 코드 입력 컴포넌트
// ====================================================================================

interface ReferralCodeInputWithLabelProps extends ReferralCodeInputProps {
  label: string;
  required?: boolean;
  error?: string;
  helpText?: string;
}

export const ReferralCodeInputWithLabel: React.FC<ReferralCodeInputWithLabelProps> = ({
  label,
  required = false,
  error,
  helpText,
  ...props
}) => {
  return (
    <div className="space-y-2">
      {/* 라벨 */}
      <label className="block text-sm font-medium text-gray-700">
        {label}
        {required && <span className="text-red-500 ml-1">*</span>}
      </label>
      
      {/* 입력 컴포넌트 */}
      <ReferralCodeInput {...props} />
      
      {/* 에러 메시지 */}
      {error && (
        <div className="text-sm text-red-600 flex items-center space-x-1">
          <svg className="w-4 h-4" fill="none" viewBox="0 0 24 24" stroke="currentColor">
            <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M12 8v4m0 4h.01M21 12a9 9 0 11-18 0 9 9 0 0118 0z" />
          </svg>
          <span>{error}</span>
        </div>
      )}
      
      {/* 도움말 텍스트 */}
      {helpText && !error && (
        <div className="text-sm text-gray-500">{helpText}</div>
      )}
    </div>
  );
};

// ====================================================================================
// 🎯 ReferralCodeInputGroup - 그룹화된 추천인 코드 입력 컴포넌트

          <p className="text-sm text-gray-600 mt-1">{description}</p>

  );

// ====================================================================================
// 🎯 기본 내보내기
// ====================================================================================

export default ReferralCodeInput;
export { ReferralCodeInputWithLabel, ReferralCodeInputGroup };
```
