  try {
    const { walletAddress, secondPassword } = req.body;
    
    const user = await User.findOne({ 
      'miningData.walletAddress': walletAddress 
    }).select('miningData.secondPassword miningData.isSecondPasswordSet');
    
    if (!user || !user.miningData.isSecondPasswordSet) {
      return res.status(400).json({
        success: false,
        message: '2차 비밀번호가 설정되지 않았습니다.'
      });
    }
    
    const bcrypt = require('bcryptjs');
    const isValid = await bcrypt.compare(secondPassword, user.miningData.secondPassword);
    
    if (!isValid) {
      return res.status(401).json({
        success: false,
        message: '잘못된 2차 비밀번호입니다.'
      });
    }
    
    res.json({
      success: true,
      message: '2차 비밀번호 인증 성공'
    });
    
  } catch (error) {
    logger.error('2차 비밀번호 검증 오류:', error);

      });

    // 누적 보상 계산 (고정값 기반)
    let totalReward = 0;
    if (user.miningData.isMiningActive && user.miningData.miningStartTime) {
      const elapsedHours = (Date.now() - user.miningData.miningStartTime.getTime()) / (1000 * 60 * 60);

      success: true,
      data: {
        totalAccumulatedReward: totalReward,
        fixedHourlyReward: user.miningData.fixedHourlyReward,
        fixedDailyReward: user.miningData.fixedDailyReward,
        isMiningActive: user.miningData.isMiningActive,
        miningStartTime: user.miningData.miningStartTime
      }
    });
    
  } catch (error) {
    logger.error('누적 보상 계산 오류:', error);

##   ️ 문제 해결 방법

### 1. **유저용 모달에서 관리자 버튼 완전 제거 (5, 6번 문제)**

**해결 방법:**
- `App.tsx`의 유저용 모달에서 다음 버튼들을 완전히 제거:
  - "   추천인 추가" 버튼
  - "🔄 추천인 초기화" 버튼  
  - "   기본 락업 초기화" 버튼
  - "🔄 추가 락업 초기화" 버튼
- 해당 버튼들이 있는 JSX 블록을 완전히 삭제

### 2. **유저용 마이닝 시작 시 관리자 창 전환 방지 (4번 문제)**

**해결 방법:**
- `App.tsx`의 마이닝 시작 핸들러에서 사용자 타입을 명확히 구분
- 유저용 모달에서 "마이닝 시작" 클릭 시 `showUserMiningBonusModal`만 유지
- `showMiningBonusModal`로 전환되지 않도록 조건문 수정

### 3. **유저용 실시간 마이닝 보상 로직 수정 (2번 문제)**

**해결 방법:**
- 유저용 모달의 실시간 마이닝 보상 섹션에서 고정값만 표시
- 관리자와 동일한 실시간 계산 로직을 제거
- 시간당 0.25 BW, 일일 6.0 BW 고정값으로만 표시

### 4. **유저용 "추천 보너스 현황" 접이식 섹션 추가 (3번 문제)**

**해결 방법:**
- 유저용 모달에 "추천 보너스 현황" 접이식 섹션 추가
- 관리자 모달의 해당 섹션을 복사하되, 관리자 전용 버튼은 제거
- 유저용 데이터만 표시하도록 수정

### 5. **관리자 모달 상태창 표시 로직 수정 (7번 문제)**

**해결 방법:**
- 관리자 모달의 마이닝 시작 핸들러에서 상태창 표시 로직 확인
- `showMiningStatusModal` 상태를 올바르게 설정
- 마이닝 시작 후 상태창이 자동으로 열리도록 수정

### 6. **관리자 창 중복 문제 해결**

**해결 방법:**
- 유저용에서 관리자 창으로 전환되는 로직 완전 제거
- 사용자 타입별로 명확한 모달 분리 유지
- `getUserType` 함수의 로직을 더 명확하게 수정

##    **정확한 문제 분석 보고서**

### **실제 문제의 근본 원인:**

**1️⃣ 마이닝 시작 버튼이 작동하지 않는 진짜 이유:**

유저용 모달과 관리자용 모달 모두에서 마이닝 시작 버튼이 작동하지 않는 이유는 **`isMiningActive` 상태가 `true`로 설정되어 있기 때문**입니다.

```typescript
if (!isMiningActive) {
  // 마이닝 시작 로직...
}
```

**현재 상황:**
- `isMiningActive = true` (이미 마이닝이 활성화된 상태)
- 따라서 `!isMiningActive = false`가 되어 마이닝 시작 로직이 실행되지 않음

**2️⃣ `isMiningActive`가 `true`인 이유:**

**가능한 원인들:**
1. **이전에 마이닝을 시작했지만 정지하지 않았음**
2. **localStorage에 저장된 마이닝 상태가 복원됨**
3. **`loadMiningState()` 함수가 이전 상태를 복원함**
4. **`startVirtualMining()` 함수에서 `setIsMiningActive(true)`가 호출되었지만 정지되지 않음**

**3️⃣ 실제 해결 방법:**

**A. 마이닝 상태 초기화:**
```typescript
// 마이닝 상태를 강제로 초기화

setMiningStopTime(null);
```

**B. localStorage 초기화:**
```typescript
// localStorage에서 마이닝 상태 제거

```

**C. 마이닝 시작 조건 수정:**
```typescript
// 현재 마이닝 중이면 정지 후 시작

}

```

**실제 문제는 `isMiningActive` 상태가 `true`로 설정되어 있어서 마이닝 시작 버튼의 조건문 `if (!isMiningActive)`이 `false`가 되어 마이닝 시작 로직이 실행되지 않는 것입니다.**
