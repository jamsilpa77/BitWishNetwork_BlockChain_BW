// ==================== 개인별 추천 보너스 KYC 연동 시스템 ====================
const PersonalReferralKYCIntegration = {
  // KYC 통과 시 스텔라 BW 토큰 마이그레이션
  migrateToStellarBW: async (walletAddress: string): Promise<{
    success: boolean;
    migratedAmount: number;
    transactionHash?: string;
  }> => {
    try {
      const personalReferralState = initializePersonalReferralBonusState(walletAddress);
      
      if (personalReferralState.referralStorageBW <= 0) {
        return { success: false, migratedAmount: 0 };
      }

      // 스텔라 블록체인에 1:1 비율로 토큰 마이그레이션
      const migrationResult = await migrateToStellarBlockchain(
        walletAddress, 
        personalReferralState.referralStorageBW
      );

      if (migrationResult.success) {
        // 마이그레이션 성공 시 로컬 상태 초기화
        const storageKey = getPersonalReferralStorageKey(walletAddress, 'personalReferralStorageBW');
        localStorage.setItem(storageKey, '0');
        
        console.log('✅ 개인별 추천 보너스 KYC 마이그레이션 완료:', {
          walletAddress,
          migratedAmount: personalReferralState.referralStorageBW,
          transactionHash: migrationResult.transactionHash
        });
      }

      return migrationResult;
    } catch (error) {
      console.error('❌ 개인별 추천 보너스 KYC 마이그레이션 실패:', error);
      return { success: false, migratedAmount: 0 };
    }
  },

  // KYC 상태 확인
  checkKYCStatus: (walletAddress: string): boolean => {
    const kycStorageKey = getPersonalReferralStorageKey(walletAddress, 'kycStatus');
    const kycStatus = localStorage.getItem(kycStorageKey);
    return kycStatus === 'verified';
  }
};
