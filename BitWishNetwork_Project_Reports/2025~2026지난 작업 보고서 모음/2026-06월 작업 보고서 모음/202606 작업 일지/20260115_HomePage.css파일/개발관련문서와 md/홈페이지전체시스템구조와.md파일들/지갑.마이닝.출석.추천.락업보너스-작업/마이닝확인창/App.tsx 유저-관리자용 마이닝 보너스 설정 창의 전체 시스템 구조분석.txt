  // 출석 보너스 (BigDecimal) - 시간 제한 적용 (유저용 출석 데이터 사용)
  const decimalAttendanceBonus = (userConsecutiveAttendanceDays > 0 && isUserAttendanceBonusTimeValid()) ? 
    new Decimal(calculateUserAttendanceBonus(userConsecutiveAttendanceDays).bonusRate) : 
    new Decimal(0);
  
  // 🎯 유저용 추천인 보너스 (BigDecimal 기반 - 완벽한 정밀도)
  const userReferralBonusData = calculateUserReferralBonusDecimal(userReferralCount);
  const decimalUserReferralBonus = userReferralBonusData.totalBonusRate;
  
  // 🎯 유저용 락업 보너스 (BigDecimal) - 등록된 경우에만 계산
  const decimalUserLockupBonus = (isUserLockupRegistered && userLockupRatio > 0) ? 
    new Decimal(calculateUserLockupBonus(userLockupRatio, userLockupMonths).totalAPY) : 
    new Decimal(0);
  
  // 총 보너스 비율 계산 (BigDecimal)
  const decimalTotalBonusRate = decimalBaseRate
    .plus(decimalAttendanceBonus)
    .plus(decimalUserReferralBonus)
    .plus(decimalUserLockupBonus);
    
  // 시간당 보상 계산
  const hourlyReward = decimalTotalBonusRate.toNumber();
  
  return {
    hourlyReward,
    userReferralBonusDetails: userReferralBonusData,
    ui: {
      attendancePct: decimalAttendanceBonus.times(100).toDecimalPlaces(6),
      referralPct: decimalUserReferralBonus.times(100).toDecimalPlaces(6),
      lockupPct: decimalUserLockupBonus.times(100).toDecimalPlaces(6),
      totalPct: decimalTotalBonusRate.times(100).toDecimalPlaces(6)
    }
  };
};
