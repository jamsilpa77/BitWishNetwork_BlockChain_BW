```typescript
//    개개인별 localStorage 키 생성 함수
const getPersonalStorageKey = (walletAddress: string, type: string) => `${type}_${walletAddress}`;

// 예시:
// userMiningState_GABC1234567890DEF
// userMiningState_XYZ9876543210ABC
// userMiningState_DEF4567890123GHI
```

#### **4.2 개개인별 데이터 로드/저장 시스템**

```typescript
// 🎯 개개인별 마이닝 데이터 로드 함수
const loadPersonalMiningData = (walletAddress: string) => {
  try {
    const personalKey = getPersonalStorageKey(walletAddress, 'userMiningState');
    const savedData = localStorage.getItem(personalKey);
    if (savedData) {
      const personalMiningState = JSON.parse(savedData);
      setUserMiningStates(prev => ({ ...prev, [walletAddress]: personalMiningState }));
      console.log(`✅ 개개인별 마이닝 데이터 로드됨: ${walletAddress}`);
    } else {
      // 새로운 개개인 데이터 초기화
      const initialMiningState = {
        isMiningActive: false,
        miningStartTime: null,
        miningStopTime: null,
        consecutiveAttendanceDays: 0,
        referralCount: 0,
        lockupRatio: 0,
        lockupMonths: 0,
        lockupBonus: 0
      };
      setUserMiningStates(prev => ({ ...prev, [walletAddress]: initialMiningState }));
      console.log(`✅ 새로운 개개인 마이닝 데이터 생성: ${walletAddress}`);
    }
  } catch (error) {
    console.error(`❌ 개개인별 마이닝 데이터 로드 실패: ${walletAddress}`, error);
  }
};
```

#### **5.2 실시간 마이닝 상태 표시**

```typescript
// 🎯 개인별 실제 계산된 채굴률 표시
{(() => {
  // 🎯 백서 기준 보너스 계산
  const attendanceBonus = AttendanceService.calculateUserAttendanceBonus(
    personalMiningState.consecutiveAttendanceDays || 0
  );
  
  const referralBonus = ReferralService.calculateUserReferralBonus(
    personalMiningState.referralCount || 0,
    walletAddress
  );
  
  const lockupBonus = LockupService.calculateLockupBonus(
    personalMiningState.lockupRatio || 0,
    personalMiningState.lockupMonths || 0
  );
  
  // 🎯 최종 채굴률 계산
  const baseRate = 0.25;
  const totalBonusRate = attendanceBonus.bonusRate + 
                        referralBonus.bonusRate.toNumber() + 
                        lockupBonus.monthlyRate;
  const finalMiningRate = baseRate * (1 + totalBonusRate);
  
  return (
    <div className="grid grid-cols-2 gap-4 mb-4">
      <div className="bg-white rounded-lg p-4 shadow-sm">
        <div className="text-2xl font-bold text-blue-600">
          {finalMiningRate.toFixed(3)} BW/시간
        </div>
        <p className="text-sm text-gray-600">최종 채굴률 (백서 기준)</p>
      </div>
      
      <div className="bg-white rounded-lg p-4 shadow-sm">
        <div className="text-2xl font-bold text-green-600">
          {(finalMiningRate * 24).toFixed(2)} BW/일
        </div>
        <p className="text-sm text-gray-600">일일 본상 (백서 기준)</p>
      </div>
    </div>
  );
})()}
```

#### **5.3 개개인별 마이닝 시작/정지 시스템**

```typescript
//    개인별 마이닝 시작
<button onClick={() => {
  if (!walletAddress) {
    alert('지갑 주소가 인증되지 않았습니다. 먼저 지갑 주소를 인증해주세요.');
    return;
  }
  const personalMiningState = userMiningStates[walletAddress];
  if (personalMiningState?.isMiningActive) {
    alert('마이닝이 이미 진행 중입니다.');
    return;
  }
  
  // 🎯 개인별 마이닝 시작
  const updatedState = {
    ...personalMiningState,
    isMiningActive: true,
    miningStartTime: new Date(),
    miningStopTime: null
  };
  setUserMiningStates(prev => ({ ...prev, [walletAddress]: updatedState }));
  savePersonalMiningData(walletAddress);
  console.log(`✅ 개인별 마이닝 시작: ${walletAddress}`);
}}>
  ⚡ 마이닝 시작
</button>
```
