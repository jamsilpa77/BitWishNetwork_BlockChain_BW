```typescript
const calculateMerchantBonus = (isMerchant: boolean): {
  bonusRate: number;
  enhancedRate: number;
  description: string;
  decimalBonusRate: Decimal;
  decimalEnhancedRate: Decimal;
} => {
  if (isMerchant) {
    const decimalBonusRate = new Decimal(1.25); // 125% (백서 정책)
    const decimalEnhancedRate = new Decimal(0.25).times(new Decimal(1).plus(decimalBonusRate));
    return {
      bonusRate: decimalBonusRate.toDecimalPlaces(8).toNumber(),
      enhancedRate: decimalEnhancedRate.toDecimalPlaces(8).toNumber(),
      description: '   가맹점 보너스 활성화 (125%)',
      decimalBonusRate,
      decimalEnhancedRate
    };
  } else {
    const decimalBonusRate = new Decimal(0); // 0%
    const decimalEnhancedRate = new Decimal(0.25); // 기본값
    return {
      bonusRate: decimalBonusRate.toDecimalPlaces(8).toNumber(),
      enhancedRate: decimalEnhancedRate.toDecimalPlaces(8).toNumber(),
      description: '❌ 가맹점 보너스 미활성화',
      decimalBonusRate,
      decimalEnhancedRate
    };
  }
};
```

```typescript
const checkKYCStatus = async (walletAddress: string) => {
  try {
    const response = await fetch(`http://localhost:4001/kyc/status/${walletAddress}`);
    const result = await response.json();
    
    if (result.success) {
      setKycStatus({
        isApplied: result.data.isApplied,
        isApproved: result.data.isApproved,
        applicationDate: result.data.applicationDate,
        approvalDate: result.data.approvalDate,
        rejectionReason: result.data.rejectionReason,
        documents: result.data.documents
      });
      return result.data;
    }
    return null;
  } catch (error) {
    console.error('KYC 상태 확인 오류:', error);
    return null;
  }
};
```

```typescript
const handleBasicLockupReset = () => {
  try {
    // 상태 초기화
    setLockupRatio(20);           // 20%로 초기화
    setLockupMonths(1);           // 1개월로 초기화
    setIsLockupRegistered(false); // 등록 상태 초기화

      months: 0,
      apy: 0,
      startDate: null,
      endDate: null
    });
    
    // localStorage 업데이트
    const currentState = JSON.parse(localStorage.getItem('miningState') || '{}');
    const updatedState = {
      ...currentState,
      lockupRatio: 20,
      lockupMonths: 1,
      isLockupRegistered: false,
      registeredLockupData: { ratio: 0, months: 0, apy: 0, startDate: null, endDate: null }
```

### **1. 기본 락업 설정 버튼 개선 (라인 8651-8748)**

```typescript
{/* 설정 등록 버튼 */}
<div className="mt-4 text-center">
  
  {/* 유저용 앱 - 테스트 지갑 생성 버튼 제거됨 */}
  
  {/* 유저용 앱 - 테스트 지갑 관리 버튼 제거됨 */}
  
  {/* 유저용 앱 - 락업 초기화 버튼 제거됨 */}
  
  {/* ==================== 관리자 테스트용 시작 ==================== */}
  {/* 관리자용: 설정 등록 + 기본 락업 초기화 버튼 */}
  <div className="flex justify-center items-center space-x-3">
    {/* 설정 등록 버튼 (1/4 크기) */}
    <button 
      onClick={async () => {
        try {
          // 락업 설정 등록 로직
        const response = await fetch('http://localhost:4001/lockup/create', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            walletAddress: authenticatedWalletAddress,
              userId: authenticatedWalletAddress, // 지갑 주소를 userId로 사용
              amount: 100, // 기본 락업 금액 (테스트용)
              lockPeriod: lockupMonths // 락업 기간을 개월로 전송
          })
        });
        
          const result = await response.json();
          
          // ✅ 성공/실패 판단 로직 개선
          if (response.ok && result.success) {
            // ✅ 성공 시에만 등록 완료 처리
            const apyValue = lockupMonths === 1 ? 2.7 : lockupMonths === 6 ? 4.2 : lockupMonths === 12 ? 5.7 : 7.2;
            
            // 🎯 시한폭탄 카운트다운을 위한 시간 정보 계산 (실제 달력 기준)
            const startDate = new Date();
            const endDate = new Date(startDate);
            endDate.setMonth(endDate.getMonth() + lockupMonths);
            
            setIsLockupRegistered(true);
            setRegisteredLockupData({
              ratio: lockupRatio,
              months: lockupMonths,
              apy: apyValue,
              startDate: startDate,    // 🎯 락업 시작 시간
              endDate: endDate         //    락업 해제 시간
            });
            
            setSuccessMessage('✅ 기본 락업 설정 등록 완료!');
            setShowSuccessModal(true);
            
            // 🎯 락업 등록 성공 시 localStorage에 확실히 저장
            try {
              const currentState = JSON.parse(localStorage.getItem('miningState') || '{}');
              const updatedState = {
                ...currentState,
                isLockupRegistered: true,
                registeredLockupData: {
                  ratio: lockupRatio,
                  months: lockupMonths,
                  apy: apyValue,
                  startDate: startDate,
                  endDate: endDate
                }
              };
              localStorage.setItem('miningState', JSON.stringify(updatedState));
            } catch (storageError) {
              console.error('❌ localStorage 저장 오류:', storageError);
            }
          } else {
            // ✅ 실패 시에만 에러 메시지 표시
            setSuccessMessage(`❌ 기본 락업 설정 실패: ${result.error || '알 수 없는 오류'}`);
            setShowSuccessModal(true);
        }
      } catch (error) {
          console.error('❌ 락업 설정 등록 오류:', error);
          // 🎯 네트워크 오류도 고급스럽게 표시
          setSuccessMessage(`네트워크 오류: ${error instanceof Error ? error.message : '알 수 없는 오류'}`);
          setShowSuccessModal(true);
        }
      }}
      disabled={isLockupRegistered}
      className={`w-1/4 px-6 py-4 font-bold rounded-xl transition-all duration-200 ${
        isLockupRegistered 
          ? 'bg-gray-400 text-gray-200 cursor-not-allowed' 
          : 'bg-gradient-to-r from-purple-500 to-pink-600 text-white hover:shadow-lg transform hover:scale-105'
      }`}
    >
      {isLockupRegistered ? '✅ 설정 완료' : '🔒 설정 등록'}
    </button>
    
    {/* 기본 락업 초기화 버튼 */}
    <button
      onClick={() => setShowBasicLockupResetModal(true)}
      className="px-4 py-2 bg-red-500 text-white rounded-lg hover:bg-red-600 transition-colors shadow-lg hover:shadow-xl transform hover:-translate-y-1"
      title="기본 락업 설정을 초기화합니다"
    >
      🔄 기본 락업 초기화
    </button>
  </div>
  {/* ==================== 관리자 테스트용 끝 ==================== */}
  
  <p className="text-xs text-gray-500 mt-2">설정 등록 버튼을 클릭해야만 락업이 실제로 등록됩니다</p>
</div>
```

**Git vs Git LFS 차이점:**

## **일반 Git:**
- **작은 파일들** (텍스트, 코드)에 최적화
- 모든 파일을 Git 히스토리에 저장
- 파일이 변경될 때마다 전체 파일을 저장
- **문제점**: 큰 파일들이 있으면 저장소가 매우 무거워짐

## **Git LFS (Large File Storage):**
- **큰 파일들** (바이너리, 이미지, 실행파일)에 최적화
- 큰 파일은 별도 서버에 저장하고, Git에는 **포인터만** 저장
- 파일이 변경되어도 포인터만 변경 (실제 파일은 한 번만 저장)
- **장점**: 저장소가 가벼워짐, 클론 속도 향상

## **실제 예시:**

**일반 Git:**
```
mongod.exe (60MB) → Git 히스토리에 60MB 저장
mongod.exe 수정 → Git 히스토리에 또 60MB 저장
총 용량: 120MB+
```

**Git LFS:**
```
mongod.exe (60MB) → LFS 서버에 저장, Git에는 포인터만 (몇 바이트)
mongod.exe 수정 → LFS 서버에 새 버전 저장, Git에는 새 포인터만
총 Git 용량: 몇 바이트
```

## **우리 프로젝트에서:**
- **코드 파일들**: 일반 Git으로 관리
- **tools 폴더**: Git LFS로 관리 (MongoDB, Node.js 등 큰 바이너리들)
