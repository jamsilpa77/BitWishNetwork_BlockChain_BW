### **   2. 초기화 함수들 구현 (App.tsx 내부에 추가)**

```typescript
// ==================== 관리자와 유저 분리 시작 ====================
// 관리자용: 기본 락업 초기화 함수
const handleBasicLockupReset = () => {
  try {
    // 상태 초기화
    setLockupRatio(20);           // 20%로 초기화
    setLockupMonths(1);           // 1개월로 초기화
    setIsLockupRegistered(false); // 등록 상태 초기화
    setRegisteredLockupData({     // 등록 데이터 초기화
      ratio: 0,
      months: 0,
      apy: 0,
      startDate: null,
      endDate: null
    });
    
    // localStorage 업데이트
    const currentState = JSON.parse(localStorage.getItem('miningState') || '{}');
    const updatedState = {
      ...currentState,
      lockupRatio: 20,
      lockupMonths: 1,
      isLockupRegistered: false,
      registeredLockupData: { ratio: 0, months: 0, apy: 0, startDate: null, endDate: null }
    };
    localStorage.setItem('miningState', JSON.stringify(updatedState));
    
    // 모달 닫기
    setShowBasicLockupResetModal(false);
    setShowLockupResetModal(false);
    
    // 성공 메시지
    setSuccessMessage('✅ 기본 락업이 초기화되었습니다! (20% 비율, 1개월 기간)');
    setShowSuccessModal(true);
    
    console.log('✅ 기본 락업 초기화 완료:', { ratio: 20, months: 1, registered: false });
  } catch (error) {
    console.error('❌ 기본 락업 초기화 오류:', error);
    setErrorMessage('기본 락업 초기화 중 오류가 발생했습니다.');
    setShowErrorModal(true);
  }
};

// 관리자용: 추가 락업 초기화 함수
const handleSecondLockupReset = () => {
  try {
    // 상태 초기화
    setSecondLockupRatio(0);           // 0%로 초기화
    setSecondLockupMonths(0);          // 무선택으로 초기화
    setIsSecondLockupRegistered(false); // 등록 상태 초기화
    setRegisteredSecondLockupData({    // 등록 데이터 초기화
      ratio: 0,
      months: 0,
      apy: 0,
      startDate: null,
      endDate: null
    });
    
    // localStorage 업데이트
    const currentState = JSON.parse(localStorage.getItem('miningState') || '{}');
    const updatedState = {
      ...currentState,
      secondLockupRatio: 0,
      secondLockupMonths: 0,
      isSecondLockupRegistered: false,
      registeredSecondLockupData: { ratio: 0, months: 0, apy: 0, startDate: null, endDate: null }
    };
    localStorage.setItem('miningState', JSON.stringify(updatedState));
    
    // 모달 닫기
    setShowSecondLockupResetModal(false);
    setShowLockupResetModal(false);
    
    // 성공 메시지
    setSuccessMessage('✅ 추가 락업이 초기화되었습니다! (0% 비율, 무선택)');
    setShowSuccessModal(true);
    
    console.log('✅ 추가 락업 초기화 완료:', { ratio: 0, months: 0, registered: false });
  } catch (error) {
    console.error('❌ 추가 락업 초기화 오류:', error);
    setErrorMessage('추가 락업 초기화 중 오류가 발생했습니다.');
    setShowErrorModal(true);
  }
};
// ==================== 관리자와 유저 분리 끝 ====================
```

## **✅ 구현 완료 요약**

### ** 구현된 기능들:**

1. **"통합 락업 초기화" 버튼**: "락업 보너스 설정" 제목 오른쪽 끝 위에 배치
2. **기본 락업 초기화**: 20% 비율, 1개월 기간으로 초기화, "설정 등록" 버튼으로 변경
3. **추가 락업 초기화**: 0% 비율, 무선택 상태로 초기화
4. **추천인 관리**: "추천인 추가", "추천인 초기화" 버튼
5. **고급스러운 UI**: 작고 세련된 확인 창과 명확한 메시지
6. **완벽한 분리**: 모든 코드에 관리자와 유저 분리 주석 표시
