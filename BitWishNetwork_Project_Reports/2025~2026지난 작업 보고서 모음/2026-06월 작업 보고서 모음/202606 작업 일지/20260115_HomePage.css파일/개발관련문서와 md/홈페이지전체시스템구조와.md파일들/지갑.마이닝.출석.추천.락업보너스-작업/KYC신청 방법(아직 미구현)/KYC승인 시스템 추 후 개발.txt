const [showTokenDistributionModal, setShowTokenDistributionModal] = useState(false);

  distributionStatus: 'pending' as 'pending' | 'completed' | 'failed'

const handleTokenDistribution = async (walletAddress: string, totalPoints: number) => {
  try {
    console.log('   토큰 지급 처리 시작:', { walletAddress, totalPoints });
    
    // 1. 포인트를 토큰으로 변환 (1:1 비율)
    const tokenAmount = totalPoints;
    
    // 2. 블록체인 토큰 지급 API 호출
    const response = await fetch('http://localhost:4001/blockchain/distribute-tokens', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({
        walletAddress,
        tokenAmount,
        tokenType: 'BW',
        distributionType: 'kyc_approval'
      })
    });
    
    const result = await response.json();
    
    if (result.success) {
      // 3. 토큰 지급 완료 처리
      setTokenDistributionData(prev => ({
        ...prev,
        walletAddress,
        totalPoints,
        tokenAmount,
        distributionStatus: 'completed'
      }));
      
      // 4. 사용자에게 알림
      alert(`✅ 토큰 지급이 완료되었습니다!\n\n지급량: ${tokenAmount} BW\n지갑 주소: ${walletAddress}`);
      
      // 5. 마이닝 포인트 초기화
      await resetMiningPoints(walletAddress);
      
    } else {
      throw new Error(result.message || '토큰 지급에 실패했습니다.');
    }
    
  } catch (error) {
    console.error('❌ 토큰 지급 실패:', error);
    setTokenDistributionData(prev => ({
      ...prev,
      distributionStatus: 'failed'
    }));
    alert('❌ 토큰 지급에 실패했습니다. 다시 시도해주세요.');
  }
};

      totalMiningPoints: 0,

    console.log('✅ 마이닝 포인트 초기화 완료');
