```typescript
const startWalletCreation = async () => {
  try {
    // 추천인 코드가 입력된 경우 자동 검증
    if (referralCodeInput && referralCodeInput.trim().length > 0) {
      console.log('🔍 추천인 코드 자동 검증 시작:', referralCodeInput);
      
      // 추천인 코드 검증
      const validationResult = await validateReferralCode(referralCodeInput);
      
      if (!validationResult.isValid) {
        console.log('❌ 추천인 코드 검증 실패');
        alert(`❌ 추천인 코드 오류: ${validationResult.errorMessage}`);
        return;
      }
      
      // 추천인 코드 인증 성공
      setUserReferralSystemState(prev => ({
        ...prev,
        referralCode: validationResult.referralCode,
        referralBonusApplied: true,
        referrerWalletAddress: validationResult.walletAddress
      }));
      
      console.log('✅ 추천인 코드 인증 성공:', validationResult);
    }
    
    // 지갑 생성 단계 진행
    setWalletStep(2);
    
    // 백엔드에서 시드문구 생성
    const response = await fetch('http://localhost:4001/wallet/generate-seedphrase', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ userId: 'user_' + Date.now() })
    });
    
    const data = await response.json();
    if (data.success) {
      setGeneratedSeedPhrase(data.seedPhrase);
      setVerificationIndices(data.positions || [0, 6, 12, 18]);
      setVerificationInputs(['', '', '', '']);
    }
  } catch (error) {
    console.error('시드문구 생성 실패:', error);
    alert('시드문구 생성에 실패했습니다.');
  }
};
```

const validateSeedPhrase = async () => {
  try {
    // 1단계: 로컬 검증
    const localValidation = validateSeedPhraseLocally();
    if (!localValidation.isValid) {
      console.error('❌ 로컬 검증 실패:', localValidation.error);
      showCustomAlert(
        'error',
        'wallet.seedPhraseVerificationFailed',
        'wallet.seedPhraseInvalidBIP39',
        'seed-phrase-verification-container'
      );
      return;
    }

    const backendValidation = await validateSeedPhraseWithBackend();
    if (!backendValidation.isValid) {

    } else {

const createWalletFromSeedPhrase = async () => {
  try {
    console.log('🔍 지갑 생성 시작...');
    
    const verificationWords: { [key: number]: string } = {};
    verificationIndices.forEach((index, i) => {
      verificationWords[index + 1] = verificationInputs[i].trim();
    });

    console.log('🔍 검증 단어:', verificationWords);
    console.log('🔍 시드문구:', generatedSeedPhrase);
    console.log('🔍 시드문구 단어 수:', generatedSeedPhrase.split(' ').length);
    console.log('🔍 검증 인덱스:', verificationIndices);

    // 로컬 검증이 통과했으므로 직접 지갑 생성 진행
    console.log('✅ 로컬 검증 통과, 지갑 생성 진행...');
    
    const response = await fetch('http://localhost:4001/wallet/create', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        seedPhrase: generatedSeedPhrase,
        userId: 'user_' + Date.now()
      })
    });
    
    const createData = await response.json();
    
    if (createData.success) {
      setPublicKey(createData.wallet.address);
      console.log('✅ 지갑 생성 성공:', createData.wallet.address);
      
      // 추천인 코드 인증 처리
      if (userReferralSystemState.referralCode) {
        try {
          await handleUserReferralJoin(userReferralSystemState.referralCode, createData.wallet.address);
          
          showCustomAlert(
            'success',
            'wallet.referralCodeAuthenticationComplete',
            'wallet.referralBonusMessage',

          );

      if (userReferralSystemState.referralCode) {
        await handleWalletCreationComplete(createData.wallet.address, userReferralSystemState.referralCode);

    } else {
      console.error('❌ 지갑 생성 실패:', createData);
      showCustomAlert(
        'error',
        'wallet.seedPhraseVerificationFailed',
        'wallet.seedPhraseInvalidBIP39',

### 4. 지갑 접근 함수

```typescript
const accessWallet = async () => {
  try {
    // 시드문구가 비어있는지 확인
    if (!seedPhrase.trim()) {
      showCustomAlert(
        'warning',
        'wallet.passwordRequiredTitle',
        'wallet.passwordRequiredMessage',
        'seed-phrase-container'
      );
      return;
    }
    
    // 시드문구로 지갑 접근
    const response = await fetch('http://localhost:4001/wallet/create', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        seedPhrase: seedPhrase.trim(),
        userId: 'user_' + Date.now()
      })
    });
    
    const data = await response.json();
    if (data.success) {
      setPublicKey(data.wallet.address);
      // 지갑 정보 표시 모달로 이동
      setShowMyWallet(false);
      setShowWalletInfo(true);
      
      // 실시간 지갑 데이터 가져오기
      setTimeout(() => fetchWalletData(), 100);
      
      console.log('✅ 지갑 접근 성공:', data.wallet.address);
    } else {
      showCustomAlert(
        'error',
        'wallet.invalidAddressTitle',
        'wallet.invalidAddressMessage',
        'seed-phrase-container'
      );
    }
  } catch (error) {
    console.error('지갑 접근 실패:', error);
    showCustomAlert(
      'error',
      'wallet.invalidAddressTitle',
      'wallet.invalidAddressMessage',
      'seed-phrase-container'
    );
  }
};

// ==================== 유저용 지갑 생성 완료 시 추천인 코드 자동 발급 useEffect ====================
useEffect(() => {
  // 지갑 생성 완료 시 (walletStep === 4) 추천인 코드 자동 발급
  if (walletStep === 4 && publicKey && !userReferralCodeGenerated) {
    const autoGenerateUserReferralCode = async () => {
      try {
        console.log('   유저용 지갑 생성 완료 - 추천인 코드 자동 발급 시작');
        
        // 추천인 코드 생성 로딩 시작
        setUserReferralCodeLoading(true);
        
        // 추천인 코드 생성
        const referralCode = await generateUserReferralCode(publicKey);
        
        // 상태 업데이트
        setUserReferralCode(referralCode);
        setUserReferralCodeGenerated(true);
        setUserReferralCodeLoading(false);
        
        console.log('✅ 유저용 추천인 코드 자동 발급 완료:', referralCode);
      } catch (error) {
        console.error('❌ 유저용 추천인 코드 자동 발급 실패:', error);
        setUserReferralCodeLoading(false);
        // 에러 발생 시에도 사용자에게 알리지 않음 (자동 발급이므로)
      }
    };
    
    autoGenerateUserReferralCode();
  }
}, [walletStep, publicKey, userReferralCodeGenerated]);
```

### 9. 지갑 데이터베이스 클래스

```typescript
class WalletDatabase {
  private wallets: WalletInfo[] = [];

  registerWallet(walletInfo: WalletInfo) {
    this.wallets.push(walletInfo);
    this.saveToLocalStorage();
  }

  getWalletByAddress(address: string): WalletInfo | undefined {
    return this.wallets.find(wallet => wallet.address === address);
  }

  getAllWallets(): WalletInfo[] {
    return this.wallets;
  }

  private saveToLocalStorage() {
    try {
      localStorage.setItem('walletDatabase', JSON.stringify(this.wallets));
    } catch (error) {
      console.error('❌ 지갑 데이터베이스 저장 실패:', error);
    }
  }

  loadFromLocalStorage() {
    try {
      const saved = localStorage.getItem('walletDatabase');
      if (saved) {
        this.wallets = JSON.parse(saved);
        console.log('✅ 지갑 데이터베이스 로드 완료:', this.wallets.length, '개');
      }
    } catch (error) {
      console.error('❌ 지갑 데이터베이스 로드 실패:', error);
    }
  }
}
```

## 📋 요약

지갑 생성 시스템은 다음과 같은 구조로 되어 있습니다:

1. **네비게이션 드롭다운** → 지갑 만들기/나의 지갑 버튼
2. **지갑 생성 모달** → 4단계 지갑 생성 과정
3. **나의 지갑 모달** → 시드문구 입력으로 지갑 접근
4. **지갑 정보 모달** → 생성된 지갑 정보 표시
5. **백엔드 API** → 시드문구 생성, 지갑 생성, 검증
6. **로컬 데이터베이스** → 지갑 정보 저장/관리
7. **추천인 시스템** → 지갑 생성 시 추천인 코드 연동

모든 코드가 `App.tsx` 파일에 집중되어 있어, 이를 별도 파일로 분리하여 관리하기 쉽게 만들어야 합니다.
