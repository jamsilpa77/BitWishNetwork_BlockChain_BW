// 24시간 유예 기간 계산 함수
const isWithin24HoursFromLastAttendance = () => {
  const koreanTime = getKoreanTime();
  const now = koreanTime.getTime();
  
  // 마지막 출석일 가져오기
  const lastAttendanceDate = getLastAttendanceDate();
  if (!lastAttendanceDate) return false;
  
  // 24시간 = 24 * 60 * 60 * 1000 = 86,400,000ms
  const twentyFourHours = 24 * 60 * 60 * 1000;
  const timeDiff = now - lastAttendanceDate.getTime();
  
  return timeDiff <= twentyFourHours;
};

// 마지막 출석일 가져오기 함수 (이미 존재하지만 확인 필요)
const getLastAttendanceDate = (): Date | null => {

  const lastDate = dates[dates.length - 1];
  return new Date(lastDate);
};

  if ((!isSecondLockupRegistered && !(secondLockupRatio > 0 && secondLockupMonths > 0)) || !isMiningActive || !miningStartTime) {
    return 0;
  }

  const elapsedTimeMs = Date.now() - miningStartTime.getTime();
  const elapsedHours = elapsedTimeMs / (1000 * 60 * 60);

```typescript
const calculateCountdown = (endDate: Date | null) => {
  if (!endDate || !(endDate instanceof Date) || isNaN(endDate.getTime())) return null;
  
  const now = new Date();
  const remainingMs = endDate.getTime() - now.getTime();
  
  if (remainingMs <= 0) {
    return { days: 0, hours: 0, minutes: 0, seconds: 0, isExpired: true };
  }
  
  const days = Math.floor(remainingMs / (1000 * 60 * 60 * 24));
  const hours = Math.floor((remainingMs % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
  const minutes = Math.floor((remainingMs % (1000 * 60 * 60)) / (1000 * 60));
  const seconds = Math.floor((remainingMs % (1000 * 60)) / 1000);
  
  return { days, hours, minutes, seconds, isExpired: false };
};
```
