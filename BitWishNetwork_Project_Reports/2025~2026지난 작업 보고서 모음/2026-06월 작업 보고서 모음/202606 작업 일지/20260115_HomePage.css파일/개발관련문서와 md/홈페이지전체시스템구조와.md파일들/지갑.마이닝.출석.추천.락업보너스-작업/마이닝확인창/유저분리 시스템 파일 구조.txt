#### **2-5. 유저 실시간 마이닝 보상 컴포넌트**
```typescript
// components/user/UserMiningReward.tsx
import { useState, useEffect } from 'react';
import { UserMiningData } from '../../types/user';

interface UserMiningRewardProps {
  userData: UserMiningData;
  onUpdate: (updates: Partial<UserMiningData>) => void;
}

export const UserMiningReward = ({ userData, onUpdate }: UserMiningRewardProps) => {
  const [tick, setTick] = useState(0);
  
  useEffect(() => {
    const interval = setInterval(() => {
      setTick(prev => prev + 1);
    }, 1000);
    
    return () => clearInterval(interval);
  }, []);
  
  const calculateRealTimeReward = () => {
    if (!userData.isMiningActive || !userData.miningStartTime) return 0;
    
    const now = new Date();
    const startTime = new Date(userData.miningStartTime);
    const elapsedSeconds = Math.floor((now.getTime() - startTime.getTime()) / 1000);
    
    // 기본 마이닝 속도: 1 BW per 10 seconds
    const baseMiningRate = 0.1; // 0.1 BW per second
    const baseReward = elapsedSeconds * baseMiningRate;
    
    // 출석 보너스 적용
    const attendanceBonus = userData.consecutiveAttendanceDays * 0.01; // 1% per day
    
    // 추천인 보너스 적용
    const referralBonus = userData.referralCount * 0.005; // 0.5% per referral
    
    // 락업 보너스 적용
    const lockupBonus = (userData.lockupRatio / 100) * 0.1; // 10% of lockup ratio
    
    const totalBonus = 1 + attendanceBonus + referralBonus + lockupBonus;
    return baseReward * totalBonus;
  };
  
  const realTimeReward = calculateRealTimeReward();
  
  return (
    <div className="bg-green-100 border border-green-300 rounded-lg p-3 mt-4">
      <h3 className="text-lg font-bold text-green-700 text-center mb-3">실시간 마이닝 보상</h3>
      <div className="grid grid-cols-2 gap-4">
        {/* 누적 보상 */}
        <div className="text-center">
          <div className="text-2xl font-bold text-green-600 mb-2">
            {realTimeReward.toFixed(4)} BW
          </div>
          <div className="text-sm text-gray-600">누적 보상</div>
        </div>
        
        {/* 시간당 보상 */}
        <div className="text-center">
          <div className="text-2xl font-bold text-green-600 mb-2">
            {(realTimeReward * 3600).toFixed(2)} BW
          </div>
          <div className="text-sm text-gray-600">시간당 예상</div>
        </div>
      </div>
      
      {/* 보너스 상세 정보 */}
      <div className="mt-4 p-3 bg-white rounded-lg">
        <h4 className="font-semibold text-gray-700 mb-2">보너스 상세</h4>
        <div className="grid grid-cols-2 gap-2 text-sm">
          <div>출석 보너스: +{userData.consecutiveAttendanceDays}%</div>
          <div>추천인 보너스: +{userData.referralCount * 0.5}%</div>
          <div>락업 보너스: +{userData.lockupRatio * 0.1}%</div>
          <div>총 보너스: +{((userData.consecutiveAttendanceDays + userData.referralCount * 0.5 + userData.lockupRatio * 0.1) * 100).toFixed(1)}%</div>
        </div>
      </div>
    </div>
  );
};
```

```typescript
// services/adminService.ts
import { UserMiningData } from '../types/user';

export class AdminService {
  static addTestReferral(walletAddress: string): void {
    // 관리자 전용 추천인 추가
    const userData = this.loadUserData(walletAddress);
    if (userData) {
      userData.referralCount += 1;
      this.saveUserData(walletAddress, userData);
    }
  }
  
  static resetTestReferral(walletAddress: string): void {
    // 관리자 전용 추천인 초기화
    const userData = this.loadUserData(walletAddress);
    if (userData) {
      userData.referralCount = 0;
      this.saveUserData(walletAddress, userData);
    }
  }
  
  static resetBasicLockup(walletAddress: string): void {
    // 관리자 전용 기본 락업 초기화
    const userData = this.loadUserData(walletAddress);
    if (userData) {
      userData.lockupRatio = 20;
      userData.lockupMonths = 1;
      this.saveUserData(walletAddress, userData);
    }
  }
  
  static resetAdditionalLockup(walletAddress: string): void {
    // 관리자 전용 추가 락업 초기화
    const userData = this.loadUserData(walletAddress);
    if (userData) {
      userData.lockupRatio = 0;
      userData.lockupMonths = 0;
      this.saveUserData(walletAddress, userData);
    }
  }
  
  static async resetMining(): Promise<void> {
    // 관리자 전용 마이닝 초기화
    try {
      // 가상 마이닝 완전 초기화
      await this.resetVirtualMining();
      
      // localStorage에서 마이닝 상태 완전 삭제
      localStorage.removeItem('miningState');
      
      // 모든 마이닝 관련 상태 초기화
      const allUserData = this.getAllUserData();
      allUserData.forEach(userData => {
        userData.isMiningActive = false;
        userData.miningStartTime = null;
        userData.totalMinedReward = 0;
        userData.dailyMiningReward = 0;
        userData.weeklyMiningReward = 0;
        userData.monthlyMiningReward = 0;
        this.saveUserData(userData.walletAddress, userData);
      });
      
      console.log('🔒 관리자 마이닝 초기화 완료');
    } catch (error) {
      console.error('❌ 관리자 마이닝 초기화 오류:', error);
      throw error;
    }
  }
  
  private static async resetVirtualMining(): Promise<void> {
    // 가상 마이닝 시스템 초기화 로직
    console.log('🔄 가상 더미 마이닝 초기화...');
  }
  
  private static loadUserData(walletAddress: string): UserMiningData | null {
    const data = localStorage.getItem(`userData_${walletAddress}`);
    return data ? JSON.parse(data) : null;
  }
  
  private static saveUserData(walletAddress: string, userData: UserMiningData): void {
    localStorage.setItem(`userData_${walletAddress}`, JSON.stringify(userData));
  }
  
  private static getAllUserData(): UserMiningData[] {
    const allData: UserMiningData[] = [];
    for (let i = 0; i < localStorage.length; i++) {
      const key = localStorage.key(i);
      if (key && key.startsWith('userData_')) {
        const data = localStorage.getItem(key);
        if (data) {
          allData.push(JSON.parse(data));
        }
      }
    }
    return allData;
  }
}
```

```typescript
// services/userService.ts
import { UserMiningData } from '../types/user';

export class UserService {
  static initializeUserData(walletAddress: string, clientIP: string): UserMiningData {
    // 유저 데이터 초기화 (0에서 시작)
    const newUserData: UserMiningData = {
      walletAddress,
      attendanceCalendar: {},
      consecutiveAttendanceDays: 1,
      referralCount: 0,
      referralCode: this.generateUniqueReferralCode(),
      lockupRatio: 20,
      lockupMonths: 1,
      isMiningActive: false,
      miningStartTime: null,
      totalMinedReward: 0,
      ipAddresses: [clientIP],
      createdAt: new Date(),
      lastActivity: new Date(),
      attendanceBonusHistory: {},
      referralBonusHistory: {},
      lockupBonusHistory: {},
      dailyMiningReward: 0,
      weeklyMiningReward: 0,
      monthlyMiningReward: 0
    };
    
    this.saveUserData(walletAddress, newUserData);
    return newUserData;
  }
  
  static loadUserData(walletAddress: string): UserMiningData | null {
    // 유저 데이터 로드
    const data = localStorage.getItem(`userData_${walletAddress}`);
    return data ? JSON.parse(data) : null;
  }
  
  static updateUserData(walletAddress: string, updates: Partial<UserMiningData>): void {
    // 유저 데이터 업데이트
    const currentData = this.loadUserData(walletAddress);
    if (currentData) {
      const updatedData = { ...currentData, ...updates, lastActivity: new Date() };
      this.saveUserData(walletAddress, updatedData);
    }
  }
  
  static calculateUserBonus(userData: UserMiningData): number {
    // 유저별 보너스 계산
    const attendanceBonus = userData.consecutiveAttendanceDays * 0.01;
    const referralBonus = userData.referralCount * 0.005;
    const lockupBonus = (userData.lockupRatio / 100) * 0.1;
    
    return 1 + attendanceBonus + referralBonus + lockupBonus;
  }
  
  private static generateUniqueReferralCode(): string {
    // 고유 추천인 코드 생성
    return `REF${Date.now().toString(36).toUpperCase()}${Math.random().toString(36).substr(2, 4).toUpperCase()}`;
  }
  
  private static saveUserData(walletAddress: string, userData: UserMiningData): void {
    localStorage.setItem(`userData_${walletAddress}`, JSON.stringify(userData));
  }
}
```

```typescript
export const generateUniqueReferralCode = (): string => {
  // 고유 추천인 코드 생성
  return `REF${Date.now().toString(36).toUpperCase()}${Math.random().toString(36).substr(2, 4).toUpperCase()}`;
};

export const calculateConsecutiveDays = (attendanceCalendar: {[key: string]: boolean}): number => {
  // 연속 출석일 계산
  const today = new Date();
  let consecutiveDays = 0;
  
  for (let i = 0; i < 365; i++) {
    const checkDate = new Date(today);
    checkDate.setDate(today.getDate() - i);
    const dateKey = checkDate.toISOString().split('T')[0];
    
    if (attendanceCalendar[dateKey]) {
      consecutiveDays++;
    } else {
      break;
    }
  }
  
  return consecutiveDays;
};

export const calculateUserAttendanceBonus = (consecutiveDays: number): number => {
  // 유저별 출석 보너스 계산
  return consecutiveDays * 0.01; // 1% per day
};
```

```typescript
// hooks/useUserData.ts
import { useState, useEffect, useCallback } from 'react';
import { UserMiningData } from '../types/user';
import { UserService } from '../services/userService';

export const useUserData = (walletAddress: string) => {
  const [userData, setUserData] = useState<UserMiningData | null>(null);
  const [isLoading, setIsLoading] = useState(true);
  
  const updateUserData = useCallback((updates: Partial<UserMiningData>) => {
    UserService.updateUserData(walletAddress, updates);
    setUserData(prev => prev ? { ...prev, ...updates } : null);
  }, [walletAddress]);
  
  useEffect(() => {
    const data = UserService.loadUserData(walletAddress);
    if (data) {
      setUserData(data);
    } else {
      const newData = UserService.initializeUserData(walletAddress, getClientIP());
      setUserData(newData);
    }
    setIsLoading(false);
  }, [walletAddress]);
  
  return { userData, updateUserData, isLoading };
};

const getClientIP = (): string => {
  // 클라이언트 IP 주소 가져오기 (실제 구현에서는 API 호출)
  return '127.0.0.1';
};
```
