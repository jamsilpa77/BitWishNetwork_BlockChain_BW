```typescript
// Node_HomePage/src/services/lockupService.ts
import { Decimal } from 'decimal.js';

export class LockupService {
  //    기본락업 보너스 계산 (백서 기준 - 50단위 부동소수점)
  static calculateBasicLockupBonus(ratio: number, months: number): {
    apy: Decimal;
    totalAPY: Decimal;
    monthlyRate: Decimal;
    enhancedRate: Decimal;
    description: string;
    lockedAmount: Decimal;
  } {
    if (ratio === 0 || months === 0) {
      return {
        apy: new Decimal(0),
        totalAPY: new Decimal(0),
        monthlyRate: new Decimal(0),
        enhancedRate: new Decimal(0.25),
        description: '락업하지 않음',
        lockedAmount: new Decimal(0)
      };
    }

    //    백서 기준 APY 계산 (50단위 부동소수점)
    let baseAPY: Decimal;
    if (months >= 36) {
      baseAPY = new Decimal(0.072); // 7.2%
    } else if (months >= 12) {
      baseAPY = new Decimal(0.057); // 5.7%
    } else if (months >= 6) {
      baseAPY = new Decimal(0.042); // 4.2%
    } else {
      baseAPY = new Decimal(0.027); // 2.7%
    }

    const ratioBonus = new Decimal(ratio).dividedBy(100); // 1% per 1% ratio
    const totalAPY = baseAPY.plus(ratioBonus);
    const monthlyRate = totalAPY.dividedBy(12);
    const enhancedRate = new Decimal(0.25).times(new Decimal(1).plus(totalAPY.times(ratio).dividedBy(100)));

    // 🎯 락업된 금액 계산 (50단위 부동소수점)
    const totalBalance = this.getTotalBalance();
    const lockedAmount = totalBalance.times(ratio).dividedBy(100);

    return {
      apy: baseAPY,
      totalAPY: totalAPY,
      monthlyRate: monthlyRate,
      enhancedRate: enhancedRate,
      description: `${months}개월 락업 (${totalAPY.times(100).toFixed(50)}% APY)`,
      lockedAmount: lockedAmount
    };
  }

  // 🎯 추가락업 보너스 계산 (백서 기준 - 50단위 부동소수점)
  static calculateAdditionalLockupBonus(ratio: number, months: number): {
    apy: Decimal;
    totalAPY: Decimal;
    monthlyRate: Decimal;
    enhancedRate: Decimal;
    description: string;
    lockedAmount: Decimal;
  } {
    if (ratio === 0 || months === 0) {
      return {
        apy: new Decimal(0),
        totalAPY: new Decimal(0),
        monthlyRate: new Decimal(0),
        enhancedRate: new Decimal(0.25),
        description: '추가락업하지 않음',
        lockedAmount: new Decimal(0)
      };
    }

    // 🎯 추가락업 APY 계산 (50단위 부동소수점)
    let baseAPY: Decimal;
    if (months >= 36) {
      baseAPY = new Decimal(0.075); // 7.5%
    } else if (months >= 12) {
      baseAPY = new Decimal(0.049); // 4.9%
    } else if (months >= 6) {
      baseAPY = new Decimal(0.035); // 3.5%
    } else {
      baseAPY = new Decimal(0.021); // 2.1%
    }

    const ratioBonus = new Decimal(ratio).dividedBy(100);
    const totalAPY = baseAPY.plus(ratioBonus);
    const monthlyRate = totalAPY.dividedBy(12);
    const enhancedRate = new Decimal(0.25).times(new Decimal(1).plus(totalAPY.times(ratio).dividedBy(100)));

    // 🎯 추가락업된 금액 계산 (50단위 부동소수점)
    const totalBalance = this.getTotalBalance();
    const lockedAmount = totalBalance.times(ratio).dividedBy(100);

    return {
      apy: baseAPY,
      totalAPY: totalAPY,
      monthlyRate: monthlyRate,
      enhancedRate: enhancedRate,
      description: `${months}개월 추가락업 (${totalAPY.times(100).toFixed(50)}% APY)`,
      lockedAmount: lockedAmount
    };
  }

  //    기본락업보관함 계산 (50단위 부동소수점)
  static calculateBasicLockupVault(
    baseReward: number,
    lockupRatio: number,
    lockupMonths: number,
    elapsedHours: number
  ): Decimal {
    const lockupData = this.calculateBasicLockupBonus(lockupRatio, lockupMonths);
    const baseRewardDecimal = new Decimal(baseReward);
    const ratioDecimal = new Decimal(lockupRatio).dividedBy(100);
    const elapsedHoursDecimal = new Decimal(elapsedHours);
    
    const hourlyLockupReward = baseRewardDecimal.times(lockupData.totalAPY);
    const totalLockupReward = hourlyLockupReward.times(elapsedHoursDecimal);
    const vaultAmount = totalLockupReward.times(ratioDecimal);
    
    return vaultAmount;
  }

  // 🎯 추가락업보관함 계산 (50단위 부동소수점)
  static calculateAdditionalLockupVault(
    baseReward: number,
    lockupRatio: number,
    lockupMonths: number,
    elapsedHours: number
  ): Decimal {
    const lockupData = this.calculateAdditionalLockupBonus(lockupRatio, lockupMonths);
    const baseRewardDecimal = new Decimal(baseReward);
    const ratioDecimal = new Decimal(lockupRatio).dividedBy(100);
    const elapsedHoursDecimal = new Decimal(elapsedHours);
    
    const hourlyLockupReward = baseRewardDecimal.times(lockupData.totalAPY);
    const totalLockupReward = hourlyLockupReward.times(elapsedHoursDecimal);
    const vaultAmount = totalLockupReward.times(ratioDecimal);
    
    return vaultAmount;
  }

  // 🎯 총 보유량 조회 (50단위 부동소수점)
  private static getTotalBalance(): Decimal {
    const balance = localStorage.getItem('totalBalance') || '0';
    return new Decimal(balance);
  }

  // 🎯 기존 호환성을 위한 calculateLockupBonus 함수
  static calculateLockupBonus(ratio: number, months: number): {apy: number; totalAPY: number; monthlyRate: number; enhancedRate: number; description: string} {
    const result = this.calculateBasicLockupBonus(ratio, months);
    return {
      apy: result.apy.toNumber(),
      totalAPY: result.totalAPY.toNumber(),
      monthlyRate: result.monthlyRate.toNumber(),
      enhancedRate: result.enhancedRate.toNumber(),
      description: result.description
    };
  }
}
```

1. **2단계 락업 구조**: 기본락업 + 추가락업
2. **16가지 변칙선택**: 각각 4×4 = 16가지 조합
3. **50단위 부동소수점**: Decimal.js 기반 정밀 계산
4. **락업 4개 항목**: 기본락업보너스/추가락업보너스/기본락업보관함/추가락업보관함
5. **개인별 데이터 저장**: 지갑 주소별 독립적 관리
6. **실시간 계산**: 설정 변경 시 즉시 반영

### ** 1단계: LockupService에 락업 해제 시스템 추가**

```typescript
// Node_HomePage/src/services/lockupService.ts
import { Decimal } from 'decimal.js';

export class LockupService {
  // ... 기존 함수들 ...

  //    락업 해제 가능 여부 확인 (50단위 부동소수점)
  static canReleaseLockup(lockupStartDate: Date, lockupMonths: number): boolean {
    const now = new Date();
    const lockupEndDate = new Date(lockupStartDate);
    lockupEndDate.setMonth(lockupEndDate.getMonth() + lockupMonths);
    
    return now >= lockupEndDate;
  }

  // 🎯 락업 남은 기간 계산 (50단위 부동소수점)
  static calculateRemainingLockupDays(lockupStartDate: Date, lockupMonths: number): {
    days: number;
    hours: number;
    minutes: number;
    seconds: number;
    totalSeconds: number;
  } {
    const now = new Date();
    const lockupEndDate = new Date(lockupStartDate);
    lockupEndDate.setMonth(lockupEndDate.getMonth() + lockupMonths);
    
    const diffTime = lockupEndDate.getTime() - now.getTime();
    const totalSeconds = Math.max(0, Math.floor(diffTime / 1000));
    const days = Math.floor(totalSeconds / (24 * 60 * 60));
    const hours = Math.floor((totalSeconds % (24 * 60 * 60)) / (60 * 60));
    const minutes = Math.floor((totalSeconds % (60 * 60)) / 60);
    const seconds = totalSeconds % 60;
    
    return {
      days,
      hours,
      minutes,
      seconds,
      totalSeconds
    };
  }

  // 🎯 락업 해제일 계산
  static calculateLockupReleaseDate(lockupStartDate: Date, lockupMonths: number): Date {
    const lockupEndDate = new Date(lockupStartDate);
    lockupEndDate.setMonth(lockupEndDate.getMonth() + lockupMonths);
    return lockupEndDate;
  }

  // 🎯 락업 해제 후 15일 이전 기간 계산
  static calculateGracePeriodEndDate(lockupEndDate: Date): Date {
    const gracePeriodEndDate = new Date(lockupEndDate);
    gracePeriodEndDate.setDate(gracePeriodEndDate.getDate() + 15);
    return gracePeriodEndDate;
  }

  // 🎯 락업 해제 버튼 상태 계산
  static getLockupReleaseButtonState(lockupStartDate: Date, lockupMonths: number): {
    disabled: boolean;
    text: string;
    icon: string;
    className: string;
    releaseDate?: string;
    remainingTime?: string;
  } {
    const now = new Date();
    const lockupEndDate = this.calculateLockupReleaseDate(lockupStartDate, lockupMonths);
    const gracePeriodEndDate = this.calculateGracePeriodEndDate(lockupEndDate);

    // 락업 기간 중
    if (now < lockupEndDate) {
      const remaining = this.calculateRemainingLockupDays(lockupStartDate, lockupMonths);
      return {
        disabled: true,
        text: '🔒 락업 잠김',
        icon: '🔒',
        className: 'bg-gray-400 text-white cursor-not-allowed',
        releaseDate: `해제일: ${lockupEndDate.toLocaleDateString()}`,
        remainingTime: `${remaining.days}일 ${remaining.hours}시간 ${remaining.minutes}분 ${remaining.seconds}초 남음`
      };
    }

    // 이전 기간 중 (15일)
    if (now < gracePeriodEndDate) {
      const remaining = this.calculateRemainingLockupDays(lockupEndDate, 0);
      return {
        disabled: true,
        text: '⏳ 이전 기간 중',
        icon: '⏳',
        className: 'bg-yellow-500 text-white cursor-not-allowed',
        releaseDate: `이전 완료일: ${gracePeriodEndDate.toLocaleDateString()}`,
        remainingTime: `${remaining.days}일 ${remaining.hours}시간 ${remaining.minutes}분 ${remaining.seconds}초 남음`
      };
    }

    // 해제 가능
    return {
      disabled: false,
      text: '🔓 락업 해제',
      icon: '🔓',
      className: 'bg-green-500 hover:bg-green-600 text-white',
      releaseDate: `해제일: ${lockupEndDate.toLocaleDateString()}`
    };
  }

  // 🎯 락업 해제 실행
  static executeLockupRelease(lockupData: any): {
    success: boolean;
    message: string;
    releasedAmount: Decimal;
  } {
    try {
      const lockupEndDate = this.calculateLockupReleaseDate(lockupData.startDate, lockupData.months);
      const gracePeriodEndDate = this.calculateGracePeriodEndDate(lockupEndDate);
      const now = new Date();

      if (now < gracePeriodEndDate) {
        return {
          success: false,
          message: '아직 이전 기간이 완료되지 않았습니다.',
          releasedAmount: new Decimal(0)
        };
      }

      // 락업 해제 실행
      const releasedAmount = new Decimal(lockupData.amount);
      
      return {
            success: true,
            message: '락업이 성공적으로 해제되었습니다.',
            releasedAmount: releasedAmount
          };
    } catch (error) {
      return {
        success: false,
        message: '락업 해제 중 오류가 발생했습니다.',
        releasedAmount: new Decimal(0)
      };
    }
  }
}
```

### ** 3단계: 실시간 락업 해제 타이머 시스템**

```typescript
// App.tsx - 실시간 락업 해제 타이머 시스템
const [lockupTimer, setLockupTimer] = useState<NodeJS.Timeout | null>(null);

// 락업 해제 타이머 시작
useEffect(() => {
  const startLockupTimer = () => {
    if (lockupTimer) {
      clearInterval(lockupTimer);
    }

    const timer = setInterval(() => {
      // 개인별 락업 상태 업데이트
      setUserMiningStates(prev => {
        const updated = { ...prev };
        Object.keys(updated).forEach(walletAddress => {
          const state = updated[walletAddress];
          if (state?.basicLockupStartDate) {
            const buttonState = LockupService.getLockupReleaseButtonState(
              new Date(state.basicLockupStartDate),
              state.basicLockupMonths || 0
            );
            
            // 락업 해제 가능 상태 업데이트
            updated[walletAddress] = {
              ...state,
              basicLockupCanRelease: !buttonState.disabled,
              basicLockupRemainingTime: buttonState.remainingTime || null
            };
          }
          
          if (state?.additionalLockupStartDate) {
            const buttonState = LockupService.getLockupReleaseButtonState(
              new Date(state.additionalLockupStartDate),
              state.additionalLockupMonths || 0
            );
            
            // 락업 해제 가능 상태 업데이트
            updated[walletAddress] = {
              ...state,
              additionalLockupCanRelease: !buttonState.disabled,
              additionalLockupRemainingTime: buttonState.remainingTime || null
            };
          }
        });
        return updated;
      });
    }, 1000); // 1초마다 업데이트

    setLockupTimer(timer);
  };

  startLockupTimer();

  return () => {
    if (lockupTimer) {
      clearInterval(lockupTimer);
    }
  };
}, [lockupTimer]);
```

## ** 완성된 락업 해제 시스템 특징**

### ** ✅ 완전 구현된 기능들**
1. **락업 해제 타이머**: 실시간 남은 시간 계산
2. **해제일 시스템**: 정확한 해제일 계산 및 표시
3. **15일 이전 기간**: 해제 후 15일 대기 기간
4. **동적 버튼 상태**: 락업 상태에 따른 버튼 활성화/비활성화
5. **실시간 업데이트**: 1초마다 상태 업데이트
6. **개인별 관리**: 지갑 주소별 독립적 락업 해제 관리
7. **50단위 부동소수점**: 정밀한 시간 계산
