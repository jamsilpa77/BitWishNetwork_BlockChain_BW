### **✅ 좌측 열: 시간당 보상 및 보너스 (라인 9262-9345)**
```typescript
{/* 좌측 열: 시간당 보상 및 보너스 */}
<div className="space-y-3">
  {/* 시간당 보상 (보너스 포함) BW */}
  <div className="text-center mr-16">
    <div className="text-2xl font-bold text-purple-600">
      {(() => {
        const rewardData = calculateRealTimeMiningRewardDecimal();
        return rewardData.hourlyReward.toFixed(8);
      })()} BW
    </div>
    <p className="text-sm text-gray-600">시간당 보상 (보너스 포함)</p>
  </div>

  {/* 출석 보너스 % */}
  <div className="text-center mr-16">
    <div className="text-lg font-semibold text-purple-600">
      {(() => {
        const rewardData = calculateRealTimeMiningRewardDecimal();
        return rewardData.ui.attendancePct.toFixed(6);
      })()}%
    </div>
    <p className="text-sm text-gray-600">출석 보너스</p>
  </div>

  {/* 기본 락업 보관함 BW */}
  <div className="text-center">
    <div className="text-lg font-semibold text-blue-600">
      {(() => {
        //    락업 초기화 시 보관함도 완전 초기화
        if (!isLockupRegistered || !isMiningActive || !miningStartTime) {
          return '0.000000';
        }
          
        // 기본 락업 보관함 계산 (실시간 반영)
        const elapsedTimeMs = Date.now() - miningStartTime.getTime();
        const elapsedHours = elapsedTimeMs / (1000 * 60 * 60);
        
        // 총 노드 보상 계산 (기본 + 출석 + 추천)
        const baseRate = 0.25;
        const attendanceBonus = (consecutiveAttendanceDays > 0 && isAttendanceBonusTimeValid()) ? 
          calculateAttendanceBonus(consecutiveAttendanceDays).bonusRate : 0;
        // 🎯 calculateRealTimeMiningRewardDecimal에서 계산된 정확한 추천 보너스 사용
        const rewardData = calculateRealTimeMiningRewardDecimal();
        const referralBonus = rewardData.referralBonus; // 이미 소수점 형태
        const totalNodeReward = baseRate * (1 + attendanceBonus + referralBonus);
        
        // 락업 보너스 (등록된 APY 사용)
        const lockupBonusRate = registeredLockupData.apy / 100;
        const lockupBonusPerHour = totalNodeReward * lockupBonusRate;
        
        // 락업된 비율만큼 보관함에 저장
        const lockupRatioDecimal = registeredLockupData.ratio / 100;
        const storedLockupBonus = lockupBonusPerHour * elapsedHours * lockupRatioDecimal;
        
        return storedLockupBonus.toFixed(6);
      })()} BW
    </div>
    <p className="text-sm text-gray-600">기본 락업 보관함</p>
  </div>

  {/* 추가 락업 보관함 BW */}
  {(isSecondLockupRegistered || (secondLockupRatio > 0 && secondLockupMonths > 0)) && (
    <div className="text-center">
      <div className="text-lg font-semibold text-purple-600">
        {calculateSecondLockupVault().toFixed(6)} BW
      </div>
      <p className="text-sm text-gray-600">추가 락업 보관함</p>
    </div>
  )}
</div>
```

## ** 5. 실시간 마이닝 보상 창 (라인 9719-9850)**
```typescript
{/* ⚡ 실시간 마이닝 보상 */}
{(isMiningActive || (!isMiningActive && miningStartTime)) && (() => {
  // ✅ 렌더링 시점에 총 경과 시간을 계산합니다.
  let totalElapsedTime = 0; // ✅ 누적 시간 초기화
  if (miningStartTime) {
    if (isMiningActive) {
      //    마이닝 진행 중: 실시간으로 경과 시간 계산
```

```typescript
//    BigDecimal 기반 실시간 마이닝 보상 계산 함수 (완벽한 정밀도)
const calculateRealTimeMiningRewardDecimal = (): {
  hourlyReward: Decimal;
  dailyReward: Decimal;
  referralBonus: number;
  ui: {
    attendancePct: number;
    referralPct: number;
    totalPct: number;
  };
} => {
  // 기본 채굴률 (BigDecimal)
  const decimalBaseRate = new Decimal(0.25);
  
  // 출석 보너스 (BigDecimal) - 시간 제한 적용
  const decimalAttendanceBonus = (consecutiveAttendanceDays > 0 && isAttendanceBonusTimeValid()) ? 
    new Decimal(calculateAttendanceBonus(consecutiveAttendanceDays).bonusRate) : 
    new Decimal(0);
  
  // 추천 보너스 (BigDecimal) - 구간별 누적 계산
  const decimalReferralBonus = new Decimal(calculateReferralBonusDecimal(referralCount).totalBonusRate);
  
  // 락업 보너스 (BigDecimal)
  const decimalLockupBonus = new Decimal(calculateLockupBonus(lockupRatio, lockupMonths).monthlyRate);
  
  // 가맹점 보너스 (BigDecimal)
  const decimalMerchantBonus = new Decimal(isMerchantRegistered ? calculateMerchantBonus(isMerchantRegistered).bonusRate : 0);
  
  // 총 보너스율 계산 (BigDecimal)
  const decimalOne = new Decimal(1);
  const decimalTotalBonusRate = decimalOne
    .plus(decimalAttendanceBonus)
    .plus(decimalReferralBonus)
    .plus(decimalLockupBonus)
    .plus(decimalMerchantBonus);
  
  // 강화된 채굴률 계산 (BigDecimal)
  const decimalEnhancedRate = decimalBaseRate.times(decimalTotalBonusRate);
  
  // 시간당 보상 계산 (BigDecimal)
  const decimalHourlyReward = decimalEnhancedRate;
  
  // 일일 보상 계산 (BigDecimal)
  const decimalDailyReward = decimalHourlyReward.times(24);
  
  return {
    hourlyReward: decimalHourlyReward,
    dailyReward: decimalDailyReward,
    referralBonus: decimalReferralBonus.toNumber(),
    ui: {
      attendancePct: decimalAttendanceBonus.times(100).toNumber(),
      referralPct: decimalReferralBonus.times(100).toNumber(),
      totalPct: decimalTotalBonusRate.minus(1).times(100).toNumber()
    }
  };
};
```

### **✅ 누적 보상 계산 함수 (라인 2637-2750)**

```typescript
// 🎯 BigDecimal 기반 누적 보상 계산 함수 (완벽한 정밀도)
const calculateAccumulatedRewardDecimal = (): {
  elapsedHours: Decimal;
  totalEnhancedRate: Decimal;
  accumulatedReward: Decimal;
  lockupVaultReward: Decimal;
  referralVaultReward: Decimal;
  totalAccumulatedReward: Decimal;
} => {
  if (!miningStartTime) {
    return {
      elapsedHours: new Decimal(0),
      totalEnhancedRate: new Decimal(0),
      accumulatedReward: new Decimal(0),
      lockupVaultReward: new Decimal(0),
      referralVaultReward: new Decimal(0),
      totalAccumulatedReward: new Decimal(0)
    };
  }
  
  // 경과 시간 계산 (BigDecimal)
  const currentTime = new Date();
  const elapsedTimeMs = currentTime.getTime() - miningStartTime.getTime();
  const decimalElapsedTimeMs = new Decimal(elapsedTimeMs);
  const decimalMsPerHour = new Decimal(1000 * 60 * 60);
  const decimalElapsedHours = decimalElapsedTimeMs.dividedBy(decimalMsPerHour);
  
  // 실시간 마이닝 보상 데이터 가져오기
  const rewardData = calculateRealTimeMiningRewardDecimal();
  
  // 기본 누적 보상 계산 (BigDecimal)
  const decimalAccumulatedReward = rewardData.hourlyReward.times(decimalElapsedHours);
  
  // 락업 보관함 보상 계산 (BigDecimal)
  const decimalLockupVaultReward = isLockupRegistered ? 
    decimalAccumulatedReward.times(new Decimal(registeredLockupData.ratio / 100)) : 
    new Decimal(0);
  
  // 추천 보관 BW 계산 (BigDecimal)
  const decimalReferralVaultReward = referralCount > 0 ? 
    new Decimal(calculateReferralStorageBW(referralCount)) : 
    new Decimal(0);
  
  // 총 누적 보상 계산 (BigDecimal)
  const decimalTotalAccumulatedReward = decimalAccumulatedReward
    .plus(decimalLockupVaultReward)
    .plus(decimalReferralVaultReward);
  
  return {
    elapsedHours: decimalElapsedHours,
    totalEnhancedRate: rewardData.hourlyReward,
    accumulatedReward: decimalAccumulatedReward,
    lockupVaultReward: decimalLockupVaultReward,
    referralVaultReward: decimalReferralVaultReward,
    totalAccumulatedReward: decimalTotalAccumulatedReward
  };
};
```
