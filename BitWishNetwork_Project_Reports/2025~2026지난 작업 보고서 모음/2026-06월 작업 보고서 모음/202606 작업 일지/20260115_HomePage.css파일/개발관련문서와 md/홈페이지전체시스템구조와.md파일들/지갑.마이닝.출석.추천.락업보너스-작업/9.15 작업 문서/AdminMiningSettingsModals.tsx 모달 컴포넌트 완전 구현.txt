interface AttendanceModalProps {
  isOpen: boolean;
  onClose: () => void;
  consecutiveDays: number;
  onAttendanceChange: (days: number) => void;
  onSave: () => void;
  attendanceCalendar: Record<string, boolean>;
}

export const AttendanceModal: React.FC<AttendanceModalProps> = ({
  isOpen,
  onClose,
  consecutiveDays,
  onAttendanceChange,
  onSave,
  attendanceCalendar
}) => {
  const { t } = useTranslation();
  const [tempDays, setTempDays] = useState(consecutiveDays);

  useEffect(() => {
    if (isOpen) {
      setTempDays(consecutiveDays);
    }
  }, [isOpen, consecutiveDays]);

  const handleSave = () => {
    onAttendanceChange(tempDays);
    onSave();
    onClose();
  };

  const calculateAttendanceBonus = (days: number) => {
    if (days <= 0) return { bonusRate: 0, enhancedRate: 0 };
    
    const baseRate = new Decimal(0.25); // 25%
    const bonusRate = baseRate.times(100);
    const enhancedRate = new Decimal(1).plus(baseRate).times(100);
    
    return {
      bonusRate: bonusRate.toNumber(),
      enhancedRate: enhancedRate.toNumber()
    };
  };

  const bonusData = calculateAttendanceBonus(tempDays);

  // 출석 달력 렌더링
  const renderAttendanceCalendar = () => {
    const today = new Date();
    const currentMonth = today.getMonth();
    const currentYear = today.getFullYear();
    const daysInMonth = new Date(currentYear, currentMonth + 1, 0).getDate();
    
    const calendarDays = [];
    for (let day = 1; day <= daysInMonth; day++) {
      const dateString = `${currentYear}-${String(currentMonth + 1).padStart(2, '0')}-${String(day).padStart(2, '0')}`;
      const isAttended = attendanceCalendar[dateString] || false;
      const isToday = day === today.getDate();
      
      calendarDays.push(
        <div
          key={day}
          className={`w-8 h-8 rounded-full flex items-center justify-center text-xs font-medium ${
            isAttended 
              ? 'bg-green-500 text-white' 
              : isToday 
                ? 'bg-blue-500 text-white' 
                : 'bg-gray-200 text-gray-600'
          }`}
        >
          {day}
        </div>
      );
    }
    
    return calendarDays;
  };

  if (!isOpen) return null;

  return (
    <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center z-[10000]">
      <div className="bg-white rounded-2xl shadow-2xl max-w-4xl w-full mx-4 max-h-[90vh] overflow-y-auto">
        {/* 헤더 */}
        <div className="p-6 border-b border-gray-200 flex justify-between items-center">
          <h2 className="text-2xl font-bold text-gray-900">📅 출석 보너스 설정</h2>
          <button 
            onClick={onClose}
            className="text-gray-400 hover:text-gray-600 transition-colors p-2 hover:bg-gray-100 rounded-full"
          >
            <svg className="w-6 h-6" fill="none" stroke="currentColor" viewBox="0 0 24 24">
              <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M6 18L18 6M6 6l12 12" />
            </svg>
          </button>
        </div>

        {/* 콘텐츠 */}
        <div className="p-6">
          <div className="grid md:grid-cols-2 gap-8">
            {/* 왼쪽: 출석 설정 */}
            <div>
              {/* 현재 연속 출석일 */}
              <div className="mb-6">
                <label className="block text-sm font-medium text-gray-700 mb-2">
                  현재 연속 출석일
                </label>
                <div className="text-3xl font-bold text-green-600">
                  {consecutiveDays}일
                </div>
              </div>

              {/* 연속 출석일 설정 */}
              <div className="mb-6">
                <label className="block text-sm font-medium text-gray-700 mb-2">
                  연속 출석일 설정 (0-365일)
                </label>
                <div className="flex items-center space-x-4">
                  <input
                    type="range"
                    min="0"
                    max="365"
                    value={tempDays}
                    onChange={(e) => setTempDays(parseInt(e.target.value))}
                    className="flex-1 h-2 bg-green-200 rounded-lg appearance-none cursor-pointer"
                    style={{
                      background: `linear-gradient(to right, #10b981 0%, #10b981 ${tempDays / 365 * 100}%, #e5e7eb ${tempDays / 365 * 100}%, #e5e7eb 100%)`
                    }}
                  />
                  <div className="text-2xl font-bold text-gray-900 min-w-[80px] text-center">
                    {tempDays}일
                  </div>
                </div>
              </div>

              {/* 보너스 계산 결과 */}
              <div className="bg-gradient-to-r from-green-50 to-emerald-50 p-6 rounded-xl mb-6">
