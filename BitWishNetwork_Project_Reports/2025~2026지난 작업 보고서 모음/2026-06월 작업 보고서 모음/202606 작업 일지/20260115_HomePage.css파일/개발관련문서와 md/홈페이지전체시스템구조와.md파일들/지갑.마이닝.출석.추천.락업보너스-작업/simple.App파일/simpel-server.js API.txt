const protectPorts = () => {

    process.env.PORT = PORT.toString();
  }
  
  // 금지된 포트 사용 시도 감지
  FORBIDDEN_PORTS.forEach(forbiddenPort => {
    if (process.env[`PORT_${forbiddenPort}`] || process.env[`SERVER_PORT_${forbiddenPort}`]) {
      console.error(`🚨 오류: 금지된 포트 ${forbiddenPort} 사용 시도가 감지되었습니다!`);
      process.exit(1);
    }
  });
  
  console.log(`🔒 포트 보호 시스템 활성화: 백엔드 ${PORT}, 프론트 ${FRONTEND_PORT}`);
  console.log(`🚫 금지된 포트: ${FORBIDDEN_PORTS.join(', ')}`);

// 🗄️ MongoDB 연결 함수 (강화된 버전)
const connectMongoDB = async (retryCount = 0) => {
  const maxRetries = 5;
  const retryDelay = 2000; // 2초
  
  try {
    console.log(`🗄️ MongoDB 연결 시도... (${retryCount + 1}/${maxRetries})`);
    
    client = new MongoClient(MONGODB_URI, {
      serverSelectionTimeoutMS: 5000,
      connectTimeoutMS: 10000,
      socketTimeoutMS: 45000,
    });
    
    await client.connect();
    db = client.db(DB_NAME);
    
    // 연결 상태 확인
    await client.db('admin').admin().ping();
    
    console.log('✅ MongoDB 연결 성공!');
    console.log(`🗄️ 데이터베이스: ${DB_NAME}`);
    console.log(`🔗 연결 URI: ${MONGODB_URI}`);
    
    // 블록체인 상태 컬렉션 초기화
    const blockchainCollection = db.collection('blockchain_status');
    const existingStatus = await blockchainCollection.findOne({});
    
    if (existingStatus) {
      // 기존 상태 복원
      blockchainStatus = existingStatus;
      console.log('🔄 기존 블록체인 상태 복원됨');
      console.log(`📊 총 블록 수: ${blockchainStatus.totalBlocks}`);
      console.log(`🔗 마지막 블록 해시: ${blockchainStatus.lastBlock?.hash || '없음'}`);
    } else {
      console.log('📝 새로운 블록체인 상태 초기화');
    }
    
    // 연결 모니터링 설정
    client.on('close', () => {
      console.log('⚠️ MongoDB 연결이 끊어졌습니다. 자동 재연결 시도...');
      setTimeout(() => connectMongoDB(), retryDelay);
    });
    
    client.on('error', (error) => {
      console.error('❌ MongoDB 연결 오류:', error.message);
      setTimeout(() => connectMongoDB(), retryDelay);
    });
    
  } catch (error) {
    console.error(`❌ MongoDB 연결 실패 (${retryCount + 1}/${maxRetries}):`, error.message);
    
    if (retryCount < maxRetries) {
      console.log(`🔄 ${retryDelay/1000}초 후 재시도...`);
      setTimeout(() => connectMongoDB(retryCount + 1), retryDelay);
    } else {
      console.log('⚠️ 최대 재시도 횟수 초과. 메모리 기반 저장으로 계속 실행...');
      console.log('💡 MongoDB 서비스가 실행 중인지 확인하세요: mongod --dbpath /data/db');
    }
  }
};

// 🔐 보안 검증 함수들
const validateAdminAccess = (walletAddress, password) => {
  // 차단된 지갑 주소 확인
  if (blockedWallets.has(walletAddress)) {
    const blockTime = blockedWallets.get(walletAddress);
    const currentTime = Date.now();
    
    if (currentTime - blockTime < BLOCK_DURATION) {
      const remainingTime = Math.ceil((BLOCK_DURATION - (currentTime - blockTime)) / (1000 * 60));
      return {
        success: false,
        error: `지갑 주소가 차단되었습니다. ${remainingTime}분 후에 다시 시도해주세요.`,
        blocked: true,
        remainingTime
      };
    } else {
      // 차단 시간이 지났으면 차단 해제
      blockedWallets.delete(walletAddress);
      loginAttempts.delete(walletAddress);
    }
  }
  
  // 지갑 주소 확인
  if (walletAddress !== ADMIN_WALLET_ADDRESS) {
    return {
      success: false,
      error: '관리자 지갑 주소가 아닙니다.',
      unauthorized: true
    };
  }
  
  // 비밀번호 해시 확인
  const inputPasswordHash = crypto.createHash('sha256').update(password).digest('hex');
  if (inputPasswordHash !== ADMIN_PASSWORD_HASH) {
    // 로그인 시도 횟수 증가
    const attempts = (loginAttempts.get(walletAddress) || 0) + 1;
    loginAttempts.set(walletAddress, attempts);
    
    if (attempts >= MAX_LOGIN_ATTEMPTS) {
      // 최대 시도 횟수 초과 시 차단
      blockedWallets.set(walletAddress, Date.now());
      loginAttempts.delete(walletAddress);
      
      return {
        success: false,
        error: `로그인 시도 횟수 초과. 지갑 주소가 1시간 동안 차단되었습니다.`,
        blocked: true,
        blockDuration: '1시간'
      };
    }
    
    return {
      success: false,
      error: `비밀번호가 올바르지 않습니다. (${attempts}/${MAX_LOGIN_ATTEMPTS})`,
      attempts,
      maxAttempts: MAX_LOGIN_ATTEMPTS
    };
  }
  
  // 인증 성공 시 로그인 시도 횟수 초기화
  loginAttempts.delete(walletAddress);
  
  return {
    success: true,
    message: '관리자 인증 성공!',
    walletAddress
  };
};

const isWalletBlocked = (walletAddress) => {
  if (blockedWallets.has(walletAddress)) {
    const blockTime = blockedWallets.get(walletAddress);
    const currentTime = Date.now();
    
    if (currentTime - blockTime < BLOCK_DURATION) {
      return true;
    } else {
      // 차단 시간이 지났으면 차단 해제
      blockedWallets.delete(walletAddress);
      loginAttempts.delete(walletAddress);
      return false;
    }
  }
  return false;
};

  // 24단어 시드문구 생성 (BIP39 표준)
  generateSeedPhrase() {
    try {
      // BIP39_WORDS 배열에서 24개 단어를 랜덤하게 선택
      const words = [];
      for (let i = 0; i < 24; i++) {
        const randomIndex = Math.floor(Math.random() * StellarWalletSystem.BIP39_WORDS.length);
        words.push(StellarWalletSystem.BIP39_WORDS[randomIndex]);
      }
      
      const seedPhrase = words.join(' ');
      
      if (words.length !== 24) {
        throw new Error('시드문구가 24단어가 아닙니다');
      }

      return {
        success: true,
        seedPhrase: seedPhrase,
        words: words,
        message: '24단어 시드문구가 성공적으로 생성되었습니다'
      };
    } catch (error) {
      return {
        success: false,
        error: error.message
      };
    }
  }

  // 💰 잔액 계산 메서드들
  calculateTotalBalance(walletId) {
    try {
      const wallet = this.wallets.get(walletId);
      if (!wallet) return 0;

      // 기본 잔액 + 채굴 노드 보상 + 출석 보너스 + 추천 보너스 + 락업 보너스
      let totalBalance = wallet.balance || 0;
      
      // 채굴 노드 보상 (마이닝 시스템에서 계산)
      const miningRewards = this.getMiningRewards(walletId);
      totalBalance += miningRewards;
      
      // 출석 보너스
      const attendanceBonus = this.getAttendanceBonus(walletId);
      totalBalance += attendanceBonus;
      
      // 추천 보너스
      const referralBonus = this.getReferralBonus(walletId);
      totalBalance += referralBonus;
      
      // 락업 보너스
      const lockupBonus = this.getLockupBonus(walletId);
      totalBalance += lockupBonus;

      return Math.round(totalBalance * 100) / 100; // 소수점 2자리까지
    } catch (error) {
      console.error('총 잔액 계산 오류:', error);
      return 0;
    }
  }

  calculateAvailableBalance(walletId) {
    try {
      const wallet = this.wallets.get(walletId);
      if (!wallet) return 0;

      const totalBalance = this.calculateTotalBalance(walletId);
      const lockedBalance = this.calculateLockedBalance(walletId);
      
      // 사용 가능한 금액 = 총 잔액 - 잠긴 금액
      const availableBalance = totalBalance - lockedBalance;
      
      return Math.max(0, Math.round(availableBalance * 100) / 100);
    } catch (error) {
      console.error('사용 가능한 잔액 계산 오류:', error);
      return 0;
    }
  }

  calculateLockedBalance(walletId) {
    try {
      const wallet = this.wallets.get(walletId);
      if (!wallet) return 0;

      let lockedBalance = 0;
      
      // 락업된 금액
      const lockupAmount = this.getLockupAmount(walletId);
      lockedBalance += lockupAmount;
      
      // 추천인 보너스 (락업 기간 동안)
      const referralLocked = this.getReferralLockedAmount(walletId);
      lockedBalance += referralLocked;

      return Math.round(lockedBalance * 100) / 100;
    } catch (error) {
      console.error('잠긴 잔액 계산 오류:', error);
      return 0;
    }
  }

  // 📜 트랜잭션 히스토리 조회
  getTransactionHistory(walletId, limit = 20, offset = 0) {
    try {
      const wallet = this.wallets.get(walletId);
      if (!wallet) return [];

      // 트랜잭션 히스토리에서 해당 지갑의 트랜잭션들 조회
      const allTransactions = Array.from(this.transactionHistory.values())
        .filter(tx => tx.fromWalletId === walletId || tx.toWalletId === walletId)
        .sort((a, b) => new Date(b.timestamp) - new Date(a.timestamp))
        .slice(offset, offset + limit);

      return allTransactions.map(tx => ({
        id: tx.id,
        type: tx.type,
        amount: tx.amount,
        from: tx.fromWalletId === walletId ? '나' : tx.fromAddress,
        to: tx.toWalletId === walletId ? '나' : tx.toAddress,
        timestamp: tx.timestamp,
        status: tx.status,
        hash: tx.hash,
        fee: tx.fee,
        message: tx.message
      }));
    } catch (error) {
      console.error('트랜잭션 히스토리 조회 오류:', error);
      return [];
    }
  }

  // 💸 BW 토큰 송금
  sendBWToken(walletId, toAddress, amount, message = '') {
    try {
      const wallet = this.wallets.get(walletId);
      if (!wallet) {
        return {
          success: false,
          error: '지갑을 찾을 수 없습니다'
        };
      }

      // 잔액 확인
      const availableBalance = this.calculateAvailableBalance(walletId);
      if (availableBalance < amount) {
        return {
          success: false,
          error: '잔액이 부족합니다'
        };
      }

      // 수수료 계산 (0.01%)
      const fee = amount * 0.0001;
      const totalAmount = amount + fee;

      if (availableBalance < totalAmount) {
        return {
          success: false,
          error: '수수료를 포함한 금액이 부족합니다'
        };
      }

      // 송금 처리
      wallet.balance -= totalAmount;
      
      // 받는 지갑 찾기
      const toWallet = Array.from(this.wallets.values())
        .find(w => w.address === toAddress);
      
      if (toWallet) {
        toWallet.balance += amount; // 수수료 제외한 금액
      }

      // 트랜잭션 기록
      const transaction = {
        id: `tx_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`,
        type: 'send',
        fromWalletId: walletId,
        toWalletId: toWallet ? toWallet.id : null,
        fromAddress: wallet.address,
        toAddress: toAddress,
        amount: amount,
        fee: fee,
        totalAmount: totalAmount,
        message: message,
        timestamp: new Date().toISOString(),
        status: 'confirmed',
        hash: `hash_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`
      };

      this.transactionHistory.set(transaction.id, transaction);

      return {
        success: true,
        message: '송금이 성공적으로 처리되었습니다',
        transactionHash: transaction.hash,
        amount: amount,
        fee: fee,
        newBalance: wallet.balance
      };
    } catch (error) {
      console.error('송금 처리 오류:', error);
      return {
        success: false,
        error: error.message
      };
    }
  }

  // 시드문구에서 스텔라 지갑 생성
  createWalletFromSeedPhrase(seedPhrase, userId) {
    try {
      // BIP39 검증 - BIP39_WORDS 배열에 있는 단어들로만 구성되었는지 확인
      const words = seedPhrase.split(' ');
      if (words.length !== 24) {
        throw new Error('시드문구는 정확히 24단어여야 합니다');
      }
      
      // 모든 단어가 BIP39_WORDS에 있는지 확인
      for (const word of words) {
        if (!StellarWalletSystem.BIP39_WORDS.includes(word)) {
          throw new Error('유효하지 않은 BIP39 단어가 포함되어 있습니다');
        }
      }

      // 시드문구에서 스텔라 키페어 생성 (올바른 방식)
      const mnemonic = seedPhrase;
      const seed = bip39.mnemonicToSeedSync(mnemonic);
      const keypair = Keypair.fromRawEd25519Seed(seed.slice(0, 32));

      const wallet = {
        id: uuidv4(),
        userId: userId,
        publicKey: keypair.publicKey(),
        secretKey: keypair.secret(), // 클라이언트에만 전달, 서버에는 저장하지 않음
        address: keypair.publicKey(), // G로 시작하는 56글자 스텔라 주소
        seedPhrase: seedPhrase,
        createdAt: new Date(),
        isFirstWallet: true,
        balance: 0,
        lockedAmount: 0,
        stakedAmount: 0,
        kycLevel: 0,
        otpEnabled: false,
        status: 'active',
        transactions: []
      };

      // 지갑 저장
      this.wallets.set(wallet.id, wallet);
      
      // 시드문구 기록 (보안을 위해 해시화하여 저장)
      const seedHash = crypto.createHash('sha256').update(seedPhrase).digest('hex');
      this.seedPhraseRecords.set(wallet.id, {
        walletId: wallet.id,
        seedHash: seedHash,
        createdAt: new Date()
      });

      return {
        success: true,
        wallet: {

        }
      };
    } catch (error) {
      return {
        success: false,
        error: error.message
      };
    }
  }

  // 랜덤 4단어 생성 (시드문구 검증용)
  generateRandomVerificationWords(seedPhrase) {
    try {
      const words = seedPhrase.split(' ');
      
      if (words.length !== 24) {
        return {
          success: false,
          error: '시드문구는 정확히 24단어여야 합니다'
        };
      }

      // 랜덤하게 4개 위치 선택 (1-24)
      const positions = [];
      while (positions.length < 4) {
        const pos = Math.floor(Math.random() * 24) + 1;
        if (!positions.includes(pos)) {
          positions.push(pos);
        }
      }

      // 선택된 위치의 단어들 반환
      const verificationWords = {};
      positions.forEach(pos => {
        verificationWords[pos] = words[pos - 1];
      });

      return {
        success: true,
        verificationWords,
        positions: positions.sort((a, b) => a - b)
      };
    } catch (error) {
      return {
        success: false,
        error: error.message
      };
    }
  }

  // 시드문구 검증 (4개 단어 확인)
  validateSeedPhraseWords(seedPhrase, verificationWords) {
    try {
      const words = seedPhrase.split(' ');
      
      if (words.length !== 24) {
        return {
          success: false,
          error: '시드문구는 정확히 24단어여야 합니다'
        };
      }

      // BIP39 검증
      if (!bip39.validateMnemonic(seedPhrase)) {
        return {
          success: false,
          error: '유효하지 않은 BIP39 시드문구입니다'
        };
      }

      const validationResults = {};
      let isValid = true;

      // 각 검증 단어 확인
      for (const [position, word] of Object.entries(verificationWords)) {
        const pos = parseInt(position) - 1; // 1-based to 0-based
        if (pos < 0 || pos >= 24) {
          validationResults[position] = {
            valid: false,
            error: '잘못된 위치입니다'
          };
          isValid = false;
          continue;
        }

        const expectedWord = words[pos];
        const isCorrect = word.toLowerCase() === expectedWord.toLowerCase();
        
        validationResults[position] = {
          valid: isCorrect,
          expected: expectedWord,
          provided: word,
          message: isCorrect ? '올바른 단어입니다' : '올바르지 않은 단어입니다'
        };

        if (!isCorrect) {
          isValid = false;
        }
      }

      return {
        success: true,
        isValid: isValid,
        validationResults: validationResults,
        message: isValid ? '모든 검증 단어가 일치합니다' : '일부 검증 단어가 일치하지 않습니다'
      };
    } catch (error) {
      return {
        success: false,
        error: error.message
      };
    }
  }

  // 지갑 잔액 조회
  getWalletBalance(walletId) {
    const wallet = this.wallets.get(walletId);
    if (!wallet) {
      return {
        success: false,
        error: '지갑을 찾을 수 없습니다'
      };
    }

    return {
      success: true,
      balance: {
        total: wallet.balance,
        available: wallet.balance - wallet.lockedAmount - wallet.stakedAmount,
        locked: wallet.lockedAmount,
        staked: wallet.stakedAmount,
        kycLevel: wallet.kycLevel,
        otpEnabled: wallet.otpEnabled
      }
    };
  }
}

// 🔐 OTP 시스템 (Google Authenticator 호환)
class OTPSystem {
  constructor() {
    this.secretKey = 'BitWishNetworkSecretKey2024';
    this.algorithm = 'sha1';
    this.digits = 6;
    this.period = 30; // 30초마다 변경
  }

  // TOTP 생성
  generateTOTP() {
    const counter = Math.floor(Date.now() / (this.period * 1000));
    const time = counter.toString(16).padStart(16, '0');
    
    // HMAC-SHA1 계산
    const hmac = crypto.createHmac(this.algorithm, this.secretKey);
    hmac.update(Buffer.from(time, 'hex'));
    const hash = hmac.digest();
    
    // 4바이트 동적 바이너리 코드 생성
    const offset = hash[hash.length - 1] & 0xf;
    const binary = ((hash[offset] & 0x7f) << 24) |
                   ((hash[offset + 1] & 0xff) << 16) |
                   ((hash[offset + 2] & 0xff) << 8) |
                   (hash[offset + 3] & 0xff);
    
    // 6자리 숫자로 변환
    const otp = binary % Math.pow(10, this.digits);
    return otp.toString().padStart(this.digits, '0');
  }

  // OTP 검증
  verifyTOTP(inputOTP, tolerance = 1) {
    const currentOTP = this.generateTOTP();
    
    if (inputOTP === currentOTP) {
      return { valid: true, message: 'OTP가 일치합니다' };
    }
    
    // 이전/다음 OTP도 허용 (시간 차이 보정)
    for (let i = 1; i <= tolerance; i++) {
      // 이전 OTP
      const prevCounter = Math.floor((Date.now() - (i * this.period * 1000)) / (this.period * 1000));
      const prevTime = prevCounter.toString(16).padStart(16, '0');
      const prevHmac = crypto.createHmac(this.algorithm, this.secretKey);
      prevHmac.update(Buffer.from(prevTime, 'hex'));
      const prevHash = prevHmac.digest();
      const prevOffset = prevHash[prevHash.length - 1] & 0xf;
      const prevBinary = ((prevHash[prevOffset] & 0x7f) << 24) |
                         ((prevHash[prevOffset + 1] & 0xff) << 16) |
                         ((prevHash[prevOffset + 2] & 0xff) << 8) |
                         (prevHash[prevOffset + 3] & 0xff);
      const prevOTP = (prevBinary % Math.pow(10, this.digits)).toString().padStart(this.digits, '0');
      
      if (inputOTP === prevOTP) {
        return { valid: true, message: 'OTP가 일치합니다 (이전 시간대)' };
      }
      
      // 다음 OTP
      const nextCounter = Math.floor((Date.now() + (i * this.period * 1000)) / (this.period * 1000));
      const nextTime = nextCounter.toString(16).padStart(16, '0');
      const nextHmac = crypto.createHmac(this.algorithm, this.secretKey);
      nextHmac.update(Buffer.from(nextTime, 'hex'));
      const nextHash = nextHmac.digest();
      const nextOffset = nextHash[nextHash.length - 1] & 0xf;
      const nextBinary = ((nextHash[nextOffset] & 0x7f) << 24) |
                         ((nextHash[nextOffset + 1] & 0xff) << 16) |
                         ((nextHash[nextOffset + 2] & 0xff) << 8) |
                         (nextHash[nextOffset + 3] & 0xff);
      const nextOTP = (nextBinary % Math.pow(10, this.digits)).toString().padStart(this.digits, '0');
      
      if (inputOTP === nextOTP) {
        return { valid: true, message: 'OTP가 일치합니다 (다음 시간대)' };
      }
    }
    
    return { valid: false, message: 'OTP가 일치하지 않습니다' };
  }

  // QR 코드 URI 생성
  generateQRURI(accountName = 'BitWishNetwork') {
    const secret = this.secretKey;
    const issuer = 'BitWishNetwork';
    const algorithm = this.algorithm;
    const digits = this.digits;
    const period = this.period;
    
    return `otpauth://totp/${encodeURIComponent(issuer)}:${encodeURIComponent(accountName)}?secret=${secret}&issuer=${encodeURIComponent(issuer)}&algorithm=${algorithm}&digits=${digits}&period=${period}`;
  }
}

// ⛏️ 마이닝 시스템
class MiningSystem {
  constructor() {
    this.difficulty = 4; // 난이도 (앞자리 0의 개수)
    this.blockReward = 10; // 블록 보상
    this.maxNonce = 1000000; // 최대 nonce 값
    this.miningSessions = new Map();
    this.globalStats = {
      totalBlocks: 0,
      totalRewards: 0,
      averageDifficulty: 4,
      totalHashRate: 0
    };
  }

  // 마이닝 세션 시작
  startMiningSession(userId, walletAddress) {
    const sessionId = uuidv4();
    const session = {
      id: sessionId,
      userId: userId,
      walletAddress: walletAddress,
      startTime: new Date(),
      status: 'mining',
      currentBlock: 0,
      totalRewards: 0,
      hashRate: 0,
      lastHashTime: new Date()
    };

    this.miningSessions.set(sessionId, session);
    
    // 마이닝 시작
    this.mineBlock(sessionId);
    
    return {
      success: true,
      sessionId: sessionId,
      message: '마이닝이 시작되었습니다'
    };
  }

  // 블록 마이닝
  mineBlock(sessionId) {
    const session = this.miningSessions.get(sessionId);
    if (!session || session.status !== 'mining') return;

    const startTime = Date.now();
    let nonce = 0;
    let hash = '';

    while (nonce < this.maxNonce) {
      const data = `${session.walletAddress}-${session.currentBlock}-${nonce}-${Date.now()}`;
      hash = this.calculateHash(data);
      
      if (this.verifyDifficulty(hash)) {
        // 블록 발견!
        const reward = this.calculateReward();
        session.totalRewards += reward;
        session.currentBlock++;
        session.hashRate = this.calculateHashRate(startTime, nonce);
        
        // 전역 통계 업데이트
        this.updateGlobalStats(reward);
        
        // 다음 블록 마이닝
        setTimeout(() => this.mineBlock(sessionId), 100);
        break;
      }
      
      nonce++;
    }
  }

  // 해시 계산
  calculateHash(data) {
    return crypto.createHash('sha256').update(data).digest('hex');
  }

  // 난이도 검증
  verifyDifficulty(hash) {
    const prefix = '0'.repeat(this.difficulty);
    return hash.startsWith(prefix);
  }

  // 보상 계산
  calculateReward() {
    return this.blockReward;
  }

  // 해시레이트 계산
  calculateHashRate(startTime, nonce) {
    const elapsedTime = (Date.now() - startTime) / 1000; // 초 단위
    if (elapsedTime === 0) return 0;
    return Math.floor(nonce / elapsedTime);
  }

  // 이전 블록 해시
  getPreviousBlockHash() {
    return crypto.createHash('sha256').update(Date.now().toString()).digest('hex');
  }

  // 전역 통계 업데이트
  updateGlobalStats(reward) {
    this.globalStats.totalBlocks++;
    this.globalStats.totalRewards += reward;
    this.globalStats.averageDifficulty = this.difficulty;
  }

  // 마이닝 세션 정지
  stopMiningSession(sessionId) {
    const session = this.miningSessions.get(sessionId);
    if (session) {
      session.status = 'stopped';
      session.endTime = new Date();
      this.miningSessions.set(sessionId, session);
      
      return {
        success: true,
        message: '마이닝이 정지되었습니다',
        session: session
      };
    }
    
    return {
      success: false,
      error: '마이닝 세션을 찾을 수 없습니다'
    };
  }

  // 🎯 지갑 주소로 마이닝 세션 찾기 및 정지
  stopMiningByWallet(walletAddress) {
    for (const [sessionId, session] of this.miningSessions) {
      if (session.walletAddress === walletAddress && session.status === 'mining') {
        session.status = 'stopped';
        session.endTime = new Date();
        this.miningSessions.set(sessionId, session);
        
        return {
          success: true,
          message: '마이닝이 정지되었습니다',
          walletAddress: walletAddress,
          sessionId: sessionId
        };
      }
    }
    
    return {
      success: false,
      error: '해당 지갑 주소의 활성 마이닝 세션을 찾을 수 없습니다'
    };
  }

  // 마이닝 세션 조회
  getMiningSession(sessionId) {
    return this.miningSessions.get(sessionId);
  }

  // 모든 마이닝 세션 조회
  getAllMiningSessions() {
    return Array.from(this.miningSessions.values());
  }

  // 전역 통계 조회
  getGlobalStats() {
    return this.globalStats;
  }
}

// 🚀 가상 더미 마이닝 시스템 (오리지널 블록체인 연결용)
class VirtualMiningSystem {
  constructor() {
    this.virtualSessions = new Map();
    this.virtualHourlyReward = 0.125; // 가상 시간당 보상
    this.virtualDailyReward = 3.0;    // 가상 일일 보상
    this.virtualBlockchain = {         // 가상 블록체인 상태
      totalBlocks: 0,
      lastBlockHash: 'virtual_hash_' + Date.now(),
      networkStatus: '가상 블록체인 연결됨',
      totalSupply: 21000000000,        // 가상 총 발행량
      circulatingSupply: 0             // 가상 유통량
    };
  }

  // 🚫 실제 블록 생성 차단 (완벽한 분리)
  startVirtualMiningSession(userId, walletAddress) {
    const sessionId = `virtual_${Date.now()}`;
    const session = {
      id: sessionId,
      userId: userId,
      walletAddress: walletAddress,
      startTime: new Date(),
      status: 'virtual_mining',
      isActive: true,
      totalRewards: 0,
      currentHourlyReward: 0,
      currentDailyReward: 0,
      lastRewardCalculation: new Date(),
      virtualBlocks: 0                 // 가상 블록 수
    };

    this.virtualSessions.set(sessionId, session);
    
    // 🎯 가상 블록체인 연결 확인
    this.connectToVirtualBlockchain();
    
    // 🧪 가상용 보상 계산 타이머 시작
    this.startVirtualRewardCalculation(sessionId);
    
    console.log(`🚀 가상 더미 마이닝 세션 시작: ${walletAddress}`);
    
    return {
      success: true,
      sessionId: sessionId,
      message: '가상 더미 마이닝이 시작되었습니다',
      hourlyReward: this.virtualHourlyReward,
      dailyReward: this.virtualDailyReward,
      blockchainStatus: this.virtualBlockchain.networkStatus
    };
  }

  // ✅ 블록체인 연결 완벽 구현
  connectToVirtualBlockchain() {
    try {
      // MongoDB 연결 상태 확인
      if (global.blockchainStatus) {
        this.virtualBlockchain.totalBlocks = global.blockchainStatus.totalBlocks || 0;
        this.virtualBlockchain.lastBlockHash = global.blockchainStatus.lastBlock?.hash || 'virtual_hash_' + Date.now();
        console.log('✅ 가상 블록체인 연결 완료');
      } else {
        console.log('⚠️ 실제 블록체인 상태 없음, 가상 상태 사용');
      }
    } catch (error) {
      console.log('⚠️ 블록체인 연결 실패, 가상 상태로 계속 진행:', error);
    }
  }

  // ✅ 가상 보상 시스템 완벽 구현
  startVirtualRewardCalculation(sessionId) {
    const session = this.virtualSessions.get(sessionId);
    if (!session) return;

    const calculateVirtualRewards = () => {
      if (session.status !== 'virtual_mining') return;

      const now = new Date();
      const elapsedHours = (now.getTime() - session.startTime.getTime()) / (1000 * 60 * 60);
      const elapsedDays = elapsedHours / 24;
      
      // 🎯 가상 보상 계산 (실제 블록체인과 동일한 수식)
      session.currentHourlyReward = elapsedHours * this.virtualHourlyReward;
      session.currentDailyReward = Math.floor(elapsedDays) * this.virtualDailyReward;
      session.totalRewards = session.currentHourlyReward + session.currentDailyReward;
      session.lastRewardCalculation = now;
      
      // 🎯 가상 블록 생성 (실제 블록은 아님)
      session.virtualBlocks = Math.floor(elapsedHours * 24); // 시간당 가상 블록
      
      this.virtualSessions.set(sessionId, session);
      
      // 🎯 가상 블록체인 상태 업데이트
      this.virtualBlockchain.totalBlocks = session.virtualBlocks;
      this.virtualBlockchain.lastBlockHash = 'virtual_hash_' + Date.now();
    };

    // 1초마다 가상 보상 계산
    const interval = setInterval(calculateVirtualRewards, 1000);
    
    // 세션에 인터벌 ID 저장 (정지 시 사용)
    session.rewardInterval = interval;
  }

  // 🛑 가상 마이닝 세션 정지
  stopVirtualMiningSession(walletAddress) {
    for (const [sessionId, session] of this.virtualSessions) {
      if (session.walletAddress === walletAddress && session.status === 'virtual_mining') {
        // 보상 계산 타이머 정지
        if (session.rewardInterval) {
          clearInterval(session.rewardInterval);
        }
        
        // 🎯 세션을 Map에서 완전히 제거
        this.virtualSessions.delete(sessionId);
        
        // ✅ 추가: virtualBlockchain 상태 초기화 (백그라운드 마이닝 완전 정지)
        this.virtualBlockchain = {
          totalBlocks: 0,
          totalTokens: '0',
          totalSupply: 21000000000,
          remainingSupply: 21000000000,
          issuancePercentage: 0,
          networkStatus: '정지됨',
          lastUpdated: new Date().toISOString(),
          recentBlocks: [],
          lastUpdate: new Date()
        };
        
        console.log(`🛑 가상 더미 마이닝 세션 정지 및 완전 삭제: ${walletAddress}`);
        console.log(`🛑 virtualBlockchain 상태 초기화 완료`);
        
        return {
          success: true,
          message: '가상 더미 마이닝이 완전히 정지되었습니다',
          walletAddress: walletAddress
        };
      }
    }
    
    return {
      success: false,
      error: '해당 지갑 주소의 활성 가상 마이닝 세션을 찾을 수 없습니다'
    };
  }

  // 🎯 가상 마이닝 상태 조회
  getVirtualMiningStatus(walletAddress) {
    for (const [sessionId, session] of this.virtualSessions) {
      if (session.walletAddress === walletAddress && session.status === 'virtual_mining') {
        return {
          success: true,
          session: session,
          blockchainStatus: this.virtualBlockchain
        };
      }
    }
    
    return {
      success: false,
      error: '해당 지갑 주소의 활성 가상 마이닝 세션을 찾을 수 없습니다'
    };
  }

  // 🎯 모든 가상 세션 정리
  clearAllVirtualSessions() {
    for (const [sessionId, session] of this.virtualSessions) {
      if (session.rewardInterval) {
        clearInterval(session.rewardInterval);
      }
    }
    this.virtualSessions.clear();
    console.log('🧹 모든 가상 마이닝 세션 정리 완료');
  }
}

      dailyReward: this.testDailyReward

  // 🧪 테스트용 보상 계산
  startTestRewardCalculation(sessionId) {
    const session = this.testSessions.get(sessionId);
    if (!session) return;

    const calculateTestRewards = () => {
      if (session.status !== 'test_mining') return;

      const now = new Date();
      const elapsedHours = (now.getTime() - session.startTime.getTime()) / (1000 * 60 * 60);
      const elapsedDays = elapsedHours / 24;
      
      // 테스트용 보상 계산
      session.currentHourlyReward = elapsedHours * this.testHourlyReward;
      session.currentDailyReward = Math.floor(elapsedDays) * this.testDailyReward;
      session.totalRewards = session.currentHourlyReward + session.currentDailyReward;
      session.lastRewardCalculation = now;
      
      this.testSessions.set(sessionId, session);
    };

    // 1초마다 테스트 보상 계산
    const interval = setInterval(calculateTestRewards, 1000);
    
    // 세션에 인터벌 ID 저장 (정지 시 사용)
    session.rewardInterval = interval;
  }

  // 🧪 테스트 마이닝 세션 정지
  stopTestMiningSession(walletAddress) {
    for (const [sessionId, session] of this.testSessions) {
      if (session.walletAddress === walletAddress && session.status === 'test_mining') {
        // 보상 계산 타이머 정지
        if (session.rewardInterval) {
          clearInterval(session.rewardInterval);
        }
        
        // 🎯 세션을 Map에서 완전히 제거 (완벽한 초기화)
        this.testSessions.delete(sessionId);
        
        console.log(`🧪 테스트 마이닝 세션 정지 및 완전 삭제: ${walletAddress}`);
        
        return {
          success: true,
          message: '테스트 마이닝이 완전히 정지되었습니다',
          walletAddress: walletAddress
        };
      }
    }
    
    return {
      success: false,
      error: '해당 지갑 주소의 활성 테스트 마이닝 세션을 찾을 수 없습니다'
    };
  }

  // 🧪 테스트 마이닝 상태 조회
  getTestMiningStatus(walletAddress) {
    for (const [sessionId, session] of this.testSessions) {

        const elapsedHours = (now.getTime() - session.startTime.getTime()) / (1000 * 60 * 60);
        const elapsedDays = elapsedHours / 24;
        
        const hourlyRewards = elapsedHours * this.testHourlyReward;
        const dailyRewards = Math.floor(elapsedDays) * this.testDailyReward;
        const totalRewards = hourlyRewards + dailyRewards;

          success: true,
          isActive: true,
          isTestMode: true, // 🧪 테스트 모드 표시
          rewards: Math.round(totalRewards * 1000) / 1000,
          hourlyRewards: Math.round(hourlyRewards * 1000) / 1000,
          dailyRewards: Math.round(dailyRewards * 1000) / 1000,
          startTime: session.startTime,
          walletAddress: walletAddress,
          message: `🧪 테스트 마이닝 중: 시간당 ${Math.round(hourlyRewards * 1000) / 1000}BW, 일일 ${Math.round(dailyRewards * 1000) / 1000}BW`
        };
      }
    }
    
    return {
      success: true,
      isActive: false,
      isTestMode: false,
      rewards: 0,
      hourlyRewards: 0,
      dailyRewards: 0,
      startTime: null,
      walletAddress: walletAddress,
      message: '테스트 마이닝이 시작되지 않았습니다'

  clearAllTestSessions() {
    for (const [sessionId, session] of this.testSessions) {
      if (session.rewardInterval) {
        clearInterval(session.rewardInterval);
      }
    }
    this.testSessions.clear();
    console.log('🧪 모든 테스트 마이닝 세션이 정리되었습니다');
  }

class GovernanceSystem {
  constructor() {
    this.proposals = new Map();
    this.votes = new Map();
    this.delegates = new Map();
    this.globalStats = {
      totalProposals: 0,
      totalVotes: 0,
      activeProposals: 0,
      totalDelegates: 0
    };
    this.governanceToken = 'BW';
  }

  // 제안 생성
  createProposal(creatorId, title, description, category, options) {
    const proposalId = uuidv4();
    const proposal = {
      id: proposalId,
      creatorId: creatorId,
      title: title,
      description: description,
      category: category,
      options: options,
      status: 'active',
      createdAt: new Date(),
      startDate: new Date(),
      endDate: new Date(Date.now() + 7 * 24 * 60 * 60 * 1000), // 7일간
      totalVotes: 0,
      results: {}
    };

    this.proposals.set(proposalId, proposal);
    this.globalStats.totalProposals++;
    this.globalStats.activeProposals++;

    return {
      success: true,
      proposalId: proposalId,
      message: '제안이 생성되었습니다'
    };
  }

  // 투표
  vote(proposalId, voterId, option, votingPower) {
    const proposal = this.proposals.get(proposalId);
    if (!proposal || proposal.status !== 'active') {
      return { success: false, error: '투표할 수 있는 제안이 아닙니다' };
    }

    if (new Date() > proposal.endDate) {
      return { success: false, error: '투표 기간이 종료되었습니다' };
    }

    const voteId = uuidv4();
    const vote = {
      id: voteId,
      proposalId: proposalId,
      voterId: voterId,
      option: option,
      votingPower: votingPower,
      createdAt: new Date()
    };

    this.votes.set(voteId, vote);
    
    // 제안 통계 업데이트
    proposal.totalVotes += votingPower;
    if (!proposal.results[option]) {
      proposal.results[option] = 0;
    }
    proposal.results[option] += votingPower;
    
    this.proposals.set(proposalId, proposal);
    this.globalStats.totalVotes++;

    return {
      success: true,
      message: '투표가 완료되었습니다'
    };
  }

  // 제안 결과 계산
  calculateProposalResults(proposalId) {
    const proposal = this.proposals.get(proposalId);
    if (!proposal) {
      return { success: false, error: '제안을 찾을 수 없습니다' };
    }

    const totalVotes = proposal.totalVotes;
    const results = {};
    
    for (const [option, votes] of Object.entries(proposal.results)) {
      results[option] = {
        votes: votes,
        percentage: totalVotes > 0 ? (votes / totalVotes * 100).toFixed(2) : 0
      };
    }

    // 승리 옵션 결정 (가장 많은 표를 받은 옵션)
    const winningOption = Object.entries(results).reduce((a, b) => 
      results[a[0]].votes > results[b[0]].votes ? a : b
    )[0];

    proposal.status = 'completed';
    proposal.finalResults = results;
    proposal.winningOption = winningOption;
    proposal.completedAt = new Date();
    
    this.proposals.set(proposalId, proposal);
    this.globalStats.activeProposals--;

    return {
      success: true,
      results: results,
      winningOption: winningOption,
      message: '제안 결과가 계산되었습니다'
    };
  }

  // 제안 조회
  getProposal(proposalId) {
    const proposal = this.proposals.get(proposalId);
    if (!proposal) {
      return { success: false, error: '제안을 찾을 수 없습니다' };
    }

    return {
      success: true,
      proposal: proposal
    };
  }

  // 모든 제안 조회
  getAllProposals() {
    return {
      success: true,
      proposals: Array.from(this.proposals.values())
    };
  }

  // 사용자 투표 조회
  getUserVotes(userId) {
    const userVotes = Array.from(this.votes.values())
      .filter(vote => vote.voterId === userId);
    
    return {
      success: true,
      votes: userVotes
    };
  }

  // 제안 투표 조회
  getProposalVotes(proposalId) {
    const proposalVotes = Array.from(this.votes.values())
      .filter(vote => vote.proposalId === proposalId);
    
    return {
      success: true,
      votes: proposalVotes
    };
  }

  // 거버넌스 통계 조회
  getGovernanceStats() {
    return {
      success: true,
      stats: this.globalStats
    };
  }

  // 카테고리별 제안 조회
  getProposalsByCategory(category) {
    const categoryProposals = Array.from(this.proposals.values())
      .filter(proposal => proposal.category === category);
    
    return {
      success: true,
      proposals: categoryProposals
    };
  }
}

    };

    if (!collection) {
      return { success: false, error: '컬렉션을 찾을 수 없습니다' };

    if (nft.owner !== fromAddress) {
      return { success: false, error: 'NFT 소유자가 아닙니다' };
    }

    const transfer = {
      id: transferId,
      nftId: nftId,
      fromAddress: fromAddress,
      toAddress: toAddress,
      transferredAt: new Date()
    };

    // 전역 통계 업데이트

    nft.isListed = true;
    nft.listedAt = new Date();
    this.nfts.set(nftId, nft);

      success: true,
      message: 'NFT가 판매 등록되었습니다'

    if (!nft) {
      return { success: false, error: 'NFT를 찾을 수 없습니다' };
    }

    if (!nft.isListed) {
      return { success: false, error: '판매 등록되지 않은 NFT입니다' };
    }

    if (nft.price !== price) {
      return { success: false, error: '가격이 일치하지 않습니다' };
    }

    // 구매 처리
    const transferId = uuidv4();
    const transfer = {
      id: transferId,
      nftId: nftId,
      fromAddress: nft.owner,
      toAddress: buyerAddress,
      price: price,
      transferredAt: new Date(),
      type: 'sale'
    };

    this.transfers.set(transferId, transfer);
    
    // NFT 소유자 변경 및 판매 상태 해제
    nft.owner = buyerAddress;
    nft.isListed = false;
    nft.price = 0;
    nft.transferredAt = new Date();
    this.nfts.set(nftId, nft);
    
    // 전역 통계 업데이트
    this.globalStats.totalTransfers++;
    this.globalStats.totalVolume += price;

    return {
      success: true,
      message: 'NFT 구매가 완료되었습니다'
    };
  }

  // NFT 조회
  getNFT(nftId) {
    const nft = this.nfts.get(nftId);
    if (!nft) {
      return { success: false, error: 'NFT를 찾을 수 없습니다' };
    }

    return {
      success: true,
      nft: nft
    };
  }

  // 컬렉션의 모든 NFT 조회

    return {
      success: true,
      nfts: walletNFTs
    };
  }

    let results = Array.from(this.nfts.values());
    
    if (query) {
      results = results.filter(nft => 
        nft.name.toLowerCase().includes(query.toLowerCase()) ||

      );
    }
    
    if (category) {
      results = results.filter(nft => {
        const collection = this.collections.get(nft.collectionId);
        return collection && collection.category === category;
      });
    }
    
    return {
      success: true,
      nfts: results
    };
  }

  // NFT 통계 조회
  getNFTStats() {
    return {
      success: true,
      stats: this.globalStats
    };
  }

class DeFiSystem {
  constructor() {

    this.loans = new Map();
    this.globalStats = {

    };

    this.createDefaultLiquidityPools();
  }

  // 기본 유동성 풀 생성
  createDefaultLiquidityPools() {
    const pools = [
      { id: 'bw-usdt', name: 'BW/USDT', tokenA: 'BW', tokenB: 'USDT', fee: 0.003 },
      { id: 'bw-eth', name: 'BW/ETH', tokenA: 'BW', tokenB: 'ETH', fee: 0.003 },
      { id: 'bw-btc', name: 'BW/BTC', tokenA: 'BW', tokenB: 'BTC', fee: 0.003 }
    ];

    pools.forEach(pool => {
      this.liquidityPools.set(pool.id, {
        ...pool,
        reserveA: 0,
        reserveB: 0,
        totalSupply: 0,
        createdAt: new Date()
      });
    });
  }

  // 유동성 풀 생성
  createLiquidityPool(tokenA, tokenB, fee) {
    const poolId = `${tokenA.toLowerCase()}-${tokenB.toLowerCase()}`;
    const pool = {
      id: poolId,
      name: `${tokenA}/${tokenB}`,
      tokenA: tokenA,
      tokenB: tokenB,
      fee: fee,
      reserveA: 0,
      reserveB: 0,
      totalSupply: 0,
      createdAt: new Date()
    };

    this.liquidityPools.set(poolId, pool);
    this.globalStats.totalLiquidity++;

    return {
      success: true,
      poolId: poolId,
      message: '유동성 풀이 생성되었습니다'
    };
  }

  // 풀 APR 계산
  calculatePoolAPR(poolId, volume24h) {
    const pool = this.liquidityPools.get(poolId);
    if (!pool) return 0;

    const dailyFee = volume24h * pool.fee;
    const annualFee = dailyFee * 365;
    const totalValue = pool.reserveA + pool.reserveB;
    
    return totalValue > 0 ? (annualFee / totalValue) * 100 : 0;
  }

  // 유동성 제공
  provideLiquidity(poolId, amountA, amountB, providerAddress) {
    const pool = this.liquidityPools.get(poolId);
    if (!pool) {
      return { success: false, error: '유동성 풀을 찾을 수 없습니다' };
    }

    // 유동성 토큰 발행
    const liquidityTokens = Math.sqrt(amountA * amountB);
    
    // 풀에 유동성 추가
    pool.reserveA += amountA;
    pool.reserveB += amountB;
    pool.totalSupply += liquidityTokens;
    
    this.liquidityPools.set(poolId, pool);
    
    return {
      success: true,
      liquidityTokens: liquidityTokens,
      message: '유동성이 제공되었습니다'
    };
  }

  // 유동성 제거
  removeLiquidity(poolId, liquidityTokens, providerAddress) {
    const pool = this.liquidityPools.get(poolId);
    if (!pool) {
      return { success: false, error: '유동성 풀을 찾을 수 없습니다' };
    }

    if (liquidityTokens > pool.totalSupply) {
      return { success: false, error: '제거할 수 있는 유동성이 부족합니다' };
    }

    // 비율에 따른 토큰 반환
    const ratio = liquidityTokens / pool.totalSupply;
    const amountA = pool.reserveA * ratio;
    const amountB = pool.reserveB * ratio;
    
    // 풀에서 유동성 제거
    pool.reserveA -= amountA;
    pool.reserveB -= amountB;
    pool.totalSupply -= liquidityTokens;
    
    this.liquidityPools.set(poolId, pool);
    
    return {
      success: true,
      amountA: amountA,
      amountB: amountB,
      message: '유동성이 제거되었습니다'
    };
  }

  // 토큰 스왑
  swap(poolId, fromToken, toToken, amount, userAddress) {
    const pool = this.liquidityPools.get(poolId);
    if (!pool) {
      return { success: false, error: '유동성 풀을 찾을 수 없습니다' };
    }

    // 스왑 수수료 계산
    const fee = amount * pool.fee;
    const amountAfterFee = amount - fee;
    
    // 스왑 후 토큰 양 계산 (상수 곱 공식)
    let toAmount;
    if (fromToken === pool.tokenA) {
      toAmount = (pool.reserveB * amountAfterFee) / (pool.reserveA + amountAfterFee);
      pool.reserveA += amount;
      pool.reserveB -= toAmount;
    } else {
      toAmount = (pool.reserveA * amountAfterFee) / (pool.reserveB + amountAfterFee);
      pool.reserveB += amount;
      pool.reserveA -= toAmount;
    }
    
    this.liquidityPools.set(poolId, pool);
    
    // 스왑 기록
    const swapId = uuidv4();
    const swap = {
      id: swapId,
      poolId: poolId,
      fromToken: fromToken,
      toToken: toToken,
      fromAmount: amount,
      toAmount: toAmount,
      fee: fee,
      userAddress: userAddress,
      timestamp: new Date()
    };
    
    this.swaps.set(swapId, swap);
    this.globalStats.totalVolume += amount;
    
    return {
      success: true,
      swapId: swapId,
      toAmount: toAmount,
      fee: fee,
      message: '스왑이 완료되었습니다'
    };
  }

  // 가격 영향 계산
  calculatePriceImpact(poolId, amount, fromToken) {
    const pool = this.liquidityPools.get(poolId);
    if (!pool) return 0;

    const reserve = fromToken === pool.tokenA ? pool.reserveA : pool.reserveB;
    return (amount / (reserve + amount)) * 100;
  }

  // 수익률 파밍 생성
  createYieldFarm(poolId, rewardToken, rewardRate, duration) {
    const farmId = uuidv4();
    const farm = {
      id: farmId,
      poolId: poolId,
      rewardToken: rewardToken,
      rewardRate: rewardRate, // 일일 보상률
      duration: duration, // 파밍 기간 (일)
      startDate: new Date(),
      endDate: new Date(Date.now() + duration * 24 * 60 * 60 * 1000),
      totalStaked: 0,
      totalRewards: 0,
      stakers: new Map(),
      status: 'active'
    };
    
    this.yieldFarms.set(farmId, farm);
    this.globalStats.totalFarms++;
    
    return {
      success: true,
      farmId: farmId,
      message: '수익률 파밍이 생성되었습니다'
    };
  }

  // 파밍에 스테이킹
  stakeInFarm(farmId, userAddress, amount) {
    const farm = this.yieldFarms.get(farmId);
    if (!farm || farm.status !== 'active') {
      return { success: false, error: '활성화된 파밍이 아닙니다' };
    }

    if (farm.stakers.has(userAddress)) {
      const existingStake = farm.stakers.get(userAddress);
      existingStake.amount += amount;
      existingStake.lastUpdate = new Date();
      farm.stakers.set(userAddress, existingStake);
    } else {
      farm.stakers.set(userAddress, {
        amount: amount,
        startDate: new Date(),
        lastUpdate: new Date(),
        totalRewards: 0
      });
    }
    
    farm.totalStaked += amount;
    this.yieldFarms.set(farmId, farm);
    
    return {
      success: true,
      message: '파밍에 스테이킹되었습니다'
    };
  }

  // 파밍 보상 수확
  harvestFarmRewards(farmId, userAddress) {
    const farm = this.yieldFarms.get(farmId);
    if (!farm) {
      return { success: false, error: '파밍을 찾을 수 없습니다' };
    }

    const staker = farm.stakers.get(userAddress);
    if (!staker) {
      return { success: false, error: '스테이커가 아닙니다' };
    }

    const now = new Date();
    const daysStaked = (now - staker.lastUpdate) / (1000 * 60 * 60 * 24);
    const reward = staker.amount * (farm.rewardRate / 100) * daysStaked;
    
    staker.totalRewards += reward;
    staker.lastUpdate = now;
    farm.stakers.set(userAddress, staker);
    farm.totalRewards += reward;
    
    this.yieldFarms.set(farmId, farm);
    
    return {
      success: true,
      reward: reward,
      message: '보상을 수확했습니다'
    };
  }

  // 대출 생성
  createLoan(borrowerAddress, collateralAmount, loanAmount, collateralToken, loanToken, interestRate, duration) {
    const loanId = uuidv4();
    const loan = {
      id: loanId,
      borrowerAddress: borrowerAddress,
      collateralAmount: collateralAmount,
      loanAmount: loanAmount,
      collateralToken: collateralToken,
      loanToken: loanToken,
      interestRate: interestRate, // 연간 이자율
      duration: duration, // 대출 기간 (일)
      startDate: new Date(),
      endDate: new Date(Date.now() + duration * 24 * 60 * 60 * 1000),
      status: 'active',
      totalInterest: 0,
      healthFactor: this.calculateHealthFactor(collateralAmount, loanAmount, interestRate)
    };
    
    this.loans.set(loanId, loan);
    this.globalStats.totalLoans++;
    
    return {
      success: true,
      loanId: loanId,
      healthFactor: loan.healthFactor,
      message: '대출이 생성되었습니다'
    };
  }

  // 건강도 계산
  calculateHealthFactor(collateralValue, loanValue, interestRate) {
    const totalDebt = loanValue * (1 + interestRate / 100);
    return collateralValue / totalDebt;
  }

  // 대출 상환
  repayLoan(loanId, amount, borrowerAddress) {
    const loan = this.loans.get(loanId);
    if (!loan || loan.borrowerAddress !== borrowerAddress) {
      return { success: false, error: '대출을 찾을 수 없거나 상환 권한이 없습니다' };
    }

    if (loan.status !== 'active') {
      return { success: false, error: '활성화된 대출이 아닙니다' };
    }

    loan.loanAmount -= amount;
    
    if (loan.loanAmount <= 0) {
      loan.status = 'repaid';
      loan.repaidAt = new Date();
    }
    
    this.loans.set(loanId, loan);
    
    return {
      success: true,
      remainingAmount: loan.loanAmount,
      message: '대출 상환이 완료되었습니다'
    };
  }

  // DeFi 통계 조회
  getDeFiStats() {
    return {
      success: true,
      stats: this.globalStats
    };
  }
}

app.post('/lockup/create', (req, res) => {
  try {
    const { walletAddress, userId, amount, lockPeriod } = req.body;
    
    if (!walletAddress || !userId || !amount || !lockPeriod) {
      return res.status(400).json({
        success: false,
        error: '지갑 주소, 사용자 ID, 금액, 락업 기간이 필요합니다'
      });
    }

    // 🎯 간단한 락업 생성 (IP 제한 없음)

    if (amount <= 0) {
      return res.status(400).json({
        success: false,
        error: '락업 금액은 0보다 커야 합니다'
      });
    }

    // 🎯 기본 락업은 잔액 차감 없이 보너스만 적용 (백서 정책)
    // 사용자 존재 확인 및 자동 생성
    let user = users.get(userId);
    if (!user) {
      // 사용자가 존재하지 않으면 자동으로 생성
      user = {
        id: userId,
        walletAddress: walletAddress,
        balance: 0,
        referral: { permanentRate: 0, signupRateApplied: false },
        createdAt: new Date()
      };
      users.set(userId, user);
    }

    // 락업 기간 검증 (백서 기반: 1, 6, 12, 36개월만 허용)
    const allowedPeriods = [1, 6, 12, 36];
    if (!allowedPeriods.includes(lockPeriod)) {
      return res.status(400).json({
        success: false,
        error: '락업 기간은 1, 6, 12, 36개월 중 하나여야 합니다'
      });
    }

    // 🎯 백서 기반 정확한 APY 값 (완벽 구현)
    const apyRates = {
      1: 2.7,     // 1개월: 2.7% APY (백서 정확값)
      6: 4.2,     // 6개월: 4.2% APY (백서 정확값)
      12: 5.7,    // 12개월: 5.7% APY (백서 정확값)
      36: 7.2     // 36개월: 7.2% APY (백서 정확값)
    };

    const apy = apyRates[lockPeriod];
    
    // 🎯 기본 락업 보너스 시스템 구현 (백서 기반)
    // 실제 토큰 락업이 아닌 채굴률 보너스로 작동
    const lockupRecord = {
      id: uuidv4(),
      walletAddress,
      userId,
      lockPeriod,           // 락업 기간 (개월)
      apy,                  // APY 값 (백서 기반)
      startDate: new Date(),
      endDate: new Date(Date.now() + lockPeriod * 30 * 24 * 60 * 60 * 1000), // 개월을 일로 변환
      status: 'active',
      type: 'bonus_system', // 🎯 보너스 시스템임을 명시
      verifiedBy: 'wallet_auth',
      createdAt: new Date()
    };

    // 🎯 락업 기록 저장 (실제 토큰 락업 없음)
    lockupRecords.set(lockupRecord.id, lockupRecord);
    
    // 🎯 사용자 락업 정보 업데이트 (잔액 차감 없음)
    if (!user.lockupInfo) {
      user.lockupInfo = {
        activeLockups: [],
        totalBonusRate: 0
      };
    }
    
    // 활성 락업 추가
    user.lockupInfo.activeLockups.push({
      id: lockupRecord.id,
      lockPeriod,
      apy,
      startDate: lockupRecord.startDate,
      endDate: lockupRecord.endDate
    });
    
    // 🎯 총 보너스율 계산 (백서 기반)
    // 기본 락업은 채굴률에 보너스를 추가하는 방식
    user.lockupInfo.totalBonusRate = apy / 100; // APY를 소수로 변환
    
    users.set(userId, user);

    // 🎯 성공 응답 (백서 기반)
    res.json({
      success: true,
      message: '기본 락업 보너스가 성공적으로 등록되었습니다',
      lockupInfo: {
        id: lockupRecord.id,
        lockPeriod,
        apy,
        bonusRate: apy / 100,
        startDate: lockupRecord.startDate,
        endDate: lockupRecord.endDate
      },
      userBalance: user.balance, // 잔액은 그대로 유지
      totalBonusRate: user.lockupInfo.totalBonusRate
    });
  } catch (error) {
    console.error('기본 락업 생성 오류:', error);
    res.status(500).json({
      success: false,
      error: '기본 락업 생성 중 오류가 발생했습니다: ' + error.message
    });
  }
});

app.post('/lockup/release', async (req, res) => {
  try {
    const { walletAddress, userId } = req.body;

    if (!walletAddress || !userId) {
      return res.status(400).json({
        success: false,
        error: '지갑 주소와 사용자 ID가 필요합니다'
      });
    }

    // 사용자 락업 정보 조회
    const user = users.get(userId);
    if (!user || !user.lockupInfo || user.lockupInfo.activeLockups.length === 0) {
      return res.status(400).json({
        success: false,
        error: '활성 락업이 없습니다'
      });
    }

    // 가장 최근 락업 정보 가져오기
    const latestLockup = user.lockupInfo.activeLockups[user.lockupInfo.activeLockups.length - 1];
    
    // 락업 해제 기록 생성
    const releaseRecord = {
      id: uuidv4(),
      walletAddress,
      userId,
      lockupId: latestLockup.id,
      releaseDate: new Date(),
      originalLockPeriod: latestLockup.lockPeriod,
      originalAPY: latestLockup.apy,
      totalAccumulatedBW: 0, // 마이닝 기간 동안 누적된 BW
      monthlyPaymentSchedule: [], // 월별 지급 스케줄
      status: 'released',
      createdAt: new Date()
    };

    // 월별 지급 스케줄 생성
    const monthlySchedule = [];
    for (let month = 1; month <= latestLockup.lockPeriod; month++) {
      monthlySchedule.push({
        month: month,
        paymentDate: new Date(Date.now() + month * 30 * 24 * 60 * 60 * 1000),
        amount: 0, // 실제 지급 시 계산
        status: 'pending', // pending, paid, overdue
        paidDate: null
      });
    }

    releaseRecord.monthlyPaymentSchedule = monthlySchedule;

    // 락업 해제 기록 저장
    lockupReleases.set(releaseRecord.id, releaseRecord);

    // 사용자 락업 정보 업데이트 (해제 상태로 변경)
    user.lockupInfo.activeLockups = user.lockupInfo.activeLockups.map(lockup => 
      lockup.id === latestLockup.id 
        ? { ...lockup, status: 'released', releaseDate: new Date() }
        : lockup
    );

    users.set(userId, user);

    res.json({
      success: true,
      message: '락업이 성공적으로 해제되었습니다',
      releaseInfo: {
        id: releaseRecord.id,
        releaseDate: releaseRecord.releaseDate,
        monthlySchedule: monthlySchedule.length,
        nextPaymentDate: monthlySchedule[0].paymentDate
      }
    });

  } catch (error) {
    console.error('락업 해제 오류:', error);
    res.status(500).json({
      success: false,
      error: '락업 해제 중 오류가 발생했습니다: ' + error.message
    });
  }
});

app.post('/lockup/claim-monthly/:walletAddress', async (req, res) => {
  try {
    const { walletAddress } = req.params;
    const { month } = req.body;

    if (!month) {
      return res.status(400).json({
        success: false,
        error: '지급할 월을 지정해주세요'
      });
    }

    // 락업 해제 기록 조회
    const releaseRecord = Array.from(lockupReleases.values())
      .find(release => release.walletAddress === walletAddress && release.status === 'released');

    if (!releaseRecord) {
      return res.status(400).json({
        success: false,
        error: '락업 해제 기록이 없습니다'
      });
    }

    // 해당 월의 지급 스케줄 조회
    const monthlySchedule = releaseRecord.monthlyPaymentSchedule
      .find(schedule => schedule.month === month);

    if (!monthlySchedule) {
      return res.status(400).json({
        success: false,
        error: '해당 월의 지급 스케줄이 없습니다'
      });
    }

    if (monthlySchedule.status !== 'pending') {
      return res.status(400).json({
        success: false,
        error: '이미 지급된 월입니다'
      });
    }

    // 지급 금액 계산
    const paymentAmount = calculateMonthlyPaymentAmount(releaseRecord, month);

    if (paymentAmount <= 0) {
      return res.status(400).json({
        success: false,
        error: '지급할 금액이 없습니다'
      });
    }

    // 사용자 잔액 업데이트
    const user = users.get(releaseRecord.userId);
    if (user) {
      user.balance = (user.balance || 0) + paymentAmount;
      users.set(releaseRecord.userId, user);
    }

    // 월별 지급 상태 업데이트
    monthlySchedule.status = 'paid';
    monthlySchedule.paidDate = new Date();
    monthlySchedule.amount = paymentAmount;

    // 락업 해제 기록 업데이트
    lockupReleases.set(releaseRecord.id, releaseRecord);

    res.json({
      success: true,
      message: `${month}개월차 지급이 완료되었습니다`,
      paymentInfo: {
        month: month,
        amount: paymentAmount,
        paidDate: monthlySchedule.paidDate,
        remainingPayments: releaseRecord.monthlyPaymentSchedule
          .filter(s => s.status === 'pending').length
      }
    });

  } catch (error) {
    console.error('월별 지급 실행 오류:', error);
    res.status(500).json({
      success: false,
      error: '월별 지급 실행 중 오류가 발생했습니다: ' + error.message
    });
  }
});

// 🎯 월별 지급 금액 계산 함수
function calculateMonthlyPaymentAmount(releaseRecord, month) {
  try {
    // 락업 기간 동안 누적된 총 BW 계산
    const totalAccumulatedBW = calculateTotalAccumulatedBW(releaseRecord);
    
    // 월별 지급 금액 = 총 누적 BW / 락업 기간
    const monthlyAmount = totalAccumulatedBW / releaseRecord.originalLockPeriod;
    
    return monthlyAmount;
  } catch (error) {
    console.error('월별 지급 금액 계산 오류:', error);
    return 0;
  }
}

// 🎯 총 누적 BW 계산 함수
function calculateTotalAccumulatedBW(releaseRecord) {
  try {
    // 락업 기간 동안의 총 마이닝 시간 계산
    const lockupDurationMs = releaseRecord.originalLockPeriod * 30 * 24 * 60 * 60 * 1000;
    const lockupDurationHours = lockupDurationMs / (1000 * 60 * 60);
    
    // 기본 채굴률
    const baseRate = 0.25;
    
    // 락업 보너스율 계산
    const lockupBonusRate = releaseRecord.originalAPY / 100;
    
    // 시간당 락업 보너스
    const hourlyLockupBonus = baseRate * lockupBonusRate;
    
    // 총 누적 BW = 시간당 락업 보너스 × 락업 기간(시간)
    const totalAccumulatedBW = hourlyLockupBonus * lockupDurationHours;
    
    return totalAccumulatedBW;
  } catch (error) {
    console.error('총 누적 BW 계산 오류:', error);
    return 0;
  }
}

app.post('/lockup/calculate', (req, res) => {
  try {
    const { amount, lockPeriod } = req.body;
    
    if (!amount || !lockPeriod) {
      return res.status(400).json({
        success: false,
        error: '금액과 락업 기간이 필요합니다'
      });
    }

    if (amount <= 0) {
      return res.status(400).json({
        success: false,
        error: '락업 금액은 0보다 커야 합니다'
      });
    }

    // 락업 보너스 계산 (기간에 따른 APY) - 새로운 정책
    const apyRates = {
      30: 1.2,    // 30일: 1.2% APY (새로운 정책)
      90: 2.1,    // 90일: 2.1% APY (새로운 정책)
      180: 3,     // 180일: 3% APY (새로운 정책)
      365: 4.2    // 365일: 4.2% APY (새로운 정책)
    };

    const apy = apyRates[lockPeriod] || 1.5; // 기본 1.5% APY (새로운 정책)
    const dailyReward = (amount * apy) / 100 / 365;
    const monthlyReward = dailyReward * 30;
    const totalReward = (amount * apy) / 100;
    const endDate = new Date(Date.now() + lockPeriod * 24 * 60 * 60 * 1000);

    res.json({
      success: true,
      calculation: {
        amount,
        lockPeriod,
        apy,
        dailyReward: parseFloat(dailyReward.toFixed(6)),
        monthlyReward: parseFloat(monthlyReward.toFixed(6)),
        totalReward: parseFloat(totalReward.toFixed(6)),
        endDate,
        lockPeriodDays: lockPeriod
      }
    });
  } catch (error) {
    res.status(500).json({
      success: false,
      error: error.message
    });
  }
});

// 🔒 백서 기반 락업 해제 실행 API
app.post('/lockup/release', (req, res) => {
  try {
    const { walletAddress, lockupId, releaseType } = req.body;
    
    if (!walletAddress || !lockupId || !releaseType) {
      return res.status(400).json({
        success: false,
        error: '지갑 주소, 락업 ID, 해제 유형이 필요합니다'
      });
    }

    const lockup = lockupRecords.get(lockupId);
    if (!lockup || lockup.walletAddress !== walletAddress) {
      return res.status(404).json({
        success: false,
        error: '락업을 찾을 수 없습니다'
      });
    }

    if (lockup.status !== 'active') {
      return res.status(400).json({
        success: false,
        error: '이미 해제된 락업입니다'
      });
    }

    const now = new Date();
    const daysRemaining = Math.ceil((lockup.endDate - now) / (1000 * 60 * 60 * 24));
    
    let releaseAmount = lockup.amount;
    let penalty = 0;
    let message = '';

    if (releaseType === 'early' && daysRemaining > 0) {
      // 조기 해제: 5% 수수료
      penalty = lockup.amount * 0.05;
      releaseAmount = lockup.amount - penalty;
      message = '조기 해제로 5% 수수료가 적용되었습니다';
    } else if (releaseType === 'normal') {
      // 정상 해제: 수수료 없음
      message = '정상 해제되었습니다';
    }

    // 사용자 잔액에 해제 금액 추가
    const user = users.get(lockup.userId);
    if (user) {
      user.balance = (user.balance || 0) + releaseAmount;
      user.lockedAmount = (user.lockedAmount || 0) - lockup.amount;
      users.set(lockup.userId, user);
    }

    // 락업 상태 업데이트
    lockup.status = 'completed';
    lockup.releaseDate = now;
    lockup.releaseAmount = releaseAmount;
    lockup.penalty = penalty;
    lockupRecords.set(lockupId, lockup);

    res.json({
      success: true,
      message,
      releaseDetails: {
        originalAmount: lockup.amount,
        releaseAmount,
        penalty,
        totalRewards: lockup.totalRewards,
        newBalance: user ? user.balance : 0
      }
    });
  } catch (error) {
    res.status(500).json({
      success: false,
      error: error.message
    });
  }
});

      return res.status(400).json({

        error: '지갑 주소, 사용자 ID, IP 주소, 시드문구가 필요합니다'
      });
    }

    // 시드문구 검증 (4단어 검증)
    const validationResult = stellarWalletSystem.validateSeedPhraseWords(seedPhrase);
    if (!validationResult.success) {
      return res.status(400).json({
        success: false,
        error: '시드문구 검증에 실패했습니다'
      });
    }

    // 기존 IP 접근 기록 확인
    const existingAccess = Array.from(ipAccess.values())
      .filter(access => access.walletAddress === walletAddress);

    // 최대 2개 IP만 허용
    if (existingAccess.length >= 2) {
      return res.status(400).json({
        success: false,
        error: '최대 2개 IP만 등록 가능합니다. 기존 IP를 삭제 후 다시 시도하세요'
      });
    }

    // 동일 IP 중복 등록 방지
    const duplicateIP = existingAccess.find(access => access.ipAddress === ipAddress);
    if (duplicateIP) {
      return res.status(400).json({
        success: false,
        error: '이미 등록된 IP 주소입니다'
      });
    }

    const accessRecord = {
      id: uuidv4(),
      walletAddress,
      userId,
      ipAddress,
      deviceInfo: deviceInfo || {},
      firstAccess: new Date(),
      lastAccess: new Date(),
      accessCount: 1,
      status: 'active',
      verifiedAt: new Date()
    };

    ipAccess.set(accessRecord.id, accessRecord);

    res.json({
      success: true,
      message: 'IP 접근이 시드문구 인증으로 등록되었습니다',
      accessRecord,
      totalRegisteredIPs: existingAccess.length + 1
    });
  } catch (error) {
    res.status(500).json({
      success: false,
      error: error.message
    });
  }
});

// 💸 트랜잭션 API
app.post('/wallet/transaction', (req, res) => {
  try {
    const { fromAddress, toAddress, amount, userId, transactionType } = req.body;
    
    if (!fromAddress || !toAddress || !amount || !userId) {
      return res.status(400).json({
        success: false,
        error: '송금자, 수신자, 금액, 사용자 ID가 필요합니다'
      });
    }

    if (amount <= 0) {
      return res.status(400).json({
        success: false,
        error: '송금 금액은 0보다 커야 합니다'
      });
    }

    // 송금자 잔액 확인
    const sender = Array.from(users.values())
      .find(user => user.walletAddress === fromAddress);
    
    if (!sender || (sender.balance || 0) < amount) {
      return res.status(400).json({
        success: false,
        error: '잔액이 부족합니다'
      });
    }

    // 수신자 확인
    const receiver = Array.from(users.values())
      .find(user => user.walletAddress === toAddress);
    
    if (!receiver) {
      return res.status(400).json({
        success: false,
        error: '수신자를 찾을 수 없습니다'
      });
    }

    // 트랜잭션 생성
    const transaction = {
      id: uuidv4(),
      fromAddress,
      toAddress,
      amount,
      userId,
      transactionType: transactionType || 'transfer',
      status: 'pending',
      createdAt: new Date(),
      fee: 0.001, // 0.1% 수수료
      totalAmount: amount + 0.001
    };

    // 송금자 잔액 차감
    sender.balance -= transaction.totalAmount;
    users.set(sender.id, sender);
    
    // 수신자 잔액 증가
    receiver.balance += amount;
    users.set(receiver.id, receiver);

    // 트랜잭션 상태를 완료로 변경
    transaction.status = 'completed';
    transaction.completedAt = new Date();
    
    // 트랜잭션 기록 저장
    transactionHistory.set(transaction.id, transaction);

    res.json({
      success: true,
      message: '트랜잭션이 완료되었습니다',
      transaction,
      newBalance: sender.balance
    });
  } catch (error) {
    res.status(500).json({
      success: false,
      error: error.message
    });
  }
});

        virtualResult.blockCreationBlocked = true; // 블록 생성 차단 상태 표시

      database: DB_NAME,
      uri: MONGODB_URI,
      collections: [],
      lastUpdated: new Date().toISOString()
    };
    
    if (db) {

        // 컬렉션 목록 가져오기
        const collections = await db.listCollections().toArray();
        mongoStatus.collections = collections.map(col => col.name);
        
        // 블록체인 상태 컬렉션 확인
        const blockchainCollection = db.collection('blockchain_status');

          lastBlockHash: blockchainDoc.lastBlock?.hash || null

        mongoStatus.error = error.message;
      }

      success: true,
      mongoStatus: mongoStatus,
      message: 'MongoDB 상태 조회 성공'
    });

    res.status(500).json({
      success: false,

// 🎯 새로운 토큰 이코노미 계산 API
app.post('/tokenomics/calculate', (req, res) => {
  try {
    const { miningHours, attendanceDays, referralCount, lockupPercentage, lockupPeriod } = req.body;
    
    if (!miningHours) {
      return res.status(400).json({
        success: false,
        error: '채굴 시간이 필요합니다'
      });
    }

    // 기본 채굴률 (새로운 정책)
    const baseMiningRate = 0.125; // 시간당 0.125 BW
    let totalMining = baseMiningRate * miningHours;

    // 출석 보너스 계산 (70% 감소 적용)
    let attendanceBonus = 0;
    if (attendanceDays >= 29) attendanceBonus = 0.06;      // 6%
    else if (attendanceDays >= 22) attendanceBonus = 0.03; // 3%
    else if (attendanceDays >= 15) attendanceBonus = 0.021; // 2.1%
    else if (attendanceDays >= 8) attendanceBonus = 0.015;  // 1.5%
    else if (attendanceDays >= 1) attendanceBonus = 0.009;  // 0.9%

    // 추천인 보너스 계산 (70% 감소 적용)
    let referralBonus = 0;
    if (referralCount <= 30) referralBonus = 0.012;        // 1.2%
    else if (referralCount <= 60) referralBonus = 0.009;   // 0.9%
    else if (referralCount <= 90) referralBonus = 0.006;   // 0.6%
    else referralBonus = 0.00375;                          // 0.375%

    const totalReferralBonus = referralBonus * referralCount;

    // 락업 보너스 계산 (새로운 정책)
    let lockupBonus = 0;
    if (lockupPercentage && lockupPeriod) {
      if (lockupPeriod >= 365) lockupBonus = 0.042;        // 4.2%
      else if (lockupPeriod >= 180) lockupBonus = 0.03;    // 3%
      else if (lockupPeriod >= 90) lockupBonus = 0.021;    // 2.1%
      else if (lockupPeriod >= 30) lockupBonus = 0.012;    // 1.2%
      else lockupBonus = 0.015;                            // 기본 1.5%
    }

    // 최종 채굴률 계산
    const totalBonus = attendanceBonus + totalReferralBonus + lockupBonus;
    const finalMiningRate = baseMiningRate * (1 + totalBonus);
    const totalReward = finalMiningRate * miningHours;

    // 반감기 적용 (2026년 4월 1일부터)
    const halvingStartDate = new Date('2026-04-01T00:00:00+09:00');
    const now = new Date();
    let halvingMultiplier = 1;

    if (now >= halvingStartDate) {
      const monthsSinceStart = Math.floor((now.getFullYear() - 2026) * 12 + (now.getMonth() - 3));
      if (monthsSinceStart > 0) {
        halvingMultiplier = Math.pow(0.95, monthsSinceStart); // 매월 5% 감소
      }
    }

    const finalRewardWithHalving = totalReward * halvingMultiplier;

    res.json({
      success: true,
      calculation: {
        baseMiningRate,
        miningHours,
        totalMining,
        bonuses: {
          attendance: {
            days: attendanceDays,
            rate: attendanceBonus,
            bonus: totalMining * attendanceBonus
          },
          referral: {
            count: referralCount,
            rate: referralBonus,
            totalRate: totalReferralBonus,
            bonus: totalMining * totalReferralBonus
          },
          lockup: {
            percentage: lockupPercentage,
            period: lockupPeriod,
            rate: lockupBonus,
            bonus: totalMining * lockupBonus
          }
        },
        totalBonus: totalBonus,
        finalMiningRate,
        totalReward,
        halving: {
          applied: now >= halvingStartDate,
          multiplier: halvingMultiplier,
          finalReward: finalRewardWithHalving
        },
        summary: {
          baseReward: totalMining,
          totalBonus: totalMining * totalBonus,
          finalReward: finalRewardWithHalving,
          efficiency: (finalRewardWithHalving / totalMining).toFixed(2) + 'x'
        }
      }
    });
  } catch (error) {
    res.status(500).json({
      success: false,
      error: error.message
    });
  }
});

  }

        const accumulatedDailyReward = Math.round((elapsedDays * 3.0) * 1000) / 1000;
        
        // 🎯 가상 추천 보관 BW 실시간 누적
        const referralCount = 3; // 가상 추천인 수
        const referralBonus = 0.375; // 가상 추천 보너스 %
        const realTimeReferralStorageBW = Math.round((referralCount * referralBonus * elapsedHours) * 1000) / 1000;
        
        res.json({
          success: true,
          walletAddress: walletAddress,
          realTimeRewards: {
            hourlyReward: accumulatedHourlyReward,
            dailyReward: accumulatedDailyReward,
            referralStorageBW: realTimeReferralStorageBW,
            elapsedHours: Math.round(elapsedHours * 1000) / 1000,
            elapsedDays: Math.round(elapsedDays * 1000) / 1000
          },
          sessionInfo: {
            startTime: session.session.startTime,
            status: session.session.status,
            isActive: session.session.isActive,
            virtualBlocks: session.session.virtualBlocks
          },
          blockchainStatus: session.blockchainStatus
        });
      } else {
        res.status(404).json({
          success: false,
          error: '활성 가상 더미 마이닝 세션을 찾을 수 없습니다'

// 🎯 추가 락업 해제 API
app.post('/second-lockup/release', async (req, res) => {
  try {
    const { walletAddress, userId } = req.body;
    
    if (!walletAddress || !userId) {
      return res.status(400).json({
        success: false,
        error: '지갑 주소와 사용자 ID가 필요합니다'
      });
    }

    // 사용자 추가 락업 정보 조회
    const user = users.get(userId);
    if (!user || !user.secondLockupInfo || user.secondLockupInfo.activeLockups.length === 0) {
      return res.status(400).json({
        success: false,
        error: '활성 추가 락업이 없습니다'
      });
    }

    // 가장 최근 추가 락업 정보 가져오기
    const latestSecondLockup = user.secondLockupInfo.activeLockups[user.secondLockupInfo.activeLockups.length - 1];

    // 추가 락업 해제 기록 생성
    const releaseRecord = {
      id: uuidv4(),
      walletAddress,
      userId,
      lockupId: latestSecondLockup.id,
      releaseDate: new Date(),
      originalLockPeriod: latestSecondLockup.months,
      originalAPY: latestSecondLockup.apy,
      originalRatio: latestSecondLockup.ratio,
      totalAccumulatedBW: 0,
      monthlyPaymentSchedule: [],
      status: 'released',
      createdAt: new Date()
    };

    // 월별 지급 스케줄 생성
    const monthlySchedule = [];
    for (let month = 1; month <= latestSecondLockup.months; month++) {
      monthlySchedule.push({
        month: month,
        paymentDate: new Date(Date.now() + month * 30 * 24 * 60 * 60 * 1000),
        amount: 0,
        status: 'pending',
        paidDate: null
      });
    }
    releaseRecord.monthlyPaymentSchedule = monthlySchedule;

    // 추가 락업 해제 기록 저장
    secondLockupReleases.set(releaseRecord.id, releaseRecord);

    // 사용자 추가 락업 정보 업데이트 (해제 상태로 변경)
    user.secondLockupInfo.activeLockups = user.secondLockupInfo.activeLockups.map(lockup => 
      lockup.id === latestSecondLockup.id 
        ? { ...lockup, status: 'released', releaseDate: new Date() }
        : lockup
    );
    users.set(userId, user);

    res.json({
      success: true,
      message: '추가 락업이 성공적으로 해제되었습니다',
      releaseInfo: {
        id: releaseRecord.id,
        releaseDate: releaseRecord.releaseDate,
        monthlySchedule: monthlySchedule.length,
        nextPaymentDate: monthlySchedule[0].paymentDate
      }
    });

  } catch (error) {
    console.error('추가 락업 해제 오류:', error);
    res.status(500).json({
      success: false,

app.get('/second-lockup/monthly-payment/:walletAddress', (req, res) => {
  try {
    const { walletAddress } = req.params;
    
    // 2차 락업 해제 기록 조회
    const releaseRecord = Array.from(secondLockupReleases.values())
      .find(release => release.walletAddress === walletAddress && release.status === 'released');
    
    if (!releaseRecord) {
      return res.json({
        success: true,
        hasRelease: false,
        message: '추가 락업 해제 기록이 없습니다'
      });
    }

    const now = new Date();
    const availablePayments = releaseRecord.monthlyPaymentSchedule
      .filter(schedule => schedule.paymentDate <= now && schedule.status === 'pending')
      .map(schedule => ({
        month: schedule.month,
        paymentDate: schedule.paymentDate,

        releaseDate: releaseRecord.releaseDate,
        totalMonths: releaseRecord.monthlyPaymentSchedule.length,
        availablePayments: availablePayments.length,
        nextPaymentDate: releaseRecord.monthlyPaymentSchedule
          .find(s => s.status === 'pending')?.paymentDate
      },
      monthlyPayments: availablePayments
    });

  } catch (error) {
    console.error('추가 락업 월별 지급 조회 오류:', error);
    res.status(500).json({
      success: false,

});

// 🎯 추가 락업 월별 지급 실행 API
app.post('/second-lockup/claim-monthly/:walletAddress', async (req, res) => {
  try {
    const { walletAddress } = req.params;
    const { month } = req.body;
    
    if (!month) {
      return res.status(400).json({
        success: false,
        error: '지급할 월을 지정해주세요'
      });
    }

    // 2차 락업 해제 기록 조회
    const releaseRecord = Array.from(secondLockupReleases.values())
      .find(release => release.walletAddress === walletAddress && release.status === 'released');
    
    if (!releaseRecord) {
      return res.status(400).json({
        success: false,
        error: '추가 락업 해제 기록이 없습니다'
      });
    }

    // 해당 월의 지급 스케줄 조회
    const monthlySchedule = releaseRecord.monthlyPaymentSchedule
      .find(schedule => schedule.month === month);
    
    if (!monthlySchedule) {
      return res.status(400).json({
        success: false,
        error: '해당 월의 지급 스케줄이 없습니다'
      });
    }

    if (monthlySchedule.status !== 'pending') {
      return res.status(400).json({
        success: false,
        error: '이미 지급된 월입니다'
      });
    }

    // 지급 금액 계산
    const paymentAmount = calculateSecondLockupMonthlyPaymentAmount(releaseRecord, month);
    if (paymentAmount <= 0) {
      return res.status(400).json({
        success: false,
        error: '지급할 금액이 없습니다'
      });
    }

    // 사용자 잔액 업데이트
    const user = users.get(releaseRecord.userId);
    if (user) {
      user.balance = (user.balance || 0) + paymentAmount;
      users.set(releaseRecord.userId, user);
    }

    // 월별 지급 상태 업데이트
    monthlySchedule.status = 'paid';
    monthlySchedule.paidDate = new Date();
    monthlySchedule.amount = paymentAmount;

    // 추가 락업 해제 기록 업데이트
    secondLockupReleases.set(releaseRecord.id, releaseRecord);

    res.json({
      success: true,
      message: `${month}개월차 추가 락업 지급이 완료되었습니다`,
      paymentInfo: {
        month: month,
        amount: paymentAmount,
        paidDate: monthlySchedule.paidDate,
        remainingPayments: releaseRecord.monthlyPaymentSchedule
          .filter(s => s.status === 'pending').length
      }
    });

  } catch (error) {
    console.error('추가 락업 월별 지급 실행 오류:', error);
    res.status(500).json({
      success: false,
      error: '추가 락업 월별 지급 실행 중 오류가 발생했습니다: ' + error.message
    });
  }
});

// 🎯 추가 락업 월별 지급 금액 계산 함수
function calculateSecondLockupMonthlyPaymentAmount(releaseRecord, month) {
  try {
    // 추가 락업 기간 동안 누적된 총 BW 계산
    const totalAccumulatedBW = calculateSecondLockupTotalAccumulatedBW(releaseRecord);
    
    // 월별 지급 금액 = 총 누적 BW / 락업 기간
    const monthlyAmount = totalAccumulatedBW / releaseRecord.originalLockPeriod;
    
    return monthlyAmount;
  } catch (error) {
    console.error('추가 락업 월별 지급 금액 계산 오류:', error);
    return 0;
  }
}

// 🎯 추가 락업 총 누적 BW 계산 함수
function calculateSecondLockupTotalAccumulatedBW(releaseRecord) {
  try {
    // 추가 락업 기간 동안의 총 마이닝 시간 계산
    const lockupDurationMs = releaseRecord.originalLockPeriod * 30 * 24 * 60 * 60 * 1000;
    const lockupDurationHours = lockupDurationMs / (1000 * 60 * 60);
    
    // 기본 채굴률 (실제 총 노드 보상 계산)
    const baseRate = 0.25; // 기본 채굴률 0.25 BW/시간
    
    // 기본 락업에서 남은 비율 계산 (예시: 기본 락업 20% 설정 시)
    const firstLockupRatio = 0.2; // 실제로는 사용자의 기본 락업 비율을 가져와야 함
    const remainingRatio = 1 - firstLockupRatio; // 80%
    
    // 추가 락업 비율 적용
    const secondLockupRatio = releaseRecord.originalRatio / 100;
    
    // 추가 락업 보너스율 계산
    const secondLockupBonusRate = releaseRecord.originalAPY / 100;
    
    // 시간당 추가 락업 보너스
    const hourlySecondLockupBonus = baseRate * remainingRatio * secondLockupRatio * secondLockupBonusRate;
    
    // 총 누적 BW = 시간당 추가 락업 보너스 × 락업 기간(시간)
    const totalAccumulatedBW = hourlySecondLockupBonus * lockupDurationHours;
    
    return totalAccumulatedBW;
  } catch (error) {
    console.error('추가 락업 총 누적 BW 계산 오류:', error);
    return 0;
  }
}
