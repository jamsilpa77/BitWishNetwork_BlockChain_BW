  try {
    // 1. 스텔라 블록체인에서 지갑 존재 확인
    const stellarAccount = await stellarServer.loadAccount(walletAddress);
    
    // 2. BitWish 토큰 잔액 확인

      balance.asset_code === 'BW'
    );
    
    // 3. 트랜잭션 히스토리 확인
    const transactions = await stellarServer.transactions()
      .forAccount(walletAddress)
      .call();
    
    return {
      exists: true,
      balance: bwBalance ? bwBalance.balance : '0',
      transactionCount: transactions.records.length,
      blockchainVerified: true
    };
    
  } catch (error) {
    return {
      exists: false,
      error: error.message,
      blockchainVerified: false
    };
  }
};
