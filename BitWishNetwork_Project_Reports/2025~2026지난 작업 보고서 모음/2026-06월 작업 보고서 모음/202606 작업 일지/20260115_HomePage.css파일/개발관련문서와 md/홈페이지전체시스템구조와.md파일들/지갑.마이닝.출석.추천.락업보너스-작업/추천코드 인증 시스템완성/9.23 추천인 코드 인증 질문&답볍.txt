```typescript
// ==================== startWalletCreation 함수 수정 ====================
const startWalletCreation = async () => {
  try {
    // 🆕 추천인 코드가 입력된 경우 자동 검증
    if (referralCodeInput && referralCodeInput.trim().length > 0) {
      console.log('🔍 추천인 코드 자동 검증 시작:', referralCodeInput);
      
      // 추천인 코드 검증
      const validationResult = await validateReferralCode(referralCodeInput);
      
      if (!validationResult.isValid) {
        console.log('❌ 추천인 코드 검증 실패');
        alert(`❌ 추천인 코드 오류: ${validationResult.errorMessage}`);
        return;
      }
      
      // 추천인 코드 상태에 저장
      setUserReferralSystemState(prev => ({
        ...prev,
        referralCode: referralCodeInput,
        referralBonusApplied: true
      }));
      
      console.log('✅ 추천인 코드 검증 성공');
    }
    
    setWalletStep(2);
    
    // 백엔드에서 시드문구 생성
    const response = await fetch('http://localhost:4001/wallet/generate-seedphrase', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({})
    });
    
    if (!response.ok) {
      throw new Error('시드문구 생성 실패');
    }
    
    const data = await response.json();
    setGeneratedSeedPhrase(data.seedPhrase);
    
    console.log('✅ 지갑 생성 시작 완료');
  } catch (error) {
    console.error('❌ 지갑 생성 시작 실패:', error);
    alert('지갑 생성에 실패했습니다. 다시 시도해주세요.');
  }
};
```

```typescript
const validateReferralCode = async (code: string): Promise<{
  isValid: boolean;
  errorMessage?: string;
}> => {
  try {
    // REF로 시작하고 11자리인지 확인
    if (!code.startsWith('REF') || code.length !== 11) {
      return {
        isValid: false,
        errorMessage: '추천인 코드 형식이 올바르지 않습니다. (REF + 8자리)'
      };
    }
    
    // 알파벳과 숫자만 포함하는지 확인
    const codePart = code.substring(3);
    const isValidFormat = /^[A-Z0-9]{8}$/.test(codePart);
    
    if (!isValidFormat) {
      return {
        isValid: false,
        errorMessage: '추천인 코드는 REF + 8자리 영문/숫자 조합이어야 합니다.'

      return {
        isValid: false,
        errorMessage: '존재하지 않는 추천인 코드입니다.'
      };
    }
    
    return {
      isValid: true
    };
  } catch (error) {
    console.error('❌ 추천인 코드 검증 실패:', error);
    return {
      isValid: false,

    };
  }
```

### **1. 영구 저장소 기반 추천코드 관리 시스템**

```typescript
// ==================== 영구 저장소 기반 추천코드 관리 시스템 ====================

// 🎯 localStorage 기반 추천코드 저장소
const REFERRAL_CODES_STORAGE_KEY = 'bitwish_referral_codes';

//    추천코드 저장소 인터페이스
interface ReferralCodeData {
  code: string;
  walletAddress: string;
  createdAt: number;
  isActive: boolean;
}

// 🎯 추천코드 저장소 관리 클래스
class ReferralCodeManager {
  // 추천코드 저장
  static saveReferralCode(walletAddress: string, code: string): void {
    try {
      const existingCodes = this.getAllReferralCodes();
      const newCodeData: ReferralCodeData = {
        code,
        walletAddress,
        createdAt: Date.now(),
        isActive: true
      };
      
      existingCodes[code] = newCodeData;
      localStorage.setItem(REFERRAL_CODES_STORAGE_KEY, JSON.stringify(existingCodes));
      
      console.log('✅ 추천코드 저장 완료:', { code, walletAddress });
    } catch (error) {
      console.error('❌ 추천코드 저장 실패:', error);
    }
  }

  // 모든 추천코드 조회
  static getAllReferralCodes(): Record<string, ReferralCodeData> {
    try {
      const stored = localStorage.getItem(REFERRAL_CODES_STORAGE_KEY);
      return stored ? JSON.parse(stored) : {};
    } catch (error) {
      console.error('❌ 추천코드 조회 실패:', error);
      return {};
    }
  }

  // 특정 추천코드로 지갑 주소 조회
  static getWalletAddressByCode(code: string): string | null {
    try {
      const allCodes = this.getAllReferralCodes();
      const codeData = allCodes[code];
      
      if (codeData && codeData.isActive) {
        return codeData.walletAddress;
      }
      
      return null;
    } catch (error) {
      console.error('❌ 지갑 주소 조회 실패:', error);
      return null;
    }
  }

  // 지갑 주소로 추천코드 조회
  static getCodeByWalletAddress(walletAddress: string): string | null {
    try {
      const allCodes = this.getAllReferralCodes();
      
      for (const [code, data] of Object.entries(allCodes)) {
        if (data.walletAddress === walletAddress && data.isActive) {
          return code;
        }
      }
      
      return null;
    } catch (error) {
      console.error('❌ 추천코드 조회 실패:', error);
      return null;
    }
  }

  // 추천코드 존재 여부 확인
  static isCodeExists(code: string): boolean {
    try {
      const allCodes = this.getAllReferralCodes();
      const codeData = allCodes[code];
      return codeData ? codeData.isActive : false;
    } catch (error) {
      console.error('❌ 추천코드 존재 확인 실패:', error);
      return false;
    }
  }
}
```

```typescript
// 🎯 추천코드 검증 함수 (영구 저장소 기반)
const validateReferralCode = async (code: string): Promise<{
  isValid: boolean;
  errorMessage?: string;
  walletAddress?: string;
}> => {
  try {
    console.log('🔍 추천코드 검증 시작:', code);

    // 1단계: 형식 검증
    if (!code.startsWith('REF') || code.length !== 11) {
      return {
        isValid: false,
        errorMessage: '추천인 코드 형식이 올바르지 않습니다. (REF + 8자리)'
      };

    // 2단계: 알파벳과 숫자만 포함하는지 확인
    const codePart = code.substring(3);
    const isValidFormat = /^[A-Z0-9]{8}$/.test(codePart);
    
    if (!isValidFormat) {
      return {
        isValid: false,
        errorMessage: '추천인 코드는 REF + 8자리 영문/숫자 조합이어야 합니다.'
      };

    const walletAddress = ReferralCodeManager.getWalletAddressByCode(code);
    
    if (!walletAddress) {
      return {
        isValid: false,
        errorMessage: '존재하지 않는 추천인 코드입니다.'
      };
    }

    console.log('✅ 추천코드 검증 성공:', { code, walletAddress });
    
    return {
      isValid: true,
      walletAddress
    };
  } catch (error) {
    console.error('❌ 추천코드 검증 실패:', error);
    return {
      isValid: false,
      errorMessage: '추천인 코드 검증 중 오류가 발생했습니다.'
    };
  }
};
```

```typescript
// ==================== 지갑 생성 완료 시 추천코드 자동 발급 시스템 ====================

        await giveInstantBonus(referrerWallet, 1.0);

      }
    }

    return { success: true, newReferralCode };
  } catch (error) {
    console.error('❌ 지갑 생성 완료 후 추천코드 시스템 실패:', error);
```

### **5. 추천인 코드 인증 버튼 클릭 시 완벽한 처리**

```typescript
// ==================== 추천인 코드 인증 버튼 클릭 시 완벽한 처리 ====================

// 🎯 추천인 코드 인증 버튼 클릭 시 처리
const handleUserReferralCodeSubmit = async (referralCode: string) => {
  try {
    console.log('   추천인 코드 인증 버튼 클릭:', referralCode);

    // 1. 추천코드 검증 (영구 저장소 기반)
    const validationResult = await validateReferralCode(referralCode);
    
    if (!validationResult.isValid) {
      alert(`❌ 추천인 코드 오류: ${validationResult.errorMessage}`);
      return { success: false, error: validationResult.errorMessage };
    }

    // 2. 인증 성공 시 상태 업데이트
    setUserReferralSystemState(prev => ({
      ...prev,
      referralCode: referralCode,
      referralBonusApplied: true,
      referrerWalletAddress: validationResult.walletAddress
    }));

    // 3. 성공 메시지 표시
    alert(`✅ 추천인 코드 인증 성공!\n\n지갑 생성 후 보상이 지급됩니다.`);
    
    console.log('✅ 추천인 코드 인증 성공:', { referralCode, referrerWallet: validationResult.walletAddress });
    
    return { success: true, data: { referralCode, referrerWallet: validationResult.walletAddress } };
  } catch (error) {
    console.error('❌ 추천인 코드 인증 실패:', error);
    alert(`❌ 추천인 코드 인증 실패: ${error.message}`);
    return { success: false, error: error.message };
  }
};
```
