## **13단계: 유저용/관리자용 락업 버튼 통합**

```typescript
// ==================== 13단계: 유저용/관리자용 락업 버튼 통합 ====================

// ==================== 유저용 마이닝 보너스 설정 모달에 락업 버튼 추가 ====================
{/* 유저용 락업 보너스 설정 버튼 */}
<button 
  onClick={() => {
    if (!authenticatedWalletAddress) {
      alert('지갑 주소가 인증되지 않았습니다. 먼저 지갑 주소를 인증해주세요.');
      return;
    }
    if (!userIsMiningActive) {
      alert('마이닝을 먼저 시작해야 락업 보너스를 설정할 수 있습니다.');
      return;
    }
    // 다른 모든 모달 닫기
    setShowMiningStartModal(false);
    setShowMiningStopModal(false);
    setShowAttendanceModal(false);
    setShowReferralModal(false);
    setShowComingSoonModal(false);
    // 유저용 락업 보너스 모달 표시
    setShowUserLockupModal(true);
  }}
  className={`px-4 py-3 font-bold rounded-lg transition-all duration-200 transform hover:scale-105 hover:rotate-1 hover:shadow-2xl ${
    userIsMiningActive
      ? 'bg-gradient-to-r from-purple-400 to-pink-500 text-white hover:shadow-lg' 
      : 'bg-gray-400 text-gray-200 cursor-not-allowed'
  }`}
  disabled={!userIsMiningActive}
  title={userIsMiningActive ? '유저용 락업 보너스를 설정할 수 있습니다' : '마이닝을 먼저 시작해야 합니다'}
>
  🔒 유저용 락업 보너스 설정 {!userIsMiningActive && '(비활성화)'}
</button>

// ==================== 관리자용 마이닝 보너스 설정 모달에 락업 버튼 추가 ====================
{/* 관리자용 락업 보너스 설정 버튼 */}
<button 
  onClick={() => {
    if (!authenticatedWalletAddress) {
      alert('지갑 주소가 인증되지 않았습니다. 먼저 지갑 주소를 인증해주세요.');
      return;
    }
    if (!isMiningActive) {
      alert('마이닝을 먼저 시작해야 락업 보너스를 설정할 수 있습니다.');
      return;
    }
    // 다른 모든 모달 닫기
    setShowMiningStartModal(false);
    setShowMiningStopModal(false);
    setShowAttendanceModal(false);
    setShowReferralModal(false);
    setShowComingSoonModal(false);
    // 관리자용 락업 보너스 모달 표시
    setShowAdminLockupModal(true);
  }}
  className={`px-4 py-3 font-bold rounded-lg transition-all duration-200 transform hover:scale-105 hover:rotate-1 hover:shadow-2xl ${
    isMiningActive
      ? 'bg-gradient-to-r from-purple-400 to-pink-500 text-white hover:shadow-lg' 
      : 'bg-gray-400 text-gray-200 cursor-not-allowed'
  }`}
  disabled={!isMiningActive}
  title={isMiningActive ? '관리자용 락업 보너스를 설정할 수 있습니다' : '마이닝을 먼저 시작해야 합니다'}
>
     관리자용 락업 보너스 설정 {!isMiningActive && '(비활성화)'}
</button>
```
## **14단계: 컴포넌트 마운트 시 데이터 로드**

```typescript
// ==================== 14단계: 컴포넌트 마운트 시 데이터 로드 ====================

useEffect(() => {
  // 유저용 락업 상태 로드
  loadUserLockupState();
  
  // 관리자용 락업 상태 로드
  loadAdminLockupState();
  
  console.log('✅ 유저용/관리자용 락업 상태 로드 완료');
}, []);

// ==================== 유저용 락업 상태 실시간 업데이트 ====================
useEffect(() => {
  if (userLockupRatio > 0 || userLockupMonths > 0) {
    saveUserLockupState();
  }
}, [userLockupRatio, userLockupMonths, isUserLockupRegistered, registeredUserLockupData, userSecondLockupRatio, userSecondLockupMonths, isUserSecondLockupRegistered, registeredUserSecondLockupData]);

// ==================== 관리자용 락업 상태 실시간 업데이트 ====================
useEffect(() => {
  if (adminLockupRatio > 0 || adminLockupMonths > 0) {
    saveAdminLockupState();
  }
}, [adminLockupRatio, adminLockupMonths, isAdminLockupRegistered, registeredAdminLockupData, adminSecondLockupRatio, adminSecondLockupMonths, isAdminSecondLockupRegistered, registeredAdminSecondLockupData]);
