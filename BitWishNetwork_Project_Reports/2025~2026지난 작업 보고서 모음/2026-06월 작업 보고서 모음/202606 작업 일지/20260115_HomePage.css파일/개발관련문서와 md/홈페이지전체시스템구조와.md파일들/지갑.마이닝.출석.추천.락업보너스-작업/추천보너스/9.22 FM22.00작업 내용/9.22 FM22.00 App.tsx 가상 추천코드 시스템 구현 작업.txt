### **1. 추천코드 인증 매개체 연결 (최우선)**

#### **1.1 App.tsx에서 추천코드 인증 로직 수정**
```typescript
// 현재: 지갑 주소 기반 추천코드 생성
const referralCode = `REF${walletAddress.slice(-8).toUpperCase()}${Date.now().toString(36)}`;

// 수정: 랜덤 추천코드 생성 + 지갑 주소 매개체 연결
const generateRandomReferralCode = (): string => {
  const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789';
  let result = 'REF';
  for (let i = 0; i < 8; i++) {
    result += chars.charAt(Math.floor(Math.random() * chars.length));
  }
  return result;
};

// 지갑 주소와 추천코드 매핑 저장
const walletReferralMapping = new Map<string, string>();
```

#### **1.2 "마이닝 보너스 설정"창 진입 시 추천코드 자동 발급**
```typescript
// 지갑 주소 인증 성공 시 추천코드 발급
const handleWalletAuthentication = (walletAddress: string) => {
  // 기존 추천코드가 있는지 확인
  let referralCode = walletReferralMapping.get(walletAddress);
  
  if (!referralCode) {
    // 새로운 추천코드 생성
    referralCode = generateRandomReferralCode();
    walletReferralMapping.set(walletAddress, referralCode);
  }
  
  // "추천 보너스"창에 추천코드 표시
  setUserReferralCode(referralCode);
};
```

```typescript
  const lockupBonus = userLockupBonus || 0;
  
  const totalBonusRate = attendanceBonus + referralBonus + lockupBonus;
  const enhancedReward = baseReward * (1 + totalBonusRate);
  
  return enhancedReward;
};

// 추천 보상 1BW 지급 시 실시간 반영
const giveReferralReward = (amount: number) => {
  setUserReferralReward(prev => prev + amount);
  // 실시간 마이닝 보상창에 즉시 반영
  updateRealTimeMiningReward();
};
```
