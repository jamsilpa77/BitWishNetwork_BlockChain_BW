### **2. 유저용 마이닝 보상 계산 함수**

```typescript
// 🎯 유저용 마이닝 보상 계산 함수 (완전 분리, 고정값 적용)
const calculateUserMiningReward = (): {
  baseRate: number;
  attendanceBonus: number;
  referralBonus: number;
  lockupBonus: number;
  merchantBonus: number;
  totalBonusRate: number;
  totalEnhancedRate: number;
  hourlyReward: number;
  dailyReward: number;
  monthlyReward: number;
  yearlyReward: number;
  fixedHourlyReward: number;
  fixedDailyReward: number;
} => {
  // 🎯 유저용 고정값 적용 (요구사항 3번)
  const fixedHourlyReward = USER_FIXED_HOURLY_REWARD;
  const fixedDailyReward = USER_FIXED_DAILY_REWARD;
  
  //    유저용 기본 채굴률 (고정값)
  const decimalBaseRate = new Decimal(fixedHourlyReward);
  
  // 🎯 유저용 출석 보너스 계산 (초기값 0)
  const decimalAttendanceBonus = new Decimal(0);
  
  // 🎯 유저용 추천 보너스 계산 (초기값 0)
  const decimalReferralBonus = new Decimal(0);
  
  // 🎯 유저용 락업 보너스 계산 (초기값 0)
  const decimalLockupBonus = new Decimal(0);
  
  // 🎯 유저용 가맹점 보너스 계산 (초기값 0)
  const decimalMerchantBonus = new Decimal(0);
  
  // 🎯 유저용 총 보너스율 계산 (모든 보너스 0)
  const decimalOne = new Decimal(1);
  const decimalTotalBonusRate = decimalOne
    .plus(decimalAttendanceBonus)
    .plus(decimalReferralBonus)
    .plus(decimalLockupBonus)
    .plus(decimalMerchantBonus);
  
  // 🎯 유저용 강화된 채굴률 계산 (고정값 사용)
  const decimalTotalEnhancedRate = decimalBaseRate;
  
  // 🎯 유저용 보상 계산 (고정값 사용)
  const decimalDailyReward = new Decimal(fixedDailyReward);
  const decimalMonthlyReward = decimalDailyReward.times(30);
  const decimalYearlyReward = decimalDailyReward.times(365);
  
  return {
    baseRate: decimalBaseRate.toDecimalPlaces(8).toNumber(),
    attendanceBonus: decimalAttendanceBonus.toDecimalPlaces(8).toNumber(),
    referralBonus: decimalReferralBonus.toDecimalPlaces(8).toNumber(),
    lockupBonus: decimalLockupBonus.toDecimalPlaces(8).toNumber(),
    merchantBonus: decimalMerchantBonus.toDecimalPlaces(8).toNumber(),
    totalBonusRate: decimalTotalBonusRate.minus(decimalOne).toDecimalPlaces(8).toNumber(),
    totalEnhancedRate: decimalTotalEnhancedRate.toDecimalPlaces(8).toNumber(),
    hourlyReward: decimalTotalEnhancedRate.toDecimalPlaces(8).toNumber(),
    dailyReward: decimalDailyReward.toDecimalPlaces(8).toNumber(),
    monthlyReward: decimalMonthlyReward.toDecimalPlaces(8).toNumber(),
    yearlyReward: decimalYearlyReward.toDecimalPlaces(8).toNumber(),
    fixedHourlyReward: fixedHourlyReward,
    fixedDailyReward: fixedDailyReward
  };
};
```

  fixedDailyReward: number;
} => {
  // 🎯 유저용 고정값 적용 (요구사항 3번)
  const fixedHourlyReward = USER_FIXED_HOURLY_REWARD;
  const fixedDailyReward = USER_FIXED_DAILY_REWARD;
  
  //    유저용 기본 채굴률 (고정값)
  const decimalBaseRate = new Decimal(fixedHourlyReward);
  
  // 🎯 유저용 출석 보너스 계산 (초기값 0)
  const decimalAttendanceBonus = new Decimal(0);
  
  // 🎯 유저용 추천 보너스 계산 (초기값 0)
  const decimalReferralBonus = new Decimal(0);
  
  // 🎯 유저용 락업 보너스 계산 (초기값 0)
  const decimalLockupBonus = new Decimal(0);
  
  // 🎯 유저용 가맹점 보너스 계산 (초기값 0)
  const decimalMerchantBonus = new Decimal(0);
  
  // 🎯 유저용 총 보너스율 계산 (모든 보너스 0)
  const decimalOne = new Decimal(1);
  const decimalTotalBonusRate = decimalOne
    .plus(decimalAttendanceBonus)
    .plus(decimalReferralBonus)
    .plus(decimalLockupBonus)
    .plus(decimalMerchantBonus);
  
  // 🎯 유저용 강화된 채굴률 계산 (고정값 사용)
  const decimalTotalEnhancedRate = decimalBaseRate;
  
  // 🎯 유저용 보상 계산 (고정값 사용)
  const decimalDailyReward = new Decimal(fixedDailyReward);
  const decimalMonthlyReward = decimalDailyReward.times(30);
  const decimalYearlyReward = decimalDailyReward.times(365);
  
  return {
    baseRate: decimalBaseRate.toDecimalPlaces(8).toNumber(),
    attendanceBonus: decimalAttendanceBonus.toDecimalPlaces(8).toNumber(),
    referralBonus: decimalReferralBonus.toDecimalPlaces(8).toNumber(),
    lockupBonus: decimalLockupBonus.toDecimalPlaces(8).toNumber(),
    merchantBonus: decimalMerchantBonus.toDecimalPlaces(8).toNumber(),
    totalBonusRate: decimalTotalBonusRate.minus(decimalOne).toDecimalPlaces(8).toNumber(),
    totalEnhancedRate: decimalTotalEnhancedRate.toDecimalPlaces(8).toNumber(),
    hourlyReward: decimalTotalEnhancedRate.toDecimalPlaces(8).toNumber(),
    dailyReward: decimalDailyReward.toDecimalPlaces(8).toNumber(),
    monthlyReward: decimalMonthlyReward.toDecimalPlaces(8).toNumber(),
    yearlyReward: decimalYearlyReward.toDecimalPlaces(8).toNumber(),
    fixedHourlyReward: fixedHourlyReward,
    fixedDailyReward: fixedDailyReward
  };
};
```

                <div className="bg-white p-4 rounded-lg shadow-sm">
                  <div className="text-sm text-gray-600">연속 출석일</div>
                  <div className="text-lg font-bold text-gray-500">{userConsecutiveAttendanceDays}일</div>
                </div>
                <div className="bg-white p-4 rounded-lg shadow-sm">
                  <div className="text-sm text-gray-600">추천인 수</div>
                  <div className="text-lg font-bold text-gray-500">{userReferralCount}명</div>
                </div>
                <div className="bg-white p-4 rounded-lg shadow-sm">
                  <div className="text-sm text-gray-600">락업 비율</div>
                  <div className="text-lg font-bold text-gray-500">{userLockupRatio}%</div>

                  <div className="text-lg font-bold text-gray-500">{userLockupMonths}개월</div>
                </div>
              </div>
            </div>
          </div>
        )}

        {/* 🎯 실시간 마이닝 보상 창 (관리자용 "🔒 초기화" 버튼 제거) */}
        <div className="bg-gradient-to-r from-green-50 to-emerald-50 border border-green-200 rounded-lg p-6 mt-6">
          <h3 className="text-lg font-bold text-green-800 mb-4 text-center">⚡ 실시간 마이닝 보상</h3>
          
          <div className="grid grid-cols-2 gap-4 mb-4">
            <div className="bg-white p-4 rounded-lg shadow-sm">
              <div className="text-sm text-gray-600">현재 마이닝 상태</div>
              <div className="text-lg font-bold text-green-600">
                {userIsMiningActive ? '🟢 활성화' : '   비활성화'}
              </div>
            </div>
            <div className="bg-white p-4 rounded-lg shadow-sm">
              <div className="text-sm text-gray-600">마이닝 시작 시간</div>
              <div className="text-lg font-bold text-green-600">
                {userMiningStartTime ? userMiningStartTime.toLocaleString() : '시작되지 않음'}
              </div>
            </div>
          </div>

          <div className="grid grid-cols-2 gap-4">
            <div className="bg-white p-4 rounded-lg shadow-sm">
              <div className="text-sm text-gray-600">시간당 보상</div>
              <div className="text-xl font-bold text-green-600">{USER_FIXED_HOURLY_REWARD.toFixed(8)} BW</div>
            </div>
            <div className="bg-white p-4 rounded-lg shadow-sm">
              <div className="text-sm text-gray-600">일일 보상</div>
              <div className="text-xl font-bold text-green-600">{USER_FIXED_DAILY_REWARD.toFixed(8)} BW</div>
            </div>
          </div>

          {/* 🎯 관리자용 "   초기화" 버튼 제거됨 (요구사항 7번) */}
        </div>
      </div>
    </div>
  </div>
)}
```
