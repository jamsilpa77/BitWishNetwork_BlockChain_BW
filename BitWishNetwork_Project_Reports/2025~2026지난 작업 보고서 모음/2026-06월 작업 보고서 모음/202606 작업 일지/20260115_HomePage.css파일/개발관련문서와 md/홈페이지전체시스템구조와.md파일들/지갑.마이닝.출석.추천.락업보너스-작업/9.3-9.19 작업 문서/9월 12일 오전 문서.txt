```typescript

const calculateLockupBonus = (lockupRatio: number, lockupMonths: number) => {

&nbsp; // 백서 기반 정확한 APY 값

&nbsp; const apyRates = {

&nbsp;   1: 0.027,     // 1개월: 2.7% APY

&nbsp;   6: 0.042,     // 6개월: 4.2% APY

&nbsp;   12: 0.057,    // 12개월: 5.7% APY

&nbsp;   36: 0.072     // 36개월: 7.2% APY

&nbsp; };

&nbsp; 

&nbsp; const baseAPY = apyRates\[lockupMonths as keyof typeof apyRates] || 0.027;

&nbsp; const periodBonus = 0; // 기간 보너스 없음

&nbsp; const totalAPY = baseAPY;

&nbsp; const monthlyRate = baseAPY / 12; // APY / 12 (월 보상률)

&nbsp; 

&nbsp; // Decimal 계산

&nbsp; const baseAPYDec = new Decimal(baseAPY);

&nbsp; const periodDec = new Decimal(periodBonus);

&nbsp; const totalAPYDec = new Decimal(totalAPY);

&nbsp; const monthlyRateDec = new Decimal(monthlyRate);

&nbsp; 

&nbsp; // 백서 공식: 기본\_채굴률 × (1 + APY × 락업\_비율)

&nbsp; const decimalBaseRate = new Decimal(0.25);

&nbsp; const decimalLockupRatio = new Decimal(lockupRatio / 100);

&nbsp; const enhancedRateDec = decimalBaseRate.times(new Decimal(1).plus(baseAPYDec.times(decimalLockupRatio)));

&nbsp; 

&nbsp; const desc = `기본 락업 ${lockupRatio}% (${lockupMonths}개월) - APY ${(baseAPY \* 100).toFixed(1)}%`;



&nbsp; return {

&nbsp;   baseAPY: baseAPYDec.toDecimalPlaces(8).toNumber(),

&nbsp;   periodBonus: periodDec.toDecimalPlaces(8).toNumber(),

&nbsp;   totalAPY: totalAPYDec.toDecimalPlaces(8).toNumber(),

&nbsp;   monthlyRate: monthlyRateDec.toDecimalPlaces(8).toNumber(),
```
