import React, { useState, useEffect } from 'react';
import Decimal from 'decimal.js';
import { useTranslation } from 'react-i18next';
import { 
  ReferralModal, 
  LockupModal, 
  AttendanceModal, 
  MerchantModal, 
  SuccessModal, 
  ResetConfirmModal 
} from './AdminMiningSettingsModals';

// CSS 애니메이션 스타일

  @keyframes fadeIn {
    from { 
      opacity: 0;

    }

  }
  
  .animate-fadeIn {
    animation: fadeIn 0.3s ease-out;
  }
`;

// 스타일 태그 추가
if (typeof document !== 'undefined') {
  const styleSheet = document.createElement('style');
  styleSheet.textContent = styles;
  document.head.appendChild(styleSheet);

// ==================== 관리자 전용 마이닝 보너스 설정 컴포넌트 시작 ====================

interface AdminMiningSettingsProps {
  isMiningActive: boolean;
  miningSettings: {
    hashRate: number;
    difficulty: number;
    blockReward: number;
    estimatedTime: number;
    totalMined: number;
    isMining: boolean;
    miningAddress: string;
    miningStartTime: string | null;
    lastMinedBlock: number | null;
  };
  onStartMining: () => void;
  onStopMining: () => void;
  onUpdateMiningStats: (stats: any) => void;
  authenticatedWalletAddress?: string;
  onClose?: () => void;
}

const AdminMiningSettings: React.FC<AdminMiningSettingsProps> = ({
  isMiningActive: propIsMiningActive,
  miningSettings: propMiningSettings,
  onStartMining,
  onStopMining,
  onUpdateMiningStats,
  authenticatedWalletAddress = '',
  onClose = () => {}
}) => {
  const { t } = useTranslation();

  // ==================== 상태 관리 시작 ====================
  // 마이닝 상태 (props와 연동)
  const [isMiningActive, setIsMiningActive] = useState(propIsMiningActive);
  const [miningStartTime, setMiningStartTime] = useState<Date | null>(
    propMiningSettings.miningStartTime ? new Date(propMiningSettings.miningStartTime) : null
  );
  const [tick, setTick] = useState(0);
  
  // props 변경 시 내부 상태 업데이트
  useEffect(() => {
    setIsMiningActive(propIsMiningActive);
    setMiningStartTime(propMiningSettings.miningStartTime ? new Date(propMiningSettings.miningStartTime) : null);
  }, [propIsMiningActive, propMiningSettings.miningStartTime]);

  // 보너스 상태
  const [consecutiveAttendanceDays, setConsecutiveAttendanceDays] = useState(1);
  const [referralCount, setReferralCount] = useState(0);
  const [lockupRatio, setLockupRatio] = useState(0);
  const [secondLockupRatio, setSecondLockupRatio] = useState(0);
  const [merchantStatus, setMerchantStatus] = useState<'none' | 'pending' | 'approved'>('none');

  // 추천인 시스템 상태
  const [referralSystemState, setReferralSystemState] = useState({
    referralBonusRate: 0,
    additionalBonusRate: 0,
    cumulativeBonusRate: 0,
    totalBonusRate: 0,
    referrerAddress: '',
    segmentInfo: [] as any[]
  });

  // 출석 달력
  const [attendanceCalendar, setAttendanceCalendar] = useState<Record<string, boolean>>({});

  // 락업 설정
  const [lockupSettings, setLockupSettings] = useState({
    ratio: 0,
    period: 0,
    isRegistered: false
  });

  const [secondLockupSettings, setSecondLockupSettings] = useState({
    ratio: 0,
    period: 0,
    isRegistered: false
  });

  // 모달 상태
  const [showReferralModal, setShowReferralModal] = useState(false);
  const [showLockupModal, setShowLockupModal] = useState(false);
  const [showAttendanceModal, setShowAttendanceModal] = useState(false);
  const [showMerchantModal, setShowMerchantModal] = useState(false);
  const [showSuccessModal, setShowSuccessModal] = useState(false);
  const [showReferralBonusDisplay, setShowReferralBonusDisplay] = useState(false);
  const [showMiningStopStatus, setShowMiningStopStatus] = useState(false);

  // 관리자 전용 모달 상태
  const [showBasicLockupResetModal, setShowBasicLockupResetModal] = useState(false);
  const [showSecondLockupResetModal, setShowSecondLockupResetModal] = useState(false);

  // 메시지 상태
  const [successMessage, setSuccessMessage] = useState('');
  const [referralBonusData, setReferralBonusData] = useState<any>(null);

  // 접이식 UI 상태
  const [isReferralBonusExpanded, setIsReferralBonusExpanded] = useState(false);
  // ==================== 상태 관리 끝 ====================

  // ==================== 유틸리티 함수들 시작 ====================
  const getKoreanTime = () => {
    const now = new Date();
    const koreanTime = new Date(now.getTime() + (9 * 60 * 60 * 1000));
    return koreanTime;
  };

  const formatDateKey = (year: number, month: number, day: number) => {
    return `${year}-${month.toString().padStart(2, '0')}-${day.toString().padStart(2, '0')}`;
  };

  const isAttendanceTimeValid = () => {
    const koreanTime = getKoreanTime();
    const hours = koreanTime.getHours();
    return hours >= 9;
  };

  const isAttendanceBonusTimeValid = () => {
    const koreanTime = getKoreanTime();
    const hours = koreanTime.getHours();
    const minutes = koreanTime.getMinutes();
    const seconds = koreanTime.getSeconds();
    
    if (hours >= 9) {
      return false;
    }
    return true;
  };

  const calculateAttendanceBonus = (days: number) => {
    if (days <= 0) return { bonusRate: 0, message: '' };
    if (days <= 7) return { bonusRate: 0.05, message: '5% 보너스' };
    if (days <= 14) return { bonusRate: 0.10, message: '10% 보너스' };
    if (days <= 30) return { bonusRate: 0.15, message: '15% 보너스' };
    if (days <= 60) return { bonusRate: 0.20, message: '20% 보너스' };
    if (days <= 90) return { bonusRate: 0.25, message: '25% 보너스' };
    return { bonusRate: 0.30, message: '30% 보너스' };
  };

  const calculateConsecutiveDaysFromCalendar = (calendar: Record<string, boolean>, currentDate: Date) => {
    const dates = Object.keys(calendar).sort();
    if (dates.length === 0) return 0;

    let consecutiveDays = 0;
    const today = new Date(currentDate);
    today.setHours(0, 0, 0, 0);

    for (let i = dates.length - 1; i >= 0; i--) {
      const date = new Date(dates[i]);
      date.setHours(0, 0, 0, 0);
      
      const diffTime = today.getTime() - date.getTime();
      const diffDays = Math.floor(diffTime / (1000 * 60 * 60 * 24));
      
      if (diffDays === consecutiveDays) {
        consecutiveDays++;
      } else {
        break;
      }
    }

    return consecutiveDays;
  };
  // ==================== 유틸리티 함수들 끝 ====================

  // ==================== 관리자 전용 테스트 함수들 시작 ====================
  const handleAddReferralDecimal = async () => {
    console.log('🧪 테스트용: handleAddReferralDecimal 함수 시작! (완벽한 백서 정책 적용)');
    
    try {
      const timestamp = Date.now();
      const randomSuffix = Math.random().toString(36).substr(2, 8).toUpperCase();
      const newReferralAddress = `G${'B'.repeat(10)}${timestamp}${randomSuffix}`.substring(0, 56);
      const newReferralCode = `REF${timestamp}${randomSuffix}`;
      
      const newReferralCount = referralCount + 1;
      
      const joinResult = {
        instantBonus: 1.0,
        permanentBonusRate: 0.012,
        additionalBonusRate: 0.001
      };

      const referralBonusData = {
        referralBonusRate: 0.012,
        additionalBonusRate: joinResult.additionalBonusRate,
        cumulativeBonusRate: new Decimal(joinResult.additionalBonusRate),
        totalBonusRate: 0.012 + joinResult.additionalBonusRate,
        segmentInfo: [{
          segment: 1,
          count: newReferralCount,
          bonusRate: joinResult.additionalBonusRate
        }]
      };

      const newReferralRelationship = {
        referralAddress: newReferralAddress,
        referralCode: newReferralCode,
        instantBonus: joinResult.instantBonus,
        permanentBonusRate: joinResult.permanentBonusRate,
        cumulativeBonusRate: referralBonusData.cumulativeBonusRate.toDecimalPlaces(8).toNumber(),
        totalBonusRate: referralBonusData.totalBonusRate,
        segmentInfo: referralBonusData.segmentInfo,
        createdAt: new Date(),
        message: t('mining.referralAddSuccess', { 
          instantBonus: joinResult.instantBonus, 
          permanentRate: (joinResult.permanentBonusRate * 100).toFixed(1),
          count: newReferralCount
        })
      };

      setReferralSystemState(prev => ({
        ...prev,
        referralBonusRate: 0.012,
        additionalBonusRate: joinResult.additionalBonusRate,
        cumulativeBonusRate: referralBonusData.cumulativeBonusRate.toNumber(),
        totalBonusRate: referralBonusData.totalBonusRate,
        segmentInfo: referralBonusData.segmentInfo,
        referrerAddress: authenticatedWalletAddress
      }));

      setReferralCount(newReferralCount);
      setReferralBonusData(newReferralRelationship);
      setShowReferralBonusDisplay(true);
      setSuccessMessage(joinResult.message);
      setShowSuccessModal(true);

      console.log(`🧪 테스트용: 새로운 추천인이 추가되었습니다! (완벽한 백서 정책 적용)`, {
        추천인주소: newReferralAddress,
        추천인코드: newReferralCode,
        즉시보너스: `${joinResult.instantBonus} BW`,
        영구보너스율: `${(joinResult.permanentBonusRate * 100).toFixed(1)}%`,
        누적보너스율: `${referralBonusData.cumulativeBonusRate.times(100).toFixed(6)}%`,
        총추천인수: newReferralCount
      });

    } catch (error) {
      console.error('🧪 테스트용: 추천인 추가 실패:', error);
      setSuccessMessage(t('mining.referralAddError') + ': ' + (error instanceof Error ? error.message : t('mining.unknownError')));
      setShowSuccessModal(true);
    }
  };

  const handleResetReferral = () => {
    console.log('🔄 추천인 초기화 함수 시작! (실시간 반영)');
    
    try {
      setReferralCount(0);
      setReferralSystemState({
        referralBonusRate: 0,
        additionalBonusRate: 0,
        cumulativeBonusRate: 0,
        totalBonusRate: 0,
        referrerAddress: '',
        segmentInfo: []
      });
      
        setSuccessMessage(t('mining.referralResetSuccess'));
        setShowSuccessModal(true);
        
        console.log(t('mining.referralResetComplete'));
      } catch (error) {
        console.error('❌ 추천인 초기화 실패:', error);
        setSuccessMessage(t('mining.referralResetError') + ': ' + (error instanceof Error ? error.message : t('mining.unknownError')));
        setShowSuccessModal(true);
      }
  };

  const handleBasicLockupReset = () => {
    try {
      setLockupRatio(0);
      setLockupSettings({
        ratio: 0,
        period: 0,
        isRegistered: false
      });
      
      setSuccessMessage(t('mining.basicLockupResetSuccess'));
      setShowSuccessModal(true);
      setShowBasicLockupResetModal(false);
      
        console.log(t('mining.basicLockupResetComplete'));
    } catch (error) {
      console.error('❌ 기본 락업 초기화 실패:', error);
      setSuccessMessage(t('mining.basicLockupResetError') + ': ' + (error instanceof Error ? error.message : t('mining.unknownError')));
      setShowSuccessModal(true);
    }
  };

  const handleSecondLockupReset = () => {
    try {
      setSecondLockupRatio(0);
      setSecondLockupSettings({
        ratio: 0,
        period: 0,
        isRegistered: false
      });
      
      setSuccessMessage(t('mining.secondLockupResetSuccess'));
      setShowSuccessModal(true);
      setShowSecondLockupResetModal(false);
      
        console.log(t('mining.secondLockupResetComplete'));
    } catch (error) {
      console.error('❌ 추가 락업 초기화 실패:', error);
      setSuccessMessage(t('mining.secondLockupResetError') + ': ' + (error instanceof Error ? error.message : t('mining.unknownError')));
      setShowSuccessModal(true);
    }
  };
  // ==================== 관리자 전용 테스트 함수들 끝 ====================

  // ==================== 마이닝 관련 함수들 시작 ====================
  const calculateRealTimeMiningRewardDecimal = () => {
    if (!isMiningActive || !miningStartTime) {
      return {
        hourlyReward: new Decimal(0),
        dailyReward: new Decimal(0),
        totalBonusRate: new Decimal(0)
      };
    }

    const baseHourlyReward = new Decimal(0.25);
    const baseDailyReward = new Decimal(6.0);

    const attendanceBonus = (consecutiveAttendanceDays > 0 && isAttendanceBonusTimeValid()) ? 
      new Decimal(calculateAttendanceBonus(consecutiveAttendanceDays).bonusRate) : 
      new Decimal(0);

    const referralBonus = new Decimal(0);
    const additionalReferralBonus = new Decimal(referralSystemState.additionalBonusRate || 0);

    const lockupBonus = lockupRatio > 0 ? 
      new Decimal(lockupRatio / 100).dividedBy(new Decimal(12)) : 
      new Decimal(0);

    const secondLockupBonus = secondLockupRatio > 0 ? 
      new Decimal(secondLockupRatio / 100).dividedBy(new Decimal(12)) : 
      new Decimal(0);

    const merchantBonus = merchantStatus === 'approved' ? new Decimal(0.25) : new Decimal(0);

    const totalBonusRate = attendanceBonus
      .plus(referralBonus)
      .plus(additionalReferralBonus)
      .plus(lockupBonus)
      .plus(secondLockupBonus)
      .plus(merchantBonus);

    const enhancedHourlyReward = baseHourlyReward.times(new Decimal(1).plus(totalBonusRate));
    const enhancedDailyReward = baseDailyReward.times(new Decimal(1).plus(totalBonusRate));

    return {
      hourlyReward: enhancedHourlyReward,
      dailyReward: enhancedDailyReward,
      totalBonusRate: totalBonusRate
    };
  };

  const calculateAccumulatedRewardDecimal = () => {
    if (!isMiningActive || !miningStartTime) {
      return new Decimal(0);
    }

    const now = new Date();
    const elapsedTime = now.getTime() - miningStartTime.getTime();
    const elapsedHours = new Decimal(elapsedTime).dividedBy(new Decimal(1000 * 60 * 60));

    const rewardData = calculateRealTimeMiningRewardDecimal();
    const accumulatedReward = rewardData.hourlyReward.times(elapsedHours);

    return accumulatedReward;
  };

  const handleMiningStart = async () => {
    try {
      // App.tsx의 onStartMining 함수 호출
      onStartMining();
      
      // 내부 상태도 업데이트
      setIsMiningActive(true);
      setMiningStartTime(new Date());
      setTick(prevTick => prevTick + 1);
      
      console.log(t('mining.startSuccess'));
    } catch (error) {
      console.error('❌ 마이닝 시작 중 오류:', error);
      alert(t('mining.startError') + ': ' + (error instanceof Error ? error.message : t('mining.unknownError')));
    }
  };

  const handleMiningStop = async () => {
    try {
      // App.tsx의 onStopMining 함수 호출
      onStopMining();
      
      // 내부 상태도 업데이트
      setIsMiningActive(false);
      setMiningStartTime(null);
      setTick(0);
      
      console.log(t('mining.stopSuccess'));
    } catch (error) {
      console.error('❌ 마이닝 정지 중 오류:', error);
      alert(t('mining.stopError') + ': ' + (error instanceof Error ? error.message : t('mining.unknownError')));
    }
  };
  // ==================== 마이닝 관련 함수들 끝 ====================

  // ==================== 출석 관련 함수들 시작 ====================
  const handleAttendanceCheck = async () => {
    if (!isAttendanceTimeValid()) {
      alert(t('mining.attendanceTimeLimit'));
      return;
    }

    try {
      const today = new Date();
      const dateKey = formatDateKey(today.getFullYear(), today.getMonth() + 1, today.getDate());
      
      if (attendanceCalendar[dateKey]) {
        alert(t('mining.alreadyAttended'));
        return;
      }

      const response = await fetch('http://localhost:4001/attendance/check-in', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          walletAddress: authenticatedWalletAddress,
          date: dateKey
        })
      });

      if (response.ok) {
        const data = await response.json();
        console.log(t('mining.attendanceCheckComplete'), data);
        
        setAttendanceCalendar(prev => ({
          ...prev,
          [dateKey]: true
        }));
        
        const newConsecutiveDays = calculateConsecutiveDaysFromCalendar({
          ...attendanceCalendar,
          [dateKey]: true
        }, today);
        
        setConsecutiveAttendanceDays(newConsecutiveDays);
        
        setSuccessMessage(t('mining.attendanceSuccess', { days: newConsecutiveDays }));
        setShowSuccessModal(true);
        
      } else {
        const errorData = await response.json();
        console.error('❌ 출석 체크 실패:', errorData);
        alert(t('mining.attendanceFailed') + ': ' + errorData.message);
      }
    } catch (error) {
      console.error('❌ 출석 체크 중 오류:', error);
      alert(t('mining.attendanceError') + ': ' + (error instanceof Error ? error.message : t('mining.unknownError')));
    }
  };

  const getAttendanceCalendarStatus = (year: number, month: number, day: number) => {
    const koreanTime = getKoreanTime();
    const hours = koreanTime.getHours();
    const minutes = koreanTime.getMinutes();
    const seconds = koreanTime.getSeconds();
    const currentDate = new Date();
    const isToday = currentDate.getFullYear() === year && currentDate.getMonth() === month - 1 && currentDate.getDate() === day;
    const dateKey = formatDateKey(year, month, day);
    const isAttended = attendanceCalendar[dateKey];

    if (isAttended) {
      return 'attended';
    } else if (isToday && (hours > 9 || (hours === 9 && minutes >= 0 && seconds >= 0))) {
      return 'active';
    } else {
      return 'inactive';
    }
  };

  const getAttendanceStatusDisplay = () => {
    const koreanTime = getKoreanTime();
    const hours = koreanTime.getHours();
    const minutes = koreanTime.getMinutes();
    const seconds = koreanTime.getSeconds();

    if (hours >= 9) {
      return false;
    } else {
      return {
        status: 'active',
        message: t('mining.attendanceActive', { days: consecutiveAttendanceDays }),
        color: 'text-green-500'
      };
    }
  };
  // ==================== 출석 관련 함수들 끝 ====================

  // ==================== 락업 관련 함수들 시작 ====================
  const handleBasicLockupSubmit = async () => {
    try {
      const response = await fetch('http://localhost:4001/lockup/create', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          walletAddress: authenticatedWalletAddress,
          ratio: lockupSettings.ratio,
          period: lockupSettings.period
        })
      });

      if (response.ok) {
        const data = await response.json();
        console.log(t('mining.basicLockupRegisterComplete'), data);
        
        setLockupRatio(lockupSettings.ratio);
        setLockupSettings(prev => ({ ...prev, isRegistered: true }));
        
        setSuccessMessage(t('mining.basicLockupSuccess', { ratio: lockupSettings.ratio, period: lockupSettings.period }));
        setShowSuccessModal(true);
        
      } else {
        const errorData = await response.json();
        console.error('❌ 기본 락업 등록 실패:', errorData);
        alert(t('mining.basicLockupFailed') + ': ' + errorData.message);
      }
    } catch (error) {
      console.error('❌ 기본 락업 등록 중 오류:', error);
      alert(t('mining.basicLockupError') + ': ' + (error instanceof Error ? error.message : t('mining.unknownError')));
    }
  };

  const handleSecondLockupSubmit = async () => {
    try {
      const response = await fetch('http://localhost:4001/second-lockup/create', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          walletAddress: authenticatedWalletAddress,
          ratio: secondLockupSettings.ratio,
          period: secondLockupSettings.period
        })
      });

      if (response.ok) {
        const data = await response.json();
        console.log(t('mining.secondLockupRegisterComplete'), data);
        
        setSecondLockupRatio(secondLockupSettings.ratio);
        setSecondLockupSettings(prev => ({ ...prev, isRegistered: true }));
        
        setSuccessMessage(t('mining.secondLockupSuccess', { ratio: secondLockupSettings.ratio, period: secondLockupSettings.period }));
        setShowSuccessModal(true);
        
      } else {
        const errorData = await response.json();
        console.error('❌ 추가 락업 등록 실패:', errorData);
        alert(t('mining.secondLockupFailed') + ': ' + errorData.message);
      }
    } catch (error) {
      console.error('❌ 추가 락업 등록 중 오류:', error);
      alert(t('mining.secondLockupError') + ': ' + (error instanceof Error ? error.message : t('mining.unknownError')));
    }
  };
  // ==================== 락업 관련 함수들 끝 ====================

  // ==================== 접이식 UI 함수들 시작 ====================
  const toggleReferralBonusExpanded = () => {
    setIsReferralBonusExpanded(!isReferralBonusExpanded);
  };
  // ==================== 접이식 UI 함수들 끝 ====================

  // ==================== useEffect 시작 ====================
  useEffect(() => {
    const checkConsecutiveDays = () => {
      const now = new Date();
      const resetTime = new Date(now.getFullYear(), now.getMonth(), now.getDate(), 9, 0, 0);
      
      const koreanTime = getKoreanTime();
      const hours = koreanTime.getHours();
      const minutes = koreanTime.getMinutes();
      const seconds = koreanTime.getSeconds();
      
      if (hours === 9 && minutes === 0 && seconds === 0) {
        console.log(t('mining.attendanceResetStart'));
        setConsecutiveAttendanceDays(0);
        console.log(t('mining.attendanceResetComplete'));
      }
      
      const dates = Object.keys(attendanceCalendar).sort();
      if (dates.length === 0) {
        setConsecutiveAttendanceDays(0);
        return;
      }
      
      const lastAttendanceDate = new Date(dates[dates.length - 1]);
      const diffTime = now.getTime() - lastAttendanceDate.getTime();
      const diffDays = Math.floor(diffTime / (1000 * 60 * 60 * 24));
      
      if (diffDays > 1) {
        setConsecutiveAttendanceDays(0);
        console.log(t('mining.consecutiveReset'));
      } else {
        const consecutiveDays = calculateConsecutiveDaysFromCalendar(attendanceCalendar, now);
        setConsecutiveAttendanceDays(consecutiveDays);
        console.log(t('mining.attendanceUpdate', { days: consecutiveDays }));
      }
    };

    if (isMiningActive && miningStartTime) {
      checkConsecutiveDays();
    }
  }, [attendanceCalendar, isMiningActive, miningStartTime]);

  useEffect(() => {
    if (isMiningActive && miningStartTime) {
      const interval = setInterval(() => {
        setTick(prevTick => prevTick + 1);
      }, 1000);
      
      return () => clearInterval(interval);
    }
  }, [isMiningActive, miningStartTime]);
  // ==================== useEffect 끝 ====================

  // ==================== 렌더링 시작 ====================
  return (
