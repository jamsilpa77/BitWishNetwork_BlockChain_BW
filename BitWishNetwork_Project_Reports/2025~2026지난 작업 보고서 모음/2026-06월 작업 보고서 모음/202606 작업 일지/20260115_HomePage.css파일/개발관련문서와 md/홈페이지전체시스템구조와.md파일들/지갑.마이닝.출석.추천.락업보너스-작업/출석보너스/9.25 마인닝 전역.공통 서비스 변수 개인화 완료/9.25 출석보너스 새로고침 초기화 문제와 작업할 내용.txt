### **🔧 해결 1: 새로고침 시 출석 데이터 즉시 로드**

**문제**: `loadAllPersonalAttendanceData()`는 실행되지만 개인별 데이터가 모달 열 때만 로드됨

**해결 코드**:
```typescript
// 개인별 출석 데이터 즉시 로드 함수 추가
const loadPersonalAttendanceDataImmediately = (walletAddress: string) => {
  try {
    const attendanceKey = getPersonalStorageKey(walletAddress, 'personalAttendanceCalendar');
    const consecutiveKey = getPersonalStorageKey(walletAddress, 'personalConsecutiveAttendanceDays');
    
    // 출석 달력 로드
    const savedCalendar = localStorage.getItem(attendanceKey);
    if (savedCalendar) {
      setPersonalAttendanceCalendars(prev => ({
        ...prev,
        [walletAddress]: JSON.parse(savedCalendar)
      }));
    }
    
    // 연속 출석일 로드
    const savedConsecutiveDays = localStorage.getItem(consecutiveKey);
    if (savedConsecutiveDays) {
      setPersonalConsecutiveAttendanceDays(prev => ({
        ...prev,
        [walletAddress]: parseInt(savedConsecutiveDays) || 0
      }));
    }
    
    console.log(`✅ 개인별 출석 데이터 즉시 로드됨: ${walletAddress}`);
  } catch (error) {
    console.error(`❌ 개인별 출석 데이터 즉시 로드 실패: ${walletAddress}`, error);
  }
};

// authenticatedWalletAddress 변경 시 즉시 로드
useEffect(() => {
  if (authenticatedWalletAddress) {
    loadPersonalAttendanceDataImmediately(authenticatedWalletAddress);
  }
}, [authenticatedWalletAddress]);
```

### **🔧 해결 2: 개인별 출석 시간 로직을 한국시간 기반으로 수정**

**문제**: `isPersonalAttendanceTimeValid()` 함수가 한국시간을 사용하지 않음

**해결 코드**:
```typescript
// 개인별 출석 시간 유효성 검증 (한국시간 기반)
const isPersonalAttendanceTimeValid = (): boolean => {
  const koreanTime = getUserKoreanTime();
  const hour = koreanTime.getHours();
  const minutes = koreanTime.getMinutes();
  const seconds = koreanTime.getSeconds();
  
  // 오전 9시부터 오후 6시까지만 유효
  if (hour < 9 || hour > 18) return false;
  
  // 정확한 시간 체크
  if (hour === 9 && (minutes > 0 || seconds > 0)) return true;
  if (hour === 18 && (minutes > 0 || seconds > 0)) return false;
  
  return true;
};

// 개인별 출석 가능 시간 범위 체크
const isPersonalAttendanceTimeRangeValid = (targetDate: Date): boolean => {
  const koreanTime = getUserKoreanTime();
  const currentTime = koreanTime.getTime();
  
  // 전날 AM 09:00:00부터 다음날 AM 08:59:59까지
  const startTime = new Date(targetDate);
  startTime.setDate(startTime.getDate() - 1);
  startTime.setHours(9, 0, 0, 0);
  
  const endTime = new Date(targetDate);
  endTime.setHours(8, 59, 59, 999);
  
  return currentTime >= startTime.getTime() && currentTime <= endTime.getTime();
};
```

### **🔧 해결 3: 날짜별 활성화 조건 구현**

**문제**: 24일/25일 날짜별 활성화 로직이 없음

**해결 코드**:
```typescript
// 개인별 출석 달력 날짜 활성화 조건
const isPersonalAttendanceDateActive = (walletAddress: string, year: number, month: number, day: number): boolean => {
  const targetDate = new Date(year, month - 1, day);
  const today = new Date();
  const koreanTime = getUserKoreanTime();
  
  // 오늘 날짜인 경우
  if (targetDate.toDateString() === today.toDateString()) {
    // 25일 날짜: 25일 정각 9시 전까지는 절대 활성화되면 안됨
    if (day === 25) {
      return koreanTime.getHours() >= 9;
    }
    // 다른 날짜: 정상 활성화
    return true;
  }
  
  // 어제 날짜인 경우
  const yesterday = new Date(today);
  yesterday.setDate(yesterday.getDate() - 1);
  if (targetDate.toDateString() === yesterday.toDateString()) {
    // 24일 날짜: 25일 오전 8시 59분 59초까지만 열려있어야 함
    if (day === 24) {
      const tomorrow = new Date(today);
      tomorrow.setHours(8, 59, 59, 999);
      return koreanTime.getTime() <= tomorrow.getTime();
    }
    // 다른 날짜: 정상 활성화
    return true;
  }
  
  // 과거 날짜: 출석 체크 완료 여부 확인
  const personalCalendar = personalAttendanceCalendars[walletAddress] || {};
  const dateKey = targetDate.toISOString().split('T')[0];
  return !personalCalendar[dateKey]; // 출석 체크 안한 날만 활성화
};
```
