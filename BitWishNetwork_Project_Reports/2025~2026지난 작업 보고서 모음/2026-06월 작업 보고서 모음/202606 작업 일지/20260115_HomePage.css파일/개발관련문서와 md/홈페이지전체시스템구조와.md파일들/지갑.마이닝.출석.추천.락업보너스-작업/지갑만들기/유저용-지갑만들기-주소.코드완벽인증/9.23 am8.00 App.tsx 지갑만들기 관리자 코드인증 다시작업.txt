// ==================== 유저용 추천인 가입 시 보상 지급 ====================
const handleUserReferralJoin = async (referralCode: string, newWalletAddress: string) => {
  try {
    console.log('🎯 유저용 추천인 코드 인증 시작:', { referralCode, newWalletAddress });
    
    // 1. 추천인 코드 유효성 검증만 수행
    const isValidCode = validateReferralCode(referralCode);
    if (!isValidCode) {
      throw new Error('유효하지 않은 추천인 코드입니다.');
    }
    
    // 2. 발급자 지갑 주소 확인
    const referrerWallet = getGeneratorAddressByCode(referralCode);
    if (!referrerWallet) {
      throw new Error('추천인 지갑 주소를 찾을 수 없습니다.');
    }
    
    // 3. 인증 성공 메시지만 표시 (보상 지급은 지갑 생성 후)
    console.log('✅ 유저용 추천인 코드 인증 성공:', { referralCode, referrerWallet });
    
    return { 
      success: true, 
      message: '추천인 코드 인증 성공! 지갑 생성 후 보상이 지급됩니다.',
      referralCode,
      referrerWallet
    };
    
  } catch (error) {
    console.error('❌ 유저용 추천인 코드 인증 실패:', error);
    return { success: false, error: error.message };
  }
};

### **2. 지갑 생성 완료 후 실제 보상 지급**

```typescript
// ==================== 지갑 생성 완료 후 추천 보상 지급 ====================
const handleWalletCreationComplete = async (newWalletAddress: string, referralCode?: string) => {
  try {
    console.log('🎯 지갑 생성 완료, 추천 보상 지급 시작:', { newWalletAddress, referralCode });
    
    if (referralCode) {
      // 1. 추천인 코드로 가입한 경우
      const referrerWallet = getGeneratorAddressByCode(referralCode);
      if (referrerWallet) {
        // 2. 가입자 즉시 보상 지급 (1BW)
        await giveInstantBonus(newWalletAddress, 1.0);
        
        // 3. 가입자 영구 보너스율 적용 (1.2%)
        await applyPermanentBonus(newWalletAddress, 0.012);
        
        // 4. 발급자 추가 보상 지급 (1BW + 1.2%)
        await giveInstantBonus(referrerWallet, 1.0);
        await applyPermanentBonus(referrerWallet, 0.012);
        
        // 5. 추천인 수 증가
        await incrementReferralCount(referrerWallet);
        
        console.log('✅ 추천인 코드 가입 시 즉시 보상 지급 완료');
        alert(`✅ 추천인 코드 가입 완료!\n\n보상 지급:\n- 추천인: 1 BW + 1.2% 보너스\n- 가입자: 1 BW + 1.2% 보너스`);
      }
    }
    
    return { success: true };
    
  } catch (error) {
    console.error('❌ 지갑 생성 완료 후 보상 지급 실패:', error);
    return { success: false, error: error.message };
  }
};
```
