## **2️⃣ 유틸리티 함수 수정**

```typescript
// 🎯 정밀도 계산 유틸리티 함수들 (50자리 정밀도 기반)
const precisionUtils = {
  // 소수점 자릿수 제한 (50자리 정밀도 기반)
  roundToPrecision: (value: number, precision: number = PRECISION_CONFIG.DISPLAY_PRECISION): number => {
    return new Decimal(value).toDecimalPlaces(precision).toNumber();
  },
  
  // 백분율 계산 정밀도 보장 (50자리 정밀도 기반)
  calculatePercentage: (value: number, total: number, precision: number = PRECISION_CONFIG.PERCENTAGE_PRECISION): number => {
    const percentage = new Decimal(value).dividedBy(total).times(100);
    return percentage.toDecimalPlaces(precision).toNumber();
  },
  
  // 보너스율 계산 정밀도 보장 (50자리 정밀도 기반)
  calculateBonusRate: (baseRate: number, bonusPercentage: number, precision: number = PRECISION_CONFIG.DISPLAY_PRECISION): number => {
    const bonusRate = new Decimal(baseRate).times(bonusPercentage).dividedBy(100);
    return bonusRate.toDecimalPlaces(precision).toNumber();
  },
  
  // 시간당 보상 계산 정밀도 보장 (50자리 정밀도 기반)
  calculateHourlyReward: (baseRate: number, totalBonusRate: number, precision: number = PRECISION_CONFIG.DISPLAY_PRECISION): number => {
    const decimalBaseRate = new Decimal(baseRate);
    const decimalOne = new Decimal(1);
    const decimalTotalBonus = new Decimal(totalBonusRate);
    
    const decimalEnhancedRate = decimalOne.plus(decimalTotalBonus).times(decimalBaseRate);
    return decimalEnhancedRate.toDecimalPlaces(precision).toNumber();
  },
  
  // 일일 보상 계산 정밀도 보장 (50자리 정밀도 기반)
  calculateDailyReward: (hourlyReward: number, precision: number = PRECISION_CONFIG.DISPLAY_PRECISION): number => {
    const dailyReward = new Decimal(hourlyReward).times(24);
    return dailyReward.toDecimalPlaces(precision).toNumber();
  },
  
  // 정확한 곱셈 연산 (50자리 정밀도 기반)
  multiply: (a: number, b: number, precision: number = PRECISION_CONFIG.DISPLAY_PRECISION): number => {
    return new Decimal(a).times(b).toDecimalPlaces(precision).toNumber();
  },
  
  // 정확한 덧셈 연산 (50자리 정밀도 기반)
  add: (a: number, b: number, precision: number = PRECISION_CONFIG.DISPLAY_PRECISION): number => {
    return new Decimal(a).plus(b).toDecimalPlaces(precision).toNumber();
  },
  
  // 정확한 나눗셈 연산 (50자리 정밀도 기반)
  divide: (a: number, b: number, precision: number = PRECISION_CONFIG.DISPLAY_PRECISION): number => {
    if (b === 0) throw new Error('Division by zero');
    return new Decimal(a).dividedBy(b).toDecimalPlaces(precision).toNumber();
  },
  
  // 정확한 뺄셈 연산 (50자리 정밀도 기반)
  subtract: (a: number, b: number, precision: number = PRECISION_CONFIG.DISPLAY_PRECISION): number => {
    return new Decimal(a).minus(b).toDecimalPlaces(precision).toNumber();
  },
  
  // 절댓값 계산 (50자리 정밀도 기반)
  abs: (value: number, precision: number = PRECISION_CONFIG.DISPLAY_PRECISION): number => {
    return new Decimal(value).abs().toDecimalPlaces(precision).toNumber();
  },
  
  // 최대값 계산 (50자리 정밀도 기반)
  max: (a: number, b: number, precision: number = PRECISION_CONFIG.DISPLAY_PRECISION): number => {
    return new Decimal(a).greaterThan(b) ? new Decimal(a).toDecimalPlaces(precision).toNumber() : new Decimal(b).toDecimalPlaces(precision).toNumber();
  },
  
  // 최소값 계산 (50자리 정밀도 기반)
  min: (a: number, b: number, precision: number = PRECISION_CONFIG.DISPLAY_PRECISION): number => {
    return new Decimal(a).lessThan(b) ? new Decimal(a).toDecimalPlaces(precision).toNumber() : new Decimal(b).toDecimalPlaces(precision).toNumber();
  }
};
```

## **5️⃣ 누적 보상 계산 함수 수정**

```typescript
//    통합된 50자리 정밀도 기반 누적 보상 계산 함수
const calculateAccumulatedRewardDecimal = (): {
  accumulatedReward: Decimal;
  elapsedHours: Decimal;
  totalEnhancedRate: Decimal;
  lockupVaultReward: Decimal;
  referralVaultReward: Decimal;
  totalAccumulatedReward: Decimal;
  precision: number;
} => {
  if (!miningStartTime) {
    return {
      accumulatedReward: new Decimal(0),
      elapsedHours: new Decimal(0),
      totalEnhancedRate: new Decimal(0),
      lockupVaultReward: new Decimal(0),
      referralVaultReward: new Decimal(0),
      totalAccumulatedReward: new Decimal(0),
      precision: PRECISION_CONFIG.DISPLAY_PRECISION
    };
  }
  
  const currentTime = new Date();
  const totalElapsedTime = currentTime.getTime() - miningStartTime.getTime();
  
  // 🎯 50자리 정밀도 기반 정확한 시간 계산
  const decimalTotalElapsedTime = new Decimal(totalElapsedTime);
  const decimalMillisecondsPerHour = new Decimal(1000 * 60 * 60);
  const decimalElapsedHours = decimalTotalElapsedTime.dividedBy(decimalMillisecondsPerHour);
  
  // 🎯 50자리 정밀도 기반 보너스 계산
  const rewardData = calculateRealTimeMiningRewardDecimal();
  const decimalTotalEnhancedRate = new Decimal(rewardData.totalEnhancedRate);
  
  // 🎯 50자리 정밀도 기반 누적 보상 계산
  const decimalAccumulatedReward = decimalElapsedHours.times(decimalTotalEnhancedRate);
  
  // 기본 락업 보관함 계산 (50자리 정밀도)
  let decimalLockupVaultReward = new Decimal(0);
  if (isLockupRegistered && isMiningActive) {
    const baseRate = new Decimal(0.25);
    const attendanceBonus = (consecutiveAttendanceDays > 0 && isAttendanceBonusTimeValid()) ? 
      new Decimal(calculateAttendanceBonus(consecutiveAttendanceDays).bonusRate) : 
      new Decimal(0);
    const referralBonus = new Decimal(referralSystemState.totalBonusRate || 0).dividedBy(100);
    const lockupBonus = new Decimal(calculateLockupBonus(lockupRatio, lockupMonths).totalAPY).dividedBy(100);
    
    const enhancedRate = baseRate
      .times(new Decimal(1).plus(attendanceBonus))
      .times(new Decimal(1).plus(referralBonus))
      .times(new Decimal(1).plus(lockupBonus));
    
    decimalLockupVaultReward = enhancedRate.times(decimalElapsedHours);
  }
  
  // 추천 보관 BW 계산 (50자리 정밀도)
  const decimalReferralVaultReward = new Decimal(calculateReferralStorageBW(referralCount));
  
  // 총 누적 보상 계산 (50자리 정밀도)
  const decimalTotalAccumulatedReward = decimalAccumulatedReward
    .plus(decimalLockupVaultReward)
    .plus(decimalReferralVaultReward);
  
  return {
    accumulatedReward: decimalAccumulatedReward.toDecimalPlaces(PRECISION_CONFIG.DISPLAY_PRECISION),
    elapsedHours: decimalElapsedHours.toDecimalPlaces(PRECISION_CONFIG.DISPLAY_PRECISION),
```

## **6️⃣ 정밀도 검증 시스템 수정**

```typescript
//    50자리 정밀도 검증 시스템
const precisionValidation = {
  // 정밀도 검증 함수 (50자리 기준)
  validatePrecision: (expected: number, actual: number, tolerance: number = 0.000000000000000001): boolean => {
    const decimalExpected = new Decimal(expected);
    const decimalActual = new Decimal(actual);
    const decimalTolerance = new Decimal(tolerance);
    const decimalError = decimalExpected.minus(decimalActual).abs();
    return decimalError.lte(decimalTolerance);
  },
  
  // 마이닝 계산 정밀도 검증 (50자리 기준)
  validateMiningCalculation: (): boolean => {
    const rewardData = calculateRealTimeMiningRewardDecimal();
    
    // 시간당 보상 검증 (50자리 정밀도)
    const expectedHourly = new Decimal(0.25).times(new Decimal(1).plus(
      new Decimal(rewardData.attendanceBonus)
        .plus(rewardData.referralBonus)
        .plus(rewardData.lockupBonus)
        .plus(rewardData.merchantBonus)
    ));
    const hourlyValid = precisionValidation.validatePrecision(
      expectedHourly.toDecimalPlaces(PRECISION_CONFIG.DISPLAY_PRECISION).toNumber(),
      rewardData.hourlyReward
    );
    
    // 일일 보상 검증 (50자리 정밀도)
    const expectedDaily = expectedHourly.times(24);
    const dailyValid = precisionValidation.validatePrecision(
      expectedDaily.toDecimalPlaces(PRECISION_CONFIG.DISPLAY_PRECISION).toNumber(),
      rewardData.dailyReward
    );
    
    return hourlyValid && dailyValid;
  },
  
  // 부동소수점 오차 분석 (50자리 기준)
  analyzeFloatingPointError: (value: number, precision: number = PRECISION_CONFIG.DISPLAY_PRECISION): void => {
    const decimalValue = new Decimal(value);
    const floatValue = parseFloat(value.toString());
    const error = decimalValue.minus(floatValue).abs();
    const relativeError = error.dividedBy(decimalValue).times(100);
    
    console.log(`🔍 50자리 정밀도 부동소수점 오차 분석 (${value}):`, {
      decimalValue: decimalValue.toFixed(PRECISION_CONFIG.DISPLAY_PRECISION),
      floatValue: floatValue.toFixed(PRECISION_CONFIG.DISPLAY_PRECISION),
      absoluteError: error.toFixed(PRECISION_CONFIG.DISPLAY_PRECISION),
      relativeError: relativeError.toFixed(PRECISION_CONFIG.PERCENTAGE_PRECISION) + '%',
      precision: PRECISION_CONFIG.DISPLAY_PRECISION
    });
  }
```
