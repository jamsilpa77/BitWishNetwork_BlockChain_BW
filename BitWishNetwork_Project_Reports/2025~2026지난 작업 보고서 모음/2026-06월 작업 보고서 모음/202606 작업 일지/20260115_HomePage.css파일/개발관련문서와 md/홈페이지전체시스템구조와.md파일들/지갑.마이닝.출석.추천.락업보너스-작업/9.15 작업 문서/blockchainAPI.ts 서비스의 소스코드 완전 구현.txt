export class BlockchainAPI {
  private baseURL: string;
  private apiKey: string;
  private timeout: number;

  constructor(baseURL: string = 'http://localhost:3000/api', apiKey?: string, timeout: number = 10000) {
    this.baseURL = baseURL;
    this.apiKey = apiKey || '';
    this.timeout = timeout;
  }

  // ====================================================================================
  // 🎯 HTTP 요청 헬퍼
  // ====================================================================================

  private async makeRequest<T>(
    endpoint: string,
    options: RequestInit = {}
  ): Promise<APIResponse<T>> {
    const url = `${this.baseURL}${endpoint}`;
    const headers: HeadersInit = {
      'Content-Type': 'application/json',
      ...options.headers,
    };

    if (this.apiKey) {
      headers['Authorization'] = `Bearer ${this.apiKey}`;
    }

    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), this.timeout);

    try {
      const response = await fetch(url, {
        ...options,
        headers,
        signal: controller.signal,
      });

      clearTimeout(timeoutId);

      if (!response.ok) {
        throw new Error(`HTTP ${response.status}: ${response.statusText}`);
      }

      const data = await response.json();
      return {
        success: true,
        data,
        timestamp: new Date().toISOString(),
      };
    } catch (error) {
      clearTimeout(timeoutId);
      return {
        success: false,
        error: error instanceof Error ? error.message : 'Unknown error',
        timestamp: new Date().toISOString(),
      };
    }
  }

  // ====================================================================================
  // 🎯 블록체인 상태 조회
  // ====================================================================================

  async getBlockchainStatus(): Promise<APIResponse<BlockchainStatus>> {
    return this.makeRequest<BlockchainStatus>('/blockchain/status');
  }

  async getNetworkStats(): Promise<APIResponse<NetworkStats>> {
    return this.makeRequest<NetworkStats>('/blockchain/network-stats');
  }

  async getMiningInfo(): Promise<APIResponse<MiningInfo>> {

  }

  // ====================================================================================
  // 🎯 블록 관련 API
  // ====================================================================================

  async getLatestBlock(): Promise<APIResponse<Block>> {
    return this.makeRequest<Block>('/blockchain/latest-block');
  }

    return this.makeRequest<Block>(`/blockchain/block/${height}`);
  }

  async getBlockByHash(hash: string): Promise<APIResponse<Block>> {
    return this.makeRequest<Block>(`/blockchain/block/hash/${hash}`);
  }

  async getBlocks(
    page: number = 1,
    limit: number = 20,
    sort: 'asc' | 'desc' = 'desc'
  ): Promise<APIResponse<{ blocks: Block[]; total: number; page: number; limit: number }>> {
    return this.makeRequest(`/blockchain/blocks?page=${page}&limit=${limit}&sort=${sort}`);
  }

  // ====================================================================================
  // 🎯 트랜잭션 관련 API
  // ====================================================================================

  async getTransaction(txId: string): Promise<APIResponse<Transaction>> {
    return this.makeRequest<Transaction>(`/blockchain/transaction/${txId}`);
  }

  async getTransactionsByAddress(
    address: string,

    limit: number = 20
  ): Promise<APIResponse<{ transactions: Transaction[]; total: number; page: number; limit: number }>> {
    return this.makeRequest(`/blockchain/transactions/address/${address}?page=${page}&limit=${limit}`);
  }

  async getTransactionsByBlock(
    blockHeight: number,

  ): Promise<APIResponse<{ transactions: Transaction[]; total: number; page: number; limit: number }>> {

  }

  async getPendingTransactions(): Promise<APIResponse<Transaction[]>> {
    return this.makeRequest<Transaction[]>('/blockchain/transactions/pending');

  async subscribeToBlocks(callback: (block: Block) => void): Promise<EventSource> {
    const eventSource = new EventSource(`${this.baseURL}/blockchain/stream/blocks`);
    
    eventSource.onmessage = (event) => {
      try {
        const block: Block = JSON.parse(event.data);
        callback(block);
      } catch (error) {
        console.error('Error parsing block data:', error);

    };

    eventSource.onerror = (error) => {
      console.error('EventSource error:', error);

    return eventSource;
  }

  async subscribeToTransactions(callback: (transaction: Transaction) => void): Promise<EventSource> {
    const eventSource = new EventSource(`${this.baseURL}/blockchain/stream/transactions`);
    
    eventSource.onmessage = (event) => {
      try {
        const transaction: Transaction = JSON.parse(event.data);
        callback(transaction);
      } catch (error) {
        console.error('Error parsing transaction data:', error);

    eventSource.onerror = (error) => {
      console.error('EventSource error:', error);

    return eventSource;
  }

  async subscribeToWallet(address: string, callback: (data: { balance: WalletBalance; transaction: Transaction }) => void): Promise<EventSource> {
    const eventSource = new EventSource(`${this.baseURL}/blockchain/stream/wallet/${address}`);
    
    eventSource.onmessage = (event) => {
      try {
        const data = JSON.parse(event.data);
        callback(data);

        console.error('Error parsing wallet data:', error);

    return this.makeRequest<{ fee: number; priority: 'low' | 'medium' | 'high' }>('/blockchain/estimate-fee', {

  }

  // ====================================================================================
  // 🎯 에러 처리 및 재시도
  // ====================================================================================

  private async retryRequest<T>(
    requestFn: () => Promise<APIResponse<T>>,
    maxRetries: number = 3,
    delay: number = 1000
  ): Promise<APIResponse<T>> {
    let lastError: Error | null = null;

    for (let i = 0; i < maxRetries; i++) {
      try {
        const result = await requestFn();
        if (result.success) {
          return result;
        }
        lastError = new Error(result.error || 'Request failed');
      } catch (error) {
        lastError = error instanceof Error ? error : new Error('Unknown error');
      }

      if (i < maxRetries - 1) {
        await new Promise(resolve => setTimeout(resolve, delay * Math.pow(2, i)));
      }
    }

    return {
      success: false,
      error: lastError?.message || 'Max retries exceeded',
      timestamp: new Date().toISOString(),
    };
  }

  // ====================================================================================
  // 🎯 설정 업데이트
  // ====================================================================================

  updateBaseURL(newBaseURL: string): void {
    this.baseURL = newBaseURL;
  }

  updateAPIKey(newAPIKey: string): void {
    this.apiKey = newAPIKey;
  }

  updateTimeout(newTimeout: number): void {
    this.timeout = newTimeout;
  }
}

// 🎯 편의 함수들
// ====================================================================================

export const formatBalance = (balance: number, decimals: number = 8): string => {
  return new Decimal(balance).toFixed(decimals);
};

export const formatAddress = (address: string, start: number = 6, end: number = 4): string => {
  if (address.length <= start + end) return address;
  return `${address.slice(0, start)}...${address.slice(-end)}`;
};

export const formatTransactionHash = (hash: string, start: number = 8, end: number = 8): string => {
  if (hash.length <= start + end) return hash;
  return `${hash.slice(0, start)}...${hash.slice(-end)}`;
};

export const calculateTransactionFee = (amount: number, feeRate: number = 0.001): number => {
  return new Decimal(amount).mul(feeRate).toNumber();
};

export const validateTransactionAmount = (amount: number, balance: number): boolean => {
  return new Decimal(amount).greaterThan(0) && new Decimal(amount).lessThanOrEqualTo(balance);
};

// ====================================================================================
// 🎯 기본 내보내기
// ====================================================================================

export default BlockchainAPI;
