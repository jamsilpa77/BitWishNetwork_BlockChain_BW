```typescript
// 유저 전용 유틸리티 함수들
import { MINING_CONSTANTS } from './constants';

export const generateUniqueReferralCode = (): string => {
  const timestamp = Date.now().toString(36).toUpperCase();
  const random = Math.random().toString(36).substr(2, 4).toUpperCase();
  return `REF${timestamp}${random}`;
};

export const calculateConsecutiveDays = (attendanceCalendar: {[key: string]: boolean}): number => {
  const today = new Date();
  let consecutiveDays = 0;
  
  for (let i = 0; i < 365; i++) {
    const checkDate = new Date(today);
    checkDate.setDate(today.getDate() - i);
    const dateKey = checkDate.toISOString().split('T')[0];
    
    if (attendanceCalendar[dateKey]) {
      consecutiveDays++;
    } else {
      break;
    }
  }
  
  return consecutiveDays;
};

export const calculateAttendanceBonus = (consecutiveDays: number): number => {
  const rates = MINING_CONSTANTS.ATTENDANCE_BONUS_RATES;
  
  if (consecutiveDays >= 30) return rates[30];
  if (consecutiveDays >= 29) return rates[29];
  if (consecutiveDays >= 22) return rates[28];
  if (consecutiveDays >= 15) return rates[21];
  if (consecutiveDays >= 8) return rates[14];
  if (consecutiveDays >= 1) return rates[7];
  
  return 0;
};

export const calculateReferralBonus = (referralCount: number): number => {
  const permanentBonus = referralCount * MINING_CONSTANTS.REFERRAL_BONUS_RATES.PERMANENT;
  const segmentRates = MINING_CONSTANTS.REFERRAL_BONUS_RATES.SEGMENTS;
  
  let segmentBonus = 0;
  if (referralCount <= 30) {
    segmentBonus = referralCount * segmentRates[30];
  } else if (referralCount <= 60) {
    segmentBonus = 30 * segmentRates[30] + (referralCount - 30) * segmentRates[60];
  } else if (referralCount <= 90) {
    segmentBonus = 30 * segmentRates[30] + 30 * segmentRates[60] + (referralCount - 60) * segmentRates[90];
  } else {
    segmentBonus = 30 * segmentRates[30] + 30 * segmentRates[60] + 30 * segmentRates[90] + (referralCount - 90) * segmentRates.MAX;
  }
  
  return permanentBonus + segmentBonus;
};

export const calculateLockupBonus = (lockupRatio: number, lockupMonths: number): number => {
  const apy = MINING_CONSTANTS.LOCKUP_APY[lockupMonths as keyof typeof MINING_CONSTANTS.LOCKUP_APY];
  if (!apy) return 0;
  
  const monthlyRate = apy / 12;
  return (lockupRatio / 100) * monthlyRate;
};
```

```typescript
// 저장소 관리 서비스
import { STORAGE_KEYS } from '../utils/constants';

export class StorageService {
  static saveUserData(walletAddress: string, data: any): void {
    try {
      const key = `${STORAGE_KEYS.USER_DATA}${walletAddress}`;
      localStorage.setItem(key, JSON.stringify(data));
    } catch (error) {
      console.error('❌ 유저 데이터 저장 오류:', error);
      throw error;
    }
  }
  
  static loadUserData(walletAddress: string): any | null {
    try {
      const key = `${STORAGE_KEYS.USER_DATA}${walletAddress}`;
      const data = localStorage.getItem(key);
      return data ? JSON.parse(data) : null;
    } catch (error) {
      console.error('❌ 유저 데이터 로드 오류:', error);
      return null;
    }
  }
  
  static saveMiningState(data: any): void {
    try {
      localStorage.setItem(STORAGE_KEYS.MINING_STATE, JSON.stringify(data));
    } catch (error) {
      console.error('❌ 마이닝 상태 저장 오류:', error);
      throw error;
    }
  }
  
  static loadMiningState(): any | null {
    try {
      const data = localStorage.getItem(STORAGE_KEYS.MINING_STATE);
      return data ? JSON.parse(data) : null;
    } catch (error) {
      console.error('❌ 마이닝 상태 로드 오류:', error);
      return null;
    }
  }
  
  static clearAllData(): void {
    try {
      Object.values(STORAGE_KEYS).forEach(key => {
        if (key.endsWith('_')) {
          // 유저 데이터 키들 삭제
          for (let i = 0; i < localStorage.length; i++) {
            const storageKey = localStorage.key(i);
            if (storageKey && storageKey.startsWith(key)) {
              localStorage.removeItem(storageKey);
            }
          }
        } else {
          localStorage.removeItem(key);
        }
      });
    } catch (error) {
      console.error('❌ 전체 데이터 삭제 오류:', error);
      throw error;
    }
  }
  
  static getAllUserKeys(): string[] {
    const keys: string[] = [];
    for (let i = 0; i < localStorage.length; i++) {
      const key = localStorage.key(i);
      if (key && key.startsWith(STORAGE_KEYS.USER_DATA)) {
        keys.push(key);
      }
    }
    return keys;
  }
}
```

```typescript
      setUserData(prev => prev ? { ...prev, ...updates } : null);
    } catch (error) {
      console.error('❌ 유저 데이터 업데이트 오류:', error);
      setError('데이터 업데이트 중 오류가 발생했습니다.');
    }
  }, [walletAddress]);
  
  const refreshUserData = useCallback(() => {
    try {
      const data = UserService.loadUserData(walletAddress);
      if (data) {
        setUserData(data);
      } else {
        // 새 유저 데이터 초기화
        const newData = UserService.initializeUserData(walletAddress, getClientIP());
        setUserData(newData);
      }
      setError(null);
    } catch (error) {
      console.error('❌ 유저 데이터 로드 오류:', error);
      setError('데이터 로드 중 오류가 발생했습니다.');
    }
  }, [walletAddress]);
  
  useEffect(() => {
    refreshUserData();
    setIsLoading(false);
  }, [walletAddress, refreshUserData]);
  
  return { 
    userData, 
    updateUserData, 
    refreshUserData, 
    isLoading, 
    error 
  };
};

const getClientIP = (): string => {
  // 클라이언트 IP 주소 가져오기 (실제 구현에서는 API 호출)
  return '127.0.0.1';
};
```

**2-2-3. `hooks/useMining.ts`**
```typescript
// 마이닝 관련 훅
import { useState, useEffect, useCallback } from 'react';
import { UserMiningData } from '../types/user';
import { MiningRewardData } from '../types/mining';
import { calculateAttendanceBonus, calculateReferralBonus, calculateLockupBonus } from '../utils/userUtils';
import { MINING_CONSTANTS } from '../utils/constants';

export const useMining = (userData: UserMiningData | null) => {
  const [miningReward, setMiningReward] = useState<MiningRewardData | null>(null);
  const [tick, setTick] = useState(0);
  
  const calculateRealTimeReward = useCallback((): MiningRewardData => {
    if (!userData || !userData.isMiningActive || !userData.miningStartTime) {
      return {
        baseRate: MINING_CONSTANTS.BASE_RATE,
        attendanceBonus: 0,
        referralBonus: 0,
        lockupBonus: 0,
        merchantBonus: 0,
        totalBonusRate: 1,
        realTimeReward: 0,
        lastUpdate: new Date()
      };
    }
    
    const now = new Date();
    const startTime = new Date(userData.miningStartTime);
    const elapsedSeconds = Math.floor((now.getTime() - startTime.getTime()) / 1000);
    
    const baseReward = elapsedSeconds * MINING_CONSTANTS.BASE_RATE;
    
    const attendanceBonus = calculateAttendanceBonus(userData.consecutiveAttendanceDays);
    const referralBonus = calculateReferralBonus(userData.referralCount);
    const lockupBonus = calculateLockupBonus(userData.lockupRatio, userData.lockupMonths);
    const merchantBonus = userData.isMerchantRegistered ? MINING_CONSTANTS.MERCHANT_BONUS : 0;
    
    const totalBonusRate = 1 + attendanceBonus + referralBonus + lockupBonus + merchantBonus;
    const realTimeReward = baseReward * totalBonusRate;
    
    return {
      baseRate: MINING_CONSTANTS.BASE_RATE,
      attendanceBonus,
      referralBonus,
      lockupBonus,
      merchantBonus,
      totalBonusRate,
      realTimeReward,
      lastUpdate: new Date()
    };
  }, [userData]);
  
  useEffect(() => {
    const interval = setInterval(() => {
      setTick(prev => prev + 1);
    }, 1000);
    
    return () => clearInterval(interval);
  }, []);
  
  useEffect(() => {
    const reward = calculateRealTimeReward();
    setMiningReward(reward);
  }, [userData, tick, calculateRealTimeReward]);
  
  const startMining = useCallback(() => {
    if (userData) {
      UserService.startMining(userData.walletAddress);
    }
  }, [userData]);
  
  const stopMining = useCallback(() => {
    if (userData) {
      UserService.stopMining(userData.walletAddress);
    }
  }, [userData]);
  
  return { 
    miningReward, 
    startMining, 
    stopMining, 
    isMiningActive: userData?.isMiningActive || false 
  };
};
```

```typescript
// 에러 바운더리 공통 컴포넌트
import React, { Component, ErrorInfo, ReactNode } from 'react';

interface Props {
  children: ReactNode;
}

interface State {
  hasError: boolean;
  error?: Error;
}

export class ErrorBoundary extends Component<Props, State> {
  constructor(props: Props) {
    super(props);
    this.state = { hasError: false };
  }

  static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error };
  }

  componentDidCatch(error: Error, errorInfo: ErrorInfo) {
    console.error('❌ 컴포넌트 오류:', error, errorInfo);
  }

  render() {
    if (this.state.hasError) {
      return (
        <div className="bg-red-100 border border-red-300 rounded-lg p-4">
          <h2 className="text-lg font-bold text-red-700 mb-2">오류가 발생했습니다</h2>
          <p className="text-red-600">페이지를 새로고침해주세요.</p>
        </div>
      );
    }

    return this.props.children;
  }
}
```

```typescript
// 유저 락업 보너스 섹션 컴포넌트
import React, { useState, useEffect } from 'react';
import { UserMiningData } from '../../types/user';
import { calculateLockupBonus } from '../../utils/userUtils';
import { MINING_CONSTANTS } from '../../utils/constants';

interface UserLockupSectionProps {
  userData: UserMiningData;
  onUpdate: (updates: Partial<UserMiningData>) => void;
}

export const UserLockupSection: React.FC<UserLockupSectionProps> = ({ userData, onUpdate }) => {
  const [basicLockupBonus, setBasicLockupBonus] = useState(0);
  const [additionalLockupBonus, setAdditionalLockupBonus] = useState(0);
  const [showLockupModal, setShowLockupModal] = useState(false);
  
  useEffect(() => {
    const basicBonus = calculateLockupBonus(userData.lockupRatio, userData.lockupMonths);
    const additionalBonus = calculateLockupBonus(userData.secondLockupRatio, userData.secondLockupMonths);
    setBasicLockupBonus(basicBonus);
    setAdditionalLockupBonus(additionalBonus);
  }, [userData.lockupRatio, userData.lockupMonths, userData.secondLockupRatio, userData.secondLockupMonths]);
  
  const handleBasicLockupRegistration = async () => {
    try {
      const response = await fetch('http://localhost:4001/lockup/create', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          walletAddress: userData.walletAddress,
          userId: userData.walletAddress,
          amount: 100,
          lockPeriod: userData.lockupMonths
        })
      });
      
      const result = await response.json();
      
      if (response.ok && result.success) {
        const apyValue = MINING_CONSTANTS.LOCKUP_APY[userData.lockupMonths as keyof typeof MINING_CONSTANTS.LOCKUP_APY];
        const startDate = new Date();
        const endDate = new Date(startDate);
        endDate.setMonth(endDate.getMonth() + userData.lockupMonths);
        
        onUpdate({
          isLockupRegistered: true,
          registeredLockupData: {
            ratio: userData.lockupRatio,
            months: userData.lockupMonths,
            isActive: true,
            apy: apyValue,
            startDate: startDate,
            endDate: endDate
          }
        });
        
        alert('✅ 기본 락업 설정 등록 완료!');
      } else {
        alert(`❌ 기본 락업 설정 실패: ${result.error || '알 수 없는 오류'}`);
      }
    } catch (error) {
      console.error('❌ 락업 설정 등록 오류:', error);
      alert('❌ 네트워크 오류가 발생했습니다.');
    }
  };
  
  const handleAdditionalLockupRegistration = async () => {
    try {
      const response = await fetch('http://localhost:4001/lockup/create', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          walletAddress: userData.walletAddress,
          userId: userData.walletAddress,
          amount: 100,
          lockPeriod: userData.secondLockupMonths
        })
      });
      
      const result = await response.json();
      
      if (response.ok && result.success) {
        const apyValue = MINING_CONSTANTS.LOCKUP_APY[userData.secondLockupMonths as keyof typeof MINING_CONSTANTS.LOCKUP_APY];
        const startDate = new Date();
        const endDate = new Date(startDate);
        endDate.setMonth(endDate.getMonth() + userData.secondLockupMonths);
        
        onUpdate({
          isSecondLockupRegistered: true,
          registeredSecondLockupData: {
            ratio: userData.secondLockupRatio,
            months: userData.secondLockupMonths,
            isActive: true,
            apy: apyValue,
            startDate: startDate,
            endDate: endDate
          }
        });
        
        alert('✅ 추가 락업 설정 등록 완료!');
      } else {
        alert(`❌ 추가 락업 설정 실패: ${result.error || '알 수 없는 오류'}`);
      }
    } catch (error) {
      console.error('❌ 추가 락업 설정 등록 오류:', error);
      alert('❌ 네트워크 오류가 발생했습니다.');
    }
  };
  
  return (
    <div className="user-lockup-section">
      <div className="bg-white border border-gray-300 rounded-lg p-6">
        <h2 className="text-xl font-bold text-gray-900 mb-4">🔒 락업 보너스</h2>
        
        {/* 기본 락업 */}
        <div className="mb-6">
          <h3 className="text-lg font-semibold text-gray-700 mb-3">기본 락업</h3>
```

```typescript
      const hours = Math.floor(elapsedMinutes / 60);
```

```typescript
import { calculateAttendanceBonus, calculateReferralBonus, calculateLockupBonus } from '../../utils/userUtils';

interface UserMiningRewardProps {
  userData: UserMiningData;
  onUpdate: (updates: Partial<UserMiningData>) => void;

  const [miningReward, setMiningReward] = useState<MiningRewardData | null>(null);
  const [tick, setTick] = useState(0);
  
  useEffect(() => {
    const interval = setInterval(() => {
      setTick(prev => prev + 1);
    }, 1000);
    
    return () => clearInterval(interval);
  }, []);
  
  useEffect(() => {
    const calculateRealTimeReward = (): MiningRewardData => {
      if (!userData.isMiningActive || !userData.miningStartTime) {
        return {
          baseRate: MINING_CONSTANTS.BASE_RATE,
          attendanceBonus: 0,
          referralBonus: 0,
          lockupBonus: 0,
          merchantBonus: 0,
          totalBonusRate: 1,
          realTimeReward: 0,
          lastUpdate: new Date()
        };

      const now = new Date();
      const startTime = new Date(userData.miningStartTime);
      const elapsedSeconds = Math.floor((now.getTime() - startTime.getTime()) / 1000);

      const totalBonusRate = 1 + attendanceBonus + referralBonus + totalLockupBonus + merchantBonus;
      const realTimeReward = baseReward * totalBonusRate;

        baseRate: MINING_CONSTANTS.BASE_RATE,

        referralBonus,
        lockupBonus: totalLockupBonus,
```

```typescript
// App.tsx - 관리자/유저 분기 로직
import React, { useState, useEffect } from 'react';
import { AdminMiningSettings } from './components/admin/AdminMiningSettings';
import { UserMiningSettings } from './components/user/UserMiningSettings';
import { LoadingSpinner } from './components/common/LoadingSpinner';
import { ErrorBoundary } from './components/common/ErrorBoundary';
import { isAdmin } from './utils/adminUtils';
import { useAdminAuth } from './hooks/useAdminAuth';

function App() {
  const [walletAddress, setWalletAddress] = useState<string>('');
  const [isLoading, setIsLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
  
  const { isAdminUser, isLoading: authLoading } = useAdminAuth(walletAddress);
  
  useEffect(() => {
    // 지갑 주소 가져오기 (실제 구현에서는 지갑 연결 로직 사용)
    const getWalletAddress = async () => {
      try {
        // 임시로 관리자 지갑 주소 사용 (실제로는 사용자 지갑 주소)
        const tempAddress = "GDG3OPHDK5WGTAG2UVEMDUUGMGIBSZMPF4RAUPVRQMG7PRNAZT5YU256";
        setWalletAddress(tempAddress);
        setIsLoading(false);
      } catch (error) {
        console.error('❌ 지갑 주소 가져오기 오류:', error);
        setError('지갑 연결에 실패했습니다.');
        setIsLoading(false);
      }
    };
    
    getWalletAddress();
  }, []);
  
  const renderMiningSettings = () => {
    if (isLoading || authLoading) {
      return <LoadingSpinner message="시스템을 초기화하는 중..." />;
    }
    
    if (error) {
      return (
        <div className="bg-red-100 border border-red-300 rounded-lg p-4">
          <h3 className="text-lg font-bold text-red-700 mb-2">오류 발생</h3>
          <p className="text-red-600">{error}</p>
        </div>
      );
    }
    
    if (!walletAddress) {
      return (
        <div className="bg-yellow-100 border border-yellow-300 rounded-lg p-4">
          <h3 className="text-lg font-bold text-yellow-700 mb-2">지갑 연결 필요</h3>
          <p className="text-yellow-600">지갑을 연결해주세요.</p>
        </div>
      );
    }
    
    if (isAdminUser) {
      return <AdminMiningSettings walletAddress={walletAddress} />;
    } else {
      return <UserMiningSettings walletAddress={walletAddress} />;
    }
  };
  
  return (
    <ErrorBoundary>
      <div className="min-h-screen bg-gray-50">
        <div className="container mx-auto px-4 py-8">
          {renderMiningSettings()}
        </div>
      </div>
    </ErrorBoundary>
  );
}

export default App;
```
