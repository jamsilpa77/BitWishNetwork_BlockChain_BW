// 🎯 설정 검증 함수
// ====================================================================================

export const validateConfig = (): { isValid: boolean; errors: string[] } => {
  const errors: string[] = [];
  
  // 토큰 설정 검증
  if (BW_TOKEN_CONFIG.totalSupply.lte(0)) {
    errors.push('Total supply must be greater than 0');
  }
  
  if (BW_TOKEN_CONFIG.decimals < 0 || BW_TOKEN_CONFIG.decimals > 18) {
    errors.push('Decimals must be between 0 and 18');
  }
  
  // 마이닝 설정 검증
  if (MINING_CONFIG.baseReward.lte(0)) {
    errors.push('Base mining reward must be greater than 0');
  }
  
  if (MINING_CONFIG.blockTime <= 0) {
    errors.push('Block time must be greater than 0');
  }
  
  // 보너스 설정 검증
  if (BONUS_CONFIG.attendance.baseBonus.lt(0)) {
    errors.push('Attendance bonus must be non-negative');
  }
  
  if (BONUS_CONFIG.referral.commissionRate.gt(1)) {
    errors.push('Referral commission rate must not exceed 100%');
  }
  
  return {
    isValid: errors.length === 0,
    errors
  };
};
