### ✅ **완벽한 함수 분리 구현**

**유저용 락업 등록 함수:**
```typescript
const handleUserLockupRegistration = async () => {
  try {
    if (!authenticatedWalletAddress) {
      setSuccessMessage('❌ 지갑 주소가 필요합니다');
      setShowSuccessModal(true);
      return;
    }

    if (userLockupRatio === 0 || userLockupMonths === 0) {
      setSuccessMessage('❌ 락업 비율과 기간을 선택해주세요');
      setShowSuccessModal(true);
      return;
    }

    const response = await fetch('http://localhost:4001/lockup/create', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        walletAddress: authenticatedWalletAddress,
        userId: authenticatedWalletAddress,
        amount: 1000,
        lockPeriod: userLockupMonths
      })
    });

    const result = await response.json();
    
    if (response.ok && result.success) {
      // 성공 시 등록 완료 처리
      setIsUserLockupRegistered(true);
      setRegisteredUserLockupData({
        ratio: userLockupRatio,
        months: userLockupMonths,
        apy: calculateUserLockupBonus(userLockupRatio, userLockupMonths).totalAPY,
        startDate: new Date(),
        endDate: new Date(Date.now() + userLockupMonths * 30 * 24 * 60 * 60 * 1000)
      });
      
      setSuccessMessage('✅ 유저용 기본 락업 설정 등록 완료!');
      setShowSuccessModal(true);
      saveUserLockupState();
    } else {
      setSuccessMessage(`❌ 유저용 기본 락업 설정 오류: ${result.message || '알 수 없는 오류'}`);
      setShowSuccessModal(true);
    }
  } catch (error) {
    console.error('유저용 기본 락업 설정 오류:', error);
    setSuccessMessage(`❌ 유저용 기본 락업 설정 오류: ${error instanceof Error ? error.message : '알 수 없는 오류'}`);
    setShowSuccessModal(true);
  }
};
```
