```typescript
// ==================== 유저용 추천인 코드 입력 시 상호간 인증 ====================
// 기존 추천인 코드 입력 UI에 추가
const handleUserReferralCodeSubmit = async (referralCode: string) => {
  try {
    console.log('   유저용 추천인 코드 입력 처리 시작:', referralCode);
    
    // 추천인 코드 검증
    const validationResult = await validateReferralCode(referralCode);
    
    if (validationResult.isValid) {
      // 상호간 인증 및 보상 지급 처리
      const joinResult = await handleUserReferralJoin(referralCode, authenticatedWalletAddress);
      
      if (joinResult.success) {
        // 성공 시 상태 업데이트
        setReferralSystemState({
          referralCode: referralCode,
          referrerAddress: validationResult.referrerAddress,
          referralBonusApplied: true,
          bonusRate: validationResult.bonusRate
        });
        
        // 추천인 수 증가 (유저용)
        setUserReferralCount(prev => prev + 1);
        
        // 보너스율 업데이트 (유저용)
        const newBonus = calculateUserReferralBonus(userReferralCount + 1);
        setUserReferralBonus(newBonus);
        
        alert(`✅ 유저용 추천인 가입 성공!\n\n보상 지급:\n- 추천인: 1 BW + 1.2% 보너스\n- 가입자: 1 BW + 1.2% 보너스`);
        
        console.log('✅ 유저용 추천인 가입 성공:', joinResult.data);
      }
    } else {
      alert(`❌ 유저용 추천인 코드 오류: ${validationResult.errorMessage}`);
    }
  } catch (error) {
    console.error('❌ 유저용 추천인 코드 입력 처리 실패:', error);
    alert('❌ 추천인 코드 처리에 실패했습니다. 다시 시도해주세요.');
  }
};

// ==================== 유저용 추천인 코드 입력 UI 업데이트 ====================
// 기존 추천인 코드 입력 폼에 추가
<div className="space-y-4">
  <div>
    <label className="block text-sm font-medium text-gray-700 mb-2">
      유저용 추천인 코드 입력
    </label>
    <input
      type="text"
      value={referralCodeInput}
      onChange={(e) => setReferralCodeInput(e.target.value)}
      placeholder="추천인 코드를 입력하세요"
      className="w-full px-3 py-2 border border-gray-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-blue-500"
    />
  </div>
  
  <button
    onClick={() => handleUserReferralCodeSubmit(referralCodeInput)}
    disabled={!referralCodeInput.trim()}
    className={`w-full py-2 px-4 rounded-lg font-semibold transition-colors ${
      referralCodeInput.trim()
        ? 'bg-blue-600 hover:bg-blue-700 text-white'
        : 'bg-gray-400 text-gray-200 cursor-not-allowed'
    }`}
  >
    {referralCodeLoading ? '처리 중...' : '유저용 추천인 코드 인증'}
  </button>
</div>
```

```typescript
        });
      }, 1000);
    });
  },
  
  // 추천인 가입 처리 모킹

      setTimeout(() => {
        resolve({
          success: true,
          message: '유저용 추천인 가입 성공',
          rewards: {
            referrerInstantBonus: 1.0,
            newUserInstantBonus: 1.0,

    });
  }

// ==================== 유저용 API 호출 함수 업데이트 ====================
const generateUserReferralCode = async (walletAddress: string): Promise<string> => {
  try {
    console.log('   유저용 추천인 코드 생성 시작:', walletAddress);
    
    // 모킹 API 사용 (실제 구현 시 백엔드 API로 교체)
    const data = await mockUserReferralAPI.generateUserReferralCode(walletAddress);
    
    if (data.success && data.referralCode) {
      console.log('✅ 유저용 추천인 코드 생성 성공:', data.referralCode);
      return data.referralCode;
    } else {
      throw new Error(data.message || '추천인 코드 생성 실패');
    }
  } catch (error) {
    console.error('❌ 유저용 추천인 코드 생성 실패:', error);
    throw error;
  }
};

const handleUserReferralJoin = async (referralCode: string, newWalletAddress: string) => {
  try {
    console.log('🎯 유저용 추천인 가입 처리 시작:', { referralCode, newWalletAddress });
    
    // 모킹 API 사용 (실제 구현 시 백엔드 API로 교체)
    const data = await mockUserReferralAPI.joinUserReferral(referralCode, newWalletAddress);
    
    if (data.success) {
      console.log('✅ 유저용 추천인 가입 성공:', data);
      
      // 보상 지급 확인
      if (data.rewards) {
        alert(`✅ 유저용 추천인 가입 성공!\n\n보상 지급:\n- 추천인: 1 BW + 1.2% 보너스\n- 가입자: 1 BW + 1.2% 보너스`);
      }
      
      return { success: true, data };
    } else {
      throw new Error(data.message || '추천인 가입 실패');
    }
  } catch (error) {
    console.error('❌ 유저용 추천인 가입 실패:', error);
    throw error;
  }
};
```
