#### **✅ 5. 정밀도 계산 유틸리티 함수들**

```typescript
const precisionUtils = {
  // 소수점 자릿수 제한 (BigDecimal 기반 - 완벽한 정밀도)
  roundToPrecision: (value: number, precision: number = 8): number => {
    return new Decimal(value).toDecimalPlaces(precision).toNumber();
  },
  
  // 보너스율 계산 정밀도 보장 (BigDecimal 기반)
  calculateBonusRate: (baseRate: number, bonusPercentage: number, precision: number = 8): number => {
    const bonusRate = new Decimal(baseRate).times(bonusPercentage).dividedBy(100);
    return bonusRate.toDecimalPlaces(precision).toNumber();
  },
  
  // 시간당 보상 계산 정밀도 보장 (BigDecimal 기반)
  calculateHourlyReward: (baseRate: number, totalBonusRate: number, precision: number = 8): number => {
    const decimalBaseRate = new Decimal(baseRate);
    const decimalOne = new Decimal(1);
    const decimalTotalBonus = new Decimal(totalBonusRate);
    
    const decimalEnhancedRate = decimalOne.plus(decimalTotalBonus).times(decimalBaseRate);
    return decimalEnhancedRate.toDecimalPlaces(precision).toNumber();
  }
```

#### **✅ 6. 정밀도 검증 시스템**

```typescript
const precisionValidation = {
  // 정밀도 검증 함수
  validatePrecision: (expected: number, actual: number, tolerance: number = 0.00000001): boolean => {
    const decimalExpected = new Decimal(expected);
    const decimalActual = new Decimal(actual);
    const decimalTolerance = new Decimal(tolerance);
    const decimalError = decimalExpected.minus(decimalActual).abs();
    return decimalError.lte(decimalTolerance);
  },
  
  // 부동소수점 오차 분석
  analyzeFloatingPointError: (value: number, precision: number = 8): void => {
    const decimalValue = new Decimal(value);
    const floatValue = parseFloat(value.toString());
    const error = decimalValue.minus(floatValue).abs();
    const relativeError = error.dividedBy(decimalValue).times(100);
    
    console.log(`🔍 부동소수점 오차 분석 (${value}):`, {
      BigDecimal값: decimalValue.toFixed(precision),
      부동소수점값: floatValue.toFixed(precision),
      절대오차: error.toFixed(precision),
      상대오차: relativeError.toFixed(6) + '%',
      정밀도: precision + '자리'
    });
  }
};
```
