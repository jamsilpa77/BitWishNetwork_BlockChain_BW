```typescript
// 전날 AM 09:00:00부터 매일 계속 출석 체크 가능
const isAdminAttendanceBonusTimeValid = (targetDate?: Date): boolean => {
  const yesterday = new Date(checkDate);
  yesterday.setDate(yesterday.getDate() - 1);
  yesterday.setHours(9, 0, 0, 0); // AM 09:00:00
  
  const tomorrow = new Date(checkDate);
  tomorrow.setDate(tomorrow.getDate() + 1);
  tomorrow.setHours(8, 59, 59, 999); // AM 08:59:59
  
  return currentTime >= yesterday.getTime() && currentTime <= tomorrow.getTime();
};
```

```typescript
const calculateUserAttendanceBonus = (consecutiveDays: number) => {
```

```typescript
  try {
```

```typescript
      setSuccessMessage('❌ 지갑 주소가 필요합니다');
      setShowSuccessModal(true);
      return;
    }

      })
    });

      // 성공 시 등록 완료 처리
      setIsUserLockupRegistered(true);
      setRegisteredUserLockupData({
        ratio: userLockupRatio,
```

**유저용 락업 보너스 계산 함수:**
```typescript
const calculateUserLockupBonus = (userLockupRatio: number, userLockupMonths: number): {
  baseAPY: number;
  periodBonus: number;
  totalAPY: number;
  monthlyRate: number;
  enhancedRate: number;
  description: string;
  lockupMultiplier: number;
  decimalBaseAPY: Decimal;
  decimalMonthlyRate: Decimal;
  decimalEnhancedRate: Decimal;
} => {
  // 구간별 차등 APY 설정
  const apyRates = {
    1: 0.027,     // 1개월: 2.7% APY
    6: 0.042,     // 6개월: 4.2% APY
    12: 0.057,    // 12개월: 5.7% APY
    36: 0.072     // 36개월: 7.2% APY
  };
  
  const baseAPY = apyRates[userLockupMonths as keyof typeof apyRates] || 0.027;
  const periodBonus = 0; // 기간 보너스 없음
  const totalAPY = baseAPY;
  const monthlyRate = baseAPY / 12; // APY / 12 (월간 비율)
  
  // Decimal 정밀도
  const baseAPYDec = new Decimal(baseAPY);
  const periodDec = new Decimal(periodBonus);
  const totalAPYDec = new Decimal(totalAPY);
  const monthlyRateDec = new Decimal(monthlyRate);
  
  // 향상 비율 계산: 기본_비율 * (1 + APY * 락업_비율)
  const decimalBaseRate = new Decimal(0.25);
  const decimalLockupRatio = new Decimal(userLockupRatio / 100);
  const enhancedRateDec = decimalBaseRate.times(new Decimal(1).plus(baseAPYDec.times(decimalLockupRatio)));
  
  return {
    baseAPY: baseAPY,
    periodBonus: periodBonus,
    totalAPY: totalAPY,
    monthlyRate: monthlyRate,
    enhancedRate: enhancedRateDec.toNumber(),
    description: `유저용 락업 ${userLockupRatio}% (${userLockupMonths}개월) - APY ${(baseAPY * 100).toFixed(1)}%`,
    lockupMultiplier: 1,
    decimalBaseAPY: baseAPYDec,
    decimalMonthlyRate: monthlyRateDec,
    decimalEnhancedRate: enhancedRateDec
  };
};
```
