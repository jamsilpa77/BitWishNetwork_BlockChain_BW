### **✅ 3-5. 연속 출석일 수 계산 함수**

```typescript
// 캘린더 기반 연속 출석일 계산 (한국 시간 기준 정밀도)
const calculateConsecutiveDaysFromCalendar = (calendar: {[key: string]: boolean}, currentDate: Date): number => {
  const dates = Object.keys(calendar).sort();
  if (dates.length === 0) return 0;
  
  let maxConsecutive = 0;
  let currentConsecutive = 0;
  
  for (let i = 0; i < dates.length; i++) {
    if (i === 0) {
      currentConsecutive = 1;
    } else {
      const prevDate = new Date(dates[i - 1]);
      const currDate = new Date(dates[i]);
      const diffTime = currDate.getTime() - prevDate.getTime();
      const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));
      
      if (diffDays === 1) {
        currentConsecutive++;
      } else {
        currentConsecutive = 1;
      }
    }
    maxConsecutive = Math.max(maxConsecutive, currentConsecutive);
  }
  
  return maxConsecutive;
};
```

const getAttendanceCalendarStatus = (year: number, month: number, day: number) => {
  const koreanTime = getKoreanTime();
  const hours = koreanTime.getHours();
  const minutes = koreanTime.getMinutes();
  const seconds = koreanTime.getSeconds();
  const currentDate = new Date();
  
  // 오늘 날짜 확인 함수
  const isToday = currentDate.getFullYear() === year && 
                  currentDate.getMonth() === month - 1 && 
                  currentDate.getDate() === day;
  
  // 출석 유효성 체크
  const dateKey = formatDateKey(year, month, day);
  const isAttended = attendanceCalendar[dateKey];
  
  if (isAttended) {
    return 'attended'; // 출석완료 - 유효성 체크
  } else if (isToday && (hours > 9 || (hours === 9 && minutes >= 0 && seconds >= 0))) {
    return 'active'; // 활성화 - AM 09:00:00부터 유효성 체크
  } else {
    return 'inactive'; // 비활성 - 출석불가
  }
};
