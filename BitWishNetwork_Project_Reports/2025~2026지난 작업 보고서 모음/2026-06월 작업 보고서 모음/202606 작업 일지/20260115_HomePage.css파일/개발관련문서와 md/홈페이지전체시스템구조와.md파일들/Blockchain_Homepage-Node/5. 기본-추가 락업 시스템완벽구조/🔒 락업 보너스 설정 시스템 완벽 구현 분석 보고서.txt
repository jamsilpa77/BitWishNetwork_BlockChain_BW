```typescript
const handleUserLockupRegistration = async () => {
  try {
    if (!authenticatedWalletAddress) {
      setSuccessMessage('❌ 지갑 주소가 필요합니다');
      setShowSuccessModal(true);
      return;
    }

    if (userLockupRatio === 0 || userLockupMonths === 0) {
      setSuccessMessage('❌ 락업 비율과 기간을 선택해주세요');
      setShowSuccessModal(true);
      return;
    }

    const response = await fetch('http://localhost:4001/lockup/create', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        walletAddress: authenticatedWalletAddress,
        userId: authenticatedWalletAddress,
        amount: 1000,
        lockPeriod: userLockupMonths
      })
    });

    const result = await response.json();
    
    if (response.ok && result.success) {
      // 성공 시 등록 완료 처리
      setIsUserLockupRegistered(true);
      setRegisteredUserLockupData({
        ratio: userLockupRatio,
        months: userLockupMonths,
        apy: calculateUserLockupBonus(userLockupRatio, userLockupMonths).totalAPY,
        startDate: new Date(),
        endDate: new Date(Date.now() + userLockupMonths * 30 * 24 * 60 * 60 * 1000)
      });
      
      setSuccessMessage('✅ 유저용 기본 락업 설정 등록 완료!');
      setShowSuccessModal(true);
      saveUserLockupState();
    } else {
      setSuccessMessage(`❌ 유저용 기본 락업 설정 오류: ${result.message || '알 수 없는 오류'}`);
      setShowSuccessModal(true);
    }
  } catch (error) {
    console.error('유저용 기본 락업 설정 오류:', error);
    setSuccessMessage(`❌ 유저용 기본 락업 설정 오류: ${error instanceof Error ? error.message : '알 수 없는 오류'}`);
    setShowSuccessModal(true);
  }
};
```

## **   4. 유저용과 관리자용 락업 보너스 계산 함수 시스템**

### ✅ **완벽한 계산 함수 분리 구현**

**유저용 락업 보너스 계산 함수:**
```typescript
const calculateUserLockupBonus = (userLockupRatio: number, userLockupMonths: number): {
  baseAPY: number;
  periodBonus: number;
  totalAPY: number;
  monthlyRate: number;
  enhancedRate: number;
  description: string;
  lockupMultiplier: number;
  decimalBaseAPY: Decimal;
  decimalMonthlyRate: Decimal;
  decimalEnhancedRate: Decimal;
} => {
  // 구간별 차등 APY 설정
  const apyRates = {
    1: 0.027,     // 1개월: 2.7% APY
    6: 0.042,     // 6개월: 4.2% APY
    12: 0.057,    // 12개월: 5.7% APY
    36: 0.072     // 36개월: 7.2% APY
  };
  
  const baseAPY = apyRates[userLockupMonths as keyof typeof apyRates] || 0.027;
  const periodBonus = 0; // 기간 보너스 없음
  const totalAPY = baseAPY;
  const monthlyRate = baseAPY / 12; // APY / 12 (월간 비율)
  
  // Decimal 정밀도
  const baseAPYDec = new Decimal(baseAPY);
  const periodDec = new Decimal(periodBonus);
  const totalAPYDec = new Decimal(totalAPY);
  const monthlyRateDec = new Decimal(monthlyRate);
  
  // 향상 비율 계산: 기본_비율 * (1 + APY * 락업_비율)
  const decimalBaseRate = new Decimal(0.25);
  const decimalLockupRatio = new Decimal(userLockupRatio / 100);
  const enhancedRateDec = decimalBaseRate.times(new Decimal(1).plus(baseAPYDec.times(decimalLockupRatio)));
  
  return {
    baseAPY: baseAPY,
    periodBonus: periodBonus,
    totalAPY: totalAPY,
    monthlyRate: monthlyRate,
    enhancedRate: enhancedRateDec.toNumber(),
    description: `유저용 락업 ${userLockupRatio}% (${userLockupMonths}개월) - APY ${(baseAPY * 100).toFixed(1)}%`,
    lockupMultiplier: 1,
    decimalBaseAPY: baseAPYDec,
    decimalMonthlyRate: monthlyRateDec,
    decimalEnhancedRate: enhancedRateDec
  };
};
```
