```typescript
const validateSeedPhraseLocally = () => {
  if (!generatedSeedPhrase) return false;
  
  const words = generatedSeedPhrase.split(' ');
  if (words.length !== 24) return false;
  
  // BIP39 표준 단어 목록 검증
  for (const word of words) {
    if (!bip39.wordlists.english.includes(word)) {
      return false;
    }
  }
  
  // 검증 입력 필드 검사
  for (let i = 0; i < 4; i++) {
    const expectedWord = words[verificationIndices[i]];
    if (verificationInputs[i] !== expectedWord) {
      return false;
    }
  }
  
  return true;
};
```

**핵심 로직:**
- 24단어 검증

```typescript
const fetchWalletData = async () => {
  try {
    if (!publicKey) return;
    
    // 1. 지갑 잔액 조회
    const balanceResponse = await fetch(`${API_BASE_URL}/api/wallet/balance`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ publicKey })
    });
    
    if (!balanceResponse.ok) {
      throw new Error('잔액 조회 실패');
    }
    
    const balanceData = await balanceResponse.json();
    
    // 2. 거래 내역 조회
    const transactionsResponse = await fetch(`${API_BASE_URL}/api/wallet/transactions`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ publicKey })
    });
    
    if (!transactionsResponse.ok) {
      throw new Error('거래 내역 조회 실패');
    }
    
    const transactionsData = await transactionsResponse.json();
    
    // 3. KYC 상태 조회
    const kycResponse = await fetch(`${API_BASE_URL}/api/kyc/status`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ publicKey })
    });
    
    if (!kycResponse.ok) {
      throw new Error('KYC 상태 조회 실패');
    }
    
    const kycData = await kycResponse.json();
    
    // 4. 통합 데이터 저장
    setWalletData({
      ...balanceData,
      transactions: transactionsData.transactions || [],
      kycStatus: kycData.status || 'pending',
      kycVerified: kycData.verified || false
    });
    
    // 5. 실시간 업데이트 상태 설정
    setLastUpdate(new Date().toLocaleString('ko-KR'));
    
  } catch (error) {
    console.error('지갑 데이터 조회 오류:', error);
    // 오류 발생 시에도 기존 데이터 유지
  }
};
```

```javascript
app.post('/api/create-wallet', async (req, res) => {
  try {
    const { seedPhrase, referralCode } = req.body;
    
    // 1. 시드 구문 유효성 검사
    if (!seedPhrase || typeof seedPhrase !== 'string') {
      return res.status(400).json({ error: '시드 구문이 필요합니다.' });
    }
    
    // 2. BIP39 표준 검증
    const words = seedPhrase.trim().split(' ');
    if (words.length !== 24) {
      return res.status(400).json({ error: '시드 구문은 24단어여야 합니다.' });
    }
    
    // 3. 중복 지갑 검사
    const existingWallet = await Wallet.findOne({ seedPhrase: seedPhrase.trim() });
    if (existingWallet) {
      return res.status(409).json({ error: '이미 존재하는 지갑입니다.' });
    }
    
    // 4. Stellar 키페어 생성
    const keypair = StellarSdk.Keypair.fromSecret(seedPhrase);
    const publicKey = keypair.publicKey();
    
    // 5. 추천인 코드 처리
    let referralData = null;
    if (referralCode && referralCode.trim()) {
      const referrer = await User.findOne({ referralCode: referralCode.trim() });
      if (referrer) {
        referralData = {
          referrerId: referrer._id,
          referrerCode: referralCode.trim(),
          referralDate: new Date()
        };
        
        // 추천인 보너스 지급 (1 BW)
        await updateUserBalance(referrer._id, 1, 'referral_bonus', '추천인 보너스');
      }
    }
    
    // 6. 새 지갑 생성
    const newWallet = new Wallet({
      publicKey,
      seedPhrase: seedPhrase.trim(),
      referralData,
      createdAt: new Date(),
      balance: {
        total: 0,
        available: 0,
        locked: 0
      },
      rewards: {
        mining: 0,
        attendance: 0,
        referral: 0,
        lockup: 0
      }
    });
    
    await newWallet.save();
    
    // 7. 사용자 데이터 생성
    const newUser = new User({
      publicKey,
      seedPhrase: seedPhrase.trim(),
      referralCode: generateReferralCode(),
      referralData,
      createdAt: new Date()
    });
    
    await newUser.save();
    
    // 8. 응답 데이터 구성
    const responseData = {
      publicKey,
      seedPhrase: seedPhrase.trim(),
      referralCode: newUser.referralCode,
      balance: newWallet.balance,
      rewards: newWallet.rewards,
      createdAt: newWallet.createdAt
    };
    
    res.status(201).json(responseData);
    
  } catch (error) {
    console.error('지갑 생성 오류:', error);
    res.status(500).json({ error: '지갑 생성 중 오류가 발생했습니다.' });
  }
});
```

```typescript
// 시드 구문 보안 검증 함수
const validateSeedPhraseSecurity = (seedPhrase: string): boolean => {
  try {
    // 1. 기본 형식 검증
    if (!seedPhrase || typeof seedPhrase !== 'string') {
      return false;
    }
    
    // 2. 공백 및 특수문자 제거
    const cleanSeedPhrase = seedPhrase.trim().toLowerCase();
    
    // 3. 24단어 검증
    const words = cleanSeedPhrase.split(/\s+/);
    if (words.length !== 24) {
      return false;
    }
    
    // 4. BIP39 표준 단어 목록 검증
    for (const word of words) {
      if (!bip39.wordlists.english.includes(word)) {
        return false;
      }
    }
    
    // 5. 중복 단어 검증
    const uniqueWords = new Set(words);
    if (uniqueWords.size !== 24) {
      return false;
    }
    
    // 6. 체크섬 검증
    if (!bip39.validateMnemonic(cleanSeedPhrase)) {
      return false;
    }
    
    return true;
  } catch (error) {
    console.error('시드 구문 보안 검증 오류:', error);
    return false;
  }
};
```

```javascript
// 데이터 암호화 함수
const encryptSensitiveData = (data) => {
  try {
    const algorithm = 'aes-256-gcm';
    const key = crypto.scryptSync(process.env.ENCRYPTION_KEY, 'salt', 32);
    const iv = crypto.randomBytes(16);
    
    const cipher = crypto.createCipher(algorithm, key);
    cipher.setAAD(Buffer.from('additional-data'));
    
    let encrypted = cipher.update(data, 'utf8', 'hex');
    encrypted += cipher.final('hex');
    
    const authTag = cipher.getAuthTag();
    
    return {
      encrypted,
      iv: iv.toString('hex'),
      authTag: authTag.toString('hex')
    };
  } catch (error) {
    console.error('암호화 오류:', error);
    throw error;
  }
};

// 데이터 복호화 함수
const decryptSensitiveData = (encryptedData) => {
  try {
    const algorithm = 'aes-256-gcm';
    const key = crypto.scryptSync(process.env.ENCRYPTION_KEY, 'salt', 32);
    
    const decipher = crypto.createDecipher(algorithm, key);
    decipher.setAAD(Buffer.from('additional-data'));
    decipher.setAuthTag(Buffer.from(encryptedData.authTag, 'hex'));
    
    let decrypted = decipher.update(encryptedData.encrypted, 'hex', 'utf8');
    decrypted += decipher.final('utf8');
    
    return decrypted;
  } catch (error) {
    console.error('복호화 오류:', error);
    throw error;
  }
};
```

```typescript
// 네트워크 오류 처리 함수
const handleNetworkError = async (error: any, retryCount: number = 0): Promise<boolean> => {
  try {
    // 1. 재시도 횟수 제한
    if (retryCount >= 3) {
      alert('네트워크 연결에 실패했습니다. 잠시 후 다시 시도해주세요.');
      return false;
    }
    
    // 2. 오류 타입별 처리
    if (error.name === 'TypeError' && error.message.includes('fetch')) {
      // 네트워크 연결 오류
      console.log(`네트워크 오류 발생. 재시도 ${retryCount + 1}/3`);
      
      // 3. 지수 백오프 재시도
      const delay = Math.pow(2, retryCount) * 1000; // 1초, 2초, 4초
      await new Promise(resolve => setTimeout(resolve, delay));
      
      return true; // 재시도 가능
    }
    
    // 4. 서버 오류 (5xx)
    if (error.status >= 500) {
      console.log(`서버 오류 발생. 재시도 ${retryCount + 1}/3`);
      
      const delay = Math.pow(2, retryCount) * 1000;
      await new Promise(resolve => setTimeout(resolve, delay));
      
      return true; // 재시도 가능
    }
    
    // 5. 클라이언트 오류 (4xx) - 재시도 불가
    if (error.status >= 400 && error.status < 500) {
      return false; // 재시도 불가
    }
    
    return false;
  } catch (handlingError) {
    console.error('네트워크 오류 처리 중 추가 오류:', handlingError);
    return false;
  }
};

// 재시도 로직이 포함된 API 호출
const fetchWithRetry = async (url: string, options: RequestInit, retryCount: number = 0): Promise<Response> => {
  try {
    const response = await fetch(url, options);
    
    if (!response.ok) {
      const error = new Error(`HTTP ${response.status}: ${response.statusText}`);
      error.status = response.status;
      throw error;
    }
    
    return response;
  } catch (error) {
    const shouldRetry = await handleNetworkError(error, retryCount);
    
    if (shouldRetry) {
      return fetchWithRetry(url, options, retryCount + 1);
    }
    
    throw error;
  }
};
```

```typescript
  // 요청 캐싱
  cache: new Map<string, { data: any; timestamp: number; ttl: number }>(),

      return cached.data;

    return null;

      if (now - value.timestamp >= value.ttl) {
        apiOptimizer.cache.delete(key);
      }
    }
  }
```

```typescript
const autoRefreshOptimizer = {
  // 적응형 새로고침 간격
  getRefreshInterval: (): number => {
    // 1. 네트워크 상태 확인
    if (navigator.onLine === false) {
      return 60000; // 오프라인 시 1분
    }
    
    // 2. 메모리 사용량 확인
    if (performance.memory) {
      const memoryUsage = performance.memory.usedJSHeapSize / performance.memory.jsHeapSizeLimit;
      if (memoryUsage > 0.8) {
        return 120000; // 메모리 사용량 높을 시 2분
      }
    }
    
    // 3. 기본 간격 (30초)
    return 30000;
  },
  
  // 스마트 새로고침
  smartRefresh: async (): Promise<void> => {
    try {
      // 1. 백그라운드에서 새로고침
      if (document.hidden) {
        return; // 탭이 숨겨져 있으면 새로고침 건너뛰기
      }
      
      // 2. 사용자 활동 확인
      const lastActivity = localStorage.getItem('lastUserActivity');
      if (lastActivity) {
        const timeSinceActivity = Date.now() - parseInt(lastActivity);
        if (timeSinceActivity > 300000) { // 5분 이상 비활성
          return; // 사용자가 비활성 상태면 새로고침 건너뛰기
        }
      }
      
      // 3. 네트워크 상태 확인
      if (!navigator.onLine) {
        return; // 오프라인 상태면 새로고침 건너뛰기
      }
      
      // 4. 새로고침 실행
      await fetchWalletDataOptimized();
      
    } catch (error) {
      console.error('스마트 새로고침 오류:', error);
    }
  },
  
  // 사용자 활동 추적
  trackUserActivity: (): void => {
    const activityEvents = ['mousedown', 'mousemove', 'keypress', 'scroll', 'touchstart'];
    
    const updateActivity = () => {
      localStorage.setItem('lastUserActivity', Date.now().toString());
    };
    
    activityEvents.forEach(event => {
      document.addEventListener(event, updateActivity, { passive: true });
    });
  }
};

// 최적화된 자동 새로고침 설정
const setupOptimizedAutoRefresh = (): void => {
  // 1. 기존 타이머 정리
  if (refreshTimer) {
    clearInterval(refreshTimer);
  }
  
  // 2. 사용자 활동 추적 시작
  autoRefreshOptimizer.trackUserActivity();
  
  // 3. 적응형 새로고침 설정
  const setupRefresh = () => {
    const interval = autoRefreshOptimizer.getRefreshInterval();
    
    refreshTimer = setInterval(() => {
      autoRefreshOptimizer.smartRefresh();
    }, interval);
    
    console.log(`자동 새로고침 간격: ${interval / 1000}초`);
  };
  
  // 4. 초기 설정
  setupRefresh();
  
  // 5. 네트워크 상태 변경 시 재설정
  window.addEventListener('online', setupRefresh);
  window.addEventListener('offline', () => {
    if (refreshTimer) {
      clearInterval(refreshTimer);
      refreshTimer = null;
    }
  });
```

### 14.5 백업 및 복구 시스템

```bash
# 백업 스크립트
#!/bin/bash
# backup.sh

# 설정 변수
BACKUP_DIR="/var/backups/bitwish-wallet"
DATE=$(date +%Y%m%d_%H%M%S)
DB_NAME="bitwish_production"
APP_DIR="/var/www/bitwish-wallet"

# 백업 디렉토리 생성
mkdir -p $BACKUP_DIR

# 1. 데이터베이스 백업
echo "데이터베이스 백업 시작..."
mongodump --db $DB_NAME --out $BACKUP_DIR/db_$DATE

# 2. 애플리케이션 파일 백업
echo "애플리케이션 파일 백업 시작..."
tar -czf $BACKUP_DIR/app_$DATE.tar.gz -C $APP_DIR .

# 3. 설정 파일 백업
echo "설정 파일 백업 시작..."
cp $APP_DIR/.env $BACKUP_DIR/env_$DATE
cp $APP_DIR/ecosystem.config.js $BACKUP_DIR/ecosystem_$DATE.js

# 4. 로그 파일 백업
echo "로그 파일 백업 시작..."
tar -czf $BACKUP_DIR/logs_$DATE.tar.gz -C $APP_DIR logs/

# 5. 오래된 백업 파일 정리 (30일 이상)
echo "오래된 백업 파일 정리..."
find $BACKUP_DIR -type f -mtime +30 -delete

# 6. 백업 완료 알림
echo "백업이 완료되었습니다: $DATE"

# 슬랙 알림 전송
curl -X POST -H 'Content-type: application/json' \
  --data "{\"text\":\"백업 완료: $DATE\"}" \
  $SLACK_WEBHOOK_URL

# 복구 스크립트
#!/bin/bash
# restore.sh

# 설정 변수
BACKUP_DIR="/var/backups/bitwish-wallet"
DB_NAME="bitwish_production"
APP_DIR="/var/www/bitwish-wallet"

# 백업 파일 목록 표시
echo "사용 가능한 백업 파일:"
ls -la $BACKUP_DIR

# 복구할 백업 선택
read -p "복구할 백업 날짜를 입력하세요 (YYYYMMDD_HHMMSS): " BACKUP_DATE

# 1. 애플리케이션 중지
echo "애플리케이션 중지..."
pm2 stop bitwish-wallet

# 2. 데이터베이스 복구
echo "데이터베이스 복구 시작..."
mongorestore --db $DB_NAME --drop $BACKUP_DIR/db_$BACKUP_DATE/$DB_NAME

# 3. 애플리케이션 파일 복구
echo "애플리케이션 파일 복구 시작..."
tar -xzf $BACKUP_DIR/app_$BACKUP_DATE.tar.gz -C $APP_DIR

# 4. 설정 파일 복구
echo "설정 파일 복구 시작..."
cp $BACKUP_DIR/env_$BACKUP_DATE $APP_DIR/.env
cp $BACKUP_DIR/ecosystem_$BACKUP_DATE.js $APP_DIR/ecosystem.config.js

# 5. 의존성 재설치
echo "의존성 재설치..."
cd $APP_DIR
npm ci --production

# 6. 애플리케이션 재시작
echo "애플리케이션 재시작..."
pm2 start ecosystem.config.js --env production

# 7. 복구 완료 알림
echo "복구가 완료되었습니다: $BACKUP_DATE"

# 슬랙 알림 전송
curl -X POST -H 'Content-type: application/json' \
  --data "{\"text\":\"복구 완료: $BACKUP_DATE\"}" \
  $SLACK_WEBHOOK_URL
```

```bash
[DEFAULT]
bantime = 3600
findtime = 600
maxretry = 3

enabled = true

logpath = /var/log/auth.log
maxretry = 3

[nginx-http-auth]
enabled = true
filter = nginx-http-auth
port = http,https
logpath = /var/log/nginx/error.log
maxretry = 3
```
