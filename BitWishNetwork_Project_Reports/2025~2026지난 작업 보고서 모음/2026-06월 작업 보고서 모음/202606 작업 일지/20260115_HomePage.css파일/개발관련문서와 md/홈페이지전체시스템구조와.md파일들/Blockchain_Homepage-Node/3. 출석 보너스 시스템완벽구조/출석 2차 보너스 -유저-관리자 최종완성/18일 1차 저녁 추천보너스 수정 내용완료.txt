```typescript
// ==================== 유저용 추천인 코드 자동 발급 ====================
const generateUserReferralCode = async (walletAddress: string): Promise<string> => {
  try {
    console.log('🎯 유저용 추천인 코드 생성 시작:', walletAddress);
    
    // 백엔드 API 호출
    const response = await fetch('http://localhost:4001/referral/generate-user', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({ walletAddress })
    });
    
    if (!response.ok) {
      throw new Error(`HTTP error! status: ${response.status}`);
    }
    
    const data = await response.json();
    
    if (data.success && data.referralCode) {
      console.log('✅ 유저용 추천인 코드 생성 성공:', data.referralCode);
      return data.referralCode;
    } else {
      throw new Error(data.message || '추천인 코드 생성 실패');
    }
  } catch (error) {
    console.error('❌ 유저용 추천인 코드 생성 실패:', error);
    throw error;
  }
};

// ==================== 유저용 추천인 코드 상태 관리 ====================
```

```typescript
  try {
    console.log('🎯 유저용 추천인 가입 처리 시작:', { referralCode, newWalletAddress });
    
    // 백엔드 API 호출
    const response = await fetch('http://localhost:4001/referral/join-user', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({ 
        referralCode, 
        newWalletAddress,
        type: 'user'
      })
    });
    
    if (!response.ok) {
      throw new Error(`HTTP error! status: ${response.status}`);
    }
    
    const data = await response.json();
    
    if (data.success) {
      console.log('✅ 유저용 추천인 가입 성공:', data);
      
      // 보상 지급 확인

      }
      
      return { success: true, data };
    } else {
      throw new Error(data.message || '추천인 가입 실패');
    }
  } catch (error) {
    console.error('❌ 유저용 추천인 가입 실패:', error);

  }
```

```typescript
  let bonusRate = 0;
```
