```typescript
// ==================== 1단계: Decimal.js 기반 정밀도 계산 시스템 ====================

// Decimal.js 전역 설정 (50자리 정밀도)
Decimal.set({
  precision: 50,
  rounding: Decimal.ROUND_HALF_UP,
  toExpNeg: -50,
  toExpPos: 50
});

// 유저용 추천인 보너스 계산 (Decimal.js 기반)
const calculateUserReferralBonusDecimal = (referralCount: number): {
  bonusRate: Decimal;
  enhancedRate: Decimal;
  description: string;
  maxBonus: number;
  instantBonus: Decimal;
  permanentBonusRate: Decimal;
  totalBonusRate: Decimal; // 총 보너스율 (영구 + 구간별)
  cumulativeBonusRate: Decimal; // 누적 보너스율
  segmentInfo: {
    currentSegment: string;
    segmentStart: number;
    segmentEnd: number;
    segmentRate: number;
    cumulativeFromPrevious: Decimal;
    additionalInCurrent: Decimal;
  };
} => {
  // 초기 보너스: 즉시 보너스 1 BW + 영구 보너스 1.2% (추천인 수)
  const decimalInstantBonus = new Decimal(1.0);
  const decimalPermanentBonusRate = new Decimal(0.012).times(referralCount); // 1.2% × N명
  
  // 구간별 누적 보너스율 계산 (BigDecimal 기반)
  let decimalCumulativeBonusRate = new Decimal(0);
  let segmentInfo: any = {};
  
  if (referralCount >= 91) {
    // 구간 4: 91명 이상 (0.75% × N명)
    // 누적: 30명(56.25%) + 30명(45%) + 30명(33.75%) + (N-90)명(0.75%)
    const segment1Bonus = new Decimal(30).times(0.01875); // 30명: 56.25%
    const segment2Bonus = new Decimal(30).times(0.015); // 31-60명: 45%
    const segment3Bonus = new Decimal(30).times(0.01125); // 61-90명: 33.75%
    const segment4Bonus = new Decimal(referralCount - 90).times(0.0075); // 91명 이상: 0.75% × (N-90)
    
    decimalCumulativeBonusRate = segment1Bonus.plus(segment2Bonus).plus(segment3Bonus).plus(segment4Bonus);
    
    segmentInfo = {
      currentSegment: '91명 이상 (대규모 커뮤니티)',
      segmentStart: 91,
      segmentEnd: Infinity,
      segmentRate: 0.0075,
      cumulativeFromPrevious: segment1Bonus.plus(segment2Bonus).plus(segment3Bonus),
      additionalInCurrent: segment4Bonus
    };
    
    return {
      bonusRate: decimalCumulativeBonusRate,
      enhancedRate: new Decimal(0.25).times(new Decimal(1).plus(decimalCumulativeBonusRate)),
      description: '🏆 대규모 커뮤니티 구축',
      maxBonus: decimalCumulativeBonusRate.times(100).toNumber(),
      instantBonus: decimalInstantBonus,
      permanentBonusRate: decimalPermanentBonusRate,
      totalBonusRate: decimalPermanentBonusRate.plus(decimalCumulativeBonusRate),
      cumulativeBonusRate: decimalCumulativeBonusRate,
      segmentInfo
    };
  } else if (referralCount >= 61) {
    // 구간 3: 61-90명 (1.125% × N명)
    // 누적: 30명(56.25%) + 30명(45%) + (N-60)명(1.125%)
    const segment1Bonus = new Decimal(30).times(0.01875); // 1-30명: 56.25%
    const segment2Bonus = new Decimal(30).times(0.015); // 31-60명: 45%
    const segment3Bonus = new Decimal(referralCount - 60).times(0.01125); // 61-90명: 1.125% × (N-60)
    
    decimalCumulativeBonusRate = segment1Bonus.plus(segment2Bonus).plus(segment3Bonus);
    
    segmentInfo = {
      currentSegment: '61-90명 (활발한 커뮤니티)',
      segmentStart: 61,
      segmentEnd: 90,
      segmentRate: 0.01125,
      cumulativeFromPrevious: segment1Bonus.plus(segment2Bonus),
      additionalInCurrent: segment3Bonus
    };
    
    return {
      bonusRate: decimalCumulativeBonusRate,
      enhancedRate: new Decimal(0.25).times(new Decimal(1).plus(decimalCumulativeBonusRate)),
      description: '🌟 활발한 커뮤니티 운영',
      maxBonus: decimalCumulativeBonusRate.times(100).toNumber(),
      instantBonus: decimalInstantBonus,
      permanentBonusRate: decimalPermanentBonusRate,
      totalBonusRate: decimalPermanentBonusRate.plus(decimalCumulativeBonusRate),
      cumulativeBonusRate: decimalCumulativeBonusRate,
      segmentInfo
    };
  } else if (referralCount >= 31) {
    // 구간 2: 31-60명 (1.5% × N명)
    // 누적: 30명(56.25%) + (N-30)명(1.5%)
    const segment1Bonus = new Decimal(30).times(0.01875); // 1-30명: 56.25%
    const segment2Bonus = new Decimal(referralCount - 30).times(0.015); // 31-60명: 1.5% × (N-30)
    
    decimalCumulativeBonusRate = segment1Bonus.plus(segment2Bonus);
    
    segmentInfo = {
      currentSegment: '31-60명 (중간 규모 커뮤니티)',
      segmentStart: 31,
      segmentEnd: 60,
      segmentRate: 0.015,
      cumulativeFromPrevious: segment1Bonus,
      additionalInCurrent: segment2Bonus
    };
    
    return {
      bonusRate: decimalCumulativeBonusRate,
      enhancedRate: new Decimal(0.25).times(new Decimal(1).plus(decimalCumulativeBonusRate)),
      description: '📈 중간 규모 커뮤니티',
      maxBonus: decimalCumulativeBonusRate.times(100).toNumber(),
      instantBonus: decimalInstantBonus,
      permanentBonusRate: decimalPermanentBonusRate,
      totalBonusRate: decimalPermanentBonusRate.plus(decimalCumulativeBonusRate),
      cumulativeBonusRate: decimalCumulativeBonusRate,
      segmentInfo
    };
  } else {
    // 구간 1: 1-30명 (1.875% × N명)
    // 누적: N명(1.875%)
    const segment1Bonus = new Decimal(referralCount).times(0.01875); // 1-30명: 1.875% × N
    
    decimalCumulativeBonusRate = segment1Bonus;
    
    segmentInfo = {
      currentSegment: '1-30명 (초기 커뮤니티)',
      segmentStart: 1,
      segmentEnd: 30,
      segmentRate: 0.01875,
      cumulativeFromPrevious: new Decimal(0),
      additionalInCurrent: segment1Bonus
    };
    
    return {
      bonusRate: decimalCumulativeBonusRate,
      enhancedRate: new Decimal(0.25).times(new Decimal(1).plus(decimalCumulativeBonusRate)),
      description: '   초기 추천 활동',
      maxBonus: decimalCumulativeBonusRate.times(100).toNumber(),
      instantBonus: decimalInstantBonus,
      permanentBonusRate: decimalPermanentBonusRate,
      totalBonusRate: decimalPermanentBonusRate.plus(decimalCumulativeBonusRate),
      cumulativeBonusRate: decimalCumulativeBonusRate,
      segmentInfo
    };
  }
};

// 관리자용 추천인 보너스 계산 (Decimal.js 기반)
const calculateAdminReferralBonusDecimal = (referralCount: number): {

    currentSegment: string;
    segmentStart: number;
    segmentEnd: number;

    cumulativeFromPrevious: Decimal;
    additionalInCurrent: Decimal;
  };
} => {
  // 관리자용은 유저용과 동일한 로직 사용 (완전 분리)
  return calculateUserReferralBonusDecimal(referralCount);
};
```
