### **✅ 4-1. 시한폭탄 카운트다운 계산 (메모장과 100% 일치)**

**메모장 요구사항:**

```typescript
// 시한폭탄 카운트다운 계산 함수
const calculateCountdown = (endDate: Date | null) => {
  if (!endDate || !(endDate instanceof Date) || isNaN(endDate.getTime())) return null;
  
  const now = new Date();
  const remainingMs = endDate.getTime() - now.getTime();
  
  if (remainingMs <= 0) {
    return { days: 0, hours: 0, minutes: 0, seconds: 0, isExpired: true };
  }
  
  const days = Math.floor(remainingMs / (1000 * 60 * 60 * 24));
  const hours = Math.floor((remainingMs % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
  const minutes = Math.floor((remainingMs % (1000 * 60 * 60)) / (1000 * 60));
  const seconds = Math.floor((remainingMs % (1000 * 60)) / 1000);
  
  return { days, hours, minutes, seconds, isExpired: false };
};
```

```typescript
const handleLockupRelease = async () => {
  try {
    if (!authenticatedWalletAddress) {
      alert('지갑 주소가 필요합니다');
      return;
    }

    if (!isLockupRegistered) {
      alert('등록된 락업이 없습니다');
      return;
    }

    const response = await fetch('http://localhost:4001/lockup/release', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        walletAddress: authenticatedWalletAddress
      })
    });

    const result = await response.json();
    
    if (response.ok && result.success) {
      // 락업 해제 성공 처리
      setIsLockupRegistered(false);
```
