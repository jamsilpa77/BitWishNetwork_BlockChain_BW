```javascript
  try {
    const { seedPhrase, protocol, network, userId } = req.body;

    if (!bip39.validateMnemonic(seedPhrase)) {

    }
    
    // ✅ 실제 저장 로직 호출
    const result = stellarWalletSystem.createWalletFromSeedPhrase(seedPhrase, userId || 'default_user');
    
    if (!result.success) {
      return res.status(400).json({
        success: false,
        error: result.error
      });
    }
    
    // ✅ 시드문구 매니저에 저장
    seedPhraseManager.saveWalletSeedPhrase(result.wallet.address, seedPhrase);
    
    // ✅ 현재 총 지갑 개수 확인

      wallet: result.wallet,

    });
```

```javascript
// StellarWalletSystem 클래스에 추가할 메서드
class StellarWalletSystem {
  // ... 기존 코드 ...
  
  // ✅ 지갑 개수 조회 메서드 추가
  getWalletCount() {
    return this.wallets.size;
  }
  
  // ✅ 모든 지갑 목록 조회 메서드 추가
  getAllWallets() {
    return Array.from(this.wallets.values());
  }
  
  // ✅ 특정 지갑 조회 메서드 추가
  getWalletById(walletId) {
    return this.wallets.get(walletId);
  }
  
  // ✅ 지갑 통계 조회 메서드 추가
  getWalletStats() {
    const wallets = Array.from(this.wallets.values());
    const totalBalance = wallets.reduce((sum, wallet) => sum + (wallet.balance || 0), 0);
    const activeWallets = wallets.filter(wallet => wallet.status === 'active').length;
    
    return {
      totalWallets: this.wallets.size,
      activeWallets: activeWallets,
      totalBalance: totalBalance,
      averageBalance: this.wallets.size > 0 ? totalBalance / this.wallets.size : 0
    };
  }
}
```
