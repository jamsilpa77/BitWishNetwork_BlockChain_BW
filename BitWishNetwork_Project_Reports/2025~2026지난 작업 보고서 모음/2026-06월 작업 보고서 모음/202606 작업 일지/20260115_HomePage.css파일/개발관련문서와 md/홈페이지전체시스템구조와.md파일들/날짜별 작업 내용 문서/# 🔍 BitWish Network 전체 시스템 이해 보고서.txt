### **API 1: 스텔라 지갑 생성**
```javascript
// POST /wallet/create-stellar-wallet
app.post('/wallet/create-stellar-wallet', async (req, res) => {
  try {
    const { seedPhrase, protocol, network } = req.body;
    
    if (!seedPhrase) {
      return res.status(400).json({
        success: false,
        error: '시드문구가 필요합니다'
      });
    }
    
    // BIP39 검증
    if (!bip39.validateMnemonic(seedPhrase)) {
      return res.status(400).json({
        success: false,
        error: '유효하지 않은 BIP39 시드문구입니다'
      });
    }
    
    // 시드문구에서 스텔라 키페어 생성
    const seed = bip39.mnemonicToSeedSync(seedPhrase);
    const keypair = Keypair.fromRawEd25519Seed(seed.slice(0, 32));
    
    // 지갑 데이터 생성
    const walletData = {
      id: uuidv4(),
      publicKey: keypair.publicKey(),
      secretKey: keypair.secret(),
      address: keypair.publicKey(),
      seedPhrase: seedPhrase,
      protocol: protocol || 'SCP',
      network: network || 'TESTNET',
      createdAt: new Date().toISOString(),
      balance: 0,
      status: 'active'
    };
    
    res.json({
      success: true,
      publicKey: walletData.publicKey,
      privateKey: walletData.privateKey,
      walletAddress: walletData.address,
      wallet: walletData,
      message: '스텔라 지갑이 성공적으로 생성되었습니다'
    });
  } catch (error) {
    console.error('스텔라 지갑 생성 오류:', error);
    res.status(500).json({
      success: false,
      error: '스텔라 지갑 생성 중 오류가 발생했습니다',
      message: error.message
    });
  }
});
```

### **API 2: 2차 비밀번호 설정**
```javascript
// POST /wallet/setup-second-password
app.post('/wallet/setup-second-password', async (req, res) => {
  try {
    const { walletAddress, password, confirmPassword } = req.body;
    
    if (!walletAddress || !password || !confirmPassword) {
      return res.status(400).json({
        success: false,
        error: '지갑 주소, 비밀번호, 비밀번호 확인이 모두 필요합니다'
      });
    }
    
    if (password !== confirmPassword) {
      return res.status(400).json({
        success: false,
        error: '비밀번호가 일치하지 않습니다'
      });
    }
    
    // 비밀번호 강도 검증
    const passwordStrength = validatePasswordStrength(password);
    if (!passwordStrength.isValid) {
      return res.status(400).json({
        success: false,
        error: '비밀번호 강도가 부족합니다',
        details: passwordStrength.errors
      });
    }
    
    // AES-256 암호화
    const encryptedPassword = crypto.createHash('sha256').update(password).digest('hex');
    
    // 2차 비밀번호 저장
    const secondPasswordData = {
      walletAddress: walletAddress,
      encryptedPassword: encryptedPassword,
      createdAt: new Date().toISOString(),
      status: 'active'
    };
    
    res.json({
      success: true,
      message: '2차 비밀번호가 성공적으로 설정되었습니다',
      walletAddress: walletAddress
    });
  } catch (error) {
    console.error('2차 비밀번호 설정 오류:', error);
    res.status(500).json({
      success: false,
      error: '2차 비밀번호 설정 중 오류가 발생했습니다',
      message: error.message
    });
  }
});

// 비밀번호 강도 검증 함수
function validatePasswordStrength(password) {
  const errors = [];
  
  if (password.length < 8) errors.push('최소 8자리 이상');
  if (!/[A-Z]/.test(password)) errors.push('대문자 포함');
  if (!/[a-z]/.test(password)) errors.push('소문자 포함');
  if (!/[0-9]/.test(password)) errors.push('숫자 포함');
  if (!/[!@#$%^&*(),.?":{}|<>]/.test(password)) errors.push('특수문자 포함');
  if (password.includes(' ')) errors.push('공백 제거');
  
  return {
    isValid: errors.length === 0,
    errors: errors
  };
}
```

### **API 7: 마이닝 중지**
```javascript
// POST /bw-mining/stop
app.post('/bw-mining/stop', async (req, res) => {
  try {
    const { walletAddress, sessionId } = req.body;
    
    if (!walletAddress) {
      return res.status(400).json({
        success: false,
        error: '지갑 주소가 필요합니다'
      });
    }
    
    // 마이닝 세션 찾기
    let targetSession = null;
    if (sessionId) {
      targetSession = activeMiningSessions.get(sessionId);
    } else {
      // 지갑 주소로 세션 찾기
      for (const [id, session] of activeMiningSessions) {
        if (session.walletAddress === walletAddress && session.isActive) {
          targetSession = session;
          break;
        }
      }
    }
    
    if (!targetSession) {
      return res.status(404).json({
        success: false,
        error: '활성 마이닝 세션을 찾을 수 없습니다'
      });
    }
    
    // 마이닝 세션 중지
    targetSession.isActive = false;
    targetSession.endTime = new Date().toISOString();
    targetSession.totalMined = calculateTotalMined(targetSession);
    
    // 세션 저장
    activeMiningSessions.set(targetSession.id, targetSession);
    
    res.json({
      success: true,
      sessionId: targetSession.id,
      totalMined: targetSession.totalMined,
      message: 'BW포인트 마이닝이 중지되었습니다'
    });
  } catch (error) {
    console.error('마이닝 중지 오류:', error);
    res.status(500).json({
      success: false,
      error: '마이닝 중지 중 오류가 발생했습니다',
      message: error.message
    });
  }
});

// 총 마이닝량 계산 함수
function calculateTotalMined(session) {
  const startTime = new Date(session.startTime);
  const endTime = new Date(session.endTime || new Date());
  const hours = (endTime - startTime) / (1000 * 60 * 60);
  return Math.round(hours * session.hourlyRate * 1000) / 1000;
}
```
