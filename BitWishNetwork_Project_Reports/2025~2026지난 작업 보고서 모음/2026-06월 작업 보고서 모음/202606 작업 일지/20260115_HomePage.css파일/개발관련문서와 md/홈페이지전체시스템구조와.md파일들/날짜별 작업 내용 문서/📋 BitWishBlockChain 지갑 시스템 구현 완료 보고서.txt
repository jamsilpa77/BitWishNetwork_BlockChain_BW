```typescript
const authenticateWithSeedPhrase = async (walletAddress: string, seedPhrase: string) => {
  // 1단계: API로 시드문구 검증
  const response = await fetch('http://localhost:4001/wallet/validate-seedphrase', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ seedPhrase: seedPhrase.trim() })
  });
  
  // 2단계: 검증 결과 처리
  if (response.ok) {
    const data = await response.json();
    return { isValid: data.success, userType: 'USER' };
  }
  return { isValid: false, userType: 'USER' };
};
```

## **📌 4단계: 실제 저장 시스템 문제 진단 및 해결**

### 4.1 **문제 진단**
```javascript
// 문제: StellarWalletSystem 클래스가 정의되었지만 인스턴스화되지 않음
class StellarWalletSystem {
  constructor() {
    this.wallets = new Map(); // ❌ 인스턴스가 없어서 undefined
  }
}

// API에서 호출 시 오류 발생
const result = stellarWalletSystem.createWalletFromSeedPhrase(); // ❌ undefined
```

### 4.2 **해결 방법**
```javascript
// simple-server.js 라인 205 이후에 추가
const seedPhraseManager = new BitWishSeedPhraseManager();
const stellarWalletSystem = new StellarWalletSystem(); // ✅ 인스턴스 생성
```

### 4.3 **API 수정**
```javascript
// POST /wallet/create-stellar-wallet API 수정
app.post('/wallet/create-stellar-wallet', async (req, res) => {
  try {
    const { seedPhrase, protocol, network, userId } = req.body;
    
    // BIP39 검증
    if (!bip39.validateMnemonic(seedPhrase)) {
      return res.status(400).json({ success: false, error: '유효하지 않은 시드문구입니다' });
    }
    
    // ✅ 실제 저장 로직 호출
    const result = stellarWalletSystem.createWalletFromSeedPhrase(seedPhrase, userId || 'default_user');
    
    if (!result.success) {
      return res.status(400).json({ success: false, error: result.error });
    }
    
    // ✅ 시드문구 매니저에 저장
    seedPhraseManager.saveWalletSeedPhrase(result.wallet.address, seedPhrase);
    
    // ✅ 현재 총 지갑 개수 확인
    const totalWallets = stellarWalletSystem.wallets.size;
    console.log(`🎉 새로운 스텔라 지갑 생성 완료! 총 지갑 개수: ${totalWallets}개`);
    
    res.json({
      success: true,
      publicKey: result.wallet.publicKey,
      privateKey: result.wallet.secretKey,
      walletAddress: result.wallet.address,
      wallet: result.wallet,
      totalWallets: totalWallets,
      message: '스텔라 지갑이 성공적으로 생성되고 저장되었습니다'
    });
  } catch (error) {
    console.error('스텔라 지갑 생성 오류:', error);
    res.status(500).json({ success: false, error: '스텔라 지갑 생성 중 오류가 발생했습니다' });
  }
});
```

```javascript
class StellarWalletSystem {
  // ✅ 지갑 개수 조회 메서드
  getWalletCount() {
    return this.wallets.size;
  }
  
  // ✅ 모든 지갑 목록 조회 메서드
  getAllWallets() {
    return Array.from(this.wallets.values());
  }
  
  // ✅ 특정 지갑 조회 메서드
  getWalletById(walletId) {
    return this.wallets.get(walletId);
  }
  
  // ✅ 지갑 통계 조회 메서드
  getWalletStats() {
    const wallets = Array.from(this.wallets.values());
    const totalBalance = wallets.reduce((sum, wallet) => sum + (wallet.balance || 0), 0);
    const activeWallets = wallets.filter(wallet => wallet.status === 'active').length;
    
    return {
      totalWallets: this.wallets.size,
      activeWallets: activeWallets,
      totalBalance: totalBalance,
      averageBalance: this.wallets.size > 0 ? totalBalance / this.wallets.size : 0
    };
  }
}
```
