import React, { useState, useEffect } from 'react';
import { useTranslation } from 'react-i18next';
import { 
  CpuChip, 
  Play, 
  Pause, 
  Square,
  Settings,
  TrendingUp,
  DollarSign,
  Clock,
  Zap,
  Shield,
  Star,
  Gift,
  Users,
  BarChart3,
  AlertTriangle,
  CheckCircle,
  XCircle,
  ArrowRight,
  RefreshCw,
  Activity
} from 'lucide-react';

// ====================================================================================
// 개인 마이닝 페이지 인터페이스 (완벽한 독립성 보장)
// ====================================================================================
interface PerfectPersonalMiningPageProps {
  walletAddress: string;
  userType: 'USER' | 'ADMIN';
  onNavigateToBonusSettings: () => void;
  onError: (error: string) => void;
  onSuccess: (message: string) => void;
}

// ====================================================================================
// 마이닝 데이터 타입 (완벽한 독립성 보장)
// ====================================================================================
interface MiningData {
  isActive: boolean;
  startTime: string | null;
  totalMined: string;
  currentHashRate: number;
  averageHashRate: number;
  miningPower: number;
  efficiency: number;
  lastBlockTime: string | null;
  blocksMined: number;
  rewardsEarned: string;
  pendingRewards: string;
  miningHistory: MiningHistoryItem[];
  bonusSettings: MiningBonusSettings;
  lastUpdated: string;
}

interface MiningHistoryItem {
  id: string;
  timestamp: string;
  hashRate: number;
  powerConsumption: number;
  efficiency: number;
  rewards: string;
  status: 'ACTIVE' | 'PAUSED' | 'STOPPED';
}

interface MiningBonusSettings {
  baseRewardRate: number;
  attendanceBonus: number;
  referralBonus: number;
  lockupBonus: number;
  halvingSystem: boolean;
  maxMiningTime: number; // 시간
  autoRestart: boolean;
  powerSavingMode: boolean;
}

// ====================================================================================
// 완벽한 개인 마이닝 페이지 컴포넌트 (완벽한 독립성 보장)
// ====================================================================================
const PerfectPersonalMiningPage: React.FC<PerfectPersonalMiningPageProps> = ({
  walletAddress,
  userType,
  onNavigateToBonusSettings,
  onError,
  onSuccess
}) => {
  // ====================================================================================
  // 다국어 번역 시스템 (완벽한 독립성 보장)
  // ====================================================================================
  const { t, i18n } = useTranslation();

  // ====================================================================================
  // 완벽한 독립 상태 관리 (전역 변수 절대 사용 금지)
  // ====================================================================================
  const [isLoading, setIsLoading] = useState(false);
  const [error, setError] = useState<string>('');
  const [success, setSuccess] = useState<string>('');
  
  // 마이닝 데이터 (완벽한 독립성 보장)
  const [miningData, setMiningData] = useState<MiningData>({
    isActive: false,
    startTime: null,
    totalMined: '0',
    currentHashRate: 0,
    averageHashRate: 0,
    miningPower: 0,
    efficiency: 0,
    lastBlockTime: null,
    blocksMined: 0,
    rewardsEarned: '0',
    pendingRewards: '0',
    miningHistory: [],
    bonusSettings: {
      baseRewardRate: 1.0,
      attendanceBonus: 0.1,
      referralBonus: 0.05,
      lockupBonus: 0.2,
      halvingSystem: true,
      maxMiningTime: 24,
      autoRestart: true,
      powerSavingMode: false
    },
    lastUpdated: ''
  });
  
  // UI 상태 (완벽한 독립성 보장)
  const [activeTab, setActiveTab] = useState<'overview' | 'history' | 'settings' | 'bonus'>('overview');
  const [showBonusSettings, setShowBonusSettings] = useState(false);
  const [miningTime, setMiningTime] = useState(0);
  const [refreshInterval, setRefreshInterval] = useState<NodeJS.Timeout | null>(null);

  // ====================================================================================
  // 완벽한 독립 API 호출 함수들 (공통 함수 절대 사용 금지)
  // ====================================================================================

  /**
   * 마이닝 데이터 로드
   * - 완벽한 독립성 보장
   * - 실시간 마이닝 상태 조회
   */
  const loadMiningData = async (): Promise<void> => {
    try {
      setIsLoading(true);
      setError('');
      
      // 백엔드 API 호출 (완벽한 독립성 보장)
      const response = await fetch('http://localhost:4001/mining/load-data', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-Request-Source': 'PerfectPersonalMiningPage',
          'X-User-Type': userType
        },
        body: JSON.stringify({
          walletAddress: walletAddress,
          userType: userType,
          timestamp: new Date().toISOString()
        })
      });

      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
      }

      const data = await response.json();
      
      if (!data.success) {
        throw new Error(data.error || 'Failed to load mining data');
      }

      setMiningData(data.miningData);
      setSuccess(t('mining.dataLoaded', '마이닝 데이터가 로드되었습니다.'));
    } catch (error) {
      console.error('마이닝 데이터 로드 실패:', error);
      const errorMessage = error instanceof Error ? error.message : t('mining.loadFailed', '데이터 로드에 실패했습니다.');
      setError(errorMessage);
      onError(errorMessage);
    } finally {
      setIsLoading(false);
    }
  };

  /**
   * 마이닝 시작
   * - 완벽한 독립성 보장
   * - BW포인트 채굴 시작
   */
  const startMining = async (): Promise<void> => {
    try {
      setIsLoading(true);
      setError('');
      
      // 백엔드 API 호출 (완벽한 독립성 보장)
      const response = await fetch('http://localhost:4001/mining/start', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-Request-Source': 'PerfectPersonalMiningPage',
          'X-User-Type': userType
        },
        body: JSON.stringify({
          walletAddress: walletAddress,
          userType: userType,
          timestamp: new Date().toISOString()
        })
      });

      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
      }

      const data = await response.json();
      
      if (!data.success) {
        throw new Error(data.error || 'Failed to start mining');
      }

      setMiningData(prev => ({
        ...prev,
        isActive: true,
        startTime: new Date().toISOString()
      }));
      
      setSuccess(t('mining.started', '마이닝이 시작되었습니다!'));
      onSuccess(t('mining.started', '마이닝이 시작되었습니다!'));
    } catch (error) {
      console.error('마이닝 시작 실패:', error);
      const errorMessage = error instanceof Error ? error.message : t('mining.startFailed', '마이닝 시작에 실패했습니다.');
      setError(errorMessage);
      onError(errorMessage);
    } finally {
      setIsLoading(false);
    }
  };

  /**
   * 마이닝 일시정지
   * - 완벽한 독립성 보장
   * - 마이닝 상태 일시정지
   */
  const pauseMining = async (): Promise<void> => {
    try {
      setIsLoading(true);
      setError('');
      
      // 백엔드 API 호출 (완벽한 독립성 보장)
      const response = await fetch('http://localhost:4001/mining/pause', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-Request-Source': 'PerfectPersonalMiningPage',
          'X-User-Type': userType
        },
        body: JSON.stringify({
          walletAddress: walletAddress,
          userType: userType,
          timestamp: new Date().toISOString()
        })
      });

      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
      }

      const data = await response.json();
      
      if (!data.success) {
        throw new Error(data.error || 'Failed to pause mining');
      }

      setMiningData(prev => ({
        ...prev,
        isActive: false
      }));
      
      setSuccess(t('mining.paused', '마이닝이 일시정지되었습니다.'));
      onSuccess(t('mining.paused', '마이닝이 일시정지되었습니다.'));
    } catch (error) {
      console.error('마이닝 일시정지 실패:', error);
      const errorMessage = error instanceof Error ? error.message : t('mining.pauseFailed', '마이닝 일시정지에 실패했습니다.');
      setError(errorMessage);
      onError(errorMessage);
    } finally {
      setIsLoading(false);
    }
  };

  /**
   * 마이닝 중지
   * - 완벽한 독립성 보장
   * - 마이닝 완전 중지
   */
  const stopMining = async (): Promise<void> => {
    try {
      setIsLoading(true);
      setError('');
      
      // 백엔드 API 호출 (완벽한 독립성 보장)
      const response = await fetch('http://localhost:4001/mining/stop', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-Request-Source': 'PerfectPersonalMiningPage',
          'X-User-Type': userType
        },
        body: JSON.stringify({
          walletAddress: walletAddress,
          userType: userType,
          timestamp: new Date().toISOString()
        })
      });

      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
      }

      const data = await response.json();
      
      if (!data.success) {
        throw new Error(data.error || 'Failed to stop mining');
      }

      setMiningData(prev => ({
        ...prev,
        isActive: false,
        startTime: null
      }));
      
      setSuccess(t('mining.stopped', '마이닝이 중지되었습니다.'));
      onSuccess(t('mining.stopped', '마이닝이 중지되었습니다.'));
    } catch (error) {
      console.error('마이닝 중지 실패:', error);
      const errorMessage = error instanceof Error ? error.message : t('mining.stopFailed', '마이닝 중지에 실패했습니다.');
      setError(errorMessage);
      onError(errorMessage);
    } finally {
      setIsLoading(false);
    }
  };

  /**
   * 보너스 설정 업데이트
   * - 완벽한 독립성 보장
   * - 마이닝 보너스 설정 변경
   */
  const updateBonusSettings = async (newSettings: Partial<MiningBonusSettings>): Promise<void> => {
    try {
      setIsLoading(true);
      setError('');
      
      // 백엔드 API 호출 (완벽한 독립성 보장)
      const response = await fetch('http://localhost:4001/mining/update-bonus-settings', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-Request-Source': 'PerfectPersonalMiningPage',
          'X-User-Type': userType
        },
        body: JSON.stringify({
          walletAddress: walletAddress,
          settings: newSettings,
          timestamp: new Date().toISOString()
        })
      });

      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
      }

      const data = await response.json();
      
      if (!data.success) {
        throw new Error(data.error || 'Failed to update bonus settings');
      }

      setMiningData(prev => ({
        ...prev,
        bonusSettings: { ...prev.bonusSettings, ...newSettings }
      }));
      
      setSuccess(t('mining.settingsUpdated', '보너스 설정이 업데이트되었습니다.'));
      onSuccess(t('mining.settingsUpdated', '보너스 설정이 업데이트되었습니다.'));
    } catch (error) {
      console.error('보너스 설정 업데이트 실패:', error);
      const errorMessage = error instanceof Error ? error.message : t('mining.settingsUpdateFailed', '설정 업데이트에 실패했습니다.');
      setError(errorMessage);
      onError(errorMessage);
    } finally {
      setIsLoading(false);
    }
  };

  // ====================================================================================
  // 완벽한 독립 유틸리티 함수들 (공통 함수 절대 사용 금지)
  // ====================================================================================

  /**
   * 마이닝 시간 계산
   * - 완벽한 독립성 보장
   * - 실시간 마이닝 시간 계산
   */
  const calculateMiningTime = (): number => {
    if (!miningData.isActive || !miningData.startTime) return 0;
    
    const startTime = new Date(miningData.startTime).getTime();
    const currentTime = new Date().getTime();
    return Math.floor((currentTime - startTime) / 1000); // 초 단위
  };

  /**
   * 시간 포맷팅
   * - 완벽한 독립성 보장
   * - 시:분:초 형식으로 변환
   */
  const formatTime = (seconds: number): string => {
    const hours = Math.floor(seconds / 3600);
    const minutes = Math.floor((seconds % 3600) / 60);
    const secs = seconds % 60;
    
    return `${hours.toString().padStart(2, '0')}:${minutes.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
  };

  /**
   * BW포인트 금액 포맷팅
   * - 완벽한 독립성 보장
   * - 50자리 정밀도 지원
   */
  const formatBWAmount = (amount: string): string => {
    try {
      const num = parseFloat(amount);
      if (isNaN(num)) return '0.00';
      return num.toFixed(2);
    } catch (error) {
      return '0.00';
    }
  };

  /**
   * 해시레이트 포맷팅
   * - 완벽한 독립성 보장
   * - H/s, KH/s, MH/s 형식으로 변환
   */
  const formatHashRate = (hashRate: number): string => {
    if (hashRate >= 1000000) {
      return `${(hashRate / 1000000).toFixed(2)} MH/s`;
    } else if (hashRate >= 1000) {
      return `${(hashRate / 1000).toFixed(2)} KH/s`;
    } else {
      return `${hashRate.toFixed(2)} H/s`;
    }
  };

  // ====================================================================================
  // 완벽한 독립 생명주기 관리 (공통 함수 절대 사용 금지)
  // ====================================================================================

  useEffect(() => {
    if (walletAddress) {
      loadMiningData();
    }
  }, [walletAddress, userType]);

  // 실시간 마이닝 시간 업데이트
  useEffect(() => {
    if (miningData.isActive) {
      const interval = setInterval(() => {
        setMiningTime(calculateMiningTime());
      }, 1000);

      setRefreshInterval(interval);

      return () => {
        if (interval) {
          clearInterval(interval);
        }
      };
    } else {
      setMiningTime(0);
    }
  }, [miningData.isActive, miningData.startTime]);

  // 실시간 데이터 새로고침
  useEffect(() => {
    if (walletAddress) {
      const interval = setInterval(() => {
        loadMiningData();
      }, 30000); // 30초마다 새로고침

      return () => {
        if (interval) {
          clearInterval(interval);
        }
      };
    }
  }, [walletAddress, userType]);

  // 컴포넌트 언마운트 시 정리
  useEffect(() => {
    return () => {
      if (refreshInterval) {
        clearInterval(refreshInterval);
      }
    };
  }, [refreshInterval]);

  // ====================================================================================
  // 완벽한 독립 렌더링 함수들 (공통 함수 절대 사용 금지)
  // ====================================================================================

  /**
   * 개요 탭 렌더링
   */
  const renderOverviewTab = () => (
    <div className="space-y-6">
      {/* 마이닝 상태 카드 */}
      <div className="bg-gradient-to-r from-blue-50 to-purple-50 rounded-lg p-6 border border-blue-200">
        <div className="flex items-center justify-between mb-4">
          <h3 className="text-lg font-semibold text-gray-900">
            {t('mining.miningStatus', '마이닝 상태')}
          </h3>
          <div className={`px-3 py-1 rounded-full text-sm font-medium ${
            miningData.isActive 
              ? 'bg-green-100 text-green-800' 
              : 'bg-gray-100 text-gray-800'
          }`}>
            {miningData.isActive ? t('mining.active', '활성') : t('mining.inactive', '비활성')}
          </div>
        </div>
        
        <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
          <div className="bg-white rounded-lg p-4">
            <div className="flex items-center space-x-3">
              <Clock className="w-5 h-5 text-blue-500" />
              <div>
                <div className="text-2xl font-bold text-gray-900">
                  {formatTime(miningTime)}
                </div>
                <div className="text-sm text-gray-500">
                  {t('mining.miningTime', '마이닝 시간')}
                </div>
              </div>
            </div>
          </div>

          <div className="bg-white rounded-lg p-4">
            <div className="flex items-center space-x-3">
              <Activity className="w-5 h-5 text-green-500" />
              <div>
                <div className="text-2xl font-bold text-gray-900">
                  {formatHashRate(miningData.currentHashRate)}
                </div>
                <div className="text-sm text-gray-500">
                  {t('mining.currentHashRate', '현재 해시레이트')}
                </div>
              </div>
            </div>
          </div>

          <div className="bg-white rounded-lg p-4">
            <div className="flex items-center space-x-3">
              <DollarSign className="w-5 h-5 text-purple-500" />
              <div>
                <div className="text-2xl font-bold text-gray-900">
                  {formatBWAmount(miningData.totalMined)} BW
                </div>
                <div className="text-sm text-gray-500">
                  {t('mining.totalMined', '총 채굴량')}
                </div>
              </div>
            </div>
          </div>
        </div>
      </div>

      {/* 마이닝 제어 버튼 */}
      <div className="bg-white rounded-lg p-6 border border-gray-200">
        <div className="flex items-center justify-between mb-4">
          <h3 className="text-lg font-semibold text-gray-900">
            {t('mining.miningControl', '마이닝 제어')}
          </h3>
          <div className="flex items-center space-x-2">
            <Settings className="w-5 h-5 text-gray-400" />
            <button
              onClick={() => setActiveTab('settings')}
              className="text-sm text-blue-600 hover:text-blue-800"
            >
              {t('mining.settings', '설정')}
            </button>
          </div>
        </div>
        
        <div className="flex space-x-4">
          {!miningData.isActive ? (
            <button
              onClick={startMining}
              disabled={isLoading}
              className="flex-1 px-6 py-3 bg-gradient-to-r from-green-600 to-blue-600 text-white rounded-lg hover:shadow-lg transition-all disabled:opacity-50"
            >
              <div className="flex items-center justify-center space-x-2">
                <Play className="w-5 h-5" />
                <span>{isLoading ? t('common.loading', '로딩...') : t('mining.startMining', '마이닝 시작')}</span>
              </div>
            </button>
          ) : (
            <>
              <button
                onClick={pauseMining}
                disabled={isLoading}
                className="flex-1 px-6 py-3 bg-gradient-to-r from-yellow-600 to-orange-600 text-white rounded-lg hover:shadow-lg transition-all disabled:opacity-50"
              >
                <div className="flex items-center justify-center space-x-2">
                  <Pause className="w-5 h-5" />
                  <span>{isLoading ? t('common.loading', '로딩...') : t('mining.pauseMining', '일시정지')}</span>
                </div>
              </button>
              <button
                onClick={stopMining}
                disabled={isLoading}
                className="flex-1 px-6 py-3 bg-gradient-to-r from-red-600 to-pink-600 text-white rounded-lg hover:shadow-lg transition-all disabled:opacity-50"
              >
                <div className="flex items-center justify-center space-x-2">
                  <Square className="w-5 h-5" />
                  <span>{isLoading ? t('common.loading', '로딩...') : t('mining.stopMining', '중지')}</span>
                </div>
              </button>
            </>
          )}
        </div>
      </div>

      {/* 보너스 설정 카드 */}
      <div className="bg-gradient-to-r from-green-50 to-blue-50 rounded-lg p-6 border border-green-200">
        <div className="flex items-center justify-between mb-4">
          <h3 className="text-lg font-semibold text-gray-900">
            {t('mining.bonusSettings', '보너스 설정')}
          </h3>
          <button
            onClick={() => setActiveTab('bonus')}
            className="px-4 py-2 bg-green-600 text-white rounded-lg hover:bg-green-700 transition-colors"
          >
            {t('mining.configureBonus', '보너스 설정')}
          </button>
        </div>
        
        <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
          <div className="text-center">
            <div className="text-2xl font-bold text-green-600">
              {miningData.bonusSettings.attendanceBonus * 100}%
            </div>
            <div className="text-sm text-gray-600">
              {t('mining.attendanceBonus', '출석 보너스')}
            </div>
          </div>
          <div className="text-center">
            <div className="text-2xl font-bold text-blue-600">
              {miningData.bonusSettings.referralBonus * 100}%
            </div>
            <div className="text-sm text-gray-600">
              {t('mining.referralBonus', '추천 보너스')}
            </div>
          </div>
          <div className="text-center">
            <div className="text-2xl font-bold text-purple-600">
              {miningData.bonusSettings.lockupBonus * 100}%
            </div>
            <div className="text-sm text-gray-600">
              {t('mining.lockupBonus', '락업 보너스')}
            </div>
          </div>
          <div className="text-center">
            <div className="text-2xl font-bold text-orange-600">
              {miningData.bonusSettings.baseRewardRate}x
            </div>
            <div className="text-sm text-gray-600">
              {t('mining.baseReward', '기본 보상')}
            </div>
          </div>
        </div>
      </div>
    </div>
  );

  /**
   * 히스토리 탭 렌더링
   */
  const renderHistoryTab = () => (
    <div className="space-y-4">
      <div className="flex items-center justify-between">
        <h3 className="text-lg font-semibold text-gray-900">
          {t('mining.miningHistory', '마이닝 히스토리')}
        </h3>
        <div className="text-sm text-gray-500">
          {t('mining.lastUpdated', '마지막 업데이트')}: {miningData.lastUpdated}
        </div>
      </div>

      {miningData.miningHistory.length === 0 ? (
        <div className="text-center py-8">
          <Activity className="w-12 h-12 text-gray-300 mx-auto mb-4" />
          <p className="text-gray-500">
            {t('mining.noHistoryYet', '아직 마이닝 히스토리가 없습니다.')}
          </p>
        </div>
      ) : (
        <div className="space-y-3">
          {miningData.miningHistory.map((item) => (
            <div key={item.id} className="bg-white rounded-lg p-4 border border-gray-200">
              <div className="flex items-center justify-between">
                <div className="flex-1">
                  <div className="flex items-center space-x-4">
                    <div className="text-sm text-gray-600">
                      {new Date(item.timestamp).toLocaleString()}
                    </div>
                    <div className="text-sm text-gray-600">
                      {formatHashRate(item.hashRate)}
                    </div>
                    <div className="text-sm text-gray-600">
                      {formatBWAmount(item.rewards)} BW
                    </div>
                  </div>
                </div>
                <div className="flex items-center space-x-3">
                  <span className={`px-2 py-1 rounded-full text-xs font-medium ${
                    item.status === 'ACTIVE' ? 'bg-green-100 text-green-800' :
                    item.status === 'PAUSED' ? 'bg-yellow-100 text-yellow-800' :
                    'bg-red-100 text-red-800'
                  }`}>
                    {t(`mining.status.${item.status}`, item.status)}
                  </span>
                </div>
              </div>
            </div>
          ))}
        </div>
      )}
    </div>
  );

  /**
   * 설정 탭 렌더링
   */
  const renderSettingsTab = () => (
    <div className="space-y-6">
      <h3 className="text-lg font-semibold text-gray-900">
        {t('mining.miningSettings', '마이닝 설정')}
      </h3>

      <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
        <div className="space-y-4">
          <div>
            <label className="block text-sm font-medium text-gray-700 mb-2">
              {t('mining.maxMiningTime', '최대 마이닝 시간 (시간)')}
            </label>
            <input
              type="number"
              min="1"
              max="168"
              value={miningData.bonusSettings.maxMiningTime}
              onChange={(e) => {
                const newSettings = {
                  ...miningData.bonusSettings,
                  maxMiningTime: parseInt(e.target.value)
                };
                setMiningData(prev => ({
                  ...prev,
                  bonusSettings: newSettings
                }));
              }}
              className="w-full px-3 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent"
            />
          </div>

          <div className="flex items-center space-x-3">
            <input
              type="checkbox"
              id="autoRestart"
              checked={miningData.bonusSettings.autoRestart}
              onChange={(e) => {
                const newSettings = {
                  ...miningData.bonusSettings,
                  autoRestart: e.target.checked
                };
                setMiningData(prev => ({
                  ...prev,
                  bonusSettings: newSettings
                }));
              }}
              className="w-4 h-4 text-blue-600 border-gray-300 rounded focus:ring-blue-500"
            />
            <label htmlFor="autoRestart" className="text-sm text-gray-700">
              {t('mining.autoRestart', '자동 재시작')}
            </label>
          </div>

          <div className="flex items-center space-x-3">
            <input
              type="checkbox"
              id="powerSavingMode"
              checked={miningData.bonusSettings.powerSavingMode}
              onChange={(e) => {
                const newSettings = {
                  ...miningData.bonusSettings,
                  powerSavingMode: e.target.checked
                };
                setMiningData(prev => ({
                  ...prev,
                  bonusSettings: newSettings
                }));
              }}
              className="w-4 h-4 text-blue-600 border-gray-300 rounded focus:ring-blue-500"
            />
            <label htmlFor="powerSavingMode" className="text-sm text-gray-700">
              {t('mining.powerSavingMode', '절전 모드')}
            </label>
          </div>
        </div>

        <div className="space-y-4">
          <div>
            <label className="block text-sm font-medium text-gray-700 mb-2">
              {t('mining.baseRewardRate', '기본 보상 비율')}
            </label>
            <input
              type="number"
              step="0.1"
              min="0.1"
              max="10"
              value={miningData.bonusSettings.baseRewardRate}
              onChange={(e) => {
                const newSettings = {
                  ...miningData.bonusSettings,
                  baseRewardRate: parseFloat(e.target.value)
                };
                setMiningData(prev => ({
                  ...prev,
                  bonusSettings: newSettings
                }));
              }}
              className="w-full px-3 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent"
            />
          </div>

          <div className="flex items-center space-x-3">
            <input
              type="checkbox"
              id="halvingSystem"
              checked={miningData.bonusSettings.halvingSystem}
              onChange={(e) => {
                const newSettings = {
                  ...miningData.bonusSettings,
                  halvingSystem: e.target.checked
                };
                setMiningData(prev => ({
                  ...prev,
                  bonusSettings: newSettings
                }));
              }}
              className="w-4 h-4 text-blue-600 border-gray-300 rounded focus:ring-blue-500"
            />
            <label htmlFor="halvingSystem" className="text-sm text-gray-700">
              {t('mining.halvingSystem', '반감 시스템')}
            </label>
          </div>
        </div>
      </div>

      <button
        onClick={() => updateBonusSettings(miningData.bonusSettings)}
        disabled={isLoading}
        className="px-4 py-2 bg-blue-600 text-white rounded-lg hover:bg-blue-700 transition-colors disabled:opacity-50"
      >
        {isLoading ? t('common.loading', '로딩...') : t('mining.saveSettings', '설정 저장')}
      </button>
    </div>
  );

  /**
   * 보너스 탭 렌더링
   */
  const renderBonusTab = () => (
    <div className="space-y-6">
      <h3 className="text-lg font-semibold text-gray-900">
        {t('mining.bonusConfiguration', '보너스 설정')}
      </h3>

      <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
        <div className="space-y-4">
          <div>
            <label className="block text-sm font-medium text-gray-700 mb-2">
              {t('mining.attendanceBonus', '출석 보너스 (%)')}
            </label>
            <input
              type="number"
              step="0.1"
              min="0"
              max="100"
              value={miningData.bonusSettings.attendanceBonus * 100}
              onChange={(e) => {
                const newSettings = {
                  ...miningData.bonusSettings,
                  attendanceBonus: parseFloat(e.target.value) / 100
                };
                setMiningData(prev => ({
                  ...prev,
                  bonusSettings: newSettings
                }));
              }}
              className="w-full px-3 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent"
            />
          </div>

          <div>
            <label className="block text-sm font-medium text-gray-700 mb-2">
              {t('mining.referralBonus', '추천 보너스 (%)')}
            </label>
            <input
              type="number"
              step="0.1"
              min="0"
              max="100"
              value={miningData.bonusSettings.referralBonus * 100}
              onChange={(e) => {
                const newSettings = {
                  ...miningData.bonusSettings,
                  referralBonus: parseFloat(e.target.value) / 100
                };
                setMiningData(prev => ({
                  ...prev,
                  bonusSettings: newSettings
                }));
              }}
              className="w-full px-3 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent"
            />
          </div>
        </div>

        <div className="space-y-4">
          <div>
            <label className="block text-sm font-medium text-gray-700 mb-2">
              {t('mining.lockupBonus', '락업 보너스 (%)')}
            </label>
            <input
              type="number"
              step="0.1"
              min="0"
              max="100"
              value={miningData.bonusSettings.lockupBonus * 100}
              onChange={(e) => {
                const newSettings = {
                  ...miningData.bonusSettings,
                  lockupBonus: parseFloat(e.target.value) / 100
                };
                setMiningData(prev => ({
                  ...prev,
                  bonusSettings: newSettings
                }));
              }}
              className="w-full px-3 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent"
            />
          </div>
        </div>
      </div>

      <button
        onClick={() => updateBonusSettings(miningData.bonusSettings)}
        disabled={isLoading}
        className="px-4 py-2 bg-green-600 text-white rounded-lg hover:bg-green-700 transition-colors disabled:opacity-50"
      >
        {isLoading ? t('common.loading', '로딩...') : t('mining.saveBonusSettings', '보너스 설정 저장')}
      </button>
    </div>
  );

  // ====================================================================================
  // 완벽한 독립 메인 렌더링 (공통 함수 절대 사용 금지)
  // ====================================================================================

  return (
    <div className="min-h-screen bg-gray-50">
      {/* 헤더 */}
      <div className="bg-white shadow-sm border-b border-gray-200">
        <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8">
          <div className="flex items-center justify-between h-16">
            <div className="flex items-center space-x-3">
              <div className="w-10 h-10 bg-gradient-to-r from-blue-500 to-purple-600 rounded-lg flex items-center justify-center">
                <CpuChip className="w-5 h-5 text-white" />
              </div>
              <div>
                <h1 className="text-xl font-bold text-gray-900">
                  {t('mining.personalMiningPage', '개인 마이닝 페이지')}
                </h1>
                <p className="text-sm text-gray-500">
                  {t('mining.subtitle', '나만의 채굴 노드 마이닝')}
                </p>
              </div>
            </div>
            <div className="flex items-center space-x-2">
              <div className={`px-2 py-1 rounded-full text-xs font-medium ${
                userType === 'ADMIN' 
                  ? 'bg-yellow-100 text-yellow-800' 
                  : 'bg-green-100 text-green-800'
              }`}>
                {userType === 'ADMIN' ? t('common.admin', '관리자') : t('common.user', '사용자')}
              </div>
            </div>
          </div>
        </div>
      </div>

      {/* 탭 네비게이션 */}
      <div className="bg-white border-b border-gray-200">
        <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8">
          <nav className="flex space-x-8">
            {[
              { id: 'overview', label: t('mining.overview', '개요'), icon: BarChart3 },
              { id: 'history', label: t('mining.history', '히스토리'), icon: Clock },
              { id: 'settings', label: t('mining.settings', '설정'), icon: Settings },
              { id: 'bonus', label: t('mining.bonus', '보너스'), icon: Gift }
            ].map((tab) => (
              <button
                key={tab.id}
                onClick={() => setActiveTab(tab.id as any)}
                className={`flex items-center space-x-2 py-4 px-1 border-b-2 font-medium text-sm transition-colors ${
                  activeTab === tab.id
                    ? 'border-blue-500 text-blue-600'
                    : 'border-transparent text-gray-500 hover:text-gray-700 hover:border-gray-300'
                }`}
              >
                <tab.icon className="w-4 h-4" />
                <span>{tab.label}</span>
              </button>
            ))}
          </nav>
        </div>
      </div>

      {/* 콘텐츠 */}
      <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-6">
        {/* 에러 메시지 */}
        {error && (
          <div className="mb-4 p-4 bg-red-50 border border-red-200 rounded-lg">
            <div className="flex items-center space-x-2">
              <XCircle className="w-5 h-5 text-red-500" />
              <span className="text-red-800">{error}</span>
            </div>
          </div>
        )}

        {/* 성공 메시지 */}
        {success && (
          <div className="mb-4 p-4 bg-green-50 border border-green-200 rounded-lg">
            <div className="flex items-center space-x-2">
              <CheckCircle className="w-5 h-5 text-green-500" />
              <span className="text-green-800">{success}</span>
            </div>
          </div>
        )}

        {/* 로딩 상태 */}
        {isLoading && (
          <div className="flex items-center justify-center py-8">
            <div className="animate-spin rounded-full h-8 w-8 border-b-2 border-blue-600"></div>
            <span className="ml-2 text-gray-600">
              {t('common.loading', '로딩...')}
            </span>
          </div>
        )}

        {/* 탭별 콘텐츠 */}
        {!isLoading && (
          <>
            {activeTab === 'overview' && renderOverviewTab()}
            {activeTab === 'history' && renderHistoryTab()}
            {activeTab === 'settings' && renderSettingsTab()}
            {activeTab === 'bonus' && renderBonusTab()}
          </>
        )}
      </div>
    </div>
  );
};

export default PerfectPersonalMiningPage;
