import React, { useState, useEffect } from 'react';
import { useTranslation } from 'react-i18next';
import { 
  Gift, 
  Users, 
  TrendingUp, 
  Star,
  Copy,
  CheckCircle,
  XCircle,
  AlertTriangle,
  ArrowRight,
  Wallet,
  CpuChip,
  DollarSign,
  Clock,
  Shield
} from 'lucide-react';

// ====================================================================================
// 추천 보너스 시스템 인터페이스 (완벽한 독립성 보장)
// ====================================================================================
interface ReferralBonusSystemProps {
  walletAddress: string;
  userType: 'USER' | 'ADMIN';
  onNavigateToMining: () => void;
  onError: (error: string) => void;
  onSuccess: (message: string) => void;
}

// ====================================================================================
// 추천 보너스 데이터 타입 (완벽한 독립성 보장)
// ====================================================================================
interface ReferralBonusData {
  referralCode: string;
  totalReferrals: number;
  activeReferrals: number;
  totalBonusEarned: string;
  pendingBonus: string;
  referralHistory: ReferralHistoryItem[];
  bonusSettings: BonusSettings;
  lastUpdated: string;
}

interface ReferralHistoryItem {
  id: string;
  referredWalletAddress: string;
  bonusAmount: string;
  status: 'PENDING' | 'CONFIRMED' | 'CANCELLED';
  createdAt: string;
  confirmedAt?: string;
}

interface BonusSettings {
  baseBonusRate: number;
  level1BonusRate: number;
  level2BonusRate: number;
  level3BonusRate: number;
  maxReferralLevel: number;
  minimumMiningTime: number; // 분
  bonusCalculationMethod: 'FIXED' | 'PERCENTAGE';
}

// ====================================================================================
// 완벽한 추천 보너스 시스템 컴포넌트 (완벽한 독립성 보장)
// ====================================================================================
const ReferralBonusSystem: React.FC<ReferralBonusSystemProps> = ({
  walletAddress,
  userType,
  onNavigateToMining,
  onError,
  onSuccess
}) => {
  // ====================================================================================
  // 다국어 번역 시스템 (완벽한 독립성 보장)
  // ====================================================================================
  const { t, i18n } = useTranslation();

  // ====================================================================================
  // 완벽한 독립 상태 관리 (전역 변수 절대 사용 금지)
  // ====================================================================================
  const [isLoading, setIsLoading] = useState(false);
  const [error, setError] = useState<string>('');
  const [success, setSuccess] = useState<string>('');
  
  // 추천 보너스 데이터 (완벽한 독립성 보장)
  const [referralData, setReferralData] = useState<ReferralBonusData>({
    referralCode: '',
    totalReferrals: 0,
    activeReferrals: 0,
    totalBonusEarned: '0',
    pendingBonus: '0',
    referralHistory: [],
    bonusSettings: {
      baseBonusRate: 0.1,
      level1BonusRate: 0.05,
      level2BonusRate: 0.03,
      level3BonusRate: 0.01,
      maxReferralLevel: 3,
      minimumMiningTime: 60,
      bonusCalculationMethod: 'PERCENTAGE'
    },
    lastUpdated: ''
  });
  
  // UI 상태 (완벽한 독립성 보장)
  const [activeTab, setActiveTab] = useState<'overview' | 'history' | 'settings'>('overview');
  const [showReferralCode, setShowReferralCode] = useState(false);
  const [referralCodeCopied, setReferralCodeCopied] = useState(false);
  const [refreshInterval, setRefreshInterval] = useState<NodeJS.Timeout | null>(null);

  // ====================================================================================
  // 완벽한 독립 API 호출 함수들 (공통 함수 절대 사용 금지)
  // ====================================================================================

  /**
   * 추천코드 발급 및 보너스 데이터 로드
   * - 완벽한 독립성 보장
   * - bitwishnetwork 인증된 시스템 연동
   */
  const loadReferralBonusData = async (): Promise<void> => {
    try {
      setIsLoading(true);
      setError('');
      
      // 백엔드 API 호출 (완벽한 독립성 보장)
      const response = await fetch('http://localhost:4001/referral/load-bonus-data', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-Request-Source': 'ReferralBonusSystem',
          'X-User-Type': userType
        },
        body: JSON.stringify({
          walletAddress: walletAddress,
          userType: userType,
          timestamp: new Date().toISOString()
        })
      });

      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
      }

      const data = await response.json();
      
      if (!data.success) {
        throw new Error(data.error || 'Failed to load referral bonus data');
      }

      setReferralData(data.referralData);
      setSuccess(t('referral.dataLoaded', '추천 보너스 데이터가 로드되었습니다.'));
    } catch (error) {
      console.error('추천 보너스 데이터 로드 실패:', error);
      const errorMessage = error instanceof Error ? error.message : t('referral.loadFailed', '데이터 로드에 실패했습니다.');
      setError(errorMessage);
      onError(errorMessage);
    } finally {
      setIsLoading(false);
    }
  };

  /**
   * 추천코드 생성 및 발급
   * - 완벽한 독립성 보장
   * - 지갑 주소 기반 고유 추천코드
   */
  const generateReferralCode = async (): Promise<string> => {
    try {
      setIsLoading(true);
      setError('');
      
      // 백엔드 API 호출 (완벽한 독립성 보장)
      const response = await fetch('http://localhost:4001/referral/generate-code', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-Request-Source': 'ReferralBonusSystem',
          'X-User-Type': userType
        },
        body: JSON.stringify({
          walletAddress: walletAddress,
          userType: userType,
          timestamp: new Date().toISOString()
        })
      });

      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
      }

      const data = await response.json();
      
      if (!data.success) {
        throw new Error(data.error || 'Failed to generate referral code');
      }

      return data.referralCode;
    } catch (error) {
      console.error('추천코드 생성 실패:', error);
      throw new Error(error instanceof Error ? error.message : t('referral.codeGenerationFailed', '추천코드 생성에 실패했습니다.'));
    } finally {
      setIsLoading(false);
    }
  };

  /**
   * 추천 보너스 설정 업데이트
   * - 완벽한 독립성 보장
   * - 관리자 전용 기능
   */
  const updateBonusSettings = async (newSettings: Partial<BonusSettings>): Promise<void> => {
    try {
      setIsLoading(true);
      setError('');
      
      if (userType !== 'ADMIN') {
        throw new Error(t('referral.adminOnly', '관리자만 설정을 변경할 수 있습니다.'));
      }
      
      // 백엔드 API 호출 (완벽한 독립성 보장)
      const response = await fetch('http://localhost:4001/referral/update-settings', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-Request-Source': 'ReferralBonusSystem',
          'X-User-Type': userType
        },
        body: JSON.stringify({
          walletAddress: walletAddress,
          settings: newSettings,
          timestamp: new Date().toISOString()
        })
      });

      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
      }

      const data = await response.json();
      
      if (!data.success) {
        throw new Error(data.error || 'Failed to update bonus settings');
      }

      setReferralData(prev => ({
        ...prev,
        bonusSettings: { ...prev.bonusSettings, ...newSettings }
      }));
      
      setSuccess(t('referral.settingsUpdated', '보너스 설정이 업데이트되었습니다.'));
      onSuccess(t('referral.settingsUpdated', '보너스 설정이 업데이트되었습니다.'));
    } catch (error) {
      console.error('보너스 설정 업데이트 실패:', error);
      const errorMessage = error instanceof Error ? error.message : t('referral.settingsUpdateFailed', '설정 업데이트에 실패했습니다.');
      setError(errorMessage);
      onError(errorMessage);
    } finally {
      setIsLoading(false);
    }
  };

  /**
   * 추천 보너스 지급 처리
   * - 완벽한 독립성 보장
   * - 실시간 보너스 계산 및 지급
   */
  const processReferralBonus = async (referralId: string): Promise<void> => {
    try {
      setIsLoading(true);
      setError('');
      
      // 백엔드 API 호출 (완벽한 독립성 보장)
      const response = await fetch('http://localhost:4001/referral/process-bonus', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-Request-Source': 'ReferralBonusSystem',
          'X-User-Type': userType
        },
        body: JSON.stringify({
          walletAddress: walletAddress,
          referralId: referralId,
          timestamp: new Date().toISOString()
        })
      });

      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
      }

      const data = await response.json();
      
      if (!data.success) {
        throw new Error(data.error || 'Failed to process referral bonus');
      }

      // 데이터 새로고침
      await loadReferralBonusData();
      
      setSuccess(t('referral.bonusProcessed', '추천 보너스가 처리되었습니다.'));
      onSuccess(t('referral.bonusProcessed', '추천 보너스가 처리되었습니다.'));
    } catch (error) {
      console.error('추천 보너스 처리 실패:', error);
      const errorMessage = error instanceof Error ? error.message : t('referral.bonusProcessFailed', '보너스 처리에 실패했습니다.');
      setError(errorMessage);
      onError(errorMessage);
    } finally {
      setIsLoading(false);
    }
  };

  // ====================================================================================
  // 완벽한 독립 유틸리티 함수들 (공통 함수 절대 사용 금지)
  // ====================================================================================

  /**
   * 추천코드 복사
   * - 완벽한 독립성 보장
   * - 클립보드 API 사용
   */
  const handleCopyReferralCode = async (): Promise<void> => {
    try {
      await navigator.clipboard.writeText(referralData.referralCode);
      setReferralCodeCopied(true);
      setSuccess(t('referral.codeCopied', '추천코드가 클립보드에 복사되었습니다.'));
      
      // 3초 후 복사 상태 해제
      setTimeout(() => {
        setReferralCodeCopied(false);
      }, 3000);
    } catch (error) {
      setError(t('referral.copyFailed', '복사에 실패했습니다. 수동으로 복사해주세요.'));
    }
  };

  /**
   * 보너스 금액 포맷팅
   * - 완벽한 독립성 보장
   * - 50자리 정밀도 지원
   */
  const formatBonusAmount = (amount: string): string => {
    try {
      const num = parseFloat(amount);
      if (isNaN(num)) return '0.00';
      return num.toFixed(2);
    } catch (error) {
      return '0.00';
    }
  };

  /**
   * 상태별 색상 반환
   * - 완벽한 독립성 보장
   * - UI 일관성 유지
   */
  const getStatusColor = (status: string): string => {
    switch (status) {
      case 'PENDING':
        return 'text-yellow-600 bg-yellow-100';
      case 'CONFIRMED':
        return 'text-green-600 bg-green-100';
      case 'CANCELLED':
        return 'text-red-600 bg-red-100';
      default:
        return 'text-gray-600 bg-gray-100';
    }
  };

  /**
   * 상태별 텍스트 반환
   * - 완벽한 독립성 보장
   * - 4개국 언어 지원
   */
  const getStatusText = (status: string): string => {
    switch (status) {
      case 'PENDING':
        return t('referral.statusPending', '대기중');
      case 'CONFIRMED':
        return t('referral.statusConfirmed', '확인됨');
      case 'CANCELLED':
        return t('referral.statusCancelled', '취소됨');
      default:
        return t('referral.statusUnknown', '알 수 없음');
    }
  };

  // ====================================================================================
  // 완벽한 독립 생명주기 관리 (공통 함수 절대 사용 금지)
  // ====================================================================================

  useEffect(() => {
    if (walletAddress) {
      loadReferralBonusData();
    }
  }, [walletAddress, userType]);

  // 실시간 데이터 새로고침 (완벽한 독립성 보장)
  useEffect(() => {
    if (walletAddress) {
      const interval = setInterval(() => {
        loadReferralBonusData();
      }, 30000); // 30초마다 새로고침

      setRefreshInterval(interval);

      return () => {
        if (interval) {
          clearInterval(interval);
        }
      };
    }
  }, [walletAddress, userType]);

  // 컴포넌트 언마운트 시 정리
  useEffect(() => {
    return () => {
      if (refreshInterval) {
        clearInterval(refreshInterval);
      }
    };
  }, [refreshInterval]);

  // ====================================================================================
  // 완벽한 독립 렌더링 함수들 (공통 함수 절대 사용 금지)
  // ====================================================================================

  /**
   * 개요 탭 렌더링
   */
  const renderOverviewTab = () => (
    <div className="space-y-6">
      {/* 추천코드 섹션 */}
      <div className="bg-gradient-to-r from-blue-50 to-purple-50 rounded-lg p-6 border border-blue-200">
        <div className="flex items-center justify-between mb-4">
          <h3 className="text-lg font-semibold text-gray-900">
            {t('referral.yourReferralCode', '나의 추천코드')}
          </h3>
          <div className="flex items-center space-x-2">
            <Shield className="w-5 h-5 text-blue-500" />
            <span className="text-sm text-blue-600">
              {t('referral.secureCode', '보안 코드')}
            </span>
          </div>
        </div>
        
        <div className="bg-white rounded-lg p-4 border border-gray-200">
          <div className="flex items-center justify-between">
            <div className="flex-1">
              <div className="font-mono text-lg text-gray-900 break-all">
                {showReferralCode ? referralData.referralCode : '••••••••••••••••'}
              </div>
              <div className="text-sm text-gray-500 mt-1">
                {t('referral.shareWithFriends', '친구들과 공유하세요!')}
              </div>
            </div>
            <div className="flex items-center space-x-2 ml-4">
              <button
                onClick={() => setShowReferralCode(!showReferralCode)}
                className="p-2 text-gray-400 hover:text-gray-600 transition-colors"
              >
                {showReferralCode ? <XCircle className="w-5 h-5" /> : <Wallet className="w-5 h-5" />}
              </button>
              <button
                onClick={handleCopyReferralCode}
                disabled={referralCodeCopied}
                className={`p-2 rounded-lg transition-all ${
                  referralCodeCopied
                    ? 'bg-green-100 text-green-600'
                    : 'bg-blue-100 text-blue-600 hover:bg-blue-200'
                }`}
              >
                <Copy className="w-5 h-5" />
              </button>
            </div>
          </div>
        </div>
      </div>

      {/* 통계 섹션 */}
      <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
        <div className="bg-white rounded-lg p-4 border border-gray-200">
          <div className="flex items-center space-x-3">
            <div className="w-10 h-10 bg-green-100 rounded-lg flex items-center justify-center">
              <Users className="w-5 h-5 text-green-600" />
            </div>
            <div>
              <div className="text-2xl font-bold text-gray-900">
                {referralData.totalReferrals}
              </div>
              <div className="text-sm text-gray-500">
                {t('referral.totalReferrals', '총 추천인')}
              </div>
            </div>
          </div>
        </div>

        <div className="bg-white rounded-lg p-4 border border-gray-200">
          <div className="flex items-center space-x-3">
            <div className="w-10 h-10 bg-blue-100 rounded-lg flex items-center justify-center">
              <TrendingUp className="w-5 h-5 text-blue-600" />
            </div>
            <div>
              <div className="text-2xl font-bold text-gray-900">
                {referralData.activeReferrals}
              </div>
              <div className="text-sm text-gray-500">
                {t('referral.activeReferrals', '활성 추천인')}
              </div>
            </div>
          </div>
        </div>

        <div className="bg-white rounded-lg p-4 border border-gray-200">
          <div className="flex items-center space-x-3">
            <div className="w-10 h-10 bg-purple-100 rounded-lg flex items-center justify-center">
              <DollarSign className="w-5 h-5 text-purple-600" />
            </div>
            <div>
              <div className="text-2xl font-bold text-gray-900">
                {formatBonusAmount(referralData.totalBonusEarned)} BW
              </div>
              <div className="text-sm text-gray-500">
                {t('referral.totalBonus', '총 보너스')}
              </div>
            </div>
          </div>
        </div>
      </div>

      {/* 대기중인 보너스 */}
      {parseFloat(referralData.pendingBonus) > 0 && (
        <div className="bg-yellow-50 border border-yellow-200 rounded-lg p-4">
          <div className="flex items-center space-x-3">
            <Clock className="w-5 h-5 text-yellow-600" />
            <div>
              <div className="font-semibold text-yellow-800">
                {t('referral.pendingBonus', '대기중인 보너스')}
              </div>
              <div className="text-yellow-700">
                {formatBonusAmount(referralData.pendingBonus)} BW
              </div>
            </div>
          </div>
        </div>
      )}

      {/* 마이닝 페이지 이동 버튼 */}
      <div className="bg-gradient-to-r from-green-50 to-blue-50 rounded-lg p-6 border border-green-200">
        <div className="text-center">
          <h3 className="text-lg font-semibold text-gray-900 mb-2">
            {t('referral.readyToMine', '마이닝 준비 완료!')}
          </h3>
          <p className="text-gray-600 mb-4">
            {t('referral.mineDescription', '지갑을 인증하고 마이닝을 시작하세요.')}
          </p>
          <button
            onClick={onNavigateToMining}
            className="px-6 py-3 bg-gradient-to-r from-green-600 to-blue-600 text-white rounded-lg hover:shadow-lg transition-all"
          >
            <div className="flex items-center space-x-2">
              <CpuChip className="w-5 h-5" />
              <span>{t('referral.goToMining', '마이닝 페이지로 이동')}</span>
              <ArrowRight className="w-4 h-4" />
            </div>
          </button>
        </div>
      </div>
    </div>
  );

  /**
   * 히스토리 탭 렌더링
   */
  const renderHistoryTab = () => (
    <div className="space-y-4">
      <div className="flex items-center justify-between">
        <h3 className="text-lg font-semibold text-gray-900">
          {t('referral.referralHistory', '추천 히스토리')}
        </h3>
        <div className="text-sm text-gray-500">
          {t('referral.lastUpdated', '마지막 업데이트')}: {referralData.lastUpdated}
        </div>
      </div>

      {referralData.referralHistory.length === 0 ? (
        <div className="text-center py-8">
          <Users className="w-12 h-12 text-gray-300 mx-auto mb-4" />
          <p className="text-gray-500">
            {t('referral.noReferralsYet', '아직 추천인이 없습니다.')}
          </p>
        </div>
      ) : (
        <div className="space-y-3">
          {referralData.referralHistory.map((item) => (
            <div key={item.id} className="bg-white rounded-lg p-4 border border-gray-200">
              <div className="flex items-center justify-between">
                <div className="flex-1">
                  <div className="font-mono text-sm text-gray-600 mb-1">
                    {item.referredWalletAddress}
                  </div>
                  <div className="text-sm text-gray-500">
                    {t('referral.bonusAmount', '보너스')}: {formatBonusAmount(item.bonusAmount)} BW
                  </div>
                </div>
                <div className="flex items-center space-x-3">
                  <span className={`px-2 py-1 rounded-full text-xs font-medium ${getStatusColor(item.status)}`}>
                    {getStatusText(item.status)}
                  </span>
                  {item.status === 'PENDING' && userType === 'ADMIN' && (
                    <button
                      onClick={() => processReferralBonus(item.id)}
                      disabled={isLoading}
                      className="px-3 py-1 bg-green-100 text-green-600 rounded-lg hover:bg-green-200 transition-colors text-sm"
                    >
                      {t('referral.process', '처리')}
                    </button>
                  )}
                </div>
              </div>
              <div className="text-xs text-gray-400 mt-2">
                {t('referral.createdAt', '생성일')}: {item.createdAt}
                {item.confirmedAt && (
                  <span className="ml-4">
                    {t('referral.confirmedAt', '확인일')}: {item.confirmedAt}
                  </span>
                )}
              </div>
            </div>
          ))}
        </div>
      )}
    </div>
  );

  /**
   * 설정 탭 렌더링 (관리자 전용)
   */
  const renderSettingsTab = () => (
    <div className="space-y-6">
      <div className="flex items-center space-x-2">
        <Shield className="w-5 h-5 text-yellow-500" />
        <h3 className="text-lg font-semibold text-gray-900">
          {t('referral.bonusSettings', '보너스 설정')}
        </h3>
        <span className="px-2 py-1 bg-yellow-100 text-yellow-800 rounded-full text-xs font-medium">
          {t('referral.adminOnly', '관리자 전용')}
        </span>
      </div>

      {userType !== 'ADMIN' ? (
        <div className="text-center py-8">
          <AlertTriangle className="w-12 h-12 text-gray-300 mx-auto mb-4" />
          <p className="text-gray-500">
            {t('referral.adminAccessRequired', '관리자 권한이 필요합니다.')}
          </p>
        </div>
      ) : (
        <div className="space-y-4">
          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
            <div>
              <label className="block text-sm font-medium text-gray-700 mb-2">
                {t('referral.baseBonusRate', '기본 보너스 비율')}
              </label>
              <input
                type="number"
                step="0.01"
                min="0"
                max="1"
                value={referralData.bonusSettings.baseBonusRate}
                onChange={(e) => {
                  const newSettings = {
                    ...referralData.bonusSettings,
                    baseBonusRate: parseFloat(e.target.value)
                  };
                  setReferralData(prev => ({
                    ...prev,
                    bonusSettings: newSettings
                  }));
                }}
                className="w-full px-3 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent"
              />
            </div>

            <div>
              <label className="block text-sm font-medium text-gray-700 mb-2">
                {t('referral.level1BonusRate', '1단계 보너스 비율')}
              </label>
              <input
                type="number"
                step="0.01"
                min="0"
                max="1"
                value={referralData.bonusSettings.level1BonusRate}
                onChange={(e) => {
                  const newSettings = {
                    ...referralData.bonusSettings,
                    level1BonusRate: parseFloat(e.target.value)
                  };
                  setReferralData(prev => ({
                    ...prev,
                    bonusSettings: newSettings
                  }));
                }}
                className="w-full px-3 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent"
              />
            </div>

            <div>
              <label className="block text-sm font-medium text-gray-700 mb-2">
                {t('referral.minimumMiningTime', '최소 마이닝 시간 (분)')}
              </label>
              <input
                type="number"
                min="1"
                value={referralData.bonusSettings.minimumMiningTime}
                onChange={(e) => {
                  const newSettings = {
                    ...referralData.bonusSettings,
                    minimumMiningTime: parseInt(e.target.value)
                  };
                  setReferralData(prev => ({
                    ...prev,
                    bonusSettings: newSettings
                  }));
                }}
                className="w-full px-3 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent"
              />
            </div>

            <div>
              <label className="block text-sm font-medium text-gray-700 mb-2">
                {t('referral.maxReferralLevel', '최대 추천 단계')}
              </label>
              <input
                type="number"
                min="1"
                max="10"
                value={referralData.bonusSettings.maxReferralLevel}
                onChange={(e) => {
                  const newSettings = {
                    ...referralData.bonusSettings,
                    maxReferralLevel: parseInt(e.target.value)
                  };
                  setReferralData(prev => ({
                    ...prev,
                    bonusSettings: newSettings
                  }));
                }}
                className="w-full px-3 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent"
              />
            </div>
          </div>

          <button
            onClick={() => updateBonusSettings(referralData.bonusSettings)}
            disabled={isLoading}
            className="px-4 py-2 bg-blue-600 text-white rounded-lg hover:bg-blue-700 transition-colors disabled:opacity-50"
          >
            {isLoading ? t('common.loading', '로딩...') : t('referral.saveSettings', '설정 저장')}
          </button>
        </div>
      )}
    </div>
  );

  // ====================================================================================
  // 완벽한 독립 메인 렌더링 (공통 함수 절대 사용 금지)
  // ====================================================================================

  return (
    <div className="bg-white rounded-lg shadow-lg border border-gray-200">
      {/* 헤더 */}
      <div className="border-b border-gray-200 px-6 py-4">
        <div className="flex items-center justify-between">
          <div className="flex items-center space-x-3">
            <div className="w-10 h-10 bg-gradient-to-r from-blue-500 to-purple-600 rounded-lg flex items-center justify-center">
              <Gift className="w-5 h-5 text-white" />
            </div>
            <div>
              <h2 className="text-xl font-bold text-gray-900">
                {t('referral.title', '추천 보너스 시스템')}
              </h2>
              <p className="text-sm text-gray-500">
                {t('referral.subtitle', 'bitwishnetwork 인증된 노드 마이닝 채굴 페이지 연동')}
              </p>
            </div>
          </div>
          <div className="flex items-center space-x-2">
            <div className={`px-2 py-1 rounded-full text-xs font-medium ${
              userType === 'ADMIN' 
                ? 'bg-yellow-100 text-yellow-800' 
                : 'bg-green-100 text-green-800'
            }`}>
              {userType === 'ADMIN' ? t('common.admin', '관리자') : t('common.user', '사용자')}
            </div>
          </div>
        </div>
      </div>

      {/* 탭 네비게이션 */}
      <div className="border-b border-gray-200">
        <nav className="flex space-x-8 px-6">
          {[
            { id: 'overview', label: t('referral.overview', '개요'), icon: TrendingUp },
            { id: 'history', label: t('referral.history', '히스토리'), icon: Clock },
            { id: 'settings', label: t('referral.settings', '설정'), icon: Shield }
          ].map((tab) => (
            <button
              key={tab.id}
              onClick={() => setActiveTab(tab.id as any)}
              className={`flex items-center space-x-2 py-4 px-1 border-b-2 font-medium text-sm transition-colors ${
                activeTab === tab.id
                  ? 'border-blue-500 text-blue-600'
                  : 'border-transparent text-gray-500 hover:text-gray-700 hover:border-gray-300'
              }`}
            >
              <tab.icon className="w-4 h-4" />
              <span>{tab.label}</span>
            </button>
          ))}
        </nav>
      </div>

      {/* 콘텐츠 */}
      <div className="p-6">
        {/* 에러 메시지 */}
        {error && (
          <div className="mb-4 p-4 bg-red-50 border border-red-200 rounded-lg">
            <div className="flex items-center space-x-2">
              <XCircle className="w-5 h-5 text-red-500" />
              <span className="text-red-800">{error}</span>
            </div>
          </div>
        )}

        {/* 성공 메시지 */}
        {success && (
          <div className="mb-4 p-4 bg-green-50 border border-green-200 rounded-lg">
            <div className="flex items-center space-x-2">
              <CheckCircle className="w-5 h-5 text-green-500" />
              <span className="text-green-800">{success}</span>
            </div>
          </div>
        )}

        {/* 로딩 상태 */}
        {isLoading && (
          <div className="flex items-center justify-center py-8">
            <div className="animate-spin rounded-full h-8 w-8 border-b-2 border-blue-600"></div>
            <span className="ml-2 text-gray-600">
              {t('common.loading', '로딩...')}
            </span>
          </div>
        )}

        {/* 탭별 콘텐츠 */}
        {!isLoading && (
          <>
            {activeTab === 'overview' && renderOverviewTab()}
            {activeTab === 'history' && renderHistoryTab()}
            {activeTab === 'settings' && renderSettingsTab()}
          </>
        )}
      </div>
    </div>
  );
};

export default ReferralBonusSystem;
