interface BWPointMiningSystemProps {
  walletAddress: string;
  userType: 'USER' | 'ADMIN';
  onError: (error: string) => void;
  onSuccess: (message: string) => void;
  onLogout: () => void;

// ====================================================================================
// BW포인트 마이닝 데이터 타입 (완벽한 독립성 보장)
// ====================================================================================
interface BWPointMiningData {
  isActive: boolean;
  startTime: string | null;
  totalBWPoints: string;
  currentBWPoints: string;
  pendingBWPoints: string;
  miningRate: string; // BW포인트/초
  hashRate: number; // 해시/초
  powerConsumption: number; // 와트
  efficiency: number; // 효율성 %
  temperature: number; // 온도 °C
  uptime: number; // 가동시간 (초)
  blocksMined: number;
  lastBlockTime: string | null;
  miningHistory: MiningHistoryItem[];
  bonusMultiplier: number;
  totalEarnings: string;
  dailyEarnings: string;
  weeklyEarnings: string;
  monthlyEarnings: string;
  lastUpdated: string;
}

interface MiningHistoryItem {
  id: string;
  timestamp: string;
  bwPointsEarned: string;
  hashRate: number;
  powerConsumption: number;
  efficiency: number;
  temperature: number;
  status: 'ACTIVE' | 'PAUSED' | 'STOPPED' | 'ERROR';
  blockHeight?: number;
  transactionHash?: string;
}

interface MiningSettings {
  maxPowerConsumption: number; // 최대 전력 소비 (와트)
  targetTemperature: number; // 목표 온도 (°C)
  autoRestart: boolean; // 자동 재시작
  powerSavingMode: boolean; // 절전 모드
  overclockMode: boolean; // 오버클럭 모드
  miningPool: string; // 마이닝 풀
  difficulty: number; // 난이도
  maxUptime: number; // 최대 가동시간 (시간)
  emergencyStop: boolean; // 비상 정지
}

  isOpen: boolean;
  onClose: () => void;
  onSave: (settings: MiningSettings) => void;
  initialSettings: MiningSettings;
}

  isOpen,

  onSave,
  initialSettings,
}) => {

  const [settings, setSettings] = useState<MiningSettings>(initialSettings);

    setSettings(initialSettings);

    const { name, value, type, checked } = e.target;

      ...prev,
      [name]: type === 'checkbox' ? checked : (type === 'number' ? parseFloat(value) || 0 : value),
    }));
  };

    onSave(settings);
    onClose();

              onChange={handleInputChange}
              className="shadow appearance-none border border-gray-700 rounded w-full py-3 px-4 text-gray-200 leading-tight focus:outline-none focus:ring-2 focus:ring-blue-500 bg-gray-800"
              min="0"
              max="2000"
            />
          </div>

          <div>
            <label className="block text-gray-300 text-sm font-bold mb-2">
              {t('bwMining.targetTemperature', '목표 온도 (°C)')}
            </label>
            <input
              type="number"
              name="targetTemperature"
              value={settings.targetTemperature}
              onChange={handleInputChange}
              className="shadow appearance-none border border-gray-700 rounded w-full py-3 px-4 text-gray-200 leading-tight focus:outline-none focus:ring-2 focus:ring-blue-500 bg-gray-800"
              min="0"
              max="100"
            />
          </div>

          <div>
            <label className="block text-gray-300 text-sm font-bold mb-2">
              {t('bwMining.miningPool', '마이닝 풀')}

            <input
              type="text"

              onChange={handleInputChange}

              min="1"
              max="100"
            />
          </div>

          <div>
            <label className="block text-gray-300 text-sm font-bold mb-2">
              {t('bwMining.maxUptime', '최대 가동시간 (시간)')}
            </label>
            <input
              type="number"
              name="maxUptime"

              onChange={handleInputChange}

// ====================================================================================
// 완벽한 BW포인트 마이닝 시스템 컴포넌트 (완벽한 독립성 보장)
// ====================================================================================
const BWPointMiningSystem: React.FC<BWPointMiningSystemProps> = ({
  walletAddress,
  userType,
  onError,
  onSuccess,
  onLogout,
}) => {
  // ====================================================================================
  // 다국어 번역 시스템 (완벽한 독립성 보장)
  // ====================================================================================
  const { t, i18n } = useTranslation();

  // ====================================================================================
  // 완벽한 독립 상태 관리 (전역 변수 절대 사용 금지)
  // ====================================================================================
  const [isLoading, setIsLoading] = useState(false);
  const [error, setError] = useState<string>('');
  const [success, setSuccess] = useState<string>('');
  
  // BW포인트 마이닝 데이터 (완벽한 독립성 보장)
  const [miningData, setMiningData] = useState<BWPointMiningData>({
    isActive: false,
    startTime: null,
    totalBWPoints: '0',
    currentBWPoints: '0',
    pendingBWPoints: '0',
    miningRate: '0',
    hashRate: 0,
    powerConsumption: 0,
    efficiency: 0,
    temperature: 0,
    uptime: 0,
    blocksMined: 0,
    lastBlockTime: null,
    miningHistory: [],
    bonusMultiplier: 1.0,
    totalEarnings: '0',
    dailyEarnings: '0',
    weeklyEarnings: '0',
    monthlyEarnings: '0',
    lastUpdated: ''
  });
  
  // 마이닝 설정 (완벽한 독립성 보장)
  const [miningSettings, setMiningSettings] = useState<MiningSettings>({
    maxPowerConsumption: 1000,
    targetTemperature: 75,
    autoRestart: true,
    powerSavingMode: false,
    overclockMode: false,
    miningPool: 'pool.bitwish.network:8080',
    difficulty: 10,
    maxUptime: 24,
    emergencyStop: false
  });
  
  // UI 상태 (완벽한 독립성 보장)
  const [activeTab, setActiveTab] = useState<'overview' | 'history' | 'settings' | 'analytics'>('overview');
  const [showSettingsModal, setShowSettingsModal] = useState(false);
  const [refreshInterval, setRefreshInterval] = useState<NodeJS.Timeout | null>(null);

  // ====================================================================================
  // 완벽한 독립 API 호출 함수들 (공통 함수 절대 사용 금지)
  // ====================================================================================

  /**
   * BW포인트 마이닝 데이터 로드
   * - 완벽한 독립성 보장
   * - 실시간 마이닝 상태 동기화
   */
  const loadMiningData = async (): Promise<void> => {
    try {
      setIsLoading(true);
      setError('');
      
      // 백엔드 API 호출 (완벽한 독립성 보장)
      const response = await fetch('http://localhost:4001/bw-mining/load-data', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-Request-Source': 'BWPointMiningSystem',
          'X-User-Type': userType
        },
        body: JSON.stringify({
          walletAddress: walletAddress,
          userType: userType,
          timestamp: new Date().toISOString()
        })
      });

      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
      }

      const data = await response.json();
      
      if (!data.success) {
        throw new Error(data.error || 'Failed to load mining data');
      }

      setMiningData(data.miningData);
      setSuccess(t('bwMining.dataLoaded', 'BW포인트 마이닝 데이터가 로드되었습니다.'));
    } catch (error) {
      console.error('BW포인트 마이닝 데이터 로드 실패:', error);
      const errorMessage = error instanceof Error ? error.message : t('bwMining.loadFailed', '데이터 로드에 실패했습니다.');
      setError(errorMessage);
      onError(errorMessage);
    } finally {
      setIsLoading(false);
    }
  };

  /**
   * BW포인트 마이닝 시작
   * - 완벽한 독립성 보장
   * - 노드 채굴형 마이닝 시작
   */
  const startMining = async (): Promise<void> => {
    try {
      setIsLoading(true);
      setError('');
      
      // 백엔드 API 호출 (완벽한 독립성 보장)
      const response = await fetch('http://localhost:4001/bw-mining/start', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-Request-Source': 'BWPointMiningSystem',
          'X-User-Type': userType
        },
        body: JSON.stringify({
          walletAddress: walletAddress,
          userType: userType,
          settings: miningSettings,
          timestamp: new Date().toISOString()
        })
      });

      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
      }

      const data = await response.json();
      
      if (!data.success) {
        throw new Error(data.error || 'Failed to start mining');
      }

      setMiningData(prev => ({
        ...prev,
        isActive: true,
        startTime: new Date().toISOString(),
        lastUpdated: new Date().toISOString()
      }));
      
      setSuccess(t('bwMining.miningStarted', 'BW포인트 마이닝이 시작되었습니다.'));
      onSuccess(t('bwMining.miningStarted', 'BW포인트 마이닝이 시작되었습니다.'));
    } catch (error) {
      console.error('BW포인트 마이닝 시작 실패:', error);
      const errorMessage = error instanceof Error ? error.message : t('bwMining.startFailed', '마이닝 시작에 실패했습니다.');
      setError(errorMessage);
      onError(errorMessage);
    } finally {
      setIsLoading(false);
    }
  };

  /**
   * BW포인트 마이닝 일시정지
   * - 완벽한 독립성 보장
   * - 마이닝 일시정지 처리
   */
  const pauseMining = async (): Promise<void> => {
    try {
      setIsLoading(true);
      setError('');
      
      // 백엔드 API 호출 (완벽한 독립성 보장)
      const response = await fetch('http://localhost:4001/bw-mining/pause', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-Request-Source': 'BWPointMiningSystem',
          'X-User-Type': userType
        },
        body: JSON.stringify({
          walletAddress: walletAddress,
          timestamp: new Date().toISOString()
        })
      });

      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
      }

      const data = await response.json();
      
      if (!data.success) {
        throw new Error(data.error || 'Failed to pause mining');
      }

      setMiningData(prev => ({
        ...prev,
        isActive: false,
        lastUpdated: new Date().toISOString()
      }));
      
      setSuccess(t('bwMining.miningPaused', 'BW포인트 마이닝이 일시정지되었습니다.'));
      onSuccess(t('bwMining.miningPaused', 'BW포인트 마이닝이 일시정지되었습니다.'));
    } catch (error) {
      console.error('BW포인트 마이닝 일시정지 실패:', error);
      const errorMessage = error instanceof Error ? error.message : t('bwMining.pauseFailed', '마이닝 일시정지에 실패했습니다.');
      setError(errorMessage);
      onError(errorMessage);
    } finally {
      setIsLoading(false);
    }
  };

  /**
   * BW포인트 마이닝 중지
   * - 완벽한 독립성 보장
   * - 마이닝 완전 중지 처리
   */
  const stopMining = async (): Promise<void> => {
    try {
      setIsLoading(true);
      setError('');
      
      // 백엔드 API 호출 (완벽한 독립성 보장)
      const response = await fetch('http://localhost:4001/bw-mining/stop', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-Request-Source': 'BWPointMiningSystem',
          'X-User-Type': userType
        },
        body: JSON.stringify({
          walletAddress: walletAddress,
          timestamp: new Date().toISOString()
        })
      });

      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
      }

      const data = await response.json();
      
      if (!data.success) {
        throw new Error(data.error || 'Failed to stop mining');
      }

      setMiningData(prev => ({
        ...prev,
        isActive: false,
        startTime: null,
        lastUpdated: new Date().toISOString()
      }));
      
      setSuccess(t('bwMining.miningStopped', 'BW포인트 마이닝이 중지되었습니다.'));
      onSuccess(t('bwMining.miningStopped', 'BW포인트 마이닝이 중지되었습니다.'));
    } catch (error) {
      console.error('BW포인트 마이닝 중지 실패:', error);
      const errorMessage = error instanceof Error ? error.message : t('bwMining.stopFailed', '마이닝 중지에 실패했습니다.');
      setError(errorMessage);
      onError(errorMessage);
    } finally {
      setIsLoading(false);
    }
  };

  /**
   * 마이닝 설정 업데이트
   * - 완벽한 독립성 보장
   * - 실시간 설정 적용
   */
  const updateMiningSettings = async (newSettings: MiningSettings): Promise<void> => {
    try {
      setIsLoading(true);
      setError('');
      
      // 백엔드 API 호출 (완벽한 독립성 보장)
      const response = await fetch('http://localhost:4001/bw-mining/update-settings', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-Request-Source': 'BWPointMiningSystem',
          'X-User-Type': userType
        },
        body: JSON.stringify({
          walletAddress: walletAddress,
          settings: newSettings,
          timestamp: new Date().toISOString()
        })
      });

      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
      }

      const data = await response.json();
      
      if (!data.success) {
        throw new Error(data.error || 'Failed to update mining settings');
      }

      setMiningSettings(newSettings);
      setSuccess(t('bwMining.settingsUpdated', '마이닝 설정이 업데이트되었습니다.'));
      onSuccess(t('bwMining.settingsUpdated', '마이닝 설정이 업데이트되었습니다.'));
    } catch (error) {
      console.error('마이닝 설정 업데이트 실패:', error);
      const errorMessage = error instanceof Error ? error.message : t('bwMining.settingsUpdateFailed', '설정 업데이트에 실패했습니다.');
      setError(errorMessage);
      onError(errorMessage);
    } finally {
      setIsLoading(false);
    }
  };

  // 완벽한 독립 유틸리티 함수들 (공통 함수 절대 사용 금지)
  // ====================================================================================

  /**
   * BW포인트 금액 포맷팅
   * - 완벽한 독립성 보장
   * - 50자리 정밀도 지원
   */
  const formatBWPoints = (amount: string): string => {
    try {
      const decimal = new Decimal(amount);
      return decimal.toFixed(10);
    } catch (error) {
      return '0.0000000000';
    }
  };

   * 해시레이트 포맷팅
   * - 완벽한 독립성 보장
   * - 단위별 표시
   */
  const formatHashRate = (hashRate: number): string => {
    if (hashRate >= 1000000000) {
      return `${(hashRate / 1000000000).toFixed(2)} GH/s`;
    } else if (hashRate >= 1000000) {
      return `${(hashRate / 1000000).toFixed(2)} MH/s`;
    } else if (hashRate >= 1000) {
      return `${(hashRate / 1000).toFixed(2)} KH/s`;
    } else {
      return `${hashRate.toFixed(2)} H/s`;
    }
  };

   * 시간 포맷팅
   * - 완벽한 독립성 보장
   * - 시:분:초 형식
   */
  const formatTime = (seconds: number): string => {
    const hours = Math.floor(seconds / 3600);
    const minutes = Math.floor((seconds % 3600) / 60);
    const secs = seconds % 60;
    return `${hours.toString().padStart(2, '0')}:${minutes.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;

    if (temperature < 70) return 'text-green-400';

              </div>
              <div className="text-sm text-gray-500">
                {t('bwMining.weeklyEarnings', '주간 수익')}
              </div>
            </div>
          </div>
        </div>

        <div className="bg-white rounded-lg p-4 border border-gray-200">
          <div className="flex items-center space-x-3">
            <div className="w-10 h-10 bg-purple-100 rounded-lg flex items-center justify-center">
              <Award className="w-5 h-5 text-purple-600" />
            </div>
            <div>
              <div className="text-2xl font-bold text-gray-900">
                {formatBWPoints(miningData.monthlyEarnings)} BW
              </div>
              <div className="text-sm text-gray-500">
                {t('bwMining.monthlyEarnings', '월간 수익')}
              </div>
            </div>
          </div>
        </div>
      </div>

      {/* 성능 분석 */}
      <div className="bg-white rounded-lg p-6 border border-gray-200">
        <h4 className="font-semibold text-gray-900 mb-4">
          {t('bwMining.performanceAnalysis', '성능 분석')}
        </h4>
        <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
          <div>
            <h5 className="text-sm font-medium text-gray-700 mb-2">
              {t('bwMining.efficiencyTrend', '효율성 트렌드')}
            </h5>
            <div className="bg-gray-50 rounded-lg p-4">
              <div className="text-3xl font-bold text-green-600 mb-2">
                {miningData.efficiency}%
              </div>
              <div className="text-sm text-gray-600">
                {t('bwMining.currentEfficiency', '현재 효율성')}
              </div>
            </div>
          </div>

          <div>
            <h5 className="text-sm font-medium text-gray-700 mb-2">
              {t('bwMining.powerEfficiency', '전력 효율성')}
            </h5>
            <div className="bg-gray-50 rounded-lg p-4">
              <div className="text-3xl font-bold text-blue-600 mb-2">
                {(miningData.hashRate / Math.max(miningData.powerConsumption, 1)).toFixed(2)} H/W
              </div>
              <div className="text-sm text-gray-600">
                {t('bwMining.hashPerWatt', '해시/와트')}
              </div>
            </div>
          </div>
        </div>
      </div>

      {/* 스텔라 토큰 마이그레이션 정보 */}
      <div className="bg-gradient-to-r from-yellow-50 to-orange-50 rounded-lg p-6 border border-yellow-200">
        <div className="flex items-center space-x-3 mb-4">
          <Star className="w-6 h-6 text-yellow-600" />
          <h4 className="text-lg font-semibold text-yellow-800">
            {t('bwMining.stellarMigration', '스텔라 토큰 마이그레이션')}
          </h4>
        </div>
        <div className="space-y-3">
          <div className="flex justify-between items-center">
            <span className="text-yellow-700">
              {t('bwMining.currentBWPoints', '현재 BW포인트')}:
            </span>
            <span className="font-bold text-yellow-800">
              {formatBWPoints(miningData.totalBWPoints)} BW
            </span>
          </div>
          <div className="flex justify-between items-center">
            <span className="text-yellow-700">
              {t('bwMining.stellarEquivalent', '스텔라 토큰 등가')}:
            </span>
            <span className="font-bold text-yellow-800">
              {formatBWPoints(miningData.totalBWPoints)} XLM
            </span>
          </div>
          <div className="text-sm text-yellow-600">
            {t('bwMining.migrationNote', 'BW포인트는 스텔라 토큰과 1:1 비율로 마이그레이션됩니다.')}
          </div>
        </div>
      </div>
    </div>
  );

  // ====================================================================================
  // 완벽한 독립 메인 렌더링 (공통 함수 절대 사용 금지)
  // ====================================================================================

  return (
    <div className="bg-white rounded-lg shadow-lg border border-gray-200">
      {/* 헤더 */}
      <div className="border-b border-gray-200 px-6 py-4">
        <div className="flex items-center justify-between">
          <div className="flex items-center space-x-3">
            <div className="w-10 h-10 bg-gradient-to-r from-blue-500 to-purple-600 rounded-lg flex items-center justify-center">
              <CpuChip className="w-5 h-5 text-white" />
            </div>
            <div>
              <h2 className="text-xl font-bold text-gray-900">
                {t('bwMining.title', 'BW포인트 마이닝 시스템')}
              </h2>
              <p className="text-sm text-gray-500">
                {t('bwMining.subtitle', '노드 채굴형 마이닝으로 BW포인트 채굴')}
              </p>
            </div>
          </div>
          <div className="flex items-center space-x-2">
            <div className={`px-2 py-1 rounded-full text-xs font-medium ${
              userType === 'ADMIN' 
                ? 'bg-yellow-100 text-yellow-800' 
                : 'bg-green-100 text-green-800'
            }`}>
              {userType === 'ADMIN' ? t('common.admin', '관리자') : t('common.user', '사용자')}
            </div>
            <button
              onClick={onLogout}
              className="bg-red-600 hover:bg-red-700 text-white font-bold py-2 px-4 rounded-lg transition-all duration-200"
            >
              {t('common.logout', '로그아웃')}
            </button>
          </div>
        </div>
      </div>

      {/* 탭 네비게이션 */}
      <div className="border-b border-gray-200">
        <nav className="flex space-x-8 px-6">
          {[
            { id: 'overview', label: t('bwMining.overview', '개요'), icon: Activity },
            { id: 'history', label: t('bwMining.history', '히스토리'), icon: Clock },
            { id: 'settings', label: t('bwMining.settings', '설정'), icon: Settings },
            { id: 'analytics', label: t('bwMining.analytics', '분석'), icon: BarChart3 }
          ].map((tab) => (
            <button
              key={tab.id}
              onClick={() => setActiveTab(tab.id as any)}
              className={`flex items-center space-x-2 py-4 px-1 border-b-2 font-medium text-sm transition-colors ${
                activeTab === tab.id
                  ? 'border-blue-500 text-blue-600'
                  : 'border-transparent text-gray-500 hover:text-gray-700 hover:border-gray-300'
              }`}
            >
              <tab.icon className="w-4 h-4" />
              <span>{tab.label}</span>
            </button>
          ))}
        </nav>
      </div>

      {/* 콘텐츠 */}
      <div className="p-6">
        {/* 에러 메시지 */}
        {error && (
          <div className="mb-4 p-4 bg-red-50 border border-red-200 rounded-lg">
            <div className="flex items-center space-x-2">
              <XCircle className="w-5 h-5 text-red-500" />
              <span className="text-red-800">{error}</span>
            </div>
          </div>
        )}

        {/* 성공 메시지 */}
        {success && (
          <div className="mb-4 p-4 bg-green-50 border border-green-200 rounded-lg">
            <div className="flex items-center space-x-2">
              <CheckCircle className="w-5 h-5 text-green-500" />
              <span className="text-green-800">{success}</span>
            </div>
          </div>
        )}

        {/* 로딩 상태 */}
        {isLoading && (
          <div className="flex items-center justify-center py-8">
            <div className="animate-spin rounded-full h-8 w-8 border-b-2 border-blue-600"></div>
            <span className="ml-2 text-gray-600">
              {t('common.loading', '로딩...')}
            </span>
          </div>
        )}

        {/* 탭별 콘텐츠 */}
        {!isLoading && (
          <>
            {activeTab === 'overview' && renderOverviewTab()}
            {activeTab === 'history' && renderHistoryTab()}
            {activeTab === 'settings' && renderSettingsTab()}
            {activeTab === 'analytics' && renderAnalyticsTab()}
          </>
        )}
      </div>

      {/* 마이닝 설정 모달 */}
      <MiningSettingsModal
        isOpen={showSettingsModal}
        onClose={() => setShowSettingsModal(false)}
        onSave={updateMiningSettings}
        initialSettings={miningSettings}
      />
    </div>
  );
};

export default BWPointMiningSystem;
