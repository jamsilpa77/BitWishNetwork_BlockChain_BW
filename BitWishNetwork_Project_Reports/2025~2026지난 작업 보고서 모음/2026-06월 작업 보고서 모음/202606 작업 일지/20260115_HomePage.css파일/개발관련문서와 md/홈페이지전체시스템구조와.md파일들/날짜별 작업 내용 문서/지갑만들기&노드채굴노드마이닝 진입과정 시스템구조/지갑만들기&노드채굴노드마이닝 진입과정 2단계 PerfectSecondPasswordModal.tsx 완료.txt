import React, { useState, useEffect } from 'react';
import { useTranslation } from 'react-i18next';
import { 
  Shield, 
  Eye, 
  EyeOff, 
  CheckCircle, 
  XCircle, 
  AlertTriangle,
  Lock,
  Key,
  Wallet,
  ArrowRight,
  ArrowLeft
} from 'lucide-react';

// ====================================================================================
// 2차 비밀번호 설정 모달 인터페이스 (완벽한 독립성 보장)
// ====================================================================================
interface PerfectSecondPasswordModalProps {
  isOpen: boolean;
  onClose: () => void;
  onPasswordSet: (walletAddress: string, password: string) => void;
  onError: (error: string) => void;
  onBackToAuth: () => void;
}

// ====================================================================================
// 2차 비밀번호 설정 단계 (완벽한 독립성 보장)
// ====================================================================================
type PerfectSecondPasswordStep = 'wallet-input' | 'password-setup' | 'confirmation' | 'complete';

// ====================================================================================
// 비밀번호 강도 타입 (완벽한 독립성 보장)
// ====================================================================================
interface PasswordStrength {
  score: number; // 0-100
  level: 'WEAK' | 'FAIR' | 'GOOD' | 'STRONG' | 'VERY_STRONG';
  feedback: string[];
  requirements: {
    length: boolean;
    uppercase: boolean;
    lowercase: boolean;
    numbers: boolean;
    symbols: boolean;
    noCommon: boolean;
  };
}

// ====================================================================================
// 완벽한 2차 비밀번호 설정 모달 컴포넌트 (완벽한 독립성 보장)
// ====================================================================================
const PerfectSecondPasswordModal: React.FC<PerfectSecondPasswordModalProps> = ({
  isOpen,
  onClose,
  onPasswordSet,
  onError,
  onBackToAuth
}) => {
  // ====================================================================================
  // 다국어 번역 시스템 (완벽한 독립성 보장)
  // ====================================================================================
  const { t, i18n } = useTranslation();

  // ====================================================================================
  // 완벽한 독립 상태 관리 (전역 변수 절대 사용 금지)
  // ====================================================================================
  const [currentStep, setCurrentStep] = useState<PerfectSecondPasswordStep>('wallet-input');
  const [isLoading, setIsLoading] = useState(false);
  const [error, setError] = useState<string>('');
  const [success, setSuccess] = useState<string>('');
  
  // 지갑 주소 입력 상태 (완벽한 독립성 보장)
  const [walletAddress, setWalletAddress] = useState('');
  const [walletAddressValid, setWalletAddressValid] = useState(false);
  const [walletAddressError, setWalletAddressError] = useState('');
  
  // 비밀번호 설정 상태 (완벽한 독립성 보장)
  const [password, setPassword] = useState('');
  const [confirmPassword, setConfirmPassword] = useState('');
  const [showPassword, setShowPassword] = useState(false);
  const [showConfirmPassword, setShowConfirmPassword] = useState(false);
  const [passwordStrength, setPasswordStrength] = useState<PasswordStrength>({
    score: 0,
    level: 'WEAK',
    feedback: [],
    requirements: {
      length: false,
      uppercase: false,
      lowercase: false,
      numbers: false,
      symbols: false,
      noCommon: false
    }
  });
  
  // 보안 상태 (완벽한 독립성 보장)
  const [securityWarnings, setSecurityWarnings] = useState<string[]>([]);
  const [userConfirmed, setUserConfirmed] = useState(false);

  // ====================================================================================
  // 완벽한 독립 검증 함수들 (공통 함수 절대 사용 금지)
  // ====================================================================================

  /**
   * 스텔라 지갑 주소 형식 검증
   * - 완벽한 독립성 보장
   * - 스텔라 SCP 프로토콜 기반
   * - 56자리 G로 시작하는 주소
   */
  const validateStellarWalletAddress = (address: string): {isValid: boolean, error?: string} => {
    try {
      const trimmedAddress = address.trim();
      
      // 기본 형식 검증
      if (!trimmedAddress) {
        return {
          isValid: false,
          error: t('password.walletAddressRequired', '지갑 주소를 입력해주세요.')
        };
      }

      // 길이 검증
      if (trimmedAddress.length !== 56) {
        return {
          isValid: false,
          error: t('password.walletAddressLengthError', '지갑 주소는 56자리여야 합니다.')
        };
      }

      // G로 시작하는지 검증
      if (!trimmedAddress.startsWith('G')) {
        return {
          isValid: false,
          error: t('password.walletAddressFormatError', '지갑 주소는 G로 시작해야 합니다.')
        };
      }

      // 유효한 문자만 포함하는지 검증 (A-Z, 0-9)
      const validCharsRegex = /^[A-Z0-9]+$/;
      if (!validCharsRegex.test(trimmedAddress)) {
        return {
          isValid: false,
          error: t('password.walletAddressCharError', '지갑 주소는 대문자와 숫자만 포함할 수 있습니다.')
        };
      }

      return { isValid: true };
    } catch (error) {
      return {
        isValid: false,
        error: t('password.walletAddressValidationError', '지갑 주소 검증 중 오류가 발생했습니다.')
      };
    }
  };

  /**
   * 비밀번호 강도 분석
   * - 완벽한 독립성 보장
   * - 6가지 요구사항 검증
   * - 실시간 피드백 제공
   */
  const analyzePasswordStrength = (password: string): PasswordStrength => {
    try {
      const requirements = {
        length: password.length >= 8,
        uppercase: /[A-Z]/.test(password),
        lowercase: /[a-z]/.test(password),
        numbers: /\d/.test(password),
        symbols: /[!@#$%^&*()_+\-=\[\]{};':"\\|,.<>\/?]/.test(password),
        noCommon: !isCommonPassword(password)
      };

      // 점수 계산 (0-100)
      let score = 0;
      const requirementCount = Object.keys(requirements).length;
      const metRequirements = Object.values(requirements).filter(Boolean).length;
      score = Math.round((metRequirements / requirementCount) * 100);

      // 길이 보너스
      if (password.length >= 12) score += 10;
      if (password.length >= 16) score += 10;

      // 복잡성 보너스
      const uniqueChars = new Set(password).size;
      if (uniqueChars >= 8) score += 10;

      // 최대 100점으로 제한
      score = Math.min(score, 100);

      // 강도 레벨 결정
      let level: PasswordStrength['level'] = 'WEAK';
      if (score >= 80) level = 'VERY_STRONG';
      else if (score >= 60) level = 'STRONG';
      else if (score >= 40) level = 'GOOD';
      else if (score >= 20) level = 'FAIR';

      // 피드백 생성
      const feedback: string[] = [];
      if (!requirements.length) {
        feedback.push(t('password.requirementLength', '최소 8자 이상이어야 합니다.'));
      }
      if (!requirements.uppercase) {
        feedback.push(t('password.requirementUppercase', '대문자를 포함해야 합니다.'));
      }
      if (!requirements.lowercase) {
        feedback.push(t('password.requirementLowercase', '소문자를 포함해야 합니다.'));
      }
      if (!requirements.numbers) {
        feedback.push(t('password.requirementNumbers', '숫자를 포함해야 합니다.'));
      }
      if (!requirements.symbols) {
        feedback.push(t('password.requirementSymbols', '특수문자를 포함해야 합니다.'));
      }
      if (!requirements.noCommon) {
        feedback.push(t('password.requirementNoCommon', '일반적인 비밀번호는 사용할 수 없습니다.'));
      }

      return {
        score,
        level,
        feedback,
        requirements
      };
    } catch (error) {
      return {
        score: 0,
        level: 'WEAK',
        feedback: [t('password.analysisError', '비밀번호 분석 중 오류가 발생했습니다.')],
        requirements: {
          length: false,
          uppercase: false,
          lowercase: false,
          numbers: false,
          symbols: false,
          noCommon: false
        }
      };
    }
  };

  /**
   * 일반적인 비밀번호 검사
   * - 완벽한 독립성 보장
   * - 보안 강화를 위한 검증
   */
  const isCommonPassword = (password: string): boolean => {
    const commonPasswords = [
      'password', '123456', '123456789', 'qwerty', 'abc123',
      'password123', 'admin', 'letmein', 'welcome', 'monkey',
      '1234567890', 'password1', 'qwerty123', 'dragon', 'master'
    ];
    
    const lowerPassword = password.toLowerCase();
    return commonPasswords.some(common => lowerPassword.includes(common));
  };

  /**
   * 비밀번호 확인 검증
   * - 완벽한 독립성 보장
   * - 일치 여부 검증
   */
  const validatePasswordConfirmation = (password: string, confirmPassword: string): {isValid: boolean, error?: string} => {
    try {
      if (!confirmPassword) {
        return {
          isValid: false,
          error: t('password.confirmPasswordRequired', '비밀번호 확인을 입력해주세요.')
        };
      }

      if (password !== confirmPassword) {
        return {
          isValid: false,
          error: t('password.passwordMismatch', '비밀번호가 일치하지 않습니다.')
        };
      }

      return { isValid: true };
    } catch (error) {
      return {
        isValid: false,
        error: t('password.confirmationError', '비밀번호 확인 중 오류가 발생했습니다.')
      };
    }
  };

  // ====================================================================================
  // 완벽한 독립 API 호출 함수들 (공통 함수 절대 사용 금지)
  // ====================================================================================

  /**
   * 지갑 주소 검증 (백엔드)
   * - 완벽한 독립성 보장
   * - 스텔라 블록체인 연동
   */
  const validateWalletAddressWithBackend = async (address: string): Promise<boolean> => {
    try {
      setIsLoading(true);
      setError('');
      
      // 백엔드 API 호출 (완벽한 독립성 보장)
      const response = await fetch('http://localhost:4001/wallet/validate-address', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-Request-Source': 'PerfectSecondPasswordModal'
        },
        body: JSON.stringify({
          walletAddress: address,
          protocol: 'SCP',
          timestamp: new Date().toISOString()
        })
      });

      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
      }

      const data = await response.json();
      
      if (!data.success) {
        throw new Error(data.error || 'Failed to validate wallet address');
      }

      return data.isValid;
    } catch (error) {
      console.error('지갑 주소 검증 실패:', error);
      throw new Error(error instanceof Error ? error.message : t('password.walletValidationFailed', '지갑 주소 검증에 실패했습니다.'));
    } finally {
      setIsLoading(false);
    }
  };

  /**
   * 2차 비밀번호 설정 저장
   * - 완벽한 독립성 보장
   * - AES-256 암호화
   */
  const saveSecondPassword = async (walletAddress: string, password: string): Promise<void> => {
    try {
      setIsLoading(true);
      setError('');
      
      // 백엔드 API 호출 (완벽한 독립성 보장)
      const response = await fetch('http://localhost:4001/wallet/save-second-password', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-Request-Source': 'PerfectSecondPasswordModal'
        },
        body: JSON.stringify({
          walletAddress: walletAddress,
          password: password,
          timestamp: new Date().toISOString()
        })
      });

      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
      }

      const data = await response.json();
      
      if (!data.success) {
        throw new Error(data.error || 'Failed to save second password');
      }

      setSuccess(t('password.saveSuccess', '2차 비밀번호가 성공적으로 설정되었습니다.'));
    } catch (error) {
      console.error('2차 비밀번호 저장 실패:', error);
      throw new Error(error instanceof Error ? error.message : t('password.saveFailed', '2차 비밀번호 저장에 실패했습니다.'));
    } finally {
      setIsLoading(false);
    }
  };

  // ====================================================================================
  // 완벽한 독립 이벤트 핸들러들 (공통 함수 절대 사용 금지)
  // ====================================================================================

  /**
   * 지갑 주소 입력 처리
   * - 완벽한 독립성 보장
   * - 실시간 검증
   */
  const handleWalletAddressInput = async (address: string) => {
    try {
      setWalletAddress(address);
      setWalletAddressError('');
      setWalletAddressValid(false);

      // 클라이언트 사이드 검증
      const validation = validateStellarWalletAddress(address);
      if (!validation.isValid) {
        setWalletAddressError(validation.error || '');
        return;
      }

      // 백엔드 검증
      const isValid = await validateWalletAddressWithBackend(address);
      if (isValid) {
        setWalletAddressValid(true);
        setSuccess(t('password.walletAddressValid', '유효한 지갑 주소입니다.'));
      } else {
        setWalletAddressError(t('password.walletAddressInvalid', '유효하지 않은 지갑 주소입니다.'));
      }
    } catch (error) {
      setWalletAddressError(error instanceof Error ? error.message : t('password.walletValidationError', '지갑 주소 검증 중 오류가 발생했습니다.'));
    }
  };

  /**
   * 비밀번호 입력 처리
   * - 완벽한 독립성 보장
   * - 실시간 강도 분석
   */
  const handlePasswordInput = (password: string) => {
    setPassword(password);
    const strength = analyzePasswordStrength(password);
    setPasswordStrength(strength);
  };

  /**
   * 비밀번호 설정 완료
   * - 완벽한 독립성 보장
   * - 최종 검증 및 저장
   */
  const handlePasswordSetupComplete = async () => {
    try {
      setIsLoading(true);
      setError('');

      // 비밀번호 강도 검증
      if (passwordStrength.score < 40) {
        setError(t('password.weakPassword', '비밀번호가 너무 약합니다. 더 강한 비밀번호를 사용해주세요.'));
        return;
      }

      // 비밀번호 확인 검증
      const confirmationValidation = validatePasswordConfirmation(password, confirmPassword);
      if (!confirmationValidation.isValid) {
        setError(confirmationValidation.error || '');
        return;
      }

      // 2차 비밀번호 저장
      await saveSecondPassword(walletAddress, password);
      
      setCurrentStep('complete');
      setSuccess(t('password.setupComplete', '2차 비밀번호 설정이 완료되었습니다!'));
    } catch (error) {
      setError(error instanceof Error ? error.message : t('password.setupFailed', '2차 비밀번호 설정에 실패했습니다.'));
    } finally {
      setIsLoading(false);
    }
  };

  /**
   * 설정 완료 처리
   * - 완벽한 독립성 보장
   * - 부모 컴포넌트에 결과 전달
   */
  const handleSetupComplete = () => {
    try {
      onPasswordSet(walletAddress, password);
      setSuccess(t('password.completeSuccess', '2차 비밀번호 설정이 완료되었습니다!'));
    } catch (error) {
      onError(error instanceof Error ? error.message : t('password.completionError', '완료 처리 중 오류가 발생했습니다.'));
    }
  };

  // ====================================================================================
  // 완벽한 독립 초기화 함수 (공통 함수 절대 사용 금지)
  // ====================================================================================

  /**
   * 모달 초기화
   * - 완벽한 독립성 보장
   * - 모든 상태 초기화
   */
  const initializeModal = () => {
    setCurrentStep('wallet-input');
    setIsLoading(false);
    setError('');
    setSuccess('');
    setWalletAddress('');
    setWalletAddressValid(false);
    setWalletAddressError('');
    setPassword('');
    setConfirmPassword('');
    setShowPassword(false);
    setShowConfirmPassword(false);
    setPasswordStrength({
      score: 0,
      level: 'WEAK',
      feedback: [],
      requirements: {
        length: false,
        uppercase: false,
        lowercase: false,
        numbers: false,
        symbols: false,
        noCommon: false
      }
    });
    setSecurityWarnings([]);
    setUserConfirmed(false);
  };

  // ====================================================================================
  // 완벽한 독립 생명주기 관리 (공통 함수 절대 사용 금지)
  // ====================================================================================

  useEffect(() => {
    if (isOpen) {
      initializeModal();
    }
  }, [isOpen]);

  // ====================================================================================
  // 완벽한 독립 렌더링 함수들 (공통 함수 절대 사용 금지)
  // ====================================================================================

  /**
   * 지갑 주소 입력 화면 렌더링
   */
  const renderWalletInputScreen = () => (
    <div className="space-y-6">
      <div className="text-center">
        <div className="w-20 h-20 bg-gradient-to-br from-blue-500 to-purple-600 rounded-full flex items-center justify-center mx-auto mb-4">
          <Wallet className="w-10 h-10 text-white" />
        </div>
        <h2 className="text-2xl font-bold text-gray-900 mb-2">
          {t('password.walletInputTitle', '지갑 주소 입력')}
        </h2>
        <p className="text-gray-600">
          {t('password.walletInputDescription', '스텔라 블록체인 오리지널 지갑 주소를 입력하세요.')}
        </p>
      </div>

      <div className="space-y-4">
        <div>
          <label className="block text-sm font-medium text-gray-700 mb-2">
            {t('password.walletAddressLabel', '지갑 주소')}
          </label>
          <input
            type="text"
            value={walletAddress}
            onChange={(e) => handleWalletAddressInput(e.target.value)}
            placeholder={t('password.walletAddressPlaceholder', 'G로 시작하는 56자리 지갑 주소를 입력하세요')}
            className={`w-full px-4 py-3 border rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent ${
              walletAddressError ? 'border-red-300' : walletAddressValid ? 'border-green-300' : 'border-gray-300'
            }`}
          />
          {walletAddressError && (
            <div className="mt-2 flex items-center space-x-2 text-red-600">
              <XCircle className="w-4 h-4" />
              <span className="text-sm">{walletAddressError}</span>
            </div>
          )}
          {walletAddressValid && (
            <div className="mt-2 flex items-center space-x-2 text-green-600">
              <CheckCircle className="w-4 h-4" />
              <span className="text-sm">{t('password.walletAddressValid', '유효한 지갑 주소입니다.')}</span>
            </div>
          )}
        </div>
      </div>

      <div className="flex space-x-3">
        <button
          onClick={onBackToAuth}
          className="flex-1 px-4 py-3 bg-gray-100 text-gray-700 rounded-lg hover:bg-gray-200 transition-colors"
        >
          <div className="flex items-center justify-center space-x-2">
            <ArrowLeft className="w-4 h-4" />
            <span>{t('common.back', '뒤로')}</span>
          </div>
        </button>
        <button
          onClick={() => setCurrentStep('password-setup')}
          disabled={!walletAddressValid}
          className="flex-1 px-4 py-3 bg-gradient-to-r from-blue-600 to-purple-600 text-white rounded-lg hover:shadow-lg transition-all disabled:opacity-50"
        >
          <div className="flex items-center justify-center space-x-2">
            <span>{t('password.next', '다음')}</span>
            <ArrowRight className="w-4 h-4" />
          </div>
        </button>
      </div>
    </div>
  );

  /**
   * 비밀번호 설정 화면 렌더링
   */
  const renderPasswordSetupScreen = () => (
    <div className="space-y-6">
      <div className="text-center">
        <div className="w-20 h-20 bg-gradient-to-br from-green-500 to-blue-600 rounded-full flex items-center justify-center mx-auto mb-4">
          <Lock className="w-10 h-10 text-white" />
        </div>
        <h2 className="text-2xl font-bold text-gray-900 mb-2">
          {t('password.passwordSetupTitle', '2차 비밀번호 설정')}
        </h2>
        <p className="text-gray-600">
          {t('password.passwordSetupDescription', '안전한 2차 비밀번호를 설정하세요.')}
        </p>
      </div>

      {/* 보안 경고 */}
      <div className="bg-red-50 border border-red-200 rounded-lg p-4">
        <div className="flex items-start space-x-3">
          <AlertTriangle className="w-5 h-5 text-red-500 mt-0.5" />
          <div>
            <h3 className="font-semibold text-red-900 mb-2">
              {t('password.securityWarningTitle', '중요한 보안 경고')}
            </h3>
            <ul className="text-sm text-red-800 space-y-1">
              <li>• {t('password.securityWarning1', '2차 비밀번호는 절대 다른 사람과 공유하지 마세요.')}</li>
              <li>• {t('password.securityWarning2', '안전한 곳에 저장하세요.')}</li>
              <li>• {t('password.securityWarning3', '분실 시 복구가 어려울 수 있습니다.')}</li>
            </ul>
          </div>
        </div>
      </div>

      <div className="space-y-4">
        {/* 비밀번호 입력 */}
        <div>
          <label className="block text-sm font-medium text-gray-700 mb-2">
            {t('password.passwordLabel', '2차 비밀번호')}
          </label>
          <div className="relative">
            <input
              type={showPassword ? 'text' : 'password'}
              value={password}
              onChange={(e) => handlePasswordInput(e.target.value)}
              placeholder={t('password.passwordPlaceholder', '강한 비밀번호를 입력하세요')}
              className="w-full px-4 py-3 pr-12 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent"
            />
            <button
              type="button"
              onClick={() => setShowPassword(!showPassword)}
              className="absolute right-3 top-1/2 transform -translate-y-1/2 text-gray-400 hover:text-gray-600"
            >
              {showPassword ? <EyeOff className="w-5 h-5" /> : <Eye className="w-5 h-5" />}
            </button>
          </div>
        </div>

        {/* 비밀번호 확인 */}
        <div>
          <label className="block text-sm font-medium text-gray-700 mb-2">
            {t('password.confirmPasswordLabel', '비밀번호 확인')}
          </label>
          <div className="relative">
            <input
              type={showConfirmPassword ? 'text' : 'password'}
              value={confirmPassword}
              onChange={(e) => setConfirmPassword(e.target.value)}
              placeholder={t('password.confirmPasswordPlaceholder', '비밀번호를 다시 입력하세요')}
              className="w-full px-4 py-3 pr-12 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent"
            />
            <button
              type="button"
              onClick={() => setShowConfirmPassword(!showConfirmPassword)}
              className="absolute right-3 top-1/2 transform -translate-y-1/2 text-gray-400 hover:text-gray-600"
            >
              {showConfirmPassword ? <EyeOff className="w-5 h-5" /> : <Eye className="w-5 h-5" />}
            </button>
          </div>
        </div>

        {/* 비밀번호 강도 표시 */}
        {password && (
          <div className="space-y-3">
            <div className="flex items-center justify-between">
              <span className="text-sm font-medium text-gray-700">
                {t('password.strengthLabel', '비밀번호 강도')}
              </span>
              <span className={`text-sm font-medium ${
                passwordStrength.level === 'VERY_STRONG' ? 'text-green-600' :
                passwordStrength.level === 'STRONG' ? 'text-blue-600' :
                passwordStrength.level === 'GOOD' ? 'text-yellow-600' :
                passwordStrength.level === 'FAIR' ? 'text-orange-600' :
                'text-red-600'
              }`}>
                {t(`password.strength.${passwordStrength.level}`, passwordStrength.level)}
              </span>
            </div>
            
            <div className="w-full bg-gray-200 rounded-full h-2">
              <div
                className={`h-2 rounded-full transition-all duration-300 ${
                  passwordStrength.level === 'VERY_STRONG' ? 'bg-green-500' :
                  passwordStrength.level === 'STRONG' ? 'bg-blue-500' :
                  passwordStrength.level === 'GOOD' ? 'bg-yellow-500' :
                  passwordStrength.level === 'FAIR' ? 'bg-orange-500' :
                  'bg-red-500'
                }`}
                style={{ width: `${passwordStrength.score}%` }}
              />
            </div>

            {/* 요구사항 체크리스트 */}
            <div className="grid grid-cols-2 gap-2 text-sm">
              {Object.entries(passwordStrength.requirements).map(([key, met]) => (
                <div key={key} className="flex items-center space-x-2">
                  {met ? (
                    <CheckCircle className="w-4 h-4 text-green-500" />
                  ) : (
                    <XCircle className="w-4 h-4 text-red-500" />
                  )}
                  <span className={met ? 'text-green-700' : 'text-red-700'}>
                    {t(`password.requirement.${key}`, key)}
                  </span>
                </div>
              ))}
            </div>

            {/* 피드백 메시지 */}
            {passwordStrength.feedback.length > 0 && (
              <div className="bg-yellow-50 border border-yellow-200 rounded-lg p-3">
                <ul className="text-sm text-yellow-800 space-y-1">
                  {passwordStrength.feedback.map((feedback, index) => (
                    <li key={index}>• {feedback}</li>
                  ))}
                </ul>
              </div>
            )}
          </div>
        )}
      </div>

      <div className="flex space-x-3">
        <button
          onClick={() => setCurrentStep('wallet-input')}
          className="flex-1 px-4 py-3 bg-gray-100 text-gray-700 rounded-lg hover:bg-gray-200 transition-colors"
        >
          <div className="flex items-center justify-center space-x-2">
            <ArrowLeft className="w-4 h-4" />
            <span>{t('common.back', '뒤로')}</span>
          </div>
        </button>
        <button
          onClick={handlePasswordSetupComplete}
          disabled={isLoading || passwordStrength.score < 40 || password !== confirmPassword}
          className="flex-1 px-4 py-3 bg-gradient-to-r from-green-600 to-blue-600 text-white rounded-lg hover:shadow-lg transition-all disabled:opacity-50"
        >
          {isLoading ? t('common.loading', '로딩...') : t('password.setupComplete', '설정 완료')}
        </button>
      </div>
    </div>
  );

  /**
   * 완료 화면 렌더링
   */
  const renderCompleteScreen = () => (
    <div className="space-y-6">
      <div className="text-center">
        <div className="w-20 h-20 bg-gradient-to-br from-green-500 to-blue-600 rounded-full flex items-center justify-center mx-auto mb-4">
          <CheckCircle className="w-10 h-10 text-white" />
        </div>
        <h2 className="text-2xl font-bold text-gray-900 mb-2">
          {t('password.completeTitle', '2차 비밀번호 설정 완료!')}
        </h2>
        <p className="text-gray-600">
          {t('password.completeDescription', '2차 비밀번호가 성공적으로 설정되었습니다.')}
        </p>
      </div>

      {/* 설정 정보 */}
      <div className="bg-gray-50 rounded-lg p-4 space-y-3">
        <div>
          <label className="text-sm font-medium text-gray-700">
            {t('password.walletAddress', '지갑 주소')}
          </label>
          <div className="mt-1 p-2 bg-white border border-gray-300 rounded font-mono text-sm break-all">
            {walletAddress}
          </div>
        </div>
        
        <div>
          <label className="text-sm font-medium text-gray-700">
            {t('password.passwordStrength', '비밀번호 강도')}
          </label>
          <div className="mt-1 p-2 bg-white border border-gray-300 rounded">
            <div className="flex items-center space-x-2">
              <span className={`px-2 py-1 rounded text-xs font-medium ${
                passwordStrength.level === 'VERY_STRONG' ? 'bg-green-100 text-green-800' :
                passwordStrength.level === 'STRONG' ? 'bg-blue-100 text-blue-800' :
                passwordStrength.level === 'GOOD' ? 'bg-yellow-100 text-yellow-800' :
                passwordStrength.level === 'FAIR' ? 'bg-orange-100 text-orange-800' :
                'bg-red-100 text-red-800'
              }`}>
                {t(`password.strength.${passwordStrength.level}`, passwordStrength.level)}
              </span>
              <span className="text-sm text-gray-600">
                ({passwordStrength.score}점)
              </span>
            </div>
          </div>
        </div>
      </div>

      {/* 성공 메시지 */}
      <div className="bg-green-50 border border-green-200 rounded-lg p-4">
        <div className="flex items-center space-x-2">
          <Shield className="w-5 h-5 text-green-500" />
          <span className="text-green-800">
            {t('password.successMessage', '이제 지갑 주소 인증에서 2차 비밀번호를 사용할 수 있습니다!')}
          </span>
        </div>
      </div>

      <button
        onClick={handleSetupComplete}
        className="w-full px-4 py-3 bg-gradient-to-r from-green-600 to-blue-600 text-white rounded-lg hover:shadow-lg transition-all"
      >
        <div className="flex items-center justify-center space-x-2">
          <span>{t('password.backToAuth', '지갑 주소 인증으로 돌아가기')}</span>
          <ArrowRight className="w-4 h-4" />
        </div>
      </button>
    </div>
  );

  // ====================================================================================
  // 완벽한 독립 메인 렌더링 (공통 함수 절대 사용 금지)
  // ====================================================================================

  if (!isOpen) return null;

  return (
    <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center z-50 p-4">
      <div className="bg-white rounded-2xl shadow-2xl w-full max-w-2xl max-h-[90vh] overflow-y-auto">
        {/* 헤더 */}
        <div className="sticky top-0 bg-white border-b border-gray-200 px-6 py-4 rounded-t-2xl">
          <div className="flex items-center justify-between">
            <h1 className="text-xl font-bold text-gray-900">
              {t('password.modalTitle', '2차 비밀번호 설정')}
            </h1>
            <button
              onClick={onClose}
              className="text-gray-400 hover:text-gray-600 transition-colors"
            >
              <XCircle className="w-6 h-6" />
            </button>
          </div>
        </div>

        {/* 콘텐츠 */}
        <div className="p-6">
          {/* 에러 메시지 */}
          {error && (
            <div className="mb-4 p-4 bg-red-50 border border-red-200 rounded-lg">
              <div className="flex items-center space-x-2">
                <XCircle className="w-5 h-5 text-red-500" />
                <span className="text-red-800">{error}</span>
              </div>
            </div>
          )}

          {/* 성공 메시지 */}
          {success && (
            <div className="mb-4 p-4 bg-green-50 border border-green-200 rounded-lg">
              <div className="flex items-center space-x-2">
                <CheckCircle className="w-5 h-5 text-green-500" />
                <span className="text-green-800">{success}</span>
              </div>
            </div>
          )}

          {/* 단계별 렌더링 */}
          {currentStep === 'wallet-input' && renderWalletInputScreen()}
          {currentStep === 'password-setup' && renderPasswordSetupScreen()}
          {currentStep === 'complete' && renderCompleteScreen()}
        </div>
      </div>
    </div>
  );
};

export default PerfectSecondPasswordModal;
