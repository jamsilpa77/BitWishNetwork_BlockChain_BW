### **1단계: 백엔드에 전체 시드문구 검증 API 추가**

**📍 위치: `simple-server.js` 2927번줄 이후 추가**

```javascript
// ====================================================================================
// POST /wallet/authenticate-with-seedphrase - 전체 시드문구로 지갑 인증
// ====================================================================================
app.post('/wallet/authenticate-with-seedphrase', async (req, res) => {
  try {
    const { walletAddress, seedPhrase } = req.body;
    
    console.log('🔍 시드문구 인증 요청:', {
      walletAddress: walletAddress || 'unknown',
      seedPhraseLength: seedPhrase ? seedPhrase.split(' ').length : 0,
      timestamp: new Date().toISOString()
    });
    
    // 입력 데이터 검증
    if (!walletAddress || !seedPhrase) {
      return res.status(400).json({
        success: false,
        error: '지갑 주소와 시드문구가 필요합니다.'
      });
    }
    
    // 시드문구 전처리
    let processedSeedPhrase = seedPhrase.trim();
    processedSeedPhrase = processedSeedPhrase.replace(/\s+/g, ' ');
    processedSeedPhrase = processedSeedPhrase.toLowerCase();
    
    // 단어 개수 확인
    const words = processedSeedPhrase.split(' ');
    if (words.length !== 24) {
      return res.status(400).json({
        success: false,
        error: `시드문구는 정확히 24단어여야 합니다. 현재 ${words.length}단어입니다.`,
        wordCount: words.length,
        expectedCount: 24
      });
    }
    
    // BIP39 유효성 검증
    const isValid = bip39.validateMnemonic(processedSeedPhrase);
    
    if (!isValid) {
      return res.status(400).json({
        success: false,
        error: '유효하지 않은 BIP39 시드문구입니다. 단어를 다시 확인해주세요.',
        details: {
          wordCount: words.length,
          processedSeedPhrase: processedSeedPhrase
        }
      });
    }
    
    // 시드문구에서 지갑 주소 생성
    const seed = bip39.mnemonicToSeedSync(processedSeedPhrase);
    const keypair = StellarSdk.Keypair.fromRawEd25519Seed(seed.slice(0, 32));
    const generatedWalletAddress = keypair.publicKey();
    
    console.log('🔍 지갑 주소 비교:', {
      provided: walletAddress,
      generated: generatedWalletAddress,
      match: walletAddress === generatedWalletAddress
    });
    
    // 지갑 주소 일치 확인
    if (walletAddress !== generatedWalletAddress) {
      return res.status(400).json({
        success: false,
        error: '시드문구와 지갑 주소가 일치하지 않습니다.',
        details: {
          providedAddress: walletAddress,
          generatedAddress: generatedWalletAddress
        }
      });
    }
    
    // 성공 응답
    res.json({
      success: true,
      message: '시드문구 인증이 성공했습니다.',
      data: {
        walletAddress: generatedWalletAddress,
        isValid: true,
        userType: 'USER' // 기본값, 필요시 데이터베이스에서 조회
      }
    });
    
  } catch (error) {
    console.error('❌ 시드문구 인증 오류:', error);
    res.status(500).json({
      success: false,
      error: '시드문구 인증 중 오류가 발생했습니다.',
      message: error.message
    });
  }
});
```

```typescript
/**
 * 시드문구 인증
 * - 완벽한 독립성 보장
 * - BIP39 검증 및 지갑 주소 일치 확인

const authenticateWithSeedPhrase = async (walletAddress: string, seedPhrase: string): Promise<{isValid: boolean, userType: 'USER' | 'ADMIN'}> => {
  try {
    setIsLoading(true);
    setError('');
    
    // 백엔드 API 호출 (완벽한 독립성 보장)
    const response = await fetch('http://localhost:4001/wallet/authenticate-with-seedphrase', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'X-Request-Source': 'PerfectWalletAuthModal'
      },
      body: JSON.stringify({
        walletAddress: walletAddress.trim(),
        seedPhrase: seedPhrase.trim()
      })
    });

    if (!response.ok) {
      const errorData = await response.json();
      throw new Error(errorData.error || `HTTP ${response.status}: ${response.statusText}`);
    }

    const result = await response.json();
    
    if (!result.success) {
      throw new Error(result.error || '시드문구 인증에 실패했습니다.');
    }

    return {
      isValid: true,
      userType: result.data.userType || 'USER'
    };

  } catch (error) {
    console.error('❌ 시드문구 인증 오류:', error);
    throw error;
  } finally {
    setIsLoading(false);
  }
};
```

### **4단계: PerfectWalletAuthModal.tsx의 시드문구 제출 핸들러 수정**

**📍 위치: `PerfectWalletAuthModal.tsx` 시드문구 제출 함수 수정**

```typescript
/**
 * 시드문구 제출 처리
 */
const handleSeedPhraseSubmit = async () => {
  try {
    if (!authData.walletAddress || !seedPhrase.trim()) {
      setError(t('auth.invalidInput', '지갑 주소와 시드문구를 입력해주세요.'));
      return;
    }

    // 시드문구 인증
    const authResult = await authenticateWithSeedPhrase(authData.walletAddress, seedPhrase);
    
    if (authResult.isValid) {
      // 성공 시 로그인 시도 기록 업데이트
      await updateLoginAttempts(authData.walletAddress, true);
      
      // 성공 처리
      setCurrentStep('success');
      setSuccess(t('auth.seedphraseSuccess', '시드문구 인증이 완료되었습니다!'));
      
      // 부모 컴포넌트에 결과 전달
      onSuccess(authData.walletAddress, authResult.userType);
    } else {
      setError(t('auth.seedphraseInvalid', '시드문구가 올바르지 않습니다.'));
    }
    
  } catch (error) {
    console.error('❌ 시드문구 인증 오류:', error);
    
    // 실패 시 로그인 시도 기록 업데이트
    await updateLoginAttempts(authData.walletAddress, false);
    
    const errorMessage = error instanceof Error ? error.message : t('auth.seedphraseError', '시드문구 인증 중 오류가 발생했습니다.');
    setError(errorMessage);
  }
};
```
