import React, { useState, useEffect } from 'react';
import { useTranslation } from 'react-i18next';
import { 
  Shield, 
  Eye, 
  EyeOff, 
  CheckCircle, 
  XCircle, 
  AlertTriangle,
  Lock,
  Key,
  Wallet,
  ArrowRight,
  ArrowLeft,
  User,
  Crown,
  CpuChip
} from 'lucide-react';

// ====================================================================================
// 지갑 주소 인증 모달 인터페이스 (완벽한 독립성 보장)
// ====================================================================================
interface PerfectWalletAuthModalProps {
  isOpen: boolean;
  onClose: () => void;
  onAuthSuccess: (walletAddress: string, userType: 'USER' | 'ADMIN') => void;
  onError: (error: string) => void;
  onNavigateToSecondPassword: () => void;
}

// ====================================================================================
// 지갑 주소 인증 단계 (완벽한 독립성 보장)
// ====================================================================================
type PerfectWalletAuthStep = 'wallet-input' | 'password-input' | 'verification' | 'success';

// ====================================================================================
// 인증 데이터 타입 (완벽한 독립성 보장)
// ====================================================================================
interface AuthData {
  walletAddress: string;
  password: string;
  userType: 'USER' | 'ADMIN' | null;
  isAuthenticated: boolean;
  lastLogin: string;
  loginAttempts: number;
  maxAttempts: number;
}

// ====================================================================================
// 완벽한 지갑 주소 인증 모달 컴포넌트 (완벽한 독립성 보장)
// ====================================================================================
const PerfectWalletAuthModal: React.FC<PerfectWalletAuthModalProps> = ({
  isOpen,
  onClose,
  onAuthSuccess,
  onError,
  onNavigateToSecondPassword
}) => {
  // ====================================================================================
  // 다국어 번역 시스템 (완벽한 독립성 보장)
  // ====================================================================================
  const { t, i18n } = useTranslation();

  // ====================================================================================
  // 완벽한 독립 상태 관리 (전역 변수 절대 사용 금지)
  // ====================================================================================
  const [currentStep, setCurrentStep] = useState<PerfectWalletAuthStep>('wallet-input');
  const [isLoading, setIsLoading] = useState(false);
  const [error, setError] = useState<string>('');
  const [success, setSuccess] = useState<string>('');
  
  // 인증 데이터 (완벽한 독립성 보장)
  const [authData, setAuthData] = useState<AuthData>({
    walletAddress: '',
    password: '',
    userType: null,
    isAuthenticated: false,
    lastLogin: '',
    loginAttempts: 0,
    maxAttempts: 5
  });
  
  // UI 상태 (완벽한 독립성 보장)
  const [showPassword, setShowPassword] = useState(false);
  const [walletAddressValid, setWalletAddressValid] = useState(false);
  const [passwordValid, setPasswordValid] = useState(false);
  const [isBlocked, setIsBlocked] = useState(false);
  const [blockTimeRemaining, setBlockTimeRemaining] = useState(0);

  // ====================================================================================
  // 완벽한 독립 검증 함수들 (공통 함수 절대 사용 금지)
  // ====================================================================================

  /**
   * 스텔라 지갑 주소 형식 검증
   * - 완벽한 독립성 보장
   * - 스텔라 SCP 프로토콜 기반
   * - 56자리 G로 시작하는 주소
   */
  const validateStellarWalletAddress = (address: string): {isValid: boolean, error?: string} => {
    try {
      const trimmedAddress = address.trim();
      
      // 기본 형식 검증
      if (!trimmedAddress) {
        return {
          isValid: false,
          error: t('auth.walletAddressRequired', '지갑 주소를 입력해주세요.')
        };
      }

      // 길이 검증
      if (trimmedAddress.length !== 56) {
        return {
          isValid: false,
          error: t('auth.walletAddressLengthError', '지갑 주소는 56자리여야 합니다.')
        };
      }

      // G로 시작하는지 검증
      if (!trimmedAddress.startsWith('G')) {
        return {
          isValid: false,
          error: t('auth.walletAddressFormatError', '지갑 주소는 G로 시작해야 합니다.')
        };
      }

      // 유효한 문자만 포함하는지 검증 (A-Z, 0-9)
      const validCharsRegex = /^[A-Z0-9]+$/;
      if (!validCharsRegex.test(trimmedAddress)) {
        return {
          isValid: false,
          error: t('auth.walletAddressCharError', '지갑 주소는 대문자와 숫자만 포함할 수 있습니다.')
        };
      }

      return { isValid: true };
    } catch (error) {
      return {
        isValid: false,
        error: t('auth.walletAddressValidationError', '지갑 주소 검증 중 오류가 발생했습니다.')
      };
    }
  };

  /**
   * 2차 비밀번호 형식 검증
   * - 완벽한 독립성 보장
   * - 최소 8자 이상
   */
  const validatePassword = (password: string): {isValid: boolean, error?: string} => {
    try {
      if (!password) {
        return {
          isValid: false,
          error: t('auth.passwordRequired', '2차 비밀번호를 입력해주세요.')
        };
      }

      if (password.length < 8) {
        return {
          isValid: false,
          error: t('auth.passwordLengthError', '2차 비밀번호는 최소 8자 이상이어야 합니다.')
        };
      }

      return { isValid: true };
    } catch (error) {
      return {
        isValid: false,
        error: t('auth.passwordValidationError', '비밀번호 검증 중 오류가 발생했습니다.')
      };
    }
  };

  // ====================================================================================
  // 완벽한 독립 API 호출 함수들 (공통 함수 절대 사용 금지)
  // ====================================================================================

  /**
   * 지갑 주소 검증 (백엔드)
   * - 완벽한 독립성 보장
   * - 스텔라 블록체인 연동
   */
  const validateWalletAddressWithBackend = async (address: string): Promise<boolean> => {
    try {
      setIsLoading(true);
      setError('');
      
      // 백엔드 API 호출 (완벽한 독립성 보장)
      const response = await fetch('http://localhost:4001/wallet/validate-address', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-Request-Source': 'PerfectWalletAuthModal'
        },
        body: JSON.stringify({
          walletAddress: address,
          protocol: 'SCP',
          timestamp: new Date().toISOString()
        })
      });

      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
      }

      const data = await response.json();
      
      if (!data.success) {
        throw new Error(data.error || 'Failed to validate wallet address');
      }

      return data.isValid;
    } catch (error) {
      console.error('지갑 주소 검증 실패:', error);
      throw new Error(error instanceof Error ? error.message : t('auth.walletValidationFailed', '지갑 주소 검증에 실패했습니다.'));
    } finally {
      setIsLoading(false);
    }
  };

  /**
   * 2차 비밀번호 인증
   * - 완벽한 독립성 보장
   * - AES-256 암호화 검증
   */
  const authenticatePassword = async (walletAddress: string, password: string): Promise<{isValid: boolean, userType: 'USER' | 'ADMIN'}> => {
    try {
      setIsLoading(true);
      setError('');
      
      // 백엔드 API 호출 (완벽한 독립성 보장)
      const response = await fetch('http://localhost:4001/wallet/authenticate-password', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-Request-Source': 'PerfectWalletAuthModal'
        },
        body: JSON.stringify({
          walletAddress: walletAddress,
          password: password,
          timestamp: new Date().toISOString()
        })
      });

      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
      }

      const data = await response.json();
      
      if (!data.success) {
        throw new Error(data.error || 'Failed to authenticate password');
      }

      return {
        isValid: data.isValid,
        userType: data.userType
      };
    } catch (error) {
      console.error('비밀번호 인증 실패:', error);
      throw new Error(error instanceof Error ? error.message : t('auth.passwordAuthFailed', '비밀번호 인증에 실패했습니다.'));
    } finally {
      setIsLoading(false);
    }
  };

  /**
   * 로그인 시도 기록 업데이트
   * - 완벽한 독립성 보장
   * - 보안 강화를 위한 시도 횟수 추적
   */
  const updateLoginAttempts = async (walletAddress: string, success: boolean): Promise<void> => {
    try {
      // 백엔드 API 호출 (완벽한 독립성 보장)
      const response = await fetch('http://localhost:4001/wallet/update-login-attempts', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-Request-Source': 'PerfectWalletAuthModal'
        },
        body: JSON.stringify({
          walletAddress: walletAddress,
          success: success,
          timestamp: new Date().toISOString()
        })
      });

      if (!response.ok) {
        console.warn('로그인 시도 기록 업데이트 실패');
      }
    } catch (error) {
      console.warn('로그인 시도 기록 업데이트 실패:', error);
    }
  };

  // ====================================================================================
  // 완벽한 독립 이벤트 핸들러들 (공통 함수 절대 사용 금지)
  // ====================================================================================

  /**
   * 지갑 주소 입력 처리
   * - 완벽한 독립성 보장
   * - 실시간 검증
   */
  const handleWalletAddressInput = async (address: string) => {
    try {
      setAuthData(prev => ({ ...prev, walletAddress: address }));
      setError('');
      setWalletAddressValid(false);

      // 클라이언트 사이드 검증
      const validation = validateStellarWalletAddress(address);
      if (!validation.isValid) {
        setError(validation.error || '');
        return;
      }

      // 백엔드 검증
      const isValid = await validateWalletAddressWithBackend(address);
      if (isValid) {
        setWalletAddressValid(true);
        setSuccess(t('auth.walletAddressValid', '유효한 지갑 주소입니다.'));
        setCurrentStep('password-input');
      } else {
        setError(t('auth.walletAddressInvalid', '유효하지 않은 지갑 주소입니다.'));
      }
    } catch (error) {
      setError(error instanceof Error ? error.message : t('auth.walletValidationError', '지갑 주소 검증 중 오류가 발생했습니다.'));
    }
  };

  /**
   * 비밀번호 입력 처리
   * - 완벽한 독립성 보장
   * - 실시간 검증
   */
  const handlePasswordInput = (password: string) => {
    setAuthData(prev => ({ ...prev, password: password }));
    setError('');

    const validation = validatePassword(password);
    if (validation.isValid) {
      setPasswordValid(true);
    } else {
      setPasswordValid(false);
      setError(validation.error || '');
    }
  };

  /**
   * 인증 처리
   * - 완벽한 독립성 보장
   * - 최종 검증 및 인증
   */
  const handleAuthentication = async () => {
    try {
      setIsLoading(true);
      setError('');

      // 차단 상태 확인
      if (isBlocked) {
        setError(t('auth.accountBlocked', '계정이 일시적으로 차단되었습니다. 잠시 후 다시 시도해주세요.'));
        return;
      }

      // 최대 시도 횟수 확인
      if (authData.loginAttempts >= authData.maxAttempts) {
        setIsBlocked(true);
        setBlockTimeRemaining(300); // 5분 차단
        setError(t('auth.maxAttemptsReached', '최대 시도 횟수를 초과했습니다. 5분 후 다시 시도해주세요.'));
        return;
      }

      // 비밀번호 인증
      const authResult = await authenticatePassword(authData.walletAddress, authData.password);
      
      if (authResult.isValid) {
        // 성공 시 로그인 시도 기록 업데이트
        await updateLoginAttempts(authData.walletAddress, true);
        
        setAuthData(prev => ({
          ...prev,
          userType: authResult.userType,
          isAuthenticated: true,
          lastLogin: new Date().toISOString(),
          loginAttempts: 0
        }));
        
        setCurrentStep('success');
        setSuccess(t('auth.authSuccess', '인증이 성공적으로 완료되었습니다!'));
      } else {
        // 실패 시 로그인 시도 기록 업데이트
        await updateLoginAttempts(authData.walletAddress, false);
        
        setAuthData(prev => ({
          ...prev,
          loginAttempts: prev.loginAttempts + 1
        }));
        
        setError(t('auth.authFailed', '인증에 실패했습니다. 지갑 주소와 2차 비밀번호를 확인해주세요.'));
      }
    } catch (error) {
      setError(error instanceof Error ? error.message : t('auth.authError', '인증 중 오류가 발생했습니다.'));
    } finally {
      setIsLoading(false);
    }
  };

  /**
   * 인증 성공 처리
   * - 완벽한 독립성 보장
   * - 부모 컴포넌트에 결과 전달
   */
  const handleAuthSuccess = () => {
    try {
      if (authData.userType) {
        onAuthSuccess(authData.walletAddress, authData.userType);
        setSuccess(t('auth.completeSuccess', '인증이 완료되었습니다!'));
      } else {
        onError(t('auth.userTypeError', '사용자 타입을 확인할 수 없습니다.'));
      }
    } catch (error) {
      onError(error instanceof Error ? error.message : t('auth.completionError', '완료 처리 중 오류가 발생했습니다.'));
    }
  };

  /**
   * 2차 비밀번호 설정으로 이동
   * - 완벽한 독립성 보장
   * - 부모 컴포넌트에 신호 전달
   */
  const handleNavigateToSecondPassword = () => {
    try {
      onNavigateToSecondPassword();
    } catch (error) {
      onError(error instanceof Error ? error.message : t('auth.navigationError', '이동 중 오류가 발생했습니다.'));
    }
  };

  // ====================================================================================
  // 완벽한 독립 초기화 함수 (공통 함수 절대 사용 금지)
  // ====================================================================================

  /**
   * 모달 초기화
   * - 완벽한 독립성 보장
   * - 모든 상태 초기화
   */
  const initializeModal = () => {
    setCurrentStep('wallet-input');
    setIsLoading(false);
    setError('');
    setSuccess('');
    setAuthData({
      walletAddress: '',
      password: '',
      userType: null,
      isAuthenticated: false,
      lastLogin: '',
      loginAttempts: 0,
      maxAttempts: 5
    });
    setShowPassword(false);
    setWalletAddressValid(false);
    setPasswordValid(false);
    setIsBlocked(false);
    setBlockTimeRemaining(0);
  };

  // ====================================================================================
  // 완벽한 독립 생명주기 관리 (공통 함수 절대 사용 금지)
  // ====================================================================================

  useEffect(() => {
    if (isOpen) {
      initializeModal();
    }
  }, [isOpen]);

  // 차단 시간 카운트다운
  useEffect(() => {
    if (isBlocked && blockTimeRemaining > 0) {
      const timer = setTimeout(() => {
        setBlockTimeRemaining(prev => {
          if (prev <= 1) {
            setIsBlocked(false);
            setAuthData(prev => ({ ...prev, loginAttempts: 0 }));
            return 0;
          }
          return prev - 1;
        });
      }, 1000);

      return () => clearTimeout(timer);
    }
  }, [isBlocked, blockTimeRemaining]);

  // ====================================================================================
  // 완벽한 독립 렌더링 함수들 (공통 함수 절대 사용 금지)
  // ====================================================================================

  /**
   * 지갑 주소 입력 화면 렌더링
   */
  const renderWalletInputScreen = () => (
    <div className="space-y-6">
      <div className="text-center">
        <div className="w-20 h-20 bg-gradient-to-br from-blue-500 to-purple-600 rounded-full flex items-center justify-center mx-auto mb-4">
          <Wallet className="w-10 h-10 text-white" />
        </div>
        <h2 className="text-2xl font-bold text-gray-900 mb-2">
          {t('auth.walletInputTitle', '지갑 주소 인증')}
        </h2>
        <p className="text-gray-600">
          {t('auth.walletInputDescription', '스텔라 블록체인 오리지널 지갑 주소를 입력하세요.')}
        </p>
      </div>

      <div className="space-y-4">
        <div>
          <label className="block text-sm font-medium text-gray-700 mb-2">
            {t('auth.walletAddressLabel', '지갑 주소')}
          </label>
          <input
            type="text"
            value={authData.walletAddress}
            onChange={(e) => handleWalletAddressInput(e.target.value)}
            placeholder={t('auth.walletAddressPlaceholder', 'G로 시작하는 56자리 지갑 주소를 입력하세요')}
            className={`w-full px-4 py-3 border rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent ${
              error && !walletAddressValid ? 'border-red-300' : walletAddressValid ? 'border-green-300' : 'border-gray-300'
            }`}
          />
        </div>
      </div>

      {/* 2차 비밀번호 설정 안내 */}
      <div className="bg-blue-50 border border-blue-200 rounded-lg p-4">
        <div className="flex items-start space-x-3">
          <AlertTriangle className="w-5 h-5 text-blue-500 mt-0.5" />
          <div>
            <h3 className="font-semibold text-blue-900 mb-1">
              {t('auth.secondPasswordNotice', '2차 비밀번호가 설정되지 않았나요?')}
            </h3>
            <p className="text-sm text-blue-800 mb-3">
              {t('auth.secondPasswordDescription', '2차 비밀번호를 먼저 설정해야 지갑 인증을 사용할 수 있습니다.')}
            </p>
            <button
              onClick={handleNavigateToSecondPassword}
              className="text-sm text-blue-600 hover:text-blue-800 underline"
            >
              {t('auth.setupSecondPassword', '2차 비밀번호 설정하기')}
            </button>
          </div>
        </div>
      </div>

      <div className="flex space-x-3">
        <button
          onClick={onClose}
          className="flex-1 px-4 py-3 bg-gray-100 text-gray-700 rounded-lg hover:bg-gray-200 transition-colors"
        >
          {t('common.cancel', '취소')}
        </button>
        <button
          onClick={() => setCurrentStep('password-input')}
          disabled={!walletAddressValid}
          className="flex-1 px-4 py-3 bg-gradient-to-r from-blue-600 to-purple-600 text-white rounded-lg hover:shadow-lg transition-all disabled:opacity-50"
        >
          <div className="flex items-center justify-center space-x-2">
            <span>{t('auth.next', '다음')}</span>
            <ArrowRight className="w-4 h-4" />
          </div>
        </button>
      </div>
    </div>
  );

  /**
   * 비밀번호 입력 화면 렌더링
   */
  const renderPasswordInputScreen = () => (
    <div className="space-y-6">
      <div className="text-center">
        <div className="w-20 h-20 bg-gradient-to-br from-green-500 to-blue-600 rounded-full flex items-center justify-center mx-auto mb-4">
          <Lock className="w-10 h-10 text-white" />
        </div>
        <h2 className="text-2xl font-bold text-gray-900 mb-2">
          {t('auth.passwordInputTitle', '2차 비밀번호 입력')}
        </h2>
        <p className="text-gray-600">
          {t('auth.passwordInputDescription', '설정한 2차 비밀번호를 입력하세요.')}
        </p>
      </div>

      <div className="space-y-4">
        {/* 지갑 주소 표시 */}
        <div className="bg-gray-50 rounded-lg p-4">
          <label className="block text-sm font-medium text-gray-700 mb-2">
            {t('auth.walletAddress', '지갑 주소')}
          </label>
          <div className="font-mono text-sm text-gray-900 break-all">
            {authData.walletAddress}
          </div>
        </div>

        {/* 비밀번호 입력 */}
        <div>
          <label className="block text-sm font-medium text-gray-700 mb-2">
            {t('auth.passwordLabel', '2차 비밀번호')}
          </label>
          <div className="relative">
            <input
              type={showPassword ? 'text' : 'password'}
              value={authData.password}
              onChange={(e) => handlePasswordInput(e.target.value)}
              placeholder={t('auth.passwordPlaceholder', '2차 비밀번호를 입력하세요')}
              className="w-full px-4 py-3 pr-12 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent"
            />
            <button
              type="button"
              onClick={() => setShowPassword(!showPassword)}
              className="absolute right-3 top-1/2 transform -translate-y-1/2 text-gray-400 hover:text-gray-600"
            >
              {showPassword ? <EyeOff className="w-5 h-5" /> : <Eye className="w-5 h-5" />}
            </button>
          </div>
        </div>

        {/* 로그인 시도 횟수 표시 */}
        {authData.loginAttempts > 0 && (
          <div className="bg-yellow-50 border border-yellow-200 rounded-lg p-3">
            <div className="flex items-center space-x-2">
              <AlertTriangle className="w-4 h-4 text-yellow-600" />
              <span className="text-sm text-yellow-800">
                {t('auth.loginAttempts', '로그인 시도')}: {authData.loginAttempts}/{authData.maxAttempts}
              </span>
            </div>
          </div>
        )}

        {/* 차단 상태 표시 */}
        {isBlocked && (
          <div className="bg-red-50 border border-red-200 rounded-lg p-3">
            <div className="flex items-center space-x-2">
              <XCircle className="w-4 h-4 text-red-600" />
              <span className="text-sm text-red-800">
                {t('auth.accountBlocked', '계정이 차단되었습니다.')} {Math.floor(blockTimeRemaining / 60)}:{(blockTimeRemaining % 60).toString().padStart(2, '0')} 후 해제
              </span>
            </div>
          </div>
        )}
      </div>

      <div className="flex space-x-3">
        <button
          onClick={() => setCurrentStep('wallet-input')}
          className="flex-1 px-4 py-3 bg-gray-100 text-gray-700 rounded-lg hover:bg-gray-200 transition-colors"
        >
          <div className="flex items-center justify-center space-x-2">
            <ArrowLeft className="w-4 h-4" />
            <span>{t('common.back', '뒤로')}</span>
          </div>
        </button>
        <button
          onClick={handleAuthentication}
          disabled={isLoading || !passwordValid || isBlocked}
          className="flex-1 px-4 py-3 bg-gradient-to-r from-green-600 to-blue-600 text-white rounded-lg hover:shadow-lg transition-all disabled:opacity-50"
        >
          {isLoading ? t('common.loading', '로딩...') : t('auth.authenticate', '인증하기')}
        </button>
      </div>
    </div>
  );

  /**
   * 성공 화면 렌더링
   */
  const renderSuccessScreen = () => (
    <div className="space-y-6">
      <div className="text-center">
        <div className="w-20 h-20 bg-gradient-to-br from-green-500 to-blue-600 rounded-full flex items-center justify-center mx-auto mb-4">
          <CheckCircle className="w-10 h-10 text-white" />
        </div>
        <h2 className="text-2xl font-bold text-gray-900 mb-2">
          {t('auth.successTitle', '인증 성공!')}
        </h2>
        <p className="text-gray-600">
          {t('auth.successDescription', '지갑 인증이 성공적으로 완료되었습니다.')}
        </p>
      </div>

      {/* 인증 정보 */}
      <div className="bg-gray-50 rounded-lg p-4 space-y-3">
        <div>
          <label className="text-sm font-medium text-gray-700">
            {t('auth.walletAddress', '지갑 주소')}
          </label>
          <div className="mt-1 p-2 bg-white border border-gray-300 rounded font-mono text-sm break-all">
            {authData.walletAddress}
          </div>
        </div>
        
        <div>
          <label className="text-sm font-medium text-gray-700">
            {t('auth.userType', '사용자 타입')}
          </label>
          <div className="mt-1 p-2 bg-white border border-gray-300 rounded">
            <div className="flex items-center space-x-2">
              {authData.userType === 'ADMIN' ? (
                <Crown className="w-4 h-4 text-yellow-500" />
              ) : (
                <User className="w-4 h-4 text-blue-500" />
              )}
              <span className={`font-medium ${
                authData.userType === 'ADMIN' ? 'text-yellow-700' : 'text-blue-700'
              }`}>
                {authData.userType === 'ADMIN' ? t('common.admin', '관리자') : t('common.user', '사용자')}
              </span>
            </div>
          </div>
        </div>
      </div>

      {/* 성공 메시지 */}
      <div className="bg-green-50 border border-green-200 rounded-lg p-4">
        <div className="flex items-center space-x-2">
          <Shield className="w-5 h-5 text-green-500" />
          <span className="text-green-800">
            {t('auth.successMessage', '이제 개인 마이닝 페이지에 접근할 수 있습니다!')}
          </span>
        </div>
      </div>

      <button
        onClick={handleAuthSuccess}
        className="w-full px-4 py-3 bg-gradient-to-r from-green-600 to-blue-600 text-white rounded-lg hover:shadow-lg transition-all"
      >
        <div className="flex items-center justify-center space-x-2">
          <CpuChip className="w-5 h-5" />
          <span>{t('auth.goToMining', '마이닝 페이지로 이동')}</span>
          <ArrowRight className="w-4 h-4" />
        </div>
      </button>
    </div>
  );

  // ====================================================================================
  // 완벽한 독립 메인 렌더링 (공통 함수 절대 사용 금지)
  // ====================================================================================

  if (!isOpen) return null;

  return (
    <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center z-50 p-4">
      <div className="bg-white rounded-2xl shadow-2xl w-full max-w-2xl max-h-[90vh] overflow-y-auto">
        {/* 헤더 */}
        <div className="sticky top-0 bg-white border-b border-gray-200 px-6 py-4 rounded-t-2xl">
          <div className="flex items-center justify-between">
            <h1 className="text-xl font-bold text-gray-900">
              {t('auth.modalTitle', '지갑 주소 인증')}
            </h1>
            <button
              onClick={onClose}
              className="text-gray-400 hover:text-gray-600 transition-colors"
            >
              <XCircle className="w-6 h-6" />
            </button>
          </div>
        </div>

        {/* 콘텐츠 */}
        <div className="p-6">
          {/* 에러 메시지 */}
          {error && (
            <div className="mb-4 p-4 bg-red-50 border border-red-200 rounded-lg">
              <div className="flex items-center space-x-2">
                <XCircle className="w-5 h-5 text-red-500" />
                <span className="text-red-800">{error}</span>
              </div>
            </div>
          )}

          {/* 성공 메시지 */}
          {success && (
            <div className="mb-4 p-4 bg-green-50 border border-green-200 rounded-lg">
              <div className="flex items-center space-x-2">
                <CheckCircle className="w-5 h-5 text-green-500" />
                <span className="text-green-800">{success}</span>
              </div>
            </div>
          )}

          {/* 단계별 렌더링 */}
          {currentStep === 'wallet-input' && renderWalletInputScreen()}
          {currentStep === 'password-input' && renderPasswordInputScreen()}
          {currentStep === 'success' && renderSuccessScreen()}
        </div>
      </div>
    </div>
  );
};

export default PerfectWalletAuthModal;
