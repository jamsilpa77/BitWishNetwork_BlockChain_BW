**2단계: 시드문구 생성 (`seedphrase`)**
```typescript
const generateSeedPhrase = async () => {
  // 1. BIP39 24단어 시드문구 생성
  const mnemonic = generateMnemonic();
  const words = mnemonic.split(' ');
  
  // 2. 스텔라 키페어 생성
  const seed = bip39.mnemonicToSeedSync(mnemonic);
  const keypair = Keypair.fromRawEd25519Seed(seed.slice(0, 32));
  
  // 3. 지갑 데이터 저장
  setWalletData({
    seedPhrase: mnemonic,
    publicKey: keypair.publicKey(),
    privateKey: keypair.secret(),
    walletAddress: keypair.publicKey()
  });
};
```

**3단계: 시드문구 검증 (`verification`)**
```typescript
const renderVerificationScreen = () => {
  // 1. 4개 단어 랜덤 선택 (3, 8, 15, 22번째)
  const verificationIndices = [3, 8, 15, 22];
  
  // 2. 사용자가 단어 입력
  const verifySeedPhrase = () => {
    const isCorrect = verificationIndices.every((index, i) => 
      verificationWords[i] === walletData.seedPhrase.split(' ')[index]
    );
    
    if (isCorrect) {
      setCurrentStep('myWallet'); // ✅ 나의 지갑으로 이동
    }
  };
};
```

```typescript
interface MyWalletInfo {
  address: string;                    // 지갑 주소
  publicKey: string;                  // 공개키
  balance: {
    XLM: number;                      // 스텔라 잔액
    BW: number;                       // BW 토큰 잔액
    total: number;                    // 총 잔액 (USD)
  };
  transactions: Transaction[];        // 거래 내역
  kycStatus: 'pending' | 'verified' | 'rejected'; // KYC 상태
  createdAt: string;                  // 생성일시
  lastActivity: string;               // 마지막 활동
  securityLevel: 'low' | 'medium' | 'high'; // 보안 수준
}

interface Transaction {
  id: string;                         // 거래 ID
  type: 'send' | 'receive';          // 거래 타입
  amount: number;                     // 금액
  currency: 'XLM' | 'BW';            // 통화
  address: string;                    // 상대방 주소
  timestamp: string;                  // 시간
  status: 'pending' | 'confirmed' | 'failed'; // 상태
  fee: number;                        // 수수료
  memo?: string;                      // 메모
}
```

### **🎯 5. API 연동**

```typescript
// myWalletService.ts
class MyWalletService {
  // 지갑 정보 조회
  async getWalletInfo(address: string): Promise<MyWalletInfo> {
    const response = await fetch(`/api/wallet/${address}`);
    return response.json();
  }
  
  // 잔액 새로고침
  async refreshBalance(address: string): Promise<Balance> {
    const response = await fetch(`/api/wallet/${address}/balance`);
    return response.json();
  }
  
  // 거래 내역 조회
  async getTransactions(address: string): Promise<Transaction[]> {
    const response = await fetch(`/api/wallet/${address}/transactions`);
    return response.json();
  }
  
  // 지갑 설정 업데이트
  async updateSettings(address: string, settings: WalletSettings): Promise<void> {
    await fetch(`/api/wallet/${address}/settings`, {
      method: 'PUT',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(settings)
    });
  }
}
```
