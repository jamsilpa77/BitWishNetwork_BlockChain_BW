## 🔥 **유저용 추천인 보너스 시스템 완벽 구현**

### **📋 1단계: 기술 명세서 분석**

```markdown
# 추천인 보너스 시스템 기술 명세서

## 1차 백서 정책 수치
- 발급자 보너스율: 2% 영구 적용 (0.02)
- 가입자 보너스율: 1회 2% 적용 (0.02)
- 즉시 보너스: 1 BW
- 기본 채굴률: 0.25 BW/시간
- KYC 연동: 즉시 보너스 지급 시 KYC 인증 필요

## 추천인 시스템 요구사항
- 추천인 코드 생성: 개인별 고유 코드
- 추천인 코드 사용: 코드 입력으로 가입
- 영구 관계: 추천인 관계 영구 유지
- 즉시 보너스: KYC 인증 후 1 BW 지급
- 보너스 계산: 실시간 보너스율 계산

## 개인별 분리 원칙
- 전역/공통 변수 완전 제거
- localStorage 기반 개인별 데이터 저장
- 완전한 컴포넌트 독립성
```

### **📋 2단계: 유저용 추천인 보너스 컴포넌트 구현**

```typescript
// Node_HomePage/src/components/mining/user/UserReferralBonus.tsx

/**
 * =====================================================
 * ⚠️  중요: 전역 모달, 공통 변수 함수 절대 포함하지 않는다
 * =====================================================
* ❌ 전역 변수 사용 금지
 * ❌ 공통 함수 사용 금지  
 * ❌ 공통 클래스 사용 금지
 * ❌ 전역 모달 사용 금지
 * ❌ 중복 코드 사용 금지
 * ❌ 다른 컴포넌트와 상태 공유 금지
 * ❌ 전역 상태 관리 라이브러리 사용 금지
 * ❌ 스텔라 관련 코드 사용 금지
 * ❌ BIP39 시드문구 사용 금지
 * 
 * ✅ 완벽한 독립성 보장
 * ✅ 자체 상태 관리만 사용
 * ✅ 자체 API 호출만 사용
 * ✅ 자체 에러 처리만 사용
 * ✅ 자체 보안 검증만 사용
 * ✅ BitWish Network 전용 시스템만 사용
 * ✅ 모든 텍스트는 4개국 언어 즉시 번역 시스템 구조로 구현
 * 유저 추천인보너스 컴포넌트 - 완전한 개인별 분리
 * - 1차 백서 정책 수치 완전 적용 (2% 영구 적용)
 * - 추천인 코드 생성 및 관리
 * - KYC 연동 시스템
 * - 50단위 부동소수점 정밀도
 * - 4개국 언어 즉시 번역
 */

import React, { useState, useEffect, useCallback } from 'react';
import { useTranslation } from 'react-i18next';
import Decimal from 'decimal.js';

// 1차 백서 정책 수치 (완전 적용)
const REFERRAL_POLICY = {
  referrerBonusRate: 0.02, // 발급자 2% 영구 적용
  refereeBonusRate: 0.02, // 가입자 1회 2% 적용
  instantBonus: 1.0, // 즉시 보너스 1 BW
  miningRate: 0.25 // 기본 채굴률 0.25 BW/시간
};

interface UserReferralBonusProps {
  walletAddress: string;
  isMiningStarted: boolean;
  onReferralChange: (referralData: any) => void;
}

const UserReferralBonus: React.FC<UserReferralBonusProps> = ({ 
  walletAddress, 
  isMiningStarted,
  onReferralChange 
}) => {
  const { t } = useTranslation();
  
  // 개인별 상태 관리 (완전 분리)
  const [referralCode, setReferralCode] = useState('');
  const [referralCount, setReferralCount] = useState(0);
  const [totalReferralBonus, setTotalReferralBonus] = useState(0);
  const [isReferrer, setIsReferrer] = useState(false);
  const [isReferee, setIsReferee] = useState(false);
  const [referrerCode, setReferrerCode] = useState('');
  const [kycStatus, setKycStatus] = useState(false);
  const [instantBonusReceived, setInstantBonusReceived] = useState(false);
  const [showReferralModal, setShowReferralModal] = useState(false);
  const [inputReferralCode, setInputReferralCode] = useState('');
  const [showKYCModal, setShowKYCModal] = useState(false);
  const [kycData, setKycData] = useState({
    name: '',
    email: '',
    phone: '',
    idNumber: ''
  });

  // 50단위 부동소수점 정밀도 계산
  const calculateWithPrecision = useCallback((value: number) => {
    return new Decimal(value).toDecimalPlaces(50);
  }, []);

  // 추천인 코드 생성
  const generateReferralCode = useCallback(() => {
    const code = `BW${walletAddress.slice(-8).toUpperCase()}`;
    setReferralCode(code);
    setIsReferrer(true);
    
    // 개인별 데이터 저장
    const referralData = {
      walletAddress,
      referralCode: code,
      isReferrer: true,
      referralCount: 0,
      totalBonus: 0,
      createdAt: new Date().toISOString()
    };
    
    localStorage.setItem(`personal_${walletAddress}_referral`, JSON.stringify(referralData));
    onReferralChange(referralData);
  }, [walletAddress, onReferralChange]);

  // 추천인 코드 사용
  const useReferralCode = useCallback(async () => {
    if (!inputReferralCode.trim()) {
      alert(t('referral.code.error.empty'));
      return;
    }

    try {
      // 백엔드 API 호출
      const response = await fetch('/api/referral/use', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          walletAddress,
          referralCode: inputReferralCode
        })
      });

      const result = await response.json();

      if (result.success) {
        setIsReferee(true);
        setReferrerCode(inputReferralCode);
        
        // 개인별 데이터 저장
        const refereeData = {
          walletAddress,
          referrerCode: inputReferralCode,
          isReferee: true,
          bonusRate: REFERRAL_POLICY.refereeBonusRate,
          createdAt: new Date().toISOString()
        };
        
        localStorage.setItem(`personal_${walletAddress}_referee`, JSON.stringify(refereeData));
        onReferralChange(refereeData);
        
        setShowReferralModal(false);
        setInputReferralCode('');
        
        alert(`✅ ${t('referral.code.success')}`);
      } else {
        alert(`❌ ${result.error || t('referral.code.error.invalid')}`);
      }
    } catch (error) {
      console.error('추천인 코드 사용 오류:', error);
      alert(`❌ ${t('referral.code.error.network')}`);
    }
  }, [walletAddress, inputReferralCode, onReferralChange, t]);

  // KYC 인증
  const handleKYCVerification = useCallback(async () => {
    if (!kycData.name || !kycData.email || !kycData.phone || !kycData.idNumber) {
      alert(t('kyc.error.incomplete'));
      return;
    }

    try {
      // 백엔드 API 호출
      const response = await fetch('/api/kyc/verify', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          walletAddress,
          kycData
        })
      });

      const result = await response.json();

      if (result.success) {
        setKycStatus(true);
        localStorage.setItem(`personal_${walletAddress}_kyc`, 'verified');
        setShowKYCModal(false);
        
        alert(`✅ ${t('kyc.success')}`);
      } else {
        alert(`❌ ${result.error || t('kyc.error.verification')}`);
      }
    } catch (error) {
      console.error('KYC 인증 오류:', error);
      alert(`❌ ${t('kyc.error.network')}`);
    }
  }, [walletAddress, kycData, t]);

  // 즉시 보너스 지급
  const claimInstantBonus = useCallback(async () => {
    if (instantBonusReceived || !kycStatus) return;

    try {
      const bonusAmount = calculateWithPrecision(REFERRAL_POLICY.instantBonus);
      
      // 백엔드 API 호출
      const response = await fetch('/api/referral/instant-bonus', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          walletAddress
        })
      });

      const result = await response.json();

      if (result.success) {
        // 개인별 데이터 업데이트
        const instantBonusData = {
          walletAddress,
          instantBonus: bonusAmount.toNumber(),
          instantBonusReceived: true,
          receivedAt: new Date().toISOString()
        };
        
        localStorage.setItem(`personal_${walletAddress}_instantBonus`, JSON.stringify(instantBonusData));
        setInstantBonusReceived(true);
        
        // 부모 컴포넌트에 알림
        onReferralChange(instantBonusData);

        alert(`✅ ${t('referral.instantBonus.success')}\n${t('referral.instantBonus.amount')}: ${bonusAmount.toNumber()} BW`);
      } else {
        alert(`❌ ${result.error || t('referral.instantBonus.error')}`);
      }
    } catch (error) {
      console.error('즉시 보너스 지급 오류:', error);
      alert(`❌ ${t('referral.instantBonus.error')}`);
    }
  }, [walletAddress, instantBonusReceived, kycStatus, onReferralChange, calculateWithPrecision, t]);

  // 추천인 보너스 계산 (50단위 정밀도)
  const calculateReferralBonus = useCallback(() => {
    if (!isReferrer) {
      return {
        bonusRate: 0,
        enhancedRate: REFERRAL_POLICY.miningRate,
        bonusAmount: 0
      };
    }

    const bonusRate = calculateWithPrecision(REFERRAL_POLICY.referrerBonusRate)
      .times(referralCount);
    const enhancedRate = calculateWithPrecision(REFERRAL_POLICY.miningRate)
      .times(calculateWithPrecision(1).plus(bonusRate));
    const bonusAmount = enhancedRate.minus(REFERRAL_POLICY.miningRate);

    return {
      bonusRate: bonusRate.toNumber(),
      enhancedRate: enhancedRate.toNumber(),
      bonusAmount: bonusAmount.toNumber()
    };
  }, [isReferrer, referralCount, calculateWithPrecision]);

  // 개인별 데이터 로드
  useEffect(() => {
    const loadPersonalData = () => {
      try {
        // 추천인 데이터 로드
        const referralData = localStorage.getItem(`personal_${walletAddress}_referral`);
        if (referralData) {
          const data = JSON.parse(referralData);
          setReferralCode(data.referralCode || '');
          setReferralCount(data.referralCount || 0);
          setTotalReferralBonus(data.totalBonus || 0);
          setIsReferrer(data.isReferrer || false);
        }

        // 추천받은 데이터 로드
        const refereeData = localStorage.getItem(`personal_${walletAddress}_referee`);
        if (refereeData) {
          const data = JSON.parse(refereeData);
          setIsReferee(data.isReferee || false);
          setReferrerCode(data.referrerCode || '');
        }

        // 즉시 보너스 데이터 로드
        const instantBonusData = localStorage.getItem(`personal_${walletAddress}_instantBonus`);
        if (instantBonusData) {
          const data = JSON.parse(instantBonusData);
          setInstantBonusReceived(data.instantBonusReceived || false);
        }

        // KYC 상태 로드
        const kycStatus = localStorage.getItem(`personal_${walletAddress}_kyc`);
        setKycStatus(kycStatus === 'verified');
      } catch (error) {
        console.error('개인별 데이터 로드 오류:', error);
      }
    };

    loadPersonalData();
  }, [walletAddress]);

  const referralBonus = calculateReferralBonus();

  return (
    <div className="bg-gradient-to-r from-blue-50 to-indigo-50 rounded-xl p-6">
      <h3 className="text-xl font-bold text-gray-800 mb-4">
        👥 {t('referral.title')}
      </h3>

      {/* 추천인 코드 섹션 */}
      <div className="mb-6">
        <h4 className="text-lg font-semibold text-blue-700 mb-3">
          🎫 {t('referral.code.title')}
        </h4>
        
        {!referralCode ? (
          <button
            onClick={generateReferralCode}
            disabled={!isMiningStarted}
            className={`w-full py-3 px-4 rounded-lg font-medium transition-colors ${
              !isMiningStarted
                ? 'bg-gray-300 text-gray-500 cursor-not-allowed'
                : 'bg-blue-500 text-white hover:bg-blue-600'
            }`}
          >
            {!isMiningStarted ? t('referral.code.miningRequired') : t('referral.code.generate')}
          </button>
        ) : (
          <div className="bg-white rounded-lg p-4">
            <div className="text-sm text-gray-600 mb-2">
              {t('referral.code.yourCode')}
            </div>
            <div className="text-2xl font-bold text-blue-600 font-mono">
              {referralCode}
            </div>
            <div className="text-xs text-gray-500 mt-1">
              {t('referral.code.description')}
            </div>
          </div>
        )}
      </div>

      {/* 추천인 코드 사용 섹션 */}
      {!isReferee && (
        <div className="mb-6">
          <h4 className="text-lg font-semibold text-green-700 mb-3">
            🎯 {t('referral.use.title')}
          </h4>
          
          <button
            onClick={() => setShowReferralModal(true)}
            disabled={!isMiningStarted}
            className={`w-full py-3 px-4 rounded-lg font-medium transition-colors ${
              !isMiningStarted
                ? 'bg-gray-300 text-gray-500 cursor-not-allowed'
                : 'bg-green-500 text-white hover:bg-green-600'
            }`}
          >
            {!isMiningStarted ? t('referral.use.miningRequired') : t('referral.use.button')}
          </button>
        </div>
      )}

      {/* 추천인 통계 */}
      {isReferrer && (
        <div className="mb-6">
          <h4 className="text-lg font-semibold text-green-700 mb-3">
            📊 {t('referral.stats.title')}
          </h4>
          
          <div className="grid grid-cols-2 gap-4">
            <div className="bg-white rounded-lg p-4 text-center">
              <div className="text-2xl font-bold text-green-600">
                {referralCount}
              </div>
              <div className="text-sm text-gray-600">
                {t('referral.stats.count')}
              </div>
            </div>
            
            <div className="bg-white rounded-lg p-4 text-center">
              <div className="text-2xl font-bold text-purple-600">
                {totalReferralBonus.toFixed(2)}
              </div>
              <div className="text-sm text-gray-600">
                {t('referral.stats.totalBonus')} (BW)
              </div>
            </div>
          </div>
        </div>
      )}

      {/* 보너스 정보 */}
      <div className="mb-6">
        <div className="bg-white rounded-lg p-4">
          <h4 className="text-lg font-semibold text-gray-700 mb-3">
            💰 {t('referral.bonus.title')}
          </h4>
          
          <div className="space-y-2">
            <div className="flex justify-between">
              <span className="text-gray-600">{t('referral.bonus.rate')}:</span>
              <span className="font-bold text-green-600">
                {referralBonus.bonusRate * 100}%
              </span>
            </div>
            
            <div className="flex justify-between">
              <span className="text-gray-600">{t('referral.bonus.enhancedRate')}:</span>
              <span className="font-bold text-blue-600">
                {referralBonus.enhancedRate.toFixed(6)} BW/시간
              </span>
            </div>
            
            <div className="flex justify-between">
              <span className="text-gray-600">{t('referral.bonus.amount')}:</span>
              <span className="font-bold text-purple-600">
                +{referralBonus.bonusAmount.toFixed(6)} BW/시간
              </span>
            </div>
          </div>
        </div>
      </div>

      {/* 즉시 보너스 섹션 */}
      <div className="mb-6">
        <h4 className="text-lg font-semibold text-yellow-700 mb-3">
          🎁 {t('referral.instantBonus.title')}
        </h4>
        
        <div className="bg-white rounded-lg p-4">
          <div className="flex items-center justify-between mb-3">
            <span className="text-gray-700">
              {t('referral.instantBonus.amount')}: {REFERRAL_POLICY.instantBonus} BW
            </span>
            <span className={`px-3 py-1 rounded-full text-sm font-medium ${
              instantBonusReceived ? 'bg-green-500 text-white' : 
              kycStatus ? 'bg-yellow-500 text-white' : 
              'bg-gray-500 text-white'
            }`}>
              {instantBonusReceived ? t('referral.instantBonus.received') : 
               kycStatus ? t('referral.instantBonus.available') : 
               t('referral.instantBonus.kycRequired')}
            </span>
          </div>
          
          {!instantBonusReceived && kycStatus && (
            <button
              onClick={claimInstantBonus}
              className="w-full py-2 px-4 bg-yellow-500 text-white rounded-lg font-medium hover:bg-yellow-600 transition-colors"
            >
              {t('referral.instantBonus.claim')}
            </button>
          )}
          
          {!kycStatus && (
            <button
              onClick={() => setShowKYCModal(true)}
              className="w-full py-2 px-4 bg-blue-500 text-white rounded-lg font-medium hover:bg-blue-600 transition-colors"
            >
              {t('kyc.verify')}
            </button>
          )}
        </div>
      </div>

      {/* 마이닝 시작 안내 */}
      {!isMiningStarted && (
        <div className="mt-4 p-3 bg-yellow-50 border border-yellow-200 rounded-lg">
          <div className="text-sm text-yellow-800">
            ⚠️ {t('referral.miningRequired')}
          </div>
        </div>
      )}

      {/* 추천인 코드 사용 모달 */}
      {showReferralModal && (
        <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center z-[10000]">
          <div className="bg-white rounded-lg shadow-xl max-w-sm w-full mx-4 transform transition-all duration-300 scale-100">
            <div className="p-6">
              <h3 className="text-lg font-bold text-gray-900 mb-4">
                🎯 {t('referral.use.title')}
              </h3>
              
              <div className="mb-4">
                <label className="block text-sm font-medium text-gray-700 mb-2">
                  {t('referral.use.codeInput')}
                </label>
                <input
                  type="text"
                  value={inputReferralCode}
                  onChange={(e) => setInputReferralCode(e.target.value.toUpperCase())}
                  placeholder="BW12345678"
                  className="w-full px-3 py-2 border border-gray-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-blue-500"
                />
              </div>
              
              <div className="flex space-x-3">
                <button 
                  onClick={() => setShowReferralModal(false)}
                  className="flex-1 px-4 py-2 bg-gray-300 text-gray-700 rounded-lg hover:bg-gray-400 transition-colors"
                >
                  {t('common.cancel')}
                </button>
                <button 
                  onClick={useReferralCode}
                  className="flex-1 px-4 py-2 bg-green-500 text-white rounded-lg hover:bg-green-600 transition-colors font-medium"
                >
                  {t('referral.use.confirm')}
                </button>
              </div>
            </div>
          </div>
        </div>
      )}

      {/* KYC 인증 모달 */}
      {showKYCModal && (
        <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center z-[10000]">
          <div className="bg-white rounded-lg shadow-xl max-w-md w-full mx-4 transform transition-all duration-300 scale-100">
            <div className="p-6">
              <h3 className="text-lg font-bold text-gray-900 mb-4">
                🔐 {t('kyc.title')}
              </h3>
              
              <div className="space-y-4">
                <div>
                  <label className="block text-sm font-medium text-gray-700 mb-1">
                    {t('kyc.name')}
                  </label>
                  <input
                    type="text"
                    value={kycData.name}
                    onChange={(e) => setKycData(prev => ({ ...prev, name: e.target.value }))}
                    className="w-full px-3 py-2 border border-gray-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-blue-500"
                  />
                </div>
                
                <div>
                  <label className="block text-sm font-medium text-gray-700 mb-1">
                    {t('kyc.email')}
                  </label>
                  <input
                    type="email"
                    value={kycData.email}
                    onChange={(e) => setKycData(prev => ({ ...prev, email: e.target.value }))}
                    className="w-full px-3 py-2 border border-gray-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-blue-500"
                  />
                </div>
                
                <div>
                  <label className="block text-sm font-medium text-gray-700 mb-1">
                    {t('kyc.phone')}
                  </label>
                  <input
                    type="tel"
                    value={kycData.phone}
                    onChange={(e) => setKycData(prev => ({ ...prev, phone: e.target.value }))}
                    className="w-full px-3 py-2 border border-gray-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-blue-500"
                  />
                </div>
                
                <div>
                  <label className="block text-sm font-medium text-gray-700 mb-1">
                    {t('kyc.idNumber')}
                  </label>
                  <input
                    type="text"
```

### **📋 4단계: 백엔드 API 구현**

```javascript
// Node_HomePage/simple-server-new.js에 추가

// 1차 백서 정책 수치 적용
const REFERRAL_POLICY = {
  referrerBonusRate: 0.02, // 발급자 2% 영구 적용
  refereeBonusRate: 0.02, // 가입자 1회 2% 적용
  instantBonus: 1.0, // 즉시 보너스 1 BW
  miningRate: 0.25 // 기본 채굴률 0.25 BW/시간
};

// 추천인 코드 생성 API
app.post('/api/referral/generate', (req, res) => {
  try {
    const { walletAddress } = req.body;
    
    // 개인별 추천인 데이터 확인
    const referralData = personalReferralData.get(walletAddress);
    if (referralData && referralData.referralCode) {
      return res.json({ success: false, error: '이미 추천인 코드가 생성되었습니다.' });
    }

    // 추천인 코드 생성
    const referralCode = `BW${walletAddress.slice(-8).toUpperCase()}`;
    
    // 추천인 데이터 저장
    const newReferralData = {
      walletAddress,
      referralCode,
      isReferrer: true,
      referralCount: 0,
      totalBonus: 0,
      createdAt: new Date().toISOString()
    };

    personalReferralData.set(walletAddress, newReferralData);

    res.json({ 
      success: true, 
      message: '추천인 코드 생성 완료!',
      data: newReferralData
    });
  } catch (error) {
    res.status(500).json({ success: false, error: error.message });
  }
});

// 추천인 코드 사용 API
app.post('/api/referral/use', (req, res) => {
  try {
    const { walletAddress, referralCode } = req.body;
    
    // 추천인 코드 검증
    const referrerData = Array.from(personalReferralData.values())
      .find(data => data.referralCode === referralCode);
    
    if (!referrerData) {
      return res.json({ success: false, error: '유효하지 않은 추천인 코드입니다.' });
    }

    // 가입자 데이터 저장
    const refereeData = {
      walletAddress,
      referrerCode: referralCode,
      referrerWalletAddress: referrerData.walletAddress,
      isReferee: true,
      bonusRate: REFERRAL_POLICY.refereeBonusRate,
      createdAt: new Date().toISOString()
    };

    personalRefereeData.set(walletAddress, refereeData);

    // 추천인 수 증가
    referrerData.referralCount += 1;
    personalReferralData.set(referrerData.walletAddress, referrerData);

    res.json({ 
      success: true, 
      message: '추천인 코드 사용 완료!',
      data: refereeData
    });
  } catch (error) {
    res.status(500).json({ success: false, error: error.message });
  }
});

// KYC 인증 API
app.post('/api/kyc/verify', (req, res) => {
  try {
    const { walletAddress, kycData } = req.body;
    
    // KYC 데이터 검증 (실제로는 더 엄격한 검증 필요)
    if (!kycData.name || !kycData.email || !kycData.phone || !kycData.idNumber) {
      return res.json({ success: false, error: '모든 필드를 입력해주세요.' });
    }

    // KYC 상태 저장
    personalKYCData.set(walletAddress, 'verified');

    res.json({ 
      success: true, 
      message: 'KYC 인증 완료!',
      data: { walletAddress, kycStatus: 'verified' }
    });
  } catch (error) {
    res.status(500).json({ success: false, error: error.message });
  }
});

// 즉시 보너스 지급 API
app.post('/api/referral/instant-bonus', (req, res) => {
  try {
    const { walletAddress } = req.body;
    
    // KYC 상태 확인
    const kycStatus = personalKYCData.get(walletAddress);
    if (!kycStatus || kycStatus !== 'verified') {
      return res.json({ success: false, error: 'KYC 인증이 필요합니다.' });
    }

    // 즉시 보너스 지급
    const instantBonusData = {
      walletAddress,
      instantBonus: REFERRAL_POLICY.instantBonus,
      instantBonusReceived: true,
      receivedAt: new Date().toISOString()
    };

    personalInstantBonusData.set(walletAddress, instantBonusData);

    res.json({ 
      success: true, 
      message: '즉시 보너스 지급 완료!',
      data: instantBonusData
    });
  } catch (error) {
    res.status(500).json({ success: false, error: error.message });
  }
});

// 통합 보상률 계산 API
app.get('/api/rewards/rate/:walletAddress', (req, res) => {
  try {
    const { walletAddress } = req.params;
    
    // 추천인 보너스 계산
    const referralData = personalReferralData.get(walletAddress);
    const referralBonus = referralData ? referralData.referralCount * REFERRAL_POLICY.referrerBonusRate : 0;
    
    // 총 보너스율 계산
    const totalBonusRate = referralBonus;
    const enhancedRate = REFERRAL_POLICY.miningRate * (1 + totalBonusRate);
    
    res.json({ 
      success: true, 
      data: {
        referralBonus,
        totalBonusRate,
        enhancedRate
      }
    });
  } catch (error) {
    res.status(500).json({ success: false, error: error.message });
  }
});
```
