```typescript
  onSuccess

  const [walletAddress, setWalletAddress] = useState('');

  const [confirmPassword, setConfirmPassword] = useState('');
```

```javascript
// 지갑 주소 검증 API
app.post('/api/bitwish/wallet/validate', async (req, res) => {
  try {
    const { address } = req.body;
    
    // BW 주소 형식 검증
    const bwAddressRegex = /^BW[a-fA-F0-9]{40}$/;
    if (!bwAddressRegex.test(address)) {
      return res.json({ success: false, error: '올바른 BitWish 지갑 주소 형식이 아닙니다.' });
    }

    // 실제 블록체인에서 주소 존재 여부 확인
    const wallet = wallets.get(address);
    if (!wallet) {
      return res.json({ success: false, error: '존재하지 않는 지갑 주소입니다.' });
    }

    res.json({ success: true, message: '유효한 지갑 주소입니다.' });
  } catch (error) {
    res.status(500).json({ success: false, error: error.message });
  }
});

// 비밀번호 설정 API
app.post('/api/bitwish/wallet/set-password', async (req, res) => {
  try {
    const { walletAddress, hashedPassword } = req.body;
    
    // 지갑 존재 확인
    const wallet = wallets.get(walletAddress);
    if (!wallet) {
      return res.json({ success: false, error: '존재하지 않는 지갑 주소입니다.' });
    }

    // 비밀번호 해시 저장
    wallet.hashedPassword = hashedPassword;
    wallet.passwordSetAt = new Date().toISOString();
    
    // 파일 시스템에 저장
    await saveWalletsToFile();
    
    // MongoDB에 저장
    await saveWalletsToMongoDB();

    res.json({ success: true, message: '비밀번호가 성공적으로 설정되었습니다.' });
  } catch (error) {
    res.status(500).json({ success: false, error: error.message });
  }
});
```

```typescript
import crypto from 'crypto';

// 강력한 암호화로 비밀번호 생성
export const generateSecurePassword = async (password: string): Promise<string> => {
  // 솔트 생성
  const salt = crypto.randomBytes(32).toString('hex');
  
  // PBKDF2를 사용한 해싱 (100,000회 반복)
  const hash = crypto.pbkdf2Sync(password, salt, 100000, 64, 'sha512').toString('hex');
  
  // 솔트 + 해시 조합
  return `${salt}:${hash}`;
};

// 비밀번호 검증
export const verifyPassword = async (password: string, hashedPassword: string): Promise<boolean> => {
  try {
    const [salt, hash] = hashedPassword.split(':');
    const verifyHash = crypto.pbkdf2Sync(password, salt, 100000, 64, 'sha512').toString('hex');
    
    return hash === verifyHash;
  } catch (error) {
    return false;
  }
};

// 비밀번호 강도 검증
export const validatePasswordStrength = (password: string): { isValid: boolean; message: string } => {
  if (password.length < 8) {
    return { isValid: false, message: '비밀번호는 8자리 이상이어야 합니다.' };
  }
  
  if (!/(?=.*[a-z])/.test(password)) {
    return { isValid: false, message: '비밀번호에 소문자가 포함되어야 합니다.' };
  }
  
  if (!/(?=.*[A-Z])/.test(password)) {
    return { isValid: false, message: '비밀번호에 대문자가 포함되어야 합니다.' };
  }
  
  if (!/(?=.*\d)/.test(password)) {
    return { isValid: false, message: '비밀번호에 숫자가 포함되어야 합니다.' };
  }
  
  if (!/(?=.*[!@#$%^&*])/.test(password)) {
    return { isValid: false, message: '비밀번호에 특수문자가 포함되어야 합니다.' };
  }
  
  return { isValid: true, message: '강력한 비밀번호입니다.' };
};
```

#### 2. `handleMiningBonusClick()` 함수
- 인증 여부 확인

- 개인 마이닝 페이지 직접 렌더링

#### 4. 개인 마이닝 페이지 렌더링
- `PerfectPersonalMiningPage` 컴포넌트 사용
- 인증된 지갑 주소 전달

## 이해 보고서

### 문제점
1. "마이닝 & 보너스" 클릭 시 인증 없이 마이닝 페이지로 이동
2. 마이닝 페이지와 "나의 지갑" 페이지에 자동 로그아웃 미구현

### 요구사항

#### 1. 마이닝 & 보너스 버튼 동작
- 클릭 시 지갑 주소 인증 모달이 먼저 표시
- 인증 성공 시에만 마이닝 페이지로 이동
- 인증 실패 시 마이닝 페이지 접근 불가

#### 2. 마이닝 페이지 세션 관리
- 접속 후 페이지 내에서는 세션 유지
- 화면 밖으로 이동 시 3분 후 자동 로그아웃
- 탭 전환, 창 최소화, 다른 앱 사용 등 포함

#### 3. 나의 지갑 페이지 세션 관리
- 접속 후 페이지 내에서는 세션 유지
- 화면 밖으로 이동 시 3분 후 자동 로그아웃
- 동일한 세션 관리 정책 적용

### 구현 방향

#### 1. 인증 모달 우선 표시
- `handleMiningBonusClick()`에서 인증 상태 확인
- 미인증 시 `BitWishWalletAuthModal` 표시
- 인증 성공 시에만 `PerfectPersonalMiningPage` 표시

#### 2. 자동 로그아웃 시스템
- `visibilitychange` 이벤트로 화면 전환 감지
- `blur`/`focus` 이벤트로 창 포커스 감지
- 3분 타이머로 자동 로그아웃 실행
- 로그아웃 시 인증 정보 삭제 및 홈페이지로 이동

#### 3. 세션 상태 관리
- 로컬 스토리지에 세션 정보 저장
- 페이지 접근 시 세션 유효성 검증
- 세션 만료 시 자동 로그아웃 처리

### 보안 고려사항
- 세션 타임아웃으로 무단 접근 방지
- 페이지 전환 시 세션 유지
- 로그아웃 시 민감 정보 삭제
- 탭 전환 감지로 보안 강화

### 사용자 경험
- 인증 모달을 통한 명확한 인증 절차
- 페이지 내에서는 세션 유지로 편의성 확보
- 화면 밖 이동 시 자동 로그아웃으로 보안 강화
- 일관된 세션 관리 정책 적용

### 구현 우선순위
1. 마이닝 & 보너스 버튼 인증 모달 우선 표시
2. 마이닝 페이지 자동 로그아웃
3. 나의 지갑 페이지 자동 로그아웃
4. 세션 상태 관리 및 검증

### 예상 효과
- 보안 강화: 무단 접근 방지
- 사용자 경험 개선: 명확한 인증 절차
- 일관성: 동일한 세션 관리 정책
- 안정성: 자동 로그아웃으로 세션 안전성 확보
