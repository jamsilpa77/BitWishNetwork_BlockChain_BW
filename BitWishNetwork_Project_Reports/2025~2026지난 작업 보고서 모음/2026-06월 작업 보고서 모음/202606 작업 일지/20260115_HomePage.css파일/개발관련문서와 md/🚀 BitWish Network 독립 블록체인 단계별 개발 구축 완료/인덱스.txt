import React, { useState } from 'react';
import { useTranslation } from 'react-i18next';
import { 
  Shield, 
  Eye, 
  EyeOff, 
  CheckCircle, 
  XCircle, 
  AlertCircle,
  Wallet,
  Key,
  Lock,
  ArrowRight,
  RefreshCw,
  User,
  Crown
} from 'lucide-react';
import BitWishSecondPasswordModal from './BitWishSecondPasswordModal';

interface BitWishWalletAuthModalProps {
  isOpen: boolean;
  onClose: () => void;
  onWalletAuthenticated: (walletAddress: string) => void;
}

const BitWishWalletAuthModal: React.FC<BitWishWalletAuthModalProps> = ({
  isOpen,
  onClose,
  onWalletAuthenticated
}) => {
  const { t } = useTranslation();
  
  // 자체 상태 관리
  const [walletAddress, setWalletAddress] = useState('');
  const [walletPassword, setWalletPassword] = useState('');
  const [showPassword, setShowPassword] = useState(false);
  const [isAuthenticating, setIsAuthenticating] = useState(false);
  const [error, setError] = useState('');
  const [success, setSuccess] = useState('');
  const [userType, setUserType] = useState<'user' | 'admin' | null>(null);
  const [showSecondPasswordModal, setShowSecondPasswordModal] = useState(false);
  
  // 비밀번호 변경 관련 상태 변수
  const [showPasswordChangeModal, setShowPasswordChangeModal] = useState(false);
  const [currentPassword, setCurrentPassword] = useState('');
  const [newPassword, setNewPassword] = useState('');
  const [confirmNewPassword, setConfirmNewPassword] = useState('');
  const [showCurrentPassword, setShowCurrentPassword] = useState(false);
  const [showNewPassword, setShowNewPassword] = useState(false);
  const [showConfirmNewPassword, setShowConfirmNewPassword] = useState(false);
  const [isChangingPassword, setIsChangingPassword] = useState(false);

  // ====================================================================================
  // 2차 비밀번호 설정 성공 처리 함수 (완전 독립)
  // ====================================================================================
  const handleSecondPasswordSuccess = () => {
    setShowSecondPasswordModal(false);
    setSuccess('2차 비밀번호가 성공적으로 설정되었습니다.');
    
    // 2초 후 메시지 초기화
    setTimeout(() => {
      setSuccess('');
    }, 2000);
  };

  // ====================================================================================
  // BitWish 지갑 인증 함수 (완전 독립)
  // ====================================================================================
  const authenticateBitWishWallet = async () => {
    try {
      setIsAuthenticating(true);
      setError('');
      setSuccess('');
      
      // 입력 검증
      if (!walletAddress.trim()) {
        setError(t('bitwish.wallet.auth.addressRequired'));
        return;
      }
      
      if (!walletPassword.trim()) {
        setError(t('bitwish.wallet.auth.passwordRequired'));
        return;
      }
      
      // BitWish 주소 형식 검증 (BW + 40자리 16진수, 총 42자리)
      const bitwishAddressRegex = /^BW[A-F0-9]{40}$/;
      if (!bitwishAddressRegex.test(walletAddress)) {
        setError(t('bitwish.wallet.auth.invalidAddress'));
        return;
      }
      
      // BitWish 지갑 인증 API 호출 (새로운 엔드포인트 사용)
      const response = await fetch('http://localhost:4001/api/bitwish/wallet/authenticate', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          walletAddress: walletAddress,
          password: walletPassword
        })
      });
      
      const result = await response.json();
      
      if (result.success) {
        // 사용자 타입 감지
        const isAdmin = result.wallet.address === 'BW0000000000000000000000000000000000000000'; // 관리자 주소
        setUserType(isAdmin ? 'admin' : 'user');
        
        setSuccess('BitWish 지갑 인증 성공');
        
        // 로컬 스토리지에 인증 정보 저장
        localStorage.setItem('bitwish_wallet_auth', JSON.stringify({
          address: result.wallet.address,
          publicKey: result.wallet.publicKey,
          networkType: 'BITWISH_MAINNET',
          authenticatedAt: new Date(),
          userType: isAdmin ? 'admin' : 'user'
        }));
        
        // 부모 컴포넌트에 인증 성공 알림
        setTimeout(() => {
          onWalletAuthenticated(result.wallet.address);
          handleClose();
        }, 1500);
        
      } else {
        setError(result.error || t('bitwish.wallet.auth.error'));
      }
    } catch (error) {
      setError(t('bitwish.wallet.auth.error'));
      console.error('BitWish 지갑 인증 오류:', error);
    } finally {
      setIsAuthenticating(false);
    }
  };

  // 모달 닫기
  const handleClose = () => {
    setWalletAddress('');
    setWalletPassword('');
    setError('');
    setSuccess('');
    setUserType(null);
    onClose();
  };

  // Enter 키 처리
  const handleKeyPress = (e: React.KeyboardEvent) => {
    if (e.key === 'Enter') {
      authenticateBitWishWallet();
    }
  };

  // ====================================================================================
  // BitWish 비밀번호 변경 함수 (완전 독립)
  // ====================================================================================
  const changeBitWishPassword = async () => {
    try {
      setIsChangingPassword(true);
      setError('');
      setSuccess('');
      
      // 입력 검증
      if (!walletAddress.trim()) {
        setError('지갑 주소가 필요합니다.');
        return;
      }
      
      if (!currentPassword.trim()) {
        setError('기존 비밀번호가 필요합니다.');
        return;
      }
      
      if (!newPassword.trim()) {
        setError('새 비밀번호가 필요합니다.');
        return;
      }
      
      if (!confirmNewPassword.trim()) {
        setError('새 비밀번호 확인이 필요합니다.');
        return;
      }
      
      // 새 비밀번호와 확인 비밀번호 일치 검증
      if (newPassword !== confirmNewPassword) {
        setError('새 비밀번호와 확인 비밀번호가 일치하지 않습니다.');
        return;
      }
      
      // 비밀번호 강도 검증
      if (newPassword.length < 8) {
        setError('새 비밀번호는 최소 8자 이상이어야 합니다.');
        return;
      }
      
      // BitWish 주소 형식 검증
      const bitwishAddressRegex = /^BW[A-F0-9]{40}$/;
      if (!bitwishAddressRegex.test(walletAddress)) {
        setError('올바른 BitWish 지갑 주소 형식이 아닙니다.');
        return;
      }
      
      // 비밀번호 변경 API 호출
      const response = await fetch('http://localhost:4001/api/bitwish/wallet/change-password', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          walletAddress: walletAddress,
          currentPassword: currentPassword,
          newPassword: newPassword
        })
      });
      
      const result = await response.json();
      
      if (result.success) {
        setSuccess('비밀번호가 성공적으로 변경되었습니다.');
        
        // 입력 필드 초기화
        setCurrentPassword('');
        setNewPassword('');
        setConfirmNewPassword('');
        
        // 1초 후 비밀번호 변경 모달 닫기
        setTimeout(() => {
          setShowPasswordChangeModal(false);
          setSuccess('');
        }, 1000);
        
      } else {
        setError(result.error || '비밀번호 변경에 실패했습니다.');
      }
      
    } catch (error) {
      console.error('비밀번호 변경 오류:', error);
      setError('비밀번호 변경 중 오류가 발생했습니다.');
    } finally {
      setIsChangingPassword(false);
    }
  };

  // 비밀번호 변경 모달 닫기 함수
  const handleClosePasswordChangeModal = () => {
    setShowPasswordChangeModal(false);
    setCurrentPassword('');
    setNewPassword('');
    setConfirmNewPassword('');
    setError('');
    setSuccess('');
    setShowCurrentPassword(false);
    setShowNewPassword(false);
    setShowConfirmNewPassword(false);
  };

  if (!isOpen) return null;

  return (
    <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center z-50 p-4">
      <div className="bg-white rounded-2xl shadow-2xl max-w-md w-full">
        {/* 헤더 */}
        <div className="flex items-center justify-between p-6 border-b border-gray-200">
          <div className="flex items-center space-x-3">
            <div className="w-10 h-10 bg-gradient-to-r from-blue-500 to-purple-600 rounded-full flex items-center justify-center">
              <Shield className="w-6 h-6 text-white" />
            </div>
            <div>
              <h2 className="text-xl font-bold text-gray-900">
                {t('bitwish.wallet.auth.title')}
              </h2>
              <p className="text-sm text-gray-500">
                {t('bitwish.wallet.auth.subtitle')}
              </p>
            </div>
          </div>
          <button
            onClick={handleClose}
            className="text-gray-400 hover:text-gray-600 transition-colors"
          >
            <XCircle className="w-6 h-6" />
          </button>
        </div>

        {/* 내용 */}
        <div className="p-6">
          <div className="space-y-6">
            <div className="text-center">
              <Wallet className="w-16 h-16 text-blue-500 mx-auto mb-4" />
              <h3 className="text-lg font-semibold text-gray-900 mb-2">
                {t('bitwish.wallet.auth.infoTitle')}
              </h3>
              <p className="text-sm text-gray-600">
                {t('bitwish.wallet.auth.infoDescription')}
              </p>
            </div>

            <div className="space-y-4">
              <div>
                <div className="flex items-center justify-between mb-2">
                  <label className="block text-sm font-medium text-gray-700">
                    {t('bitwish.wallet.auth.addressLabel')}
                  </label>
                  <button
                    onClick={() => setShowSecondPasswordModal(true)}
                    className="text-sm text-blue-600 hover:text-blue-800 underline"
                  >
                    2차 비밀번호 설정
                  </button>
                </div>
                <div className="relative">
                  <input
                    type="text"
                    value={walletAddress}
                    onChange={(e) => setWalletAddress(e.target.value.toUpperCase())}
                    onKeyPress={handleKeyPress}
                    placeholder={t('bitwish.wallet.auth.addressPlaceholder')}
                    className="w-full px-4 py-3 pl-12 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent font-mono text-sm"
                  />
                  <div className="absolute left-3 top-1/2 transform -translate-y-1/2 text-gray-400">
                    <Key className="w-5 h-5" />
                  </div>
                </div>
                <p className="text-xs text-gray-500 mt-1">
                  예: BW1234567890ABCDEF1234567890ABCDEF12345678
                </p>
              </div>

              <div>
                <div className="flex items-center justify-between mb-2">
                  <label className="block text-sm font-medium text-gray-700">
                    {t('bitwish.wallet.auth.passwordLabel')}
                  </label>
                  <button
                    onClick={() => setShowPasswordChangeModal(true)}
                    className="text-sm text-red-600 hover:text-red-800 underline"
                  >
                    비밀번호 변경
                  </button>
                </div>
                <div className="relative">
                  <input
                    type={showPassword ? 'text' : 'password'}
                    value={walletPassword}
                    onChange={(e) => setWalletPassword(e.target.value)}
                    onKeyPress={handleKeyPress}
                    placeholder={t('bitwish.wallet.auth.passwordPlaceholder')}
                    className="w-full px-4 py-3 pl-12 pr-12 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent"
                  />
                  <div className="absolute left-3 top-1/2 transform -translate-y-1/2 text-gray-400">
                    <Lock className="w-5 h-5" />
                  </div>
                  <button
                    type="button"
                    onClick={() => setShowPassword(!showPassword)}
                    className="absolute right-3 top-1/2 transform -translate-y-1/2 text-gray-400 hover:text-gray-600"
                  >
                    {showPassword ? <EyeOff className="w-5 h-5" /> : <Eye className="w-5 h-5" />}
                  </button>
                </div>
              </div>
            </div>

            {/* 에러 메시지 */}
            {error && (
              <div className="flex items-center space-x-2 text-red-600 bg-red-50 p-3 rounded-lg">
                <AlertCircle className="w-5 h-5" />
                <span className="text-sm">{error}</span>
              </div>
            )}

            {/* 성공 메시지 */}
            {success && (
              <div className="flex items-center space-x-2 text-green-600 bg-green-50 p-3 rounded-lg">
                <CheckCircle className="w-5 h-5" />
                <span className="text-sm">{success}</span>
              </div>
            )}

            {/* 사용자 타입 표시 */}
            {userType && (
              <div className={`flex items-center space-x-2 p-3 rounded-lg ${
                userType === 'admin' 
                  ? 'text-purple-600 bg-purple-50' 
                  : 'text-blue-600 bg-blue-50'
              }`}>
                {userType === 'admin' ? (
                  <Crown className="w-5 h-5" />
                ) : (
                  <User className="w-5 h-5" />
                )}
                <span className="text-sm font-medium">
                  {userType === 'admin' 
                    ? t('bitwish.wallet.auth.adminUser')
                    : t('bitwish.wallet.auth.regularUser')
                  }
                </span>
              </div>
            )}

            {/* 인증 버튼 */}
            <button
              onClick={authenticateBitWishWallet}
              disabled={isAuthenticating || !walletAddress || !walletPassword}
              className="w-full bg-gradient-to-r from-blue-500 to-purple-600 text-white py-3 px-6 rounded-lg font-medium hover:from-blue-600 hover:to-purple-700 disabled:opacity-50 disabled:cursor-not-allowed transition-all duration-200 flex items-center justify-center space-x-2"
            >
              {isAuthenticating ? (
                <>
                  <RefreshCw className="w-5 h-5 animate-spin" />
                  <span>{t('bitwish.wallet.auth.authenticating')}</span>
                </>
              ) : (
                <>
                  <span>{t('bitwish.wallet.auth.button')}</span>
                  <ArrowRight className="w-5 h-5" />
                </>
              )}
            </button>

            {/* 도움말 */}
            <div className="text-center">
              <p className="text-xs text-gray-500">
                {t('bitwish.wallet.auth.help')}
              </p>
            </div>
          </div>
        </div>
      </div>

      {/* 2차 비밀번호 설정 모달 */}
      <BitWishSecondPasswordModal
        isOpen={showSecondPasswordModal}
        onClose={() => setShowSecondPasswordModal(false)}
        onSuccess={handleSecondPasswordSuccess}
      />

      {/* BitWish 비밀번호 변경 모달 */}
      {showPasswordChangeModal && (
        <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center z-50">
          <div className="bg-white rounded-lg shadow-xl w-full max-w-md mx-4">
            {/* 헤더 */}
            <div className="flex items-center justify-between p-6 border-b border-gray-200">
              <div className="flex items-center space-x-3">
                <div className="w-10 h-10 bg-gradient-to-r from-red-500 to-pink-600 rounded-full flex items-center justify-center">
                  <Shield className="w-6 h-6 text-white" />
                </div>
                <div>
                  <h2 className="text-xl font-bold text-gray-900">
                    BitWish 비밀번호 변경
                  </h2>
                  <p className="text-sm text-gray-500">
                    기존 비밀번호를 새로운 비밀번호로 변경하세요
                  </p>
                </div>
              </div>
              <button
                onClick={handleClosePasswordChangeModal}
                className="text-gray-400 hover:text-gray-600 transition-colors"
              >
                <XCircle className="w-6 h-6" />
              </button>
            </div>

            {/* 내용 */}
            <div className="p-6">
              <div className="space-y-4">
                {/* 기존 지갑 주소 */}
                <div>
                  <label className="block text-sm font-medium text-gray-700 mb-2">
                    BitWish 지갑 주소
                  </label>
                  <div className="relative">
                    <input
                      type="text"
                      value={walletAddress}
                      readOnly
                      className="w-full px-4 py-3 pl-12 border border-gray-300 rounded-lg bg-gray-50 font-mono text-sm"
                    />
                    <div className="absolute left-3 top-1/2 transform -translate-y-1/2 text-gray-400">
                      <Key className="w-5 h-5" />
                    </div>
                  </div>
                </div>

                {/* 기존 비밀번호 */}
                <div>
                  <label className="block text-sm font-medium text-gray-700 mb-2">
                    기존 비밀번호
                  </label>
                  <div className="relative">
                    <input
                      type={showCurrentPassword ? 'text' : 'password'}
                      value={currentPassword}
                      onChange={(e) => setCurrentPassword(e.target.value)}
                      placeholder="기존 비밀번호를 입력하세요"
                      className="w-full px-4 py-3 pl-12 pr-12 border border-gray-300 rounded-lg focus:ring-2 focus:ring-red-500 focus:border-transparent"
                    />
                    <div className="absolute left-3 top-1/2 transform -translate-y-1/2 text-gray-400">
                      <Lock className="w-5 h-5" />
                    </div>
                    <button
                      type="button"
                      onClick={() => setShowCurrentPassword(!showCurrentPassword)}
                      className="absolute right-3 top-1/2 transform -translate-y-1/2 text-gray-400 hover:text-gray-600"
                    >
                      {showCurrentPassword ? <EyeOff className="w-5 h-5" /> : <Eye className="w-5 h-5" />}
                    </button>
                  </div>
                </div>

                {/* 새 비밀번호 */}
                <div>
                  <label className="block text-sm font-medium text-gray-700 mb-2">
                    새 비밀번호
                  </label>
                  <div className="relative">
                    <input
                      type={showNewPassword ? 'text' : 'password'}
                      value={newPassword}
                      onChange={(e) => setNewPassword(e.target.value)}
                      placeholder="새 비밀번호를 입력하세요"
                      className="w-full px-4 py-3 pl-12 pr-12 border border-gray-300 rounded-lg focus:ring-2 focus:ring-red-500 focus:border-transparent"
                    />
                    <div className="absolute left-3 top-1/2 transform -translate-y-1/2 text-gray-400">
                      <Lock className="w-5 h-5" />
                    </div>
                    <button
                      type="button"
                      onClick={() => setShowNewPassword(!showNewPassword)}
                      className="absolute right-3 top-1/2 transform -translate-y-1/2 text-gray-400 hover:text-gray-600"
                    >
                      {showNewPassword ? <EyeOff className="w-5 h-5" /> : <Eye className="w-5 h-5" />}
                    </button>
                  </div>
                </div>

                {/* 새 비밀번호 확인 */}
                <div>
                  <label className="block text-sm font-medium text-gray-700 mb-2">
                    새 비밀번호 확인
                  </label>
                  <div className="relative">
                    <input
                      type={showConfirmNewPassword ? 'text' : 'password'}
                      value={confirmNewPassword}
                      onChange={(e) => setConfirmNewPassword(e.target.value)}
                      placeholder="새 비밀번호를 다시 입력하세요"
                      className="w-full px-4 py-3 pl-12 pr-12 border border-gray-300 rounded-lg focus:ring-2 focus:ring-red-500 focus:border-transparent"
                    />
                    <div className="absolute left-3 top-1/2 transform -translate-y-1/2 text-gray-400">
                      <Lock className="w-5 h-5" />
                    </div>
                    <button
                      type="button"
                      onClick={() => setShowConfirmNewPassword(!showConfirmNewPassword)}
                      className="absolute right-3 top-1/2 transform -translate-y-1/2 text-gray-400 hover:text-gray-600"
                    >
                      {showConfirmNewPassword ? <EyeOff className="w-5 h-5" /> : <Eye className="w-5 h-5" />}
                    </button>
                  </div>
                </div>
              </div>

              {/* 에러 메시지 */}
              {error && (
                <div className="flex items-center space-x-2 text-red-600 bg-red-50 p-3 rounded-lg mt-4">
                  <AlertCircle className="w-5 h-5" />
                  <span className="text-sm">{error}</span>
                </div>
              )}

              {/* 성공 메시지 */}
              {success && (
                <div className="flex items-center space-x-2 text-green-600 bg-green-50 p-3 rounded-lg mt-4">
                  <CheckCircle className="w-5 h-5" />
                  <span className="text-sm">{success}</span>
                </div>
              )}

              {/* 버튼 */}
              <div className="flex space-x-3 mt-6">
                <button
                  onClick={handleClosePasswordChangeModal}
                  className="flex-1 px-4 py-3 bg-gray-500 text-white font-semibold rounded-lg hover:bg-gray-600 transition-colors"
                >
                  취소
                </button>
                <button
                  onClick={changeBitWishPassword}
                  disabled={isChangingPassword}
                  className="flex-1 px-4 py-3 bg-gradient-to-r from-red-500 to-pink-600 text-white font-semibold rounded-lg hover:from-red-600 hover:to-pink-700 transition-all duration-200 disabled:opacity-50 disabled:cursor-not-allowed"
                >
                  {isChangingPassword ? '변경 중...' : '확인'}
                </button>
              </div>
            </div>
          </div>
        </div>
      )}
    </div>
  );
};

export default BitWishWalletAuthModal;
