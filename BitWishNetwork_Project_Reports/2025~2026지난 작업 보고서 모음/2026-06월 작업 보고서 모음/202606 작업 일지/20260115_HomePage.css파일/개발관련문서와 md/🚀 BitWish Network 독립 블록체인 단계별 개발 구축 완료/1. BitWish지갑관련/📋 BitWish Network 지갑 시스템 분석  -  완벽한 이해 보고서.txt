const generateBitWishWalletAddress = async (seedPhrase: string): Promise<string> => {
  // 시드문구를 SHA-256으로 해시화
  const encoder = new TextEncoder();
  const data = encoder.encode(seedPhrase);
  const hashBuffer = await crypto.subtle.digest('SHA-256', data);
  const hashArray = Array.from(new Uint8Array(hashBuffer));
  const hashHex = hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
  
  // BitWish 주소: BW + 40자리 16진수
  return 'BW' + hashHex.substring(0, 40).toUpperCase();
}

```typescript
// simple-server.js (228-242행)
async function generateBitWishAddressFromSeed(seedPhrase) {
  try {
    // 시드문구를 SHA-256으로 해시화
    const encoder = new TextEncoder();
    const data = encoder.encode(seedPhrase);
    const hashBuffer = await crypto.subtle.digest('SHA-256', data);
    const hashArray = Array.from(new Uint8Array(hashBuffer));
    const hashHex = hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
    
    // BitWish 주소: BW + 40자리 16진수
    return 'BW' + hashHex.substring(0, 40).toUpperCase();
  } catch (error) {
    console.error('BitWish address generation error:', error);
    throw error;
  }
}
```
