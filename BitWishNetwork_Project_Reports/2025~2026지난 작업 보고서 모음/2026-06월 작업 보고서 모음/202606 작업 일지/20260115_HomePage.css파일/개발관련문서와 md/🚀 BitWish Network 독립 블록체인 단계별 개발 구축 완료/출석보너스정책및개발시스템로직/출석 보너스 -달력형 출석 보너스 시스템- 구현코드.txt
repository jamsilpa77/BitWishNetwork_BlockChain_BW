### **📋 2단계: 유저용 달력형 출석 보너스 컴포넌트 구현**

```typescript
// Node_HomePage/src/components/mining/user/UserAttendanceBonus.tsx

/**
 * =====================================================
 * ⚠️  중요: 전역 모달, 공통 변수 함수 절대 포함하지 않는다
 * =====================================================
* ❌ 전역 변수 사용 금지
 * ❌ 공통 함수 사용 금지  
 * ❌ 공통 클래스 사용 금지
 * ❌ 전역 모달 사용 금지
 * ❌ 중복 코드 사용 금지
 * ❌ 다른 컴포넌트와 상태 공유 금지
 * ❌ 전역 상태 관리 라이브러리 사용 금지
 * ❌ 스텔라 관련 코드 사용 금지
 * ❌ BIP39 시드문구 사용 금지
 * 
 * ✅ 완벽한 독립성 보장
 * ✅ 자체 상태 관리만 사용
 * ✅ 자체 API 호출만 사용
 * ✅ 자체 에러 처리만 사용
 * ✅ 자체 보안 검증만 사용
 * ✅ BitWish Network 전용 시스템만 사용
 * ✅ 모든 텍스트는 4개국 언어 즉시 번역 시스템 구조로 구현
 * 유저 달력형 출석보너스 컴포넌트 - 완전한 개인별 분리
 * - 1차 백서 정책 수치 완전 적용 (5% 보너스)
 * - 달력형 UI 시스템
 * - AM 09:00:00 시간 제한 로직
 * - 24시간 유예 기간 시스템
 * - 50단위 부동소수점 정밀도
 * - 4개국 언어 즉시 번역
 */

import React, { useState, useEffect, useCallback } from 'react';
import { useTranslation } from 'react-i18next';
import Decimal from 'decimal.js';

// 1차 백서 정책 수치 (완전 적용)
const ATTENDANCE_POLICY = {
  bonusRate: 0.05, // 5% 보너스 (백서 기준)
  checkInTime: 9, // AM 09:00:00부터 출석 체크 가능
  gracePeriod: 24, // 24시간 유예 기간 (시간)
  miningRate: 0.25 // 기본 채굴률 0.25 BW/시간
};

interface UserAttendanceBonusProps {
  walletAddress: string;
  isMiningStarted: boolean;
  onAttendanceChange: (attendanceData: any) => void;
}

const UserAttendanceBonus: React.FC<UserAttendanceBonusProps> = ({ 
  walletAddress, 
  isMiningStarted,
  onAttendanceChange 
}) => {
  const { t } = useTranslation();
  
  // 개인별 상태 관리 (완전 분리)
  const [currentMonth, setCurrentMonth] = useState(new Date());
  const [attendanceData, setAttendanceData] = useState<Map<string, any>>(new Map());
  const [isCheckedIn, setIsCheckedIn] = useState(false);
  const [lastCheckInTime, setLastCheckInTime] = useState<Date | null>(null);
  const [canCheckIn, setCanCheckIn] = useState(false);
  const [timeUntilNextCheckIn, setTimeUntilNextCheckIn] = useState<string>('');
  const [attendanceStreak, setAttendanceStreak] = useState(0);
  const [totalAttendanceDays, setTotalAttendanceDays] = useState(0);
  const [bonusStatus, setBonusStatus] = useState<'active' | 'expired' | 'inactive'>('inactive');

  // 50단위 부동소수점 정밀도 계산
  const calculateWithPrecision = useCallback((value: number) => {
    return new Decimal(value).toDecimalPlaces(50);
  }, []);

  // 한국 시간 가져오기
  const getKoreanTime = useCallback(() => {
    const now = new Date();
    const koreanTime = new Date(now.getTime() + (9 * 60 * 60 * 1000)); // UTC+9
    return koreanTime;
  }, []);

  // 출석 체크 가능 시간 확인 (백서 기준)
  const checkAttendanceTime = useCallback(() => {
    const koreanTime = getKoreanTime();
    const hours = koreanTime.getHours();
    
    // AM 09:00:00부터 출석 체크 가능
    if (hours >= ATTENDANCE_POLICY.checkInTime) {
      setCanCheckIn(true);
      return true;
    }
    
    // 24시간 유예 기간 내라면 출석 체크 가능
    if (lastCheckInTime) {
      const timeDiff = koreanTime.getTime() - lastCheckInTime.getTime();
      const hoursDiff = timeDiff / (1000 * 60 * 60);
      
      if (hoursDiff < ATTENDANCE_POLICY.gracePeriod) {
        setCanCheckIn(true);
        return true;
      }
    }
    
    setCanCheckIn(false);
    return false;
  }, [lastCheckInTime, getKoreanTime]);

  // 출석 보너스 적용 시간 확인 (백서 기준)
  const checkBonusStatus = useCallback(() => {
    if (!lastCheckInTime) {
      setBonusStatus('inactive');
      return;
    }

    const koreanTime = getKoreanTime();
    const timeDiff = koreanTime.getTime() - lastCheckInTime.getTime();
    const hoursDiff = timeDiff / (1000 * 60 * 60);
    
    // 24시간 유예 기간 내라면 보너스 활성
    if (hoursDiff < ATTENDANCE_POLICY.gracePeriod) {
      setBonusStatus('active');
    } else {
      setBonusStatus('expired');
    }
  }, [lastCheckInTime, getKoreanTime]);

  // 다음 출석 체크까지 남은 시간 계산
  const calculateTimeUntilNextCheckIn = useCallback(() => {
    if (!lastCheckInTime) {
      setTimeUntilNextCheckIn('');
      return;
    }

    const koreanTime = getKoreanTime();
    const nextCheckInTime = new Date(lastCheckInTime);
    nextCheckInTime.setDate(nextCheckInTime.getDate() + 1);
    nextCheckInTime.setHours(ATTENDANCE_POLICY.checkInTime, 0, 0, 0);
    
    const timeDiff = nextCheckInTime.getTime() - koreanTime.getTime();
    
    if (timeDiff <= 0) {
      setTimeUntilNextCheckIn('');
      return;
    }

    const hours = Math.floor(timeDiff / (1000 * 60 * 60));
    const minutes = Math.floor((timeDiff % (1000 * 60 * 60)) / (1000 * 60));
    const seconds = Math.floor((timeDiff % (1000 * 60)) / 1000);

    setTimeUntilNextCheckIn(`${hours.toString().padStart(2, '0')}:${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`);
  }, [lastCheckInTime, getKoreanTime]);

  // 출석 보너스 계산 (50단위 정밀도)
  const calculateAttendanceBonus = useCallback(() => {
    if (bonusStatus !== 'active') {
      return {
        bonusRate: 0,
        enhancedRate: ATTENDANCE_POLICY.miningRate,
        bonusAmount: 0
      };
    }

    const bonusRate = calculateWithPrecision(ATTENDANCE_POLICY.bonusRate);
    const enhancedRate = calculateWithPrecision(ATTENDANCE_POLICY.miningRate)
      .times(calculateWithPrecision(1).plus(bonusRate));
    const bonusAmount = enhancedRate.minus(ATTENDANCE_POLICY.miningRate);

    return {
      bonusRate: bonusRate.toNumber(),
      enhancedRate: enhancedRate.toNumber(),
      bonusAmount: bonusAmount.toNumber()
    };
  }, [bonusStatus, calculateWithPrecision]);

  // 달력 데이터 생성
  const generateCalendarData = useCallback(() => {
    const year = currentMonth.getFullYear();
    const month = currentMonth.getMonth();
    
    const firstDay = new Date(year, month, 1);
    const lastDay = new Date(year, month + 1, 0);
    const startDate = new Date(firstDay);
    startDate.setDate(startDate.getDate() - firstDay.getDay());
    
    const calendarDays = [];
    const currentDate = new Date(startDate);
    
    for (let i = 0; i < 42; i++) {
      const dateKey = currentDate.toISOString().split('T')[0];
      const isCurrentMonth = currentDate.getMonth() === month;
      const isToday = currentDate.toDateString() === new Date().toDateString();
      const isAttended = attendanceData.has(dateKey);
      
      calendarDays.push({
        date: new Date(currentDate),
        dateKey,
        isCurrentMonth,
        isToday,
        isAttended
      });
      
      currentDate.setDate(currentDate.getDate() + 1);
    }
    
    return calendarDays;
  }, [currentMonth, attendanceData]);

  // 출석 체크 실행
  const handleCheckIn = useCallback(async () => {
    if (!canCheckIn || !isMiningStarted) return;

    try {
      const now = getKoreanTime();
      const dateKey = now.toISOString().split('T')[0];
      
      const attendanceRecord = {
        walletAddress,
        checkInTime: now.toISOString(),
        bonusRate: ATTENDANCE_POLICY.bonusRate,
        streak: attendanceStreak + 1,
        totalDays: totalAttendanceDays + 1
      };

      // 개인별 데이터 저장
      const newAttendanceData = new Map(attendanceData);
      newAttendanceData.set(dateKey, attendanceRecord);
      setAttendanceData(newAttendanceData);
      
      localStorage.setItem(`personal_${walletAddress}_attendance`, JSON.stringify(Array.from(newAttendanceData.entries())));
      
      // 상태 업데이트
      setIsCheckedIn(true);
      setLastCheckInTime(now);
      setAttendanceStreak(prev => prev + 1);
      setTotalAttendanceDays(prev => prev + 1);
      setBonusStatus('active');
      
      // 부모 컴포넌트에 알림
      onAttendanceChange(attendanceRecord);

      // 성공 알림
      alert(`✅ ${t('attendance.checkIn.success')}\n${t('attendance.checkIn.bonus')}: ${ATTENDANCE_POLICY.bonusRate * 100}%`);
    } catch (error) {
      console.error('출석 체크 오류:', error);
      alert(`❌ ${t('attendance.checkIn.error')}`);
    }
  }, [walletAddress, canCheckIn, isMiningStarted, attendanceStreak, totalAttendanceDays, onAttendanceChange, getKoreanTime, t, attendanceData]);

  // 개인별 데이터 로드
  useEffect(() => {
    const loadPersonalData = () => {
      try {
        const savedData = localStorage.getItem(`personal_${walletAddress}_attendance`);
        if (savedData) {
          const data = JSON.parse(savedData);
          const attendanceMap = new Map(data);
          setAttendanceData(attendanceMap);
          
          // 오늘 출석 여부 확인
          const today = new Date().toISOString().split('T')[0];
          if (attendanceMap.has(today)) {
            setIsCheckedIn(true);
            setLastCheckInTime(new Date(attendanceMap.get(today).checkInTime));
          }
          
          // 통계 계산
          let streak = 0;
          let totalDays = 0;
          const sortedDates = Array.from(attendanceMap.keys()).sort().reverse();
          
          for (const dateKey of sortedDates) {
            totalDays++;
            if (streak === 0) {
              const date = new Date(dateKey);
              const today = new Date();
              const diffTime = today.getTime() - date.getTime();
              const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));
              
              if (diffDays <= 1) {
                streak++;
              } else {
                break;
              }
            }
          }
          
          setAttendanceStreak(streak);
          setTotalAttendanceDays(totalDays);
        }
      } catch (error) {
        console.error('개인별 데이터 로드 오류:', error);
      }
    };

    loadPersonalData();
  }, [walletAddress]);

  // 실시간 업데이트
  useEffect(() => {
    const interval = setInterval(() => {
      checkAttendanceTime();
      checkBonusStatus();
      calculateTimeUntilNextCheckIn();
    }, 1000);

    return () => clearInterval(interval);
  }, [checkAttendanceTime, checkBonusStatus, calculateTimeUntilNextCheckIn]);

  const attendanceBonus = calculateAttendanceBonus();
  const calendarDays = generateCalendarData();

  return (
    <div className="bg-gradient-to-r from-green-50 to-emerald-50 rounded-xl p-6">
      <h3 className="text-xl font-bold text-gray-800 mb-4">
        📅 {t('attendance.title')}
      </h3>

      {/* 출석 상태 표시 */}
      <div className="mb-6">
        <div className={`rounded-lg p-4 ${bonusStatus === 'active' ? 'bg-green-100 border-2 border-green-300' : 'bg-gray-100 border-2 border-gray-300'}`}>
          <div className="flex items-center justify-between mb-2">
            <span className="text-lg font-semibold">
              {bonusStatus === 'active' ? '✅' : bonusStatus === 'expired' ? '⏰' : '❌'} 
              {t('attendance.status.title')}
            </span>
            <span className={`px-3 py-1 rounded-full text-sm font-medium ${
              bonusStatus === 'active' ? 'bg-green-500 text-white' : 
              bonusStatus === 'expired' ? 'bg-yellow-500 text-white' : 
              'bg-gray-500 text-white'
            }`}>
              {bonusStatus === 'active' ? t('attendance.status.active') : 
               bonusStatus === 'expired' ? t('attendance.status.expired') : 
               t('attendance.status.inactive')}
            </span>
          </div>
          
          {lastCheckInTime && (
            <div className="text-sm text-gray-600">
              {t('attendance.lastCheckIn')}: {lastCheckInTime.toLocaleString('ko-KR')}
            </div>
          )}
        </div>
      </div>

      {/* 달력 섹션 */}
      <div className="mb-6">
        <div className="bg-white rounded-lg p-4">
          {/* 달력 헤더 */}
          <div className="flex items-center justify-between mb-4">
            <button
              onClick={() => setCurrentMonth(prev => new Date(prev.getFullYear(), prev.getMonth() - 1))}
              className="p-2 hover:bg-gray-100 rounded-lg transition-colors"
            >
              ←
            </button>
            <h4 className="text-lg font-semibold text-gray-700">
              {currentMonth.getFullYear()}년 {currentMonth.getMonth() + 1}월
            </h4>
            <button
              onClick={() => setCurrentMonth(prev => new Date(prev.getFullYear(), prev.getMonth() + 1))}
              className="p-2 hover:bg-gray-100 rounded-lg transition-colors"
            >
              →
            </button>
          </div>

          {/* 달력 그리드 */}
          <div className="grid grid-cols-7 gap-1">
            {/* 요일 헤더 */}
            {['일', '월', '화', '수', '목', '금', '토'].map(day => (
              <div key={day} className="text-center text-sm font-medium text-gray-500 p-2">
                {day}
              </div>
            ))}
            
            {/* 날짜 셀 */}
            {calendarDays.map((day, index) => (
              <div
                key={index}
                className={`text-center p-2 rounded-lg transition-colors ${
                  !day.isCurrentMonth ? 'text-gray-300' :
                  day.isToday ? 'bg-blue-500 text-white font-bold' :
                  day.isAttended ? 'bg-green-500 text-white font-bold' :
                  'text-gray-700 hover:bg-gray-100'
                }`}
              >
                <div className="text-sm">
                  {day.date.getDate()}
                </div>
                {day.isAttended && (
                  <div className="text-xs">✓</div>
                )}
              </div>
            ))}
          </div>
        </div>
      </div>

      {/* 출석 통계 */}
      <div className="grid grid-cols-2 gap-4 mb-6">
        <div className="bg-white rounded-lg p-4 text-center">
          <div className="text-2xl font-bold text-blue-600">
            {attendanceStreak}
          </div>
          <div className="text-sm text-gray-600">
            {t('attendance.streak')}
          </div>
        </div>
        
        <div className="bg-white rounded-lg p-4 text-center">
          <div className="text-2xl font-bold text-purple-600">
            {totalAttendanceDays}
          </div>
          <div className="text-sm text-gray-600">
            {t('attendance.totalDays')}
          </div>
        </div>
      </div>

      {/* 보너스 정보 */}
      <div className="mb-6">
        <div className="bg-white rounded-lg p-4">
          <h4 className="text-lg font-semibold text-gray-700 mb-3">
            💰 {t('attendance.bonus.title')}
          </h4>
          
          <div className="space-y-2">
            <div className="flex justify-between">
              <span className="text-gray-600">{t('attendance.bonus.rate')}:</span>
              <span className="font-bold text-green-600">
                {attendanceBonus.bonusRate * 100}%
              </span>
            </div>
            
            <div className="flex justify-between">
              <span className="text-gray-600">{t('attendance.bonus.enhancedRate')}:</span>
              <span className="font-bold text-blue-600">
                {attendanceBonus.enhancedRate.toFixed(6)} BW/시간
              </span>
            </div>
            
            <div className="flex justify-between">
              <span className="text-gray-600">{t('attendance.bonus.amount')}:</span>
              <span className="font-bold text-purple-600">
                +{attendanceBonus.bonusAmount.toFixed(6)} BW/시간
              </span>
            </div>
          </div>
        </div>
      </div>

      {/* 다음 출석 체크까지 남은 시간 */}
      {timeUntilNextCheckIn && (
        <div className="mb-6">
          <div className="bg-blue-50 border-2 border-blue-200 rounded-lg p-4">
            <div className="text-center">
              <div className="text-sm text-blue-700 mb-1">
                {t('attendance.nextCheckIn')}
              </div>
              <div className="text-2xl font-bold text-blue-800 font-mono">
                {timeUntilNextCheckIn}
              </div>
            </div>
          </div>
        </div>
      )}

      {/* 출석 체크 버튼 */}
      <button
        onClick={handleCheckIn}
        disabled={!canCheckIn || !isMiningStarted || isCheckedIn}
        className={`w-full py-3 px-4 rounded-lg font-medium transition-colors ${
          !canCheckIn || !isMiningStarted || isCheckedIn
            ? 'bg-gray-300 text-gray-500 cursor-not-allowed'
            : 'bg-green-500 text-white hover:bg-green-600'
        }`}
      >
        {!isMiningStarted ? t('attendance.checkIn.miningRequired') :
         isCheckedIn ? t('attendance.checkIn.checked') :
         canCheckIn ? t('attendance.checkIn.button') :
         t('attendance.checkIn.waiting')}
      </button>

      {/* 마이닝 시작 안내 */}
      {!isMiningStarted && (
        <div className="mt-4 p-3 bg-yellow-50 border border-yellow-200 rounded-lg">
          <div className="text-sm text-yellow-800">
            ⚠️ {t('attendance.miningRequired')}
          </div>
        </div>
      )}
    </div>
  );
};

export default UserAttendanceBonus;
```

### **📋 4단계: 백엔드 API 구현**

```javascript
// Node_HomePage/simple-server-new.js에 추가

// 1차 백서 정책 수치 적용
const ATTENDANCE_POLICY = {
  bonusRate: 0.05, // 5% 보너스 (백서 기준)
  checkInTime: 9, // AM 09:00:00부터 출석 체크 가능
  gracePeriod: 24 // 24시간 유예 기간 (시간)
};

// 출석 체크 API
app.post('/attendance/check-in', (req, res) => {
  try {
    const { walletAddress } = req.body;
    
    // 개인별 출석 데이터 확인
    const attendanceData = personalAttendanceData.get(walletAddress);
    if (attendanceData && attendanceData.isCheckedIn) {
      return res.json({ success: false, error: '이미 출석 체크를 완료했습니다.' });
    }

    // 출석 체크 시간 확인
    const koreanTime = new Date(Date.now() + (9 * 60 * 60 * 1000));
    const hours = koreanTime.getHours();
    
    if (hours < ATTENDANCE_POLICY.checkInTime) {
      return res.json({ success: false, error: '출석 체크 시간이 아닙니다. (AM 09:00:00부터 가능)' });
    }

    // 출석 데이터 저장
    const newAttendanceData = {
      walletAddress,
      isCheckedIn: true,
      checkInTime: new Date().toISOString(),
      bonusRate: ATTENDANCE_POLICY.bonusRate,
      streak: (attendanceData?.streak || 0) + 1,
      totalDays: (attendanceData?.totalDays || 0) + 1
    };

    personalAttendanceData.set(walletAddress, newAttendanceData);

    res.json({ 
      success: true, 
      message: '출석 체크 완료!',
      data: newAttendanceData
    });
  } catch (error) {
    res.status(500).json({ success: false, error: error.message });
  }
});

// 출석 상태 조회 API
app.get('/attendance/status/:walletAddress', (req, res) => {
  try {
    const { walletAddress } = req.params;
    const attendanceData = personalAttendanceData.get(walletAddress) || {};
    
    res.json({ success: true, data: attendanceData });
  } catch (error) {
    res.status(500).json({ success: false, error: error.message });
  }
});
```
