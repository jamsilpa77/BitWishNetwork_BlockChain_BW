### **📋 2단계: 유저용 락업 보너스 컴포넌트 구현**

```typescript
// Node_HomePage/src/components/mining/user/UserLockupBonus.tsx

/**
 * =====================================================
 * ⚠️  중요: 전역 모달, 공통 변수 함수 절대 포함하지 않는다
 * =====================================================
 * ❌ 전역 변수 사용 금지
 * ❌ 공통 함수 사용 금지  
 * ❌ 공통 클래스 사용 금지
 * ❌ 전역 모달 사용 금지
 * ❌ 중복 코드 사용 금지
 * ❌ 다른 컴포넌트와 상태 공유 금지
 * ❌ 전역 상태 관리 라이브러리 사용 금지
 * ❌ 스텔라 관련 코드 사용 금지
 * ❌ BIP39 시드문구 사용 금지
 * 
 * ✅ 완벽한 독립성 보장
 * ✅ 자체 상태 관리만 사용
 * ✅ 자체 API 호출만 사용
 * ✅ 자체 에러 처리만 사용
 * ✅ 자체 보안 검증만 사용
 * ✅ BitWish Network 전용 시스템만 사용
 * ✅ 모든 텍스트는 4개국 언어 즉시 번역 시스템 구조로 구현 
 * 유저 락업보너스 컴포넌트 - 완전한 개인별 분리
 * - 1차 백서 정책 수치 완전 적용
 * - 슬라이더 + 시한폭탄 카운터다운 시스템
 * - 50단위 부동소수점 정밀도
 * - 4개국 언어 즉시 번역
 */

import React, { useState, useEffect, useCallback } from 'react';
import { useTranslation } from 'react-i18next';
import Decimal from 'decimal.js';

// 1차 백서 정책 수치 (완전 적용)
const LOCKUP_POLICY = {
  basicLockup: {
    ratios: [20, 50, 75, 100], // 4단계
    periods: [1, 6, 12, 36], // 4단계 (개월)
    apy: [2.7, 4.2, 5.7, 7.2] // 4단계 APY
  },
  additionalLockup: {
    ratios: [25, 50, 75, 100], // 4단계
    periods: [3, 6, 12, 36], // 4단계 (개월)
    apy: [2.1, 3.5, 4.9, 7.5] // 4단계 APY
  },
  miningRate: 0.25 // 기본 채굴률 0.25 BW/시간
};

interface UserLockupBonusProps {
  walletAddress: string;
  onLockupChange: (lockupData: any) => void;
}

const UserLockupBonus: React.FC<UserLockupBonusProps> = ({ 
  walletAddress, 
  onLockupChange 
}) => {
  const { t } = useTranslation();
  
  // 개인별 상태 관리 (완전 분리)
  const [basicLockupRatio, setBasicLockupRatio] = useState(20);
  const [basicLockupPeriod, setBasicLockupPeriod] = useState(1);
  const [additionalLockupRatio, setAdditionalLockupRatio] = useState(0);
  const [additionalLockupPeriod, setAdditionalLockupPeriod] = useState(0);
  const [isBasicLockupRegistered, setIsBasicLockupRegistered] = useState(false);
  const [isAdditionalLockupRegistered, setIsAdditionalLockupRegistered] = useState(false);
  const [lockupStartTime, setLockupStartTime] = useState<Date | null>(null);
  const [lockupEndTime, setLockupEndTime] = useState<Date | null>(null);
  const [countdown, setCountdown] = useState({ days: 0, hours: 0, minutes: 0, seconds: 0 });
  const [isCountdownActive, setIsCountdownActive] = useState(false);

  // 50단위 부동소수점 정밀도 계산
  const calculateWithPrecision = useCallback((value: number) => {
    return new Decimal(value).toDecimalPlaces(50);
  }, []);

  // 기본락업 APY 계산 (1차 백서 기준)
  const calculateBasicLockupAPY = useCallback(() => {
    const periodIndex = LOCKUP_POLICY.basicLockup.periods.indexOf(basicLockupPeriod);
    return LOCKUP_POLICY.basicLockup.apy[periodIndex] || 2.7;
  }, [basicLockupPeriod]);

  // 추가락업 APY 계산 (1차 백서 기준)
  const calculateAdditionalLockupAPY = useCallback(() => {
    const periodIndex = LOCKUP_POLICY.additionalLockup.periods.indexOf(additionalLockupPeriod);
    return LOCKUP_POLICY.additionalLockup.apy[periodIndex] || 2.1;
  }, [additionalLockupPeriod]);

  // 락업 보너스 계산 (50단위 정밀도)
  const calculateLockupBonus = useCallback(() => {
    if (!isBasicLockupRegistered) return { basic: 0, additional: 0, total: 0 };

    const basicAPY = calculateBasicLockupAPY();
    const additionalAPY = calculateAdditionalLockupAPY();
    
    // 기본락업 보너스 계산
    const basicBonus = calculateWithPrecision(LOCKUP_POLICY.miningRate)
      .times(basicLockupRatio / 100)
      .times(basicAPY / 100);

    // 추가락업 보너스 계산 (남은 채굴률 기반)
    const remainingRate = calculateWithPrecision(1).minus(basicLockupRatio / 100);
    const additionalBonus = isAdditionalLockupRegistered 
      ? remainingRate
          .times(additionalLockupRatio / 100)
          .times(additionalAPY / 100)
      : calculateWithPrecision(0);

    const totalBonus = basicBonus.plus(additionalBonus);

    return {
      basic: basicBonus.toNumber(),
      additional: additionalBonus.toNumber(),
      total: totalBonus.toNumber()
    };
  }, [
    isBasicLockupRegistered,
    isAdditionalLockupRegistered,
    basicLockupRatio,
    additionalLockupRatio,
    calculateBasicLockupAPY,
    calculateAdditionalLockupAPY,
    calculateWithPrecision
  ]);

  // 시한폭탄 카운터다운 계산
  const calculateCountdown = useCallback(() => {
    if (!lockupEndTime) return;

    const now = new Date();
    const endTime = new Date(lockupEndTime);
    const diff = endTime.getTime() - now.getTime();

    if (diff <= 0) {
      setIsCountdownActive(false);
      setCountdown({ days: 0, hours: 0, minutes: 0, seconds: 0 });
      return;
    }

    const days = Math.floor(diff / (1000 * 60 * 60 * 24));
    const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
    const minutes = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60));
    const seconds = Math.floor((diff % (1000 * 60)) / 1000);

    setCountdown({ days, hours, minutes, seconds });
    setIsCountdownActive(true);
  }, [lockupEndTime]);

  // 카운터다운 실시간 업데이트
  useEffect(() => {
    if (!isCountdownActive) return;

    const interval = setInterval(calculateCountdown, 1000);
    return () => clearInterval(interval);
  }, [isCountdownActive, calculateCountdown]);

  // 기본락업 등록
  const handleBasicLockupRegister = useCallback(() => {
    const endTime = new Date();
    endTime.setMonth(endTime.getMonth() + basicLockupPeriod);
    
    setLockupStartTime(new Date());
    setLockupEndTime(endTime);
    setIsBasicLockupRegistered(true);
    
    // 개인별 데이터 저장
    const lockupData = {
      basicLockupRatio,
      basicLockupPeriod,
      basicAPY: calculateBasicLockupAPY(),
      startTime: new Date().toISOString(),
      endTime: endTime.toISOString()
    };
    
    localStorage.setItem(`personal_${walletAddress}_basicLockup`, JSON.stringify(lockupData));
    onLockupChange(lockupData);
  }, [walletAddress, basicLockupRatio, basicLockupPeriod, calculateBasicLockupAPY, onLockupChange]);

  // 추가락업 등록
  const handleAdditionalLockupRegister = useCallback(() => {
    if (!isBasicLockupRegistered) return;
    
    setIsAdditionalLockupRegistered(true);
    
    // 개인별 데이터 저장
    const additionalData = {
      additionalLockupRatio,
      additionalLockupPeriod,
      additionalAPY: calculateAdditionalLockupAPY()
    };
    
    localStorage.setItem(`personal_${walletAddress}_additionalLockup`, JSON.stringify(additionalData));
    onLockupChange({ ...additionalData, isAdditionalLockupRegistered: true });
  }, [walletAddress, additionalLockupRatio, additionalLockupPeriod, calculateAdditionalLockupAPY, isBasicLockupRegistered, onLockupChange]);

  // 락업 해제
  const handleLockupRelease = useCallback(() => {
    setIsBasicLockupRegistered(false);
    setIsAdditionalLockupRegistered(false);
    setLockupStartTime(null);
    setLockupEndTime(null);
    setIsCountdownActive(false);
    
    // 개인별 데이터 삭제
    localStorage.removeItem(`personal_${walletAddress}_basicLockup`);
    localStorage.removeItem(`personal_${walletAddress}_additionalLockup`);
    onLockupChange({ isLockupRegistered: false });
  }, [walletAddress, onLockupChange]);

  // 개인별 데이터 로드
  useEffect(() => {
    const loadPersonalData = () => {
      try {
        // 기본락업 데이터 로드
        const basicLockupData = localStorage.getItem(`personal_${walletAddress}_basicLockup`);
        if (basicLockupData) {
          const data = JSON.parse(basicLockupData);
          setBasicLockupRatio(data.basicLockupRatio || 20);
          setBasicLockupPeriod(data.basicLockupPeriod || 1);
          setIsBasicLockupRegistered(true);
          setLockupStartTime(new Date(data.startTime));
          setLockupEndTime(new Date(data.endTime));
        }

        // 추가락업 데이터 로드
        const additionalLockupData = localStorage.getItem(`personal_${walletAddress}_additionalLockup`);
        if (additionalLockupData) {
          const data = JSON.parse(additionalLockupData);
          setAdditionalLockupRatio(data.additionalLockupRatio || 0);
          setAdditionalLockupPeriod(data.additionalLockupPeriod || 0);
          setIsAdditionalLockupRegistered(true);
        }
      } catch (error) {
        console.error('개인별 데이터 로드 오류:', error);
      }
    };

    loadPersonalData();
  }, [walletAddress]);

  const lockupBonus = calculateLockupBonus();

  return (
    <div className="bg-gradient-to-r from-purple-50 to-indigo-50 rounded-xl p-6">
      <h3 className="text-xl font-bold text-gray-800 mb-4">
        🔒 {t('lockup.title')}
      </h3>

      {/* 기본락업 슬라이더 시스템 */}
      <div className="mb-6">
        <h4 className="text-lg font-semibold text-purple-700 mb-3">
          {t('lockup.basic.title')}
        </h4>
        
        {/* 비율 슬라이더 */}
        <div className="mb-4">
          <label className="block text-sm font-medium text-gray-700 mb-2">
            {t('lockup.basic.ratio')}: {basicLockupRatio}%
          </label>
          <input
            type="range"
            min="20"
            max="100"
            step="25"
            value={basicLockupRatio}
            onChange={(e) => setBasicLockupRatio(Number(e.target.value))}
            className="w-full h-2 bg-gray-200 rounded-lg appearance-none cursor-pointer slider"
            disabled={isBasicLockupRegistered}
          />
          <div className="flex justify-between text-xs text-gray-500 mt-1">
            <span>20%</span>
            <span>50%</span>
            <span>75%</span>
            <span>100%</span>
          </div>
        </div>

        {/* 기간 선택 */}
        <div className="mb-4">
          <label className="block text-sm font-medium text-gray-700 mb-2">
            {t('lockup.basic.period')}
          </label>
          <div className="grid grid-cols-4 gap-2">
            {LOCKUP_POLICY.basicLockup.periods.map((period) => (
              <button
                key={period}
                onClick={() => setBasicLockupPeriod(period)}
                disabled={isBasicLockupRegistered}
                className={`px-3 py-2 rounded-lg text-sm font-medium transition-colors ${
                  basicLockupPeriod === period
                    ? 'bg-purple-500 text-white'
                    : 'bg-white text-gray-700 hover:bg-purple-100'
                } ${isBasicLockupRegistered ? 'opacity-50 cursor-not-allowed' : ''}`}
              >
                {period}{t('lockup.months')}
              </button>
            ))}
          </div>
        </div>

        {/* APY 표시 */}
        <div className="mb-4">
          <div className="bg-white rounded-lg p-3">
            <div className="text-sm text-gray-600">
              {t('lockup.basic.apy')}: <span className="font-bold text-purple-600">
                {calculateBasicLockupAPY()}%
              </span>
            </div>
            <div className="text-sm text-gray-600">
              {t('lockup.basic.bonus')}: <span className="font-bold text-green-600">
                {lockupBonus.basic.toFixed(6)} BW/시간
              </span>
            </div>
          </div>
        </div>

        {/* 등록 버튼 */}
        <button
          onClick={handleBasicLockupRegister}
          disabled={isBasicLockupRegistered}
          className={`w-full py-2 px-4 rounded-lg font-medium transition-colors ${
            isBasicLockupRegistered
              ? 'bg-gray-300 text-gray-500 cursor-not-allowed'
              : 'bg-purple-500 text-white hover:bg-purple-600'
          }`}
        >
          {isBasicLockupRegistered ? t('lockup.basic.registered') : t('lockup.basic.register')}
        </button>
      </div>

      {/* 추가락업 슬라이더 시스템 */}
      {isBasicLockupRegistered && (
        <div className="mb-6">
          <h4 className="text-lg font-semibold text-indigo-700 mb-3">
            {t('lockup.additional.title')}
          </h4>
          
          {/* 비율 슬라이더 */}
          <div className="mb-4">
            <label className="block text-sm font-medium text-gray-700 mb-2">
              {t('lockup.additional.ratio')}: {additionalLockupRatio}%
            </label>
            <input
              type="range"
              min="25"
              max="100"
              step="25"
              value={additionalLockupRatio}
              onChange={(e) => setAdditionalLockupRatio(Number(e.target.value))}
              className="w-full h-2 bg-gray-200 rounded-lg appearance-none cursor-pointer slider"
              disabled={isAdditionalLockupRegistered}
            />
            <div className="flex justify-between text-xs text-gray-500 mt-1">
              <span>25%</span>
              <span>50%</span>
              <span>75%</span>
              <span>100%</span>
            </div>
          </div>

          {/* 기간 선택 */}
          <div className="mb-4">
            <label className="block text-sm font-medium text-gray-700 mb-2">
              {t('lockup.additional.period')}
            </label>
            <div className="grid grid-cols-4 gap-2">
              {LOCKUP_POLICY.additionalLockup.periods.map((period) => (
                <button
                  key={period}
                  onClick={() => setAdditionalLockupPeriod(period)}
                  disabled={isAdditionalLockupRegistered}
                  className={`px-3 py-2 rounded-lg text-sm font-medium transition-colors ${
                    additionalLockupPeriod === period
                      ? 'bg-indigo-500 text-white'
                      : 'bg-white text-gray-700 hover:bg-indigo-100'
                  } ${isAdditionalLockupRegistered ? 'opacity-50 cursor-not-allowed' : ''}`}
                >
                  {period}{t('lockup.months')}
                </button>
              ))}
            </div>
          </div>

          {/* APY 표시 */}
          <div className="mb-4">
            <div className="bg-white rounded-lg p-3">
              <div className="text-sm text-gray-600">
                {t('lockup.additional.apy')}: <span className="font-bold text-indigo-600">
                  {calculateAdditionalLockupAPY()}%
                </span>
              </div>
              <div className="text-sm text-gray-600">
                {t('lockup.additional.bonus')}: <span className="font-bold text-green-600">
                  {lockupBonus.additional.toFixed(6)} BW/시간
                </span>
              </div>
            </div>
          </div>

          {/* 등록 버튼 */}
          <button
            onClick={handleAdditionalLockupRegister}
            disabled={isAdditionalLockupRegistered}
            className={`w-full py-2 px-4 rounded-lg font-medium transition-colors ${
              isAdditionalLockupRegistered
                ? 'bg-gray-300 text-gray-500 cursor-not-allowed'
                : 'bg-indigo-500 text-white hover:bg-indigo-600'
            }`}
          >
            {isAdditionalLockupRegistered ? t('lockup.additional.registered') : t('lockup.additional.register')}
          </button>
        </div>
      )}

      {/* 시한폭탄 카운터다운 */}
      {isCountdownActive && (
        <div className="mb-6">
          <h4 className="text-lg font-semibold text-red-700 mb-3">
            ⏰ {t('lockup.countdown.title')}
          </h4>
          <div className="bg-red-50 border-2 border-red-200 rounded-lg p-4">
            <div className="grid grid-cols-4 gap-4 text-center">
              <div className="bg-red-500 text-white rounded-lg p-3">
                <div className="text-2xl font-bold">{countdown.days}</div>
                <div className="text-sm">{t('lockup.countdown.days')}</div>
              </div>
              <div className="bg-red-500 text-white rounded-lg p-3">
                <div className="text-2xl font-bold">{countdown.hours}</div>
                <div className="text-sm">{t('lockup.countdown.hours')}</div>
              </div>
              <div className="bg-red-500 text-white rounded-lg p-3">
                <div className="text-2xl font-bold">{countdown.minutes}</div>
                <div className="text-sm">{t('lockup.countdown.minutes')}</div>
              </div>
              <div className="bg-red-500 text-white rounded-lg p-3">
                <div className="text-2xl font-bold">{countdown.seconds}</div>
                <div className="text-sm">{t('lockup.countdown.seconds')}</div>
              </div>
            </div>
          </div>
        </div>
      )}

      {/* 총 보너스 표시 */}
      <div className="mb-6">
        <div className="bg-gradient-to-r from-green-50 to-emerald-50 rounded-lg p-4">
          <h4 className="text-lg font-semibold text-green-700 mb-2">
            💰 {t('lockup.total.bonus')}
          </h4>
          <div className="text-2xl font-bold text-green-600">
            {lockupBonus.total.toFixed(6)} BW/시간
          </div>
          <div className="text-sm text-gray-600 mt-1">
            {t('lockup.total.description')}
          </div>
        </div>
      </div>

      {/* 락업 해제 버튼 */}
      {(isBasicLockupRegistered || isAdditionalLockupRegistered) && (
        <button
          onClick={handleLockupRelease}
          className="w-full py-2 px-4 bg-red-500 text-white rounded-lg font-medium hover:bg-red-600 transition-colors"
        >
          🔓 {t('lockup.release')}
        </button>
      )}
    </div>
  );
};

export default UserLockupBonus;
```

### **📋 4단계: 백엔드 API 구현**

```javascript
// Node_HomePage/simple-server-new.js에 추가

// 1차 백서 정책 수치 적용
const LOCKUP_POLICY = {
  basicLockup: {
    ratios: [20, 50, 75, 100],
    periods: [1, 6, 12, 36],
    apy: [2.7, 4.2, 5.7, 7.2]
  },
  additionalLockup: {
    ratios: [25, 50, 75, 100],
    periods: [3, 6, 12, 36],
    apy: [2.1, 3.5, 4.9, 7.5]
  },
  miningRate: 0.25
};

// 락업 APY 계산 API
app.get('/api/lockup/calculate', (req, res) => {
  try {
    const { type, period } = req.query;
    
    let apy = 0;
    if (type === 'basic') {
      const periodIndex = LOCKUP_POLICY.basicLockup.periods.indexOf(Number(period));
      apy = LOCKUP_POLICY.basicLockup.apy[periodIndex] || 2.7;
    } else if (type === 'additional') {
      const periodIndex = LOCKUP_POLICY.additionalLockup.periods.indexOf(Number(period));
      apy = LOCKUP_POLICY.additionalLockup.apy[periodIndex] || 2.1;
    }
    
    res.json({ success: true, apy });
  } catch (error) {
    res.status(500).json({ success: false, error: error.message });
  }
});

// 락업 생성 API
app.post('/api/lockup/create', (req, res) => {
  try {
    const { walletAddress, type, ratio, period } = req.body;
    
    // 개인별 락업 데이터 저장
    const lockupData = {
      walletAddress,
      type,
      ratio,
      period,
      apy: type === 'basic' 
        ? LOCKUP_POLICY.basicLockup.apy[LOCKUP_POLICY.basicLockup.periods.indexOf(period)]
        : LOCKUP_POLICY.additionalLockup.apy[LOCKUP_POLICY.additionalLockup.periods.indexOf(period)],
      startTime: new Date().toISOString(),
      endTime: new Date(Date.now() + period * 30 * 24 * 60 * 60 * 1000).toISOString()
    };
    
    // 개인별 데이터 저장소에 저장
    if (!personalLockupData.has(walletAddress)) {
      personalLockupData.set(walletAddress, {});
    }
    personalLockupData.get(walletAddress)[type] = lockupData;
    
    res.json({ success: true, data: lockupData });
  } catch (error) {
    res.status(500).json({ success: false, error: error.message });
  }
});

// 락업 상태 조회 API
app.get('/api/lockup/status/:walletAddress', (req, res) => {
  try {
    const { walletAddress } = req.params;
    const lockupData = personalLockupData.get(walletAddress) || {};
    
    res.json({ success: true, data: lockupData });
  } catch (error) {
    res.status(500).json({ success: false, error: error.message });
  }
});

// 락업 해제 API
app.post('/api/lockup/release', (req, res) => {
  try {
    const { walletAddress, type } = req.body;
    
    if (personalLockupData.has(walletAddress)) {
      if (type) {
        delete personalLockupData.get(walletAddress)[type];
      } else {
        personalLockupData.delete(walletAddress);
      }
    }
    
    res.json({ success: true, message: '락업이 해제되었습니다.' });
  } catch (error) {
    res.status(500).json({ success: false, error: error.message });
  }
});

// 통합 보상률 계산 API
app.get('/api/rewards/rate/:walletAddress', (req, res) => {
  try {
    const { walletAddress } = req.params;
    
    // 락업 보너스 계산
    const lockupData = personalLockupData.get(walletAddress) || {};
    let lockupBonus = 0;
    
    if (lockupData.basicLockup) {
      const basicBonus = LOCKUP_POLICY.miningRate * 
        (lockupData.basicLockup.ratio / 100) * 
        (lockupData.basicLockup.apy / 100);
      lockupBonus += basicBonus;
    }
    
    if (lockupData.additionalLockup) {
      const remainingRate = 1 - (lockupData.basicLockup?.ratio || 0) / 100;
      const additionalBonus = LOCKUP_POLICY.miningRate * 
        remainingRate * 
        (lockupData.additionalLockup.ratio / 100) * 
        (lockupData.additionalLockup.apy / 100);
      lockupBonus += additionalBonus;
    }
    
    // 총 보너스율 계산
    const totalBonusRate = lockupBonus / LOCKUP_POLICY.miningRate;
    const enhancedRate = LOCKUP_POLICY.miningRate * (1 + totalBonusRate);
    
    res.json({ 
      success: true, 
      data: {
        lockupBonus,
        totalBonusRate,
        enhancedRate
      }
    });
  } catch (error) {
    res.status(500).json({ success: false, error: error.message });
  }
});
```
