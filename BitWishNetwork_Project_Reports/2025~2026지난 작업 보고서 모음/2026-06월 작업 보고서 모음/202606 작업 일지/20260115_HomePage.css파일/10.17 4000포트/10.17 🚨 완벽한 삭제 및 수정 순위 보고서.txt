    const data = await response.json();
    
    if (data.success) {
      // ✅ 가상 BW 실시간 업데이트 (50자리 정밀도)
      setMiningData(prev => ({
        ...prev,
        totalMined: data.data.totalMined,
        miningTime: data.data.miningTime,
        currentHashRate: data.data.hashRate,
        lastUpdated: new Date().toISOString()
      }));
    }
  } catch (error) {
    console.error('❌ 가상 마이닝 업데이트 오류:', error);
  }
};

// ✅ 실시간 업데이트 useEffect 추가
useEffect(() => {
  if (miningData?.isActive) {
    const interval = setInterval(() => {
      updateVirtualMining();
    }, 5000); // 5초마다 업데이트
    
    return () => clearInterval(interval);
  }
}, [miningData?.isActive, walletAddress]);

```javascript
// ✅ 관리자 마이닝 정지 API 경로 통일
app.post('/api/admin/bitwish/mining/stop', async (req, res) => {
  try {
    const { walletAddress } = req.body;

    if (!walletAddress || !walletAddress.startsWith('BW')) {
      return res.status(400).json({
        success: false,
        message: '유효하지 않은 지갑 주소입니다.'
      });
    }

    const db = bitWishMongoDB.db;
    if (!db) {
      return res.status(500).json({
        success: false,
        message: 'MongoDB 연결이 필요합니다.'
      });
    }

    const miningCollection = db.collection('admin_user_mining_data');

    // ✅ 관리자 가상 마이닝 데이터 조회 (userType='admin' 필터)
    const miningData = await miningCollection.findOne({ 
      walletAddress,
      userType: 'admin'
    });

    if (!miningData) {
      return res.status(404).json({
        success: false,
        message: '관리자 가상 마이닝 데이터를 찾을 수 없습니다.'
      });
    }

    if (!miningData.miningStatus.isActive) {
      return res.status(400).json({
        success: false,
        message: '가상 마이닝이 활성화되지 않았습니다.'
      });
    }

    const now = new Date();
    const nowString = now.toISOString().replace('T', ' ').substring(0, 19);

    // ✅ 50자리 정밀도 계산 (블록체인 완전 분리)
    const Decimal = require('decimal.js');
    Decimal.set({ precision: 50 });

    // ✅ 가상 BW 마이닝 시간 계산
    const startTime = new Date(miningData.miningStartTime);
    const elapsedMs = now - startTime;
    const elapsedHours = elapsedMs / (1000 * 60 * 60);

    // ✅ 가상 BW 계산 (50자리 정밀도)
    const totalBasicMined = (elapsedHours * 0.25).toFixed(50);
    const totalAttendanceBonus = miningData.miningBreakdown.attendance.isCheckedToday ? 
      (parseFloat(totalBasicMined) * 0.05).toFixed(50) : "0.00000000000000000000000000000000000000000000000000";
    
    const activeReferrals = miningData.miningBreakdown.referralPermanent.activeReferrals || 0;
    const totalReferralPermanent = (parseFloat(totalBasicMined) * 0.02 * activeReferrals).toFixed(50);
    const referralOneTimeBonus = miningData.miningBreakdown.referralOneTime.total || "0.00000000000000000000000000000000000000000000000000";

    const totalMined = (
      parseFloat(totalBasicMined) + 
      parseFloat(totalAttendanceBonus) + 
      parseFloat(totalReferralPermanent) + 
      parseFloat(referralOneTimeBonus)
    ).toFixed(50);

    // ✅ 관리자 가상 마이닝 데이터 업데이트 (완전한 독립성)
    await miningCollection.updateOne(
      { walletAddress, userType: 'admin' },
      {
        $set: {
          'miningStatus.isActive': false,
          'miningStatus.isPaused': false,
          'miningStatus.lastUpdateTime': nowString,
          'totalMined.amount': totalMined,
          'miningBreakdown.basic.total': totalBasicMined,
          'miningBreakdown.attendance.total': totalAttendanceBonus,
          'miningBreakdown.referralPermanent.total': totalReferralPermanent,
          updatedAt: now
        }
      }
    );

    res.json({
      success: true,
      message: '관리자 가상 BW 마이닝이 중지되었습니다.',
      data: {
        walletAddress: walletAddress,
        miningStopTime: nowString,
        totalMined: totalMined,
        systemType: 'VIRTUAL_BW_ONLY',
        isTestMode: true,
        blockchainConnection: false
      }
    });

  } catch (error) {
    console.error('❌ 관리자 가상 마이닝 정지 오류:', error);
    res.status(500).json({
      success: false,
      message: '관리자 가상 마이닝 정지 중 오류가 발생했습니다.',
      error: error.message
    });
  }
});
```

### **6. 관리자 전용 출석 체크 함수 구현**

```typescript
// ✅ 추가할 코드 - 관리자 전용 출석 체크
const handleAdminAttendanceCheck = useCallback(async () => {
  try {
    const response = await fetch('http://localhost:4001/api/admin/bitwish/test/attendance/check', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ 
        walletAddress,
        adminData: { accessLevel: 'ADMIN' }
      })
    });

    const result = await response.json();
    
    if (result.success) {
      showAdminModal('success', '출석 체크 완료', '관리자 출석 보너스가 지급되었습니다.');
    } else {
      showAdminModal('error', '출석 체크 실패', result.message || '출석 체크에 실패했습니다.');
    }
  } catch (error) {
    showAdminModal('error', 'API 오류', '출석 체크 중 오류가 발생했습니다.');
  }
}, [walletAddress, showAdminModal]);
```
