## **📌 Part 1: Multer 설정 (simple-server.cjs 상단에 추가)**

```javascript
// ========================================
// Multer 파일 업로드 설정
// ========================================

const multer = require('multer');
const path = require('path');
const fs = require('fs');

// KYC 파일 저장 설정
const kycStorage = multer.diskStorage({
  destination: function (req, file, cb) {
    const walletAddress = req.body.walletAddress || 'unknown';
    const uploadDir = path.join(__dirname, 'uploads', 'kyc', walletAddress);
    
    // 디렉토리 생성 (없으면)
    if (!fs.existsSync(uploadDir)) {
      fs.mkdirSync(uploadDir, { recursive: true });
    }
    
    cb(null, uploadDir);
  },
  filename: function (req, file, cb) {
    const documentType = file.fieldname; // idCard, fullBodyPhoto, selfieWithID
    const timestamp = new Date().toISOString().replace(/[:.]/g, '-').substring(0, 19);
    const ext = path.extname(file.originalname);
    const fileName = `${documentType}_${timestamp}${ext}`;
    cb(null, fileName);
  }
});

// 파일 필터 (JPG/PNG만 허용)
const fileFilter = (req, file, cb) => {
  const allowedTypes = ['image/jpeg', 'image/jpg', 'image/png'];
  
  if (allowedTypes.includes(file.mimetype)) {
    cb(null, true);
  } else {
    cb(new Error('JPG 또는 PNG 파일만 업로드 가능합니다.'), false);
  }
};

// Multer 인스턴스 생성
const uploadKYC = multer({
  storage: kycStorage,
  fileFilter: fileFilter,
  limits: {
    fileSize: 10 * 1024 * 1024 // 10MB
  }
}).fields([
  { name: 'idCard', maxCount: 1 },
  { name: 'fullBodyPhoto', maxCount: 1 },
  { name: 'selfieWithID', maxCount: 1 }
]);

console.log('✅ Multer KYC 파일 업로드 시스템 초기화 완료');
```

## **📌 Part 2: Backend API 6-1 - KYC 신청 (simple-server.cjs에 추가)**

```javascript
// ========================================
// API 6-1: POST /api/admin/bitwish/kyc/submit
// 설명: KYC 신청 (파일 업로드 포함, 50자리 정밀도)
// ========================================

app.post('/api/admin/bitwish/kyc/submit', (req, res) => {
  uploadKYC(req, res, async (err) => {
    try {
      // ========== 1. Multer 에러 처리 ==========
      if (err instanceof multer.MulterError) {
        if (err.code === 'LIMIT_FILE_SIZE') {
          return res.status(400).json({
            success: false,
            message: '파일 크기는 10MB를 초과할 수 없습니다.'
          });
        }
        return res.status(400).json({
          success: false,
          message: `파일 업로드 오류: ${err.message}`
        });
      } else if (err) {
        return res.status(400).json({
          success: false,
          message: err.message
        });
      }

      const { walletAddress } = req.body;
      const personalInfo = JSON.parse(req.body.personalInfo || '{}');

      // ========== 2. 입력 검증 ==========
      if (!walletAddress || !walletAddress.startsWith('BW')) {
        return res.status(400).json({
          success: false,
          message: '유효하지 않은 지갑 주소입니다.'
        });
      }

      if (!personalInfo.name || !personalInfo.birthDate || !personalInfo.phone) {
        return res.status(400).json({
          success: false,
          message: '필수 개인정보를 모두 입력해주세요.'
        });
      }

      // ========== 3. 파일 업로드 확인 ==========
      const files = req.files;
      if (!files || !files.idCard || !files.fullBodyPhoto || !files.selfieWithID) {
        return res.status(400).json({
          success: false,
          message: '신분증, 전신사진, 셀카를 모두 업로드해주세요.'
        });
      }

      const db = bitWishMongoDB.db;
      if (!db) {
        return res.status(500).json({
          success: false,
          message: 'MongoDB 연결이 필요합니다.'
        });
      }

      const miningCollection = db.collection('admin_user_mining_data');

      // ========== 4. 기존 데이터 조회 (userType='user' 필터) ==========
      const userData = await miningCollection.findOne({ 
        walletAddress,
        userType: 'user'
      });

      if (!userData) {
        return res.status(404).json({
          success: false,
          message: '마이닝 데이터를 찾을 수 없습니다.'
        });
      }

      // 이미 KYC 신청했는지 확인
      if (userData.kyc && userData.kyc.status !== 'none') {
        return res.status(400).json({
          success: false,
          message: `이미 KYC 신청 완료 (상태: ${userData.kyc.status})`
        });
      }

      const now = new Date();
      const nowString = now.toISOString().replace('T', ' ').substring(0, 19);

      // ========== 5. 파일 정보 저장 ==========
      const documentsInfo = {
        idCard: {
          uploaded: true,
          fileName: files.idCard[0].filename,
          filePath: path.join('uploads', 'kyc', walletAddress, files.idCard[0].filename),
          uploadedAt: nowString
        },
        fullBodyPhoto: {
          uploaded: true,
          fileName: files.fullBodyPhoto[0].filename,
          filePath: path.join('uploads', 'kyc', walletAddress, files.fullBodyPhoto[0].filename),
          uploadedAt: nowString
        },
        selfieWithID: {
          uploaded: true,
          fileName: files.selfieWithID[0].filename,
          filePath: path.join('uploads', 'kyc', walletAddress, files.selfieWithID[0].filename),
          uploadedAt: nowString
        }
      };

      // ========== 6. KYC 데이터 구성 ==========
      const kycData = {
        status: 'pending',
        submittedAt: nowString,
        approvedDate: null,
        rejectedDate: null,
        rejectionReason: null,
        documents: documentsInfo,
        personalInfo: {
          name: personalInfo.name,
          birthDate: personalInfo.birthDate,
          address: personalInfo.address || '',
          phone: personalInfo.phone,
          email: personalInfo.email || ''
        },
        migrationStartDate: null,
        countdown: {
          days: 0,
          hours: 0,
          minutes: 0,
          seconds: 0,
          totalSeconds: 0,
          isActive: false
        }
      };

      // ========== 7. MongoDB 업데이트 (userType='user' 필터) ==========
      await miningCollection.updateOne(
        { 
          walletAddress,
          userType: 'user'
        },
        {
          $set: {
            kyc: kycData,
            updatedAt: now
          }
        }
      );

      // ========== 8. 응답 ==========
      res.json({
        success: true,
        message: 'KYC 신청이 완료되었습니다. 승인까지 24-48시간이 소요됩니다.',
        data: {
          walletAddress: walletAddress,
          status: 'pending',
          submittedAt: nowString,
          documents: {
            idCard: documentsInfo.idCard.fileName,
            fullBodyPhoto: documentsInfo.fullBodyPhoto.fileName,
            selfieWithID: documentsInfo.selfieWithID.fileName
          }
        }
      });

    } catch (error) {
      console.error('❌ KYC 신청 오류:', error);
      res.status(500).json({
        success: false,
        message: 'KYC 신청 중 오류가 발생했습니다.',
        error: error.message
      });
    }
  });
});
```

## **📌 Part 4: Backend API 6-3 - 전체 KYC 신청 조회 (관리자용)**

```javascript
// ========================================
// API 6-3: GET /api/admin/bitwish/kyc/all-applications
// 설명: 전체 KYC 신청 조회 (관리자 전용)
// ========================================

app.get('/api/admin/bitwish/kyc/all-applications', async (req, res) => {
  try {
    const { adminWallet } = req.query;

    // ========== 관리자 검증 ==========
    if (!adminWallet || !adminWallet.startsWith('BW')) {
      return res.status(403).json({
        success: false,
        message: '관리자 권한이 필요합니다.'
      });
    }

    const db = bitWishMongoDB.db;
    if (!db) {
      return res.status(500).json({
        success: false,
        message: 'MongoDB 연결이 필요합니다.'
      });
    }

    const miningCollection = db.collection('admin_user_mining_data');

    // ========== 관리자 확인 (userType='admin' 필터) ==========
    const adminData = await miningCollection.findOne({ 
      walletAddress: adminWallet,
      userType: 'admin'
    });

    if (!adminData) {
      return res.status(403).json({
        success: false,
        message: '관리자 권한이 없습니다.'
      });
    }

    // ========== 전체 KYC 신청 조회 (userType='user'만) ==========
    const allUsers = await miningCollection.find({ 
      userType: 'user',
      'kyc.status': { $exists: true, $ne: 'none' }
    }).toArray();

    const applications = allUsers.map(user => ({
      walletAddress: user.walletAddress,
      status: user.kyc.status,
      submittedAt: user.kyc.submittedAt,
      approvedDate: user.kyc.approvedDate,
      rejectedDate: user.kyc.rejectedDate,
      personalInfo: user.kyc.personalInfo,
      documents: {
        idCard: user.kyc.documents.idCard.filePath,
        fullBodyPhoto: user.kyc.documents.fullBodyPhoto.filePath,
        selfieWithID: user.kyc.documents.selfieWithID.filePath
      }
    }));

    // 상태별 통계
    const stats = {
      total: applications.length,
      pending: applications.filter(a => a.status === 'pending').length,
      approved: applications.filter(a => a.status === 'approved').length,
      rejected: applications.filter(a => a.status === 'rejected').length
    };

    res.json({
      success: true,
      stats: stats,
      applications: applications
    });

  } catch (error) {
    console.error('❌ KYC 신청 조회 오류:', error);
    res.status(500).json({
      success: false,
      message: 'KYC 신청 조회 중 오류가 발생했습니다.',
      error: error.message
    });
```

## **📌 Part 5: Backend API 6-4 - KYC 승인 (관리자용, 50자리 정밀도)**

```javascript
// ========================================
// API 6-4: POST /api/admin/bitwish/kyc/approve
// 설명: KYC 승인 처리 (관리자 전용, 50자리 정밀도)
// ========================================

app.post('/api/admin/bitwish/kyc/approve', async (req, res) => {
  try {
    const { adminWallet, walletAddress } = req.body;

    // ========== 1. 입력 검증 ==========
    if (!adminWallet || !adminWallet.startsWith('BW')) {
      return res.status(403).json({
        success: false,
        message: '관리자 권한이 필요합니다.'
      });
    }

    if (!walletAddress || !walletAddress.startsWith('BW')) {
      return res.status(400).json({
        success: false,
        message: '유효하지 않은 지갑 주소입니다.'
      });
    }

    const db = bitWishMongoDB.db;
    if (!db) {
      return res.status(500).json({
        success: false,
        message: 'MongoDB 연결이 필요합니다.'
      });
    }

    const miningCollection = db.collection('admin_user_mining_data');

    // ========== 2. 관리자 확인 (userType='admin' 필터) ==========
    const adminData = await miningCollection.findOne({ 
      walletAddress: adminWallet,
      userType: 'admin'
    });

    if (!adminData) {
      return res.status(403).json({
        success: false,
        message: '관리자 권한이 없습니다.'
      });
    }

    // ========== 3. 유저 데이터 조회 (userType='user' 필터) ==========
    const userData = await miningCollection.findOne({ 
      walletAddress,
      userType: 'user'
    });

    if (!userData) {
      return res.status(404).json({
        success: false,
        message: '유저 데이터를 찾을 수 없습니다.'
      });
    }

    if (!userData.kyc || userData.kyc.status !== 'pending') {
      return res.status(400).json({
        success: false,
        message: 'KYC 승인 대기 상태가 아닙니다.'
      });
    }

    const now = new Date();
    const nowString = now.toISOString().replace('T', ' ').substring(0, 19);

    // ========== 4. 마이그레이션 시작일 설정 (승인일 + 15일) ==========
    const migrationDate = new Date(now);
    migrationDate.setDate(migrationDate.getDate() + 15);
    const migrationDateString = migrationDate.toISOString().substring(0, 10);

    // ========== 5. 15일 카운트다운 계산 ==========
    const totalSeconds = 15 * 24 * 60 * 60; // 15일
    const days = 15;
    const hours = 0;
    const minutes = 0;
    const seconds = 0;

    // ========== 6. KYC 승인 업데이트 (userType='user' 필터) ==========
    await miningCollection.updateOne(
      { 
        walletAddress,
        userType: 'user'
      },
      {
        $set: {
          'kyc.status': 'approved',
          'kyc.approvedDate': nowString,
          'kyc.migrationStartDate': migrationDateString,
          'kyc.countdown.days': days,
          'kyc.countdown.hours': hours,
          'kyc.countdown.minutes': minutes,
          'kyc.countdown.seconds': seconds,
          'kyc.countdown.totalSeconds': totalSeconds,
          'kyc.countdown.isActive': true,
          updatedAt: now
        }
      }
    );

    // ========== 7. 추천 보너스 활성화 처리 (50자리 정밀도) ==========
    if (userData.referralInfo && userData.referralInfo.usedReferralCode) {
      const referrerWallet = userData.referralInfo.referrerWallet;

      // 7-1. 가입자 1회 2% 보너스 계산
      let oneTime2PercentAmount = "0.00000000000000000000000000000000000000000000000000";
      
      if (!userData.referralInfo.oneTime2PercentApplied) {
        const currentBasicMining = parseFloat(userData.miningBreakdown?.basic?.total || 0);
        oneTime2PercentAmount = (currentBasicMining * 0.02).toFixed(50);
      }

      // 7-2. 가입자 보너스 활성화
      const refereeOneTimeBonusTotal = (
        parseFloat(userData.referralInfo.oneTimeBonusReceived || 0) + 
        parseFloat(oneTime2PercentAmount)
      ).toFixed(50);

      const refereeReferralOneTimeTotal = (
        parseFloat(userData.miningBreakdown?.referralOneTime?.total || 0) + 
        parseFloat(refereeOneTimeBonusTotal)
      ).toFixed(50);

      await miningCollection.updateOne(
        { 
          walletAddress,
          userType: 'user'
        },
        {
          $set: {
            'referralInfo.oneTimeBonusStatus': 'activated',
            'referralInfo.oneTime2PercentApplied': true,
            'referralInfo.oneTime2PercentAmount': oneTime2PercentAmount,
            'miningBreakdown.referralOneTime.total': refereeReferralOneTimeTotal,
            'miningBreakdown.referralOneTime.count': 1,
            'miningBreakdown.referralOneTime.perReferral': refereeOneTimeBonusTotal,
            updatedAt: now
          }
        }
      );

      // 7-3. 추천인 보관함 KYC 상태 업데이트
      const referrerData = await miningCollection.findOne({ 
        walletAddress: referrerWallet,
        userType: 'user'
      });

      if (referrerData) {
        const referrerReferees = referrerData.referralBonusVault?.referees || {};
        
        if (referrerReferees[walletAddress]) {
          referrerReferees[walletAddress].kycStatus = 'approved';
          referrerReferees[walletAddress].kycApprovedDate = nowString;
          referrerReferees[walletAddress].canMigrate = true;
          referrerReferees[walletAddress].migrationDate = migrationDateString;
        }

        // 총 보관함 금액 재계산 (50자리 정밀도)
        let totalVault = 0;
        for (const refKey in referrerReferees) {
          totalVault += parseFloat(referrerReferees[refKey].totalFromThisReferee || 0);
        }
        const totalVaultStr = totalVault.toFixed(50);

        await miningCollection.updateOne(
          { 
            walletAddress: referrerWallet,
            userType: 'user'
          },
          {
            $set: {
              'referralBonusVault.totalVault': totalVaultStr,
              'referralBonusVault.referees': referrerReferees,
              updatedAt: now
            }
          }
        );
      }
    }

    // ========== 8. 응답 ==========
    res.json({
      success: true,
      message: 'KYC 승인이 완료되었습니다.',
      data: {
        walletAddress: walletAddress,
        status: 'approved',
        approvedDate: nowString,
        migrationStartDate: migrationDateString,
        countdownDays: 15
      }
    });

  } catch (error) {
    console.error('❌ KYC 승인 오류:', error);
    res.status(500).json({
      success: false,
      message: 'KYC 승인 중 오류가 발생했습니다.',
      error: error.message
    });
  }
});
```

```javascript
app.post('/api/admin/bitwish/kyc/reject', async (req, res) => {
  try {
    const { adminWallet, walletAddress, rejectionReason } = req.body;

    // ========== 1. 입력 검증 ==========
    if (!adminWallet || !adminWallet.startsWith('BW')) {
      return res.status(403).json({
        success: false,
        message: '관리자 권한이 필요합니다.'
      });
    }

    if (!walletAddress || !walletAddress.startsWith('BW')) {
      return res.status(400).json({
        success: false,
        message: '유효하지 않은 지갑 주소입니다.'
      });
    }

    if (!rejectionReason) {
      return res.status(400).json({
        success: false,
        message: '거부 사유를 입력해주세요.'
      });
    }

    const db = bitWishMongoDB.db;
    if (!db) {
      return res.status(500).json({
        success: false,
        message: 'MongoDB 연결이 필요합니다.'
      });
    }

    const miningCollection = db.collection('admin_user_mining_data');

    // ========== 2. 관리자 확인 (userType='admin' 필터) ==========
    const adminData = await miningCollection.findOne({ 
      walletAddress: adminWallet,
      userType: 'admin'
    });

    if (!adminData) {
      return res.status(403).json({
        success: false,
        message: '관리자 권한이 없습니다.'
      });
    }

    // ========== 3. 유저 데이터 조회 (userType='user' 필터) ==========
    const userData = await miningCollection.findOne({ 
      walletAddress,
      userType: 'user'
    });

    if (!userData) {
      return res.status(404).json({
        success: false,
        message: '유저 데이터를 찾을 수 없습니다.'
      });
    }

    if (!userData.kyc || userData.kyc.status !== 'pending') {
      return res.status(400).json({
        success: false,
        message: 'KYC 승인 대기 상태가 아닙니다.'
      });
    }

    const now = new Date();
    const nowString = now.toISOString().replace('T', ' ').substring(0, 19);

    // ========== 4. KYC 거부 업데이트 (userType='user' 필터) ==========
    await miningCollection.updateOne(
      { 
        walletAddress,
        userType: 'user'
      },
      {
        $set: {
          'kyc.status': 'rejected',
          'kyc.rejectedDate': nowString,
          'kyc.rejectionReason': rejectionReason,
          updatedAt: now
        }
      }
    );

    // ========== 5. 응답 ==========
    res.json({
      success: true,
      message: 'KYC가 거부되었습니다.',
      data: {
        walletAddress: walletAddress,
        status: 'rejected',
        rejectedDate: nowString,
        rejectionReason: rejectionReason
      }
    });

  } catch (error) {
    console.error('❌ KYC 거부 오류:', error);
    res.status(500).json({
      success: false,
      message: 'KYC 거부 중 오류가 발생했습니다.',
      error: error.message
    });
  }
});
```

```typescript
              <td style={{ padding: '12px' }}>{app.submittedAt}</td>
              <td style={{ padding: '12px' }}>
                {app.status === 'pending' && (
                  <div style={{ display: 'flex', gap: '5px' }}>
                    <button 
                      onClick={() => handleApproveKYC(app.walletAddress)}
                      style={{
                        padding: '6px 12px',
                        background: '#4CAF50',
                        color: 'white',
                        border: 'none',
                        borderRadius: '4px',
                        cursor: 'pointer'
                      }}
                    >
                      ✅ 승인
                    </button>
                    <button 
                      onClick={() => setSelectedKYCUser(app)}
                      style={{
                        padding: '6px 12px',
                        background: '#f44336',
                        color: 'white',
                        border: 'none',
                        borderRadius: '4px',
                        cursor: 'pointer'
                      }}
                    >
                      ❌ 거부
                    </button>
                  </div>
                )}
              </td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  </div>
)}

{/* KYC 거부 모달 */}
{selectedKYCUser && (
  <div style={{
    position: 'fixed',
    top: 0,
    left: 0,
    right: 0,
    bottom: 0,
    background: 'rgba(0,0,0,0.5)',
    display: 'flex',
    alignItems: 'center',
    justifyContent: 'center',
    zIndex: 1000
  }}>
    <div style={{
      background: 'white',
      padding: '30px',
      borderRadius: '10px',
      maxWidth: '500px',
      width: '90%'
    }}>
      <h3 style={{ marginTop: 0 }}>KYC 거부</h3>
      <p>지갑: {selectedKYCUser.walletAddress}</p>
      <textarea
        placeholder="거부 사유를 입력하세요..."
        value={rejectionReason}
        onChange={(e) => setRejectionReason(e.target.value)}
        style={{
          width: '100%',
          minHeight: '100px',
          padding: '10px',
          border: '1px solid #ddd',
          borderRadius: '5px',
          marginBottom: '15px'
        }}
      />
      <div style={{ display: 'flex', gap: '10px' }}>
        <button 
          onClick={() => handleRejectKYC(selectedKYCUser.walletAddress)}
          disabled={loading}
          style={{
            flex: 1,
            padding: '12px',
            background: '#f44336',
            color: 'white',
            border: 'none',
            borderRadius: '5px',
            cursor: loading ? 'not-allowed' : 'pointer'
          }}
        >
          확인
        </button>
        <button 
          onClick={() => {
            setSelectedKYCUser(null);
            setRejectionReason('');
          }}
          style={{
            flex: 1,
            padding: '12px',
            background: '#757575',
            color: 'white',
            border: 'none',
            borderRadius: '5px',
            cursor: 'pointer'
          }}
        >
          취소
        </button>
      </div>
    </div>
  </div>
)}
```
