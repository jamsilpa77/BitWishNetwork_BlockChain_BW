## **📌 Part 1: Backend API 7-1 - 실시간 카운트다운 업데이트 (simple-server.cjs에 추가)**

```javascript
// ========================================
// API 7-1: POST /api/admin/bitwish/countdown/update
// 설명: 실시간 카운트다운 업데이트 (Days, Hours, Minutes, Seconds)
// ========================================

app.post('/api/admin/bitwish/countdown/update', async (req, res) => {
  try {
    const { walletAddress } = req.body;

    // ========== 1. 입력 검증 ==========
    if (!walletAddress || !walletAddress.startsWith('BW')) {
      return res.status(400).json({
        success: false,
        message: '유효하지 않은 지갑 주소입니다.'
      });
    }

    const db = bitWishMongoDB.db;
    if (!db) {
      return res.status(500).json({
        success: false,
        message: 'MongoDB 연결이 필요합니다.'
      });
    }

    const miningCollection = db.collection('admin_user_mining_data');

    // ========== 2. 유저 데이터 조회 (userType='user' 필터) ==========
    const userData = await miningCollection.findOne({ 
      walletAddress,
      userType: 'user'
    });

    if (!userData) {
      return res.status(404).json({
        success: false,
        message: '유저 데이터를 찾을 수 없습니다.'
      });
    }

    // KYC 승인 확인
    if (!userData.kyc || userData.kyc.status !== 'approved') {
      return res.status(400).json({
        success: false,
        message: 'KYC가 승인되지 않았습니다.'
      });
    }

    // 카운트다운 활성화 확인
    if (!userData.kyc.countdown || !userData.kyc.countdown.isActive) {
      return res.status(400).json({
        success: false,
        message: '카운트다운이 활성화되지 않았습니다.'
      });
    }

    const now = new Date();
    const migrationStartDate = new Date(userData.kyc.migrationStartDate + 'T00:00:00');

    // ========== 3. 남은 시간 계산 ==========
    const timeDiff = migrationStartDate - now;

    // 이미 만료된 경우
    if (timeDiff <= 0) {
      await miningCollection.updateOne(
        { 
          walletAddress,
          userType: 'user'
        },
        {
          $set: {
            'kyc.countdown.days': 0,
            'kyc.countdown.hours': 0,
            'kyc.countdown.minutes': 0,
            'kyc.countdown.seconds': 0,
            'kyc.countdown.totalSeconds': 0,
            'kyc.countdown.isActive': false,
            'kyc.countdown.isExpired': true,
            'kyc.countdown.expiredAt': now.toISOString().replace('T', ' ').substring(0, 19),
            updatedAt: now
          }
        }
      );

      return res.json({
        success: true,
        message: '카운트다운이 만료되었습니다.',
        data: {
          days: 0,
          hours: 0,
          minutes: 0,
          seconds: 0,
          totalSeconds: 0,
          isActive: false,
          isExpired: true,
          migrationStartDate: userData.kyc.migrationStartDate
        }
      });
    }

    // ========== 4. 시간 계산 (일, 시간, 분, 초) ==========
    const totalSeconds = Math.floor(timeDiff / 1000);
    const days = Math.floor(totalSeconds / (24 * 60 * 60));
    const hours = Math.floor((totalSeconds % (24 * 60 * 60)) / (60 * 60));
    const minutes = Math.floor((totalSeconds % (60 * 60)) / 60);
    const seconds = totalSeconds % 60;

    // ========== 5. MongoDB 업데이트 (userType='user' 필터) ==========
    await miningCollection.updateOne(
      { 
        walletAddress,
        userType: 'user'
      },
      {
        $set: {
          'kyc.countdown.days': days,
          'kyc.countdown.hours': hours,
          'kyc.countdown.minutes': minutes,
          'kyc.countdown.seconds': seconds,
          'kyc.countdown.totalSeconds': totalSeconds,
          'kyc.countdown.lastUpdate': now.toISOString().replace('T', ' ').substring(0, 19),
          updatedAt: now
        }
      }
    );

    // ========== 6. 응답 ==========
    res.json({
      success: true,
      data: {
        days: days,
        hours: hours,
        minutes: minutes,
        seconds: seconds,
        totalSeconds: totalSeconds,
        isActive: true,
        isExpired: false,
        migrationStartDate: userData.kyc.migrationStartDate,
        lastUpdate: now.toISOString().replace('T', ' ').substring(0, 19)
      }
    });

  } catch (error) {
    console.error('❌ 카운트다운 업데이트 오류:', error);
    res.status(500).json({
      success: false,
      message: '카운트다운 업데이트 중 오류가 발생했습니다.',
      error: error.message
    });
  }
});
```

## **📌 Part 3: Backend - 자동 카운트다운 만료 체크 시스템 (simple-server.cjs에 추가)**

```javascript
// ========================================
// 자동 카운트다운 만료 체크 시스템 (1시간마다)
// 설명: 만료된 카운트다운을 자동으로 감지하고 알림
// ========================================

setInterval(async () => {
  try {
    console.log('🔔 카운트다운 만료 체크 시작...');

    const db = bitWishMongoDB.db;
    if (!db) {
      console.error('❌ MongoDB 연결 없음');
      return;
    }

    const miningCollection = db.collection('admin_user_mining_data');
    const now = new Date();

    // ========== 1. 활성 카운트다운 조회 (userType='user'만) ==========
    const activeCountdowns = await miningCollection.find({ 
      userType: 'user',
      'kyc.countdown.isActive': true
    }).toArray();

    let expiredCount = 0;

    for (const user of activeCountdowns) {
      const migrationStartDate = new Date(user.kyc.migrationStartDate + 'T00:00:00');
      const timeDiff = migrationStartDate - now;

      // ========== 2. 만료된 카운트다운 처리 ==========
      if (timeDiff <= 0) {
        await miningCollection.updateOne(
          { 
            walletAddress: user.walletAddress,
            userType: 'user'
          },
          {
            $set: {
              'kyc.countdown.days': 0,
              'kyc.countdown.hours': 0,
              'kyc.countdown.minutes': 0,
              'kyc.countdown.seconds': 0,
              'kyc.countdown.totalSeconds': 0,
              'kyc.countdown.isActive': false,
              'kyc.countdown.isExpired': true,
              'kyc.countdown.expiredAt': now.toISOString().replace('T', ' ').substring(0, 19),
              updatedAt: now
            }
          }
        );

        expiredCount++;
        console.log(`✅ 카운트다운 만료: ${user.walletAddress}`);
      }
    }

    if (expiredCount > 0) {
      console.log(`✅ 총 ${expiredCount}개의 카운트다운이 만료되었습니다.`);
    } else {
      console.log('✅ 만료된 카운트다운 없음');
    }

  } catch (error) {
    console.error('❌ 카운트다운 만료 체크 오류:', error);
  }
}, 60 * 60 * 1000); // 1시간마다 실행

console.log('✅ 카운트다운 자동 만료 체크 시스템 시작 (1시간마다)');
```

## **📌 Part 4: Backend API 7-3 - 만료 알림 조회 (simple-server.cjs에 추가)**

```javascript
// ========================================
// API 7-3: GET /api/admin/bitwish/countdown/notifications
// 설명: 카운트다운 만료 알림 조회 (유저용)
// ========================================

app.get('/api/admin/bitwish/countdown/notifications', async (req, res) => {
  try {
    const { walletAddress } = req.query;

    if (!walletAddress || !walletAddress.startsWith('BW')) {
      return res.status(400).json({
        success: false,
        message: '유효하지 않은 지갑 주소입니다.'
      });
    }

    const db = bitWishMongoDB.db;
    if (!db) {
      return res.status(500).json({
        success: false,
        message: 'MongoDB 연결이 필요합니다.'
      });
    }

    const miningCollection = db.collection('admin_user_mining_data');

    // ========== 데이터 조회 (userType='user' 필터) ==========
    const userData = await miningCollection.findOne({ 
      walletAddress,
      userType: 'user'
    });

    if (!userData) {
      return res.status(404).json({
        success: false,
        message: '유저 데이터를 찾을 수 없습니다.'
      });
    }

    const notifications = [];

    // ========== 만료 알림 확인 ==========
    if (userData.kyc?.countdown?.isExpired) {
      notifications.push({
        type: 'countdown_expired',
        title: '마이그레이션 가능',
        message: '15일 카운트다운이 완료되었습니다. 이제 마이그레이션이 가능합니다.',
        expiredAt: userData.kyc.countdown.expiredAt,
        migrationStartDate: userData.kyc.migrationStartDate,
        isRead: false
      });
    }

    // ========== 3일 남음 알림 ==========
    if (userData.kyc?.countdown?.isActive && userData.kyc.countdown.days <= 3 && userData.kyc.countdown.days > 0) {
      notifications.push({
        type: 'countdown_warning',
        title: '마이그레이션 임박',
        message: `마이그레이션까지 ${userData.kyc.countdown.days}일 ${userData.kyc.countdown.hours}시간 남았습니다.`,
        daysRemaining: userData.kyc.countdown.days,
        isRead: false
      });
    }

    res.json({
      success: true,
      notifications: notifications,
      count: notifications.length
    });

  } catch (error) {
    console.error('❌ 알림 조회 오류:', error);
    res.status(500).json({
      success: false,
      message: '알림 조회 중 오류가 발생했습니다.',
      error: error.message
    });
  }
});
```

```typescript
        </div>
      )}

      {!countdown.isActive && !countdown.isExpired && (
        <div style={{
          background: '#f5f5f5',
          padding: '20px',
          borderRadius: '10px',
          textAlign: 'center',
          color: '#666'
        }}>
          <div style={{ fontSize: '18px', marginBottom: '10px' }}>
            ⏸️ {t.countdownNotActive}
          </div>
          <div style={{ fontSize: '14px' }}>
            {t.kycApprovalRequired}
          </div>
        </div>
      )}
    </div>
  );
};

// ========== UI 렌더링 함수: 알림 섹션 ==========
const renderNotificationsSection = () => {
  if (notifications.length === 0) return null;

  return (
    <div style={{ 
      border: '2px solid #f44336', 
      padding: '15px', 
      borderRadius: '8px',
      background: 'white',
      marginBottom: '20px'
    }}>
      <h3 style={{ color: '#f44336', marginTop: 0 }}>
        🔔 {t.notifications} ({notifications.length})
      </h3>

      {notifications.map((notification, index) => (
        <div 
          key={index}
          style={{
            background: notification.type === 'countdown_expired' ? '#E8F5E9' : '#FFF3E0',
            padding: '15px',
            borderRadius: '8px',
            marginBottom: index < notifications.length - 1 ? '10px' : 0,
            border: `2px solid ${notification.type === 'countdown_expired' ? '#4CAF50' : '#FF9800'}`
          }}
        >
          <div style={{ 
            fontWeight: 'bold', 
            marginBottom: '5px',
            color: notification.type === 'countdown_expired' ? '#2E7D32' : '#E65100'
          }}>
            {notification.type === 'countdown_expired' ? '✅' : '⚠️'} {notification.title}
          </div>
          <div style={{ fontSize: '14px', color: '#666' }}>
            {notification.message}
          </div>
          {notification.expiredAt && (
            <div style={{ fontSize: '12px', color: '#999', marginTop: '5px' }}>
              {notification.expiredAt}
            </div>
          )}
        </div>
      ))}
    </div>
  );
};

// ========== 최종 UI에 추가 (KYC 섹션 밑에) ==========
{/* 알림 */}
{miningData && renderNotificationsSection()}

{/* 15일 카운트다운 (알림 밑에 추가) */}
{miningData && renderCountdownSection()}
```

```typescript
// ========== 상태에 추가 ==========
const [countdownStats, setCountdownStats] = useState<any>(null);

// ========== 카운트다운 통계 계산 함수 ==========
const handleCalculateCountdownStats = () => {
  if (allUsers.length === 0) return;

  let totalActive = 0;
  let totalExpired = 0;
  let expiringSoon = 0; // 3일 이내

  for (const user of allUsers) {
    if (user.kyc?.countdown?.isActive) {
      totalActive++;
      if (user.kyc.countdown.days <= 3) {
        expiringSoon++;
      }
    }
    if (user.kyc?.countdown?.isExpired) {
      totalExpired++;
    }
  }

  setCountdownStats({
    totalActive,
    totalExpired,
    expiringSoon,
    totalUsers: allUsers.length
  });
};

useEffect(() => {
```
