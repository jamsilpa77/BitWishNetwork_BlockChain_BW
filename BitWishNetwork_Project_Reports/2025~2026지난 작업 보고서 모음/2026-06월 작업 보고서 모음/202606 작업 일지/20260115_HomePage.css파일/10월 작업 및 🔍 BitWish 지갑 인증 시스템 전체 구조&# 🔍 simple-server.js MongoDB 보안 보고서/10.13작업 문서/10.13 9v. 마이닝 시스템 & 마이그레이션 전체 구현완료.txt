      setError('지갑 주소는 BW로 시작해야 합니다');
      return;
    }

        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ 
          walletAddress,
          userType: 'admin' // 관리자 전용
        })
      });

      const data = await response.json();

      if (data.success) {
        await handleUpdateMining();
      } else {
        setError(data.message || 'Mining start failed');
      }
    } catch (err: any) {
      setError(err.message || 'Network error');
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {

  }, [walletAddress]);

  // ========== 50자리 숫자 표시 함수 ==========
  const formatNumber = (num: string) => {
    const parsed = parseFloat(num);
    if (isNaN(parsed)) return '0.00000000';
    return parsed.toFixed(8);
  };

  // ========== 출석 달력 렌더링 ==========
  const renderAttendanceCalendar = () => {
    if (!miningData) return null;

    const now = new Date();
    const currentYear = now.getFullYear();
    const currentMonth = now.getMonth() + 1;
    const currentMonthKey = `${currentYear}-${String(currentMonth).padStart(2, '0')}`;
    
    const monthlyCheckIns = miningData.breakdown?.attendance?.monthlyCheckIns || {};
    const currentMonthData = monthlyCheckIns[currentMonthKey] || { days: [], totalDays: 0 };
    
    const daysInMonth = new Date(currentYear, currentMonth, 0).getDate();
    const days = [];

    for (let i = 1; i <= daysInMonth; i++) {
      const isChecked = currentMonthData.days.includes(i);
      const isToday = i === now.getDate();
      
      days.push(
        <div
          key={i}
          style={{
            width: '30px',
            height: '30px',
            display: 'flex',
            alignItems: 'center',
            justifyContent: 'center',
            margin: '2px',
            borderRadius: '5px',
            background: isChecked ? '#4CAF50' : (isToday ? '#FFC107' : '#f5f5f5'),
            color: isChecked ? 'white' : (isToday ? 'white' : '#333'),
            fontWeight: isToday ? 'bold' : 'normal',
            fontSize: '12px'
          }}
        >
          {i}
        </div>
      );
    }

    return (
      <div style={{ 
        border: '2px solid #4CAF50', 
        padding: '15px', 
        borderRadius: '8px',
        background: 'white',
        marginBottom: '20px'
      }}>
        <h3 style={{ color: '#4CAF50', marginTop: 0 }}>
          📅 {t.attendanceCalendar} ({currentMonthKey})
        </h3>
        <p>
          {t.checkInDays}: <strong>{currentMonthData.totalDays}</strong>일
        </p>
        <div style={{ display: 'flex', flexWrap: 'wrap', marginTop: '10px' }}>
          {days}
        </div>
        <div style={{ marginTop: '10px', fontSize: '12px', color: '#666' }}>
          <span style={{ background: '#4CAF50', color: 'white', padding: '2px 8px', borderRadius: '3px', marginRight: '10px' }}>
            ■ 출석
          </span>
          <span style={{ background: '#FFC107', color: 'white', padding: '2px 8px', borderRadius: '3px', marginRight: '10px' }}>
            ■ 오늘
          </span>
          <span style={{ background: '#f5f5f5', color: '#333', padding: '2px 8px', borderRadius: '3px' }}>
            ■ 미출석
          </span>
        </div>
      </div>
    );
  };

  // ========== 추천 보너스 섹션 렌더링 ==========
  const renderReferralSection = () => {
    if (!referralInfo) return null;

    return (
      <div style={{ 
        border: '2px solid #9C27B0', 
        padding: '15px', 
        borderRadius: '8px',
        background: 'white',
        marginBottom: '20px'
      }}>
        <h3 style={{ color: '#9C27B0', marginTop: 0 }}>
          👥 {t.referralSystem}
        </h3>

        <div style={{ 
          background: '#F3E5F5', 
          padding: '15px', 
          borderRadius: '5px',
          marginBottom: '15px'
        }}>
          <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center' }}>
            <div>
              <strong>{t.myReferralCode}:</strong><br/>
              <span style={{ 
                fontSize: '18px', 
                fontFamily: 'monospace',
                color: '#9C27B0',
                fontWeight: 'bold'
              }}>
                {referralInfo.myReferralCode || 'BWREF-XXXXXXXX'}
              </span>
            </div>
            <button 
              onClick={handleCopyReferralCode}
              style={{
                padding: '10px 20px',
                background: '#9C27B0',
                color: 'white',
                border: 'none',
                borderRadius: '5px',
                cursor: 'pointer'
              }}
            >
              📋 {t.copyCode}
            </button>
          </div>
          <p style={{ fontSize: '12px', color: '#666', marginBottom: 0, marginTop: '10px' }}>
            사용 횟수: {referralInfo.myReferralCodeUsedCount}회
          </p>
        </div>

        <div style={{ 
          display: 'grid', 
          gridTemplateColumns: '1fr 1fr', 
          gap: '10px',
          marginBottom: '15px'
        }}>
          <div style={{ background: '#F5F5F5', padding: '10px', borderRadius: '5px' }}>
            <strong>{t.activeReferrals}:</strong><br/>
            <span style={{ fontSize: '20px', color: '#9C27B0' }}>
              {referralInfo.activeReferrals}명
            </span>
          </div>
          <div style={{ background: '#F5F5F5', padding: '10px', borderRadius: '5px' }}>
            <strong>{t.permanentBonus}:</strong><br/>
            <span style={{ fontSize: '20px', color: '#4CAF50' }}>
              {(referralInfo.activeReferrals * 2).toFixed(0)}%
            </span>
          </div>
        </div>

        <div style={{
          background: '#FFF3E0',
          padding: '10px',
          borderRadius: '5px'
        }}>
          <strong>🔒 {t.bonusVault}:</strong><br/>
          <span style={{ fontSize: '18px', color: '#FF9800' }}>
            {formatNumber(referralInfo.referralBonusVault.totalVault)} BW
          </span>
        </div>
      </div>
    );
  };

  // ========== KYC 섹션 렌더링 ==========
  const renderKYCSection = () => {
    if (!kycStatus) return null;

    const statusColor = {
      'none': '#757575',
      'pending': '#FF9800',
      'approved': '#4CAF50',
      'rejected': '#f44336'
    }[kycStatus.status] || '#757575';

    const statusText = {
      'none': t.kycNone,
      'pending': t.kycPending,
      'approved': t.kycApproved,
      'rejected': t.kycRejected
    }[kycStatus.status] || t.kycNone;

    return (
      <div style={{ 
        border: '2px solid #2196F3', 
        padding: '15px', 
        borderRadius: '8px',
        background: 'white',
        marginBottom: '20px'
      }}>
        <h3 style={{ color: '#2196F3', marginTop: 0 }}>
          🔐 {t.kycSystem}
        </h3>

        <div style={{
          background: '#E3F2FD',
          padding: '10px',
          borderRadius: '5px'
        }}>
          <div style={{ marginBottom: '5px' }}>
            <strong>{t.kycStatus}:</strong>{' '}
            <span style={{ 
              color: statusColor,
              fontWeight: 'bold',
              fontSize: '16px'
            }}>
              {statusText}
            </span>
          </div>
          
          {kycStatus.status === 'approved' && (
            <>
              <div style={{ fontSize: '14px', color: '#666' }}>
                {t.migrationStartDate}: <strong>{kycStatus.migrationStartDate}</strong>
              </div>
              <div style={{ fontSize: '14px', color: '#666' }}>
                {t.countdownDays}: <strong>{kycStatus.countdown?.days || 0}일</strong>
              </div>
            </>
          )}
        </div>
      </div>
    );
  };

  // ========== 카운트다운 섹션 렌더링 ==========
  const renderCountdownSection = () => {
    if (!countdown) return null;

    return (
      <div style={{ 
        border: '2px solid #FF9800', 
        padding: '15px', 
        borderRadius: '8px',
        background: 'white',
        marginBottom: '20px'
      }}>
        <h3 style={{ color: '#FF9800', marginTop: 0 }}>
          ⏰ {t.migrationCountdown}
        </h3>

        {countdown.isActive && !countdown.isExpired && (
          <div style={{
            background: 'linear-gradient(135deg, #667eea 0%, #764ba2 100%)',
            padding: '20px',
            borderRadius: '10px',
            color: 'white',
            textAlign: 'center'
          }}>
            <div style={{ 
              display: 'grid', 
              gridTemplateColumns: '1fr 1fr 1fr 1fr', 
              gap: '10px'
            }}>
              <div style={{
                background: 'rgba(255,255,255,0.2)',
                padding: '15px 10px',
                borderRadius: '8px'
              }}>
                <div style={{ fontSize: '32px', fontWeight: 'bold' }}>
                  {String(countdown.days).padStart(2, '0')}
                </div>
                <div style={{ fontSize: '12px', marginTop: '5px' }}>
                  {t.daysRemaining}
                </div>
              </div>
              
              <div style={{
                background: 'rgba(255,255,255,0.2)',
                padding: '15px 10px',
                borderRadius: '8px'
              }}>
                <div style={{ fontSize: '32px', fontWeight: 'bold' }}>
                  {String(countdown.hours).padStart(2, '0')}
                </div>
                <div style={{ fontSize: '12px', marginTop: '5px' }}>
                  {t.hoursRemaining}
                </div>
              </div>
              
              <div style={{
                background: 'rgba(255,255,255,0.2)',
                padding: '15px 10px',
                borderRadius: '8px'
              }}>
                <div style={{ fontSize: '32px', fontWeight: 'bold' }}>
                  {String(countdown.minutes).padStart(2, '0')}
                </div>
                <div style={{ fontSize: '12px', marginTop: '5px' }}>
                  {t.minutesRemaining}
                </div>
              </div>
              
              <div style={{
                background: 'rgba(255,255,255,0.2)',
                padding: '15px 10px',
                borderRadius: '8px'
              }}>
                <div style={{ fontSize: '32px', fontWeight: 'bold' }}>
                  {String(countdown.seconds).padStart(2, '0')}
                </div>
                <div style={{ fontSize: '12px', marginTop: '5px' }}>
                  {t.secondsRemaining}
                </div>
              </div>
            </div>
          </div>
        )}

        {countdown.isExpired && (
          <div style={{
            background: '#4CAF50',
            padding: '20px',
            borderRadius: '10px',
            color: 'white',
            textAlign: 'center'
          }}>
            <div style={{ fontSize: '24px', fontWeight: 'bold' }}>
              ✅ {t.countdownExpired}
            </div>
            <div style={{ fontSize: '16px', marginTop: '10px' }}>
              🎉 {t.migrationReady}
            </div>
          </div>
        )}
      </div>
    );
  };

  // ========== 마이그레이션 섹션 렌더링 ==========
  const renderMigrationSection = () => {
    if (!miningData) return null;

    return (
      <div style={{ 
        border: '2px solid #4CAF50', 
        padding: '15px', 
        borderRadius: '8px',
        background: 'white',
        marginBottom: '20px'
      }}>
        <h3 style={{ color: '#4CAF50', marginTop: 0 }}>
          🚀 {t.migrationSystem}
        </h3>

        <div style={{
          background: 'linear-gradient(135deg, #667eea 0%, #764ba2 100%)',
          padding: '20px',
          borderRadius: '10px',
          marginBottom: '15px',
          color: 'white',
          textAlign: 'center'
        }}>
          <div style={{ fontSize: '14px', opacity: 0.9, marginBottom: '10px' }}>
            💰 {t.availableAmount}
          </div>
          <div style={{ fontSize: '32px', fontWeight: 'bold' }}>
            {formatNumber(miningData.walletBalance?.availableAmount?.amount || '0')} BW
          </div>
        </div>

        {migrationHistory.length > 0 && (
          <>
            <h4 style={{ borderBottom: '1px solid #ddd', paddingBottom: '5px' }}>
              📜 {t.migrationHistory} ({migrationHistory.length})
            </h4>
            <div style={{ maxHeight: '200px', overflowY: 'auto' }}>
              {migrationHistory.map((record, index) => (
                <div 
                  key={index}
                  style={{
                    background: '#E8F5E9',
                    padding: '10px',
                    borderRadius: '5px',
                    marginBottom: '8px',
                    fontSize: '12px'
                  }}
                >
                  <div style={{ fontWeight: 'bold', color: '#2E7D32' }}>
                    {record.month} - {formatNumber(record.amount)} BW
                  </div>
                  <div style={{ color: '#666', marginTop: '3px' }}>
                    {record.migratedAt}
                  </div>
                </div>
              ))}
            </div>
          </>
        )}
      </div>
    );
  };

  // ========== 월별 보관함 섹션 렌더링 ==========
  const renderMonthlyVaultSection = () => {
    if (!miningData) return null;

    const monthlyMining = miningData.monthlyMining || {};
    const months = Object.keys(monthlyMining).sort().reverse(); // 최신 월 먼저

    if (months.length === 0) return null;

    return (
      <div style={{ 
        border: '2px solid #FF6F00', 
        padding: '15px', 
        borderRadius: '8px',
        background: 'white',
        marginBottom: '20px'
      }}>
        <h3 style={{ color: '#FF6F00', marginTop: 0 }}>
          📦 {t.monthlyVault} ({months.length}개월)
        </h3>

        <div style={{ maxHeight: '300px', overflowY: 'auto' }}>
          {months.map((month) => {
            const data = monthlyMining[month];
            return (
              <div 
                key={month}
                style={{
                  background: '#FFF3E0',
                  padding: '12px',
                  borderRadius: '8px',
                  marginBottom: '10px',
                  border: '1px solid #FF9800'
                }}
              >
                <div style={{ fontWeight: 'bold', marginBottom: '5px', color: '#E65100' }}>
                  📅 {month}
                </div>
                <div style={{ fontSize: '18px', fontWeight: 'bold', color: '#FF6F00', marginBottom: '8px' }}>
                  {formatNumber(data.amount)} BW
                </div>
                <div style={{ fontSize: '11px', color: '#666' }}>
                  • 기본: {formatNumber(data.breakdown.basic)} BW<br/>
                  • 출석: {formatNumber(data.breakdown.attendance)} BW<br/>
                  • 추천: {formatNumber(data.breakdown.referral)} BW
                </div>
                <div style={{ fontSize: '10px', color: '#999', marginTop: '5px' }}>
                  기간: {data.startDate} ~ {data.endDate}
                </div>
              </div>
            );
          })}
        </div>
      </div>
    );
  };

  // ========== UI 렌더링 ==========
  return (
    <div style={{ padding: '20px', maxWidth: '1000px', margin: '0 auto' }}>
      {/* 언어 선택 */}
      <div style={{ marginBottom: '20px', textAlign: 'right' }}>
        <button onClick={() => setLanguage('ko')} style={{ margin: '0 5px', padding: '5px 10px' }}>한국어</button>
        <button onClick={() => setLanguage('en')} style={{ margin: '0 5px', padding: '5px 10px' }}>English</button>
        <button onClick={() => setLanguage('zh')} style={{ margin: '0 5px', padding: '5px 10px' }}>中文</button>
        <button onClick={() => setLanguage('ja')} style={{ margin: '0 5px', padding: '5px 10px' }}>日本語</button>
      </div>

      {/* 제목 */}
      <h1 style={{ borderBottom: '3px solid #E65100', paddingBottom: '10px' }}>
        🛡️ {t.title}
        <span style={{ fontSize: '14px', color: '#666', marginLeft: '15px' }}>
          ⭐ {t.precision}
        </span>
      </h1>

      {/* 전체 통계 */}
      {stats && (
        <div style={{ 
          background: '#FBE9E7', 
          padding: '15px', 
          borderRadius: '8px', 
          marginBottom: '20px',
          border: '2px solid #FF5722'
        }}>
          <h3 style={{ marginTop: 0 }}>📊 전체 통계</h3>
          <div style={{ display: 'grid', gridTemplateColumns: '1fr 1fr 1fr', gap: '10px' }}>
            <div>
              <strong>{t.totalIssued}:</strong><br/>
              {formatNumber(stats.totalIssued)} BW
            </div>
            <div>
              <strong>{t.remainingSupply}:</strong><br/>
              {formatNumber(stats.remainingSupply)} BW
            </div>
            <div>
              <strong>{t.issuanceRate}:</strong><br/>
              {parseFloat(stats.issuanceRate).toFixed(4)}%
            </div>
          </div>
          <p style={{ fontSize: '12px', color: '#666', marginBottom: 0 }}>
            🔄 {t.autoUpdate}
          </p>
        </div>
      )}

      {/* 지갑 주소 입력 */}
      <div style={{ marginBottom: '20px' }}>
        <input
          type="text"
          placeholder={t.walletPlaceholder}
          value={walletAddress}
          onChange={(e) => setWalletAddress(e.target.value)}
          style={{ 
            width: '100%', 
            padding: '12px', 
            fontSize: '16px',
            border: '2px solid #E65100',
            borderRadius: '5px'
          }}
        />
      </div>

      {/* 버튼 */}
      <div style={{ marginBottom: '20px', display: 'flex', gap: '10px' }}>
        <button 
          onClick={handleStartMining} 
          disabled={loading}
          style={{ 
            flex: 1,
            padding: '12px 20px',
            fontSize: '16px',
            background: '#FF5722',
            color: 'white',
            border: 'none',
            borderRadius: '5px',
            cursor: loading ? 'not-allowed' : 'pointer'
          }}
        >
          {loading ? t.loading : t.startMining}
        </button>
        <button 
          onClick={handleUpdateMining} 
          disabled={loading}
          style={{ 
            flex: 1,
            padding: '12px 20px',
            fontSize: '16px',
            background: '#FF9800',
            color: 'white',
            border: 'none',
            borderRadius: '5px',
            cursor: loading ? 'not-allowed' : 'pointer'
          }}
        >
          {loading ? t.loading : t.updateMining}
        </button>
        <button 
          onClick={handleCheckIn} 
          disabled={loading}
          style={{ 
            flex: 1,
            padding: '12px 20px',
            fontSize: '16px',
            background: '#4CAF50',
            color: 'white',
            border: 'none',
            borderRadius: '5px',
            cursor: loading ? 'not-allowed' : 'pointer'
          }}
        >
          {loading ? t.loading : `📅 ${t.checkIn}`}
        </button>
      </div>

      {/* 오류 메시지 */}
      {error && (
        <div style={{ 
          color: 'white', 
          background: '#f44336',
          padding: '15px',
          borderRadius: '5px',
          marginBottom: '20px' 
        }}>
          ❌ {t.error}: {error}
        </div>
      )}

      {/* 채굴 데이터 표시 */}
      {miningData && (
        <div style={{ 
          border: '2px solid #FF5722', 
          padding: '20px', 
          borderRadius: '8px',
          background: 'white',
          marginBottom: '20px'
        }}>
          <h2 style={{ color: '#FF5722', marginTop: 0 }}>
            💰 {t.totalMined}: {formatNumber(miningData.totalMined)} BW
          </h2>
          <p style={{ fontSize: '18px', color: '#333' }}>
            📅 {t.currentMonth}: {formatNumber(miningData.currentMonthMining)} BW
          </p>
          
          <h3 style={{ borderBottom: '1px solid #ddd', paddingBottom: '5px' }}>상세 내역:</h3>
          <ul style={{ lineHeight: '1.8' }}>
            <li>⛏️ {t.basicMining}: <strong>{formatNumber(miningData.breakdown.basic)}</strong> BW</li>
            <li>📅 {t.attendanceBonus}: <strong>{formatNumber(miningData.breakdown.attendance)}</strong> BW</li>
            <li>👥 {t.referralBonus}: <strong>
              {formatNumber(
                (parseFloat(miningData.breakdown.referralPermanent) + 
                 parseFloat(miningData.breakdown.referralOneTime)).toFixed(50)
              )}
            </strong> BW</li>
          </ul>

          <div style={{ 
            background: '#f5f5f5', 
            padding: '10px', 
            borderRadius: '5px',
            marginTop: '15px'
          }}>
            <p style={{ fontSize: '12px', color: '#666', margin: 0 }}>
              🕐 {t.lastUpdate}: {miningData.lastUpdate}
            </p>
            <p style={{ fontSize: '11px', color: '#999', margin: '5px 0 0 0' }}>
              ⭐ BitWish 50자리 부동소수점 정밀 계산 적용 (관리자 전용)
            </p>
          </div>
        </div>
      )}

      {/* 출석 달력 */}
      {miningData && renderAttendanceCalendar()}

      {/* 추천 보너스 시스템 */}
      {miningData && renderReferralSection()}

      {/* KYC 시스템 */}
      {miningData && renderKYCSection()}

      {/* 15일 카운트다운 */}
      {miningData && renderCountdownSection()}

      {/* 월별 채굴 보관함 */}
      {miningData && renderMonthlyVaultSection()}

      {/* 마이그레이션 시스템 */}
      {miningData && renderMigrationSection()}
    </div>
  );
};

export default AdminPersonalMiningPage;
