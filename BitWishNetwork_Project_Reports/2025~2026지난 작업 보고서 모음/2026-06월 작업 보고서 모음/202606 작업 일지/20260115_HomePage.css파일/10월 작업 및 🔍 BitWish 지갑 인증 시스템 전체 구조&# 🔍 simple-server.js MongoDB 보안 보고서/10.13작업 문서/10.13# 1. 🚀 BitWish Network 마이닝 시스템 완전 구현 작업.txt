## 📌 API 2-1: 마이닝 시작

```javascript
// ========================================
// API: POST /api/admin/bitwish/mining/start
// 설명: 마이닝 시작 + 블록 생성 + 수수료 분배
// ========================================

app.post('/api/admin/bitwish/mining/start', async (req, res) => {
  try {
    const { walletAddress } = req.body;

    // ========== 1. 입력 검증 ==========
    if (!walletAddress || !walletAddress.startsWith('BW')) {
      return res.status(400).json({
        success: false,
        message: '유효하지 않은 지갑 주소입니다.'
      });
    }

    console.log(`🚀 마이닝 시작 요청: ${walletAddress}`);

    const db = await getMongoDBConnection();
    const miningCollection = db.collection('admin_user_mining_data');
    const blockCollection = db.collection('admin_block_records');
    const feeCollection = db.collection('admin_fee_collection');
    const systemWalletsCollection = db.collection('admin_system_wallets');
    const referralCodesCollection = db.collection('admin_referral_codes');

    // ========== 2. 기존 마이닝 데이터 확인 ==========
    const existingMining = await miningCollection.findOne({ walletAddress });

    if (existingMining && existingMining.miningStatus.isActive) {
      return res.status(400).json({
        success: false,
        message: '이미 마이닝이 진행 중입니다.'
      });
    }

    const now = new Date();
    const nowString = now.toISOString().replace('T', ' ').substring(0, 19);

    // ========== 3. 블록 생성 (실제 블록체인 기록) ==========
    const totalBlocks = await blockCollection.countDocuments();
    const newBlockNumber = totalBlocks + 1;
    const previousBlock = await blockCollection.findOne({}, { sort: { blockNumber: -1 } });
    const previousHash = previousBlock ? previousBlock.blockHash : "0x" + "0".repeat(64);
```

## 📌 API 2-2: 실시간 채굴량 업데이트

```javascript
// ========================================
// API: POST /api/admin/bitwish/mining/update
// 설명: 실시간 채굴량 계산 및 업데이트
// ========================================

app.post('/api/admin/bitwish/mining/update', async (req, res) => {
  try {
    const { walletAddress } = req.body;

    if (!walletAddress) {
      return res.status(400).json({
        success: false,
        message: '지갑 주소가 필요합니다.'
      });
    }

    const db = await getMongoDBConnection();
    const miningCollection = db.collection('admin_user_mining_data');

    const miningData = await miningCollection.findOne({ walletAddress });

    if (!miningData || !miningData.miningStatus.isActive) {
      return res.status(400).json({
        success: false,
        message: '활성화된 마이닝이 없습니다.'
      });
    }

    // ========== 1. 경과 시간 계산 ==========
    const startTime = new Date(miningData.miningStartTime);
    const now = new Date();
    const elapsedMs = now - startTime;
    const elapsedHours = elapsedMs / (1000 * 60 * 60);

    // ========== 2. 기본 채굴량 계산 (0.25 BW/시간) ==========
    const basicMined = elapsedHours * 0.25;

    // ========== 3. 출석 보너스 계산 (5%) ==========
    let attendanceBonus = 0;
    if (miningData.miningBreakdown.attendance.isCheckedToday) {
      attendanceBonus = basicMined * 0.05;
    }

    // ========== 4. 추천 영구 보너스 계산 (2% × 활성 추천인 수) ==========
    const activeReferrals = miningData.miningBreakdown.referralPermanent.activeReferrals || 0;
    const referralPermanentBonus = basicMined * 0.02 * activeReferrals;

    // ========== 5. 추천 일회성 보너스 (변동 없음) ==========
    const referralOneTimeBonus = parseFloat(miningData.miningBreakdown.referralOneTime.total || 0);

    // ========== 6. 총 채굴량 ==========
    const totalMined = basicMined + attendanceBonus + referralPermanentBonus + referralOneTimeBonus;

    // ========== 7. 월별 보관 로직 ==========
    const currentMonth = now.toISOString().substring(0, 7); // "2025-01"
    const monthStart = `${currentMonth}-01`;
    const lastDayOfMonth = new Date(now.getFullYear(), now.getMonth() + 1, 0).getDate();
    const monthEnd = `${currentMonth}-${lastDayOfMonth}`;

    // ========== 8. 데이터베이스 업데이트 ==========
    const updateData = {
      $set: {
        'totalMined.amount': totalMined.toFixed(8),
        'miningBreakdown.basic.total': basicMined.toFixed(8),
        'miningBreakdown.attendance.total': attendanceBonus.toFixed(8),
        'miningBreakdown.referralPermanent.total': referralPermanentBonus.toFixed(8),
        'miningStatus.lastUpdateTime': now.toISOString().replace('T', ' ').substring(0, 19),
        'walletBalance.myBW.amount': totalMined.toFixed(8),
        updatedAt: now
      }
    };

    // 월별 보관 업데이트
    const currentMonthMining = basicMined + attendanceBonus + referralPermanentBonus + referralOneTimeBonus;
    
    updateData.$set[`monthlyMining.${currentMonth}`] = {
      amount: currentMonthMining.toFixed(8),
      saved: true,
      startDate: monthStart,
      endDate: monthEnd,
      breakdown: {
        basic: basicMined.toFixed(8),
        attendance: attendanceBonus.toFixed(8),
        referral: (referralPermanentBonus + referralOneTimeBonus).toFixed(8)
      }
    };

    await miningCollection.updateOne({ walletAddress }, updateData);

    // ========== 9. 응답 ==========
    res.json({
      success: true,
      message: '채굴량이 업데이트되었습니다.',
      data: {
        walletAddress: walletAddress,
        totalMined: totalMined.toFixed(8),
        breakdown: {
          basic: basicMined.toFixed(8),
          attendance: attendanceBonus.toFixed(8),
          referralPermanent: referralPermanentBonus.toFixed(8),
          referralOneTime: referralOneTimeBonus.toFixed(8)
        },
        elapsedHours: elapsedHours.toFixed(2),
        currentMonth: currentMonth,
        monthlyMining: currentMonthMining.toFixed(8)
      }
    });

  } catch (error) {
    console.error('❌ 채굴량 업데이트 오류:', error);
    res.status(500).json({
      success: false,
      message: '채굴량 업데이트 중 오류가 발생했습니다.',
      error: error.message
    });
  }
});
```

# 📋 **Phase 3: Backend API - 출석 보너스 시스템**

## 📌 API 3-1: 출석 체크

```javascript
// ========================================
// API: POST /api/admin/bitwish/attendance/checkin
// 설명: 출석 체크 (하루 1회, 5% 보너스)
// ⚠️ 전역 변수/함수 일절 사용 안 함
// ⚠️ BitWish Network 전용
// ========================================

app.post('/api/admin/bitwish/attendance/checkin', async (req, res) => {
  try {
    const { walletAddress } = req.body;

    // ========== 1. 입력 검증 (자체 검증) ==========
    if (!walletAddress || !walletAddress.startsWith('BW')) {
      return res.status(400).json({
        success: false,
        message: '유효하지 않은 지갑 주소입니다.'
      });
    }

    console.log(`📅 출석 체크 요청: ${walletAddress}`);

    const db = await getMongoDBConnection();
    const miningCollection = db.collection('admin_user_mining_data');

    // ========== 2. 마이닝 데이터 조회 ==========
    const miningData = await miningCollection.findOne({ walletAddress });

    if (!miningData) {
      return res.status(404).json({
        success: false,
        message: '마이닝 데이터를 찾을 수 없습니다.'
      });
    }

    if (!miningData.miningStatus.isActive) {
      return res.status(400).json({
        success: false,
        message: '마이닝이 활성화되지 않았습니다.'
      });
    }

    // ========== 3. 오늘 날짜 확인 ==========
    const now = new Date();
    const nowString = now.toISOString().replace('T', ' ').substring(0, 19);
    const today = now.toISOString().substring(0, 10); // "2025-01-12"
    const currentMonth = now.toISOString().substring(0, 7); // "2025-01"
    const todayDay = now.getDate(); // 12

    // ========== 4. 중복 출석 체크 ==========
    const lastCheckIn = miningData.miningBreakdown.attendance.lastCheckIn;
    if (lastCheckIn) {
      const lastCheckInDate = lastCheckIn.substring(0, 10);
      if (lastCheckInDate === today) {
        return res.status(400).json({
          success: false,
          message: '오늘 이미 출석하셨습니다.'
        });
      }
    }

    // ========== 5. 월별 출석 기록 업데이트 ==========
    const monthlyCheckIns = miningData.miningBreakdown.attendance.monthlyCheckIns || {};
    
    if (!monthlyCheckIns[currentMonth]) {
      monthlyCheckIns[currentMonth] = {
        days: [],
        totalDays: 0,
        bonus: "0.00000000"
      };
    }

    // 오늘 날짜 추가
    if (!monthlyCheckIns[currentMonth].days.includes(todayDay)) {
      monthlyCheckIns[currentMonth].days.push(todayDay);
      monthlyCheckIns[currentMonth].totalDays = monthlyCheckIns[currentMonth].days.length;
    }

    // ========== 6. 출석 보너스 계산 (기본 채굴량의 5%) ==========
    const startTime = new Date(miningData.miningStartTime);
    const elapsedMs = now - startTime;
    const elapsedHours = elapsedMs / (1000 * 60 * 60);
    const basicMined = elapsedHours * 0.25;
    const attendanceBonus = basicMined * 0.05;

    monthlyCheckIns[currentMonth].bonus = attendanceBonus.toFixed(8);

    // ========== 7. 데이터베이스 업데이트 ==========
    await miningCollection.updateOne(
      { walletAddress },
      {
        $set: {
          'miningBreakdown.attendance.lastCheckIn': nowString,
          'miningBreakdown.attendance.isCheckedToday': true,
          'miningBreakdown.attendance.monthlyCheckIns': monthlyCheckIns,
          'miningBreakdown.attendance.total': attendanceBonus.toFixed(8),
          updatedAt: now
        }
      }
    );

    console.log(`✅ 출석 완료: ${walletAddress} (${today})`);

    // ========== 8. 응답 ==========
    res.json({
      success: true,
      message: '출석 체크가 완료되었습니다.',
      data: {
        walletAddress: walletAddress,
        checkInDate: today,
        checkInTime: nowString,
        currentMonth: currentMonth,
        monthlyCheckIns: monthlyCheckIns[currentMonth].totalDays,
        bonus: attendanceBonus.toFixed(8)
      }
    });

  } catch (error) {
    console.error('❌ 출석 체크 오류:', error);
    res.status(500).json({
      success: false,
      message: '출석 체크 중 오류가 발생했습니다.',
      error: error.message
    });
  }
});
```

## 📌 API 3-2: 출석 현황 조회

```javascript
// ========================================
// API: GET /api/admin/bitwish/attendance/status/:walletAddress
// 설명: 출석 현황 조회 (월별 출석 기록)
// ⚠️ 전역 변수/함수 일절 사용 안 함
// ========================================

app.get('/api/admin/bitwish/attendance/status/:walletAddress', async (req, res) => {
  try {
    const { walletAddress } = req.params;

    if (!walletAddress || !walletAddress.startsWith('BW')) {
      return res.status(400).json({
        success: false,
        message: '유효하지 않은 지갑 주소입니다.'
      });
    }

    const db = await getMongoDBConnection();
    const miningCollection = db.collection('admin_user_mining_data');

    const miningData = await miningCollection.findOne({ walletAddress });

    if (!miningData) {
      return res.status(404).json({
        success: false,
        message: '마이닝 데이터를 찾을 수 없습니다.'
      });
    }

    // ========== 현재 날짜 정보 ==========
    const now = new Date();
    const today = now.toISOString().substring(0, 10);
    const currentMonth = now.toISOString().substring(0, 7);

    // ========== 출석 데이터 ==========
    const attendanceData = miningData.miningBreakdown.attendance;
    const lastCheckIn = attendanceData.lastCheckIn ? attendanceData.lastCheckIn.substring(0, 10) : null;
    const isCheckedToday = lastCheckIn === today;

    // ========== 월별 출석 기록 ==========
    const monthlyCheckIns = attendanceData.monthlyCheckIns || {};
    const currentMonthData = monthlyCheckIns[currentMonth] || {
      days: [],
      totalDays: 0,
      bonus: "0.00000000"
    };

    res.json({
      success: true,
      data: {
        walletAddress: walletAddress,
        today: today,
        isCheckedToday: isCheckedToday,
        lastCheckIn: lastCheckIn,
        currentMonth: currentMonth,
        monthlyCheckIns: currentMonthData.totalDays,
        checkInDays: currentMonthData.days.sort((a, b) => a - b),
        totalBonus: attendanceData.total || "0.00000000",
        rate: attendanceData.rate || "0.05"
      }
    });

  } catch (error) {
    console.error('❌ 출석 현황 조회 오류:', error);
    res.status(500).json({
      success: false,
      message: '출석 현황 조회 중 오류가 발생했습니다.',
      error: error.message
    });
  }
});
```

## 📌 API 3-3: 자정 초기화 (출석 상태 리셋)

```javascript
// ========================================
// API: POST /api/admin/bitwish/attendance/reset-daily
// 설명: 매일 자정 출석 상태 초기화 (크론 작업)
// ⚠️ 전역 변수/함수 일절 사용 안 함
// ========================================

app.post('/api/admin/bitwish/attendance/reset-daily', async (req, res) => {
  try {
    console.log(`🔄 출석 상태 자정 초기화 시작`);

    const db = await getMongoDBConnection();
    const miningCollection = db.collection('admin_user_mining_data');

    // ========== 모든 활성 마이닝 데이터 조회 ==========
    const allMiningData = await miningCollection.find({
      'miningStatus.isActive': true
    }).toArray();

    if (allMiningData.length === 0) {
      return res.json({
        success: true,
        message: '초기화할 데이터가 없습니다.',
        count: 0
      });
    }

    // ========== 일괄 초기화 ==========
    const now = new Date();
    let resetCount = 0;

    for (const data of allMiningData) {
      await miningCollection.updateOne(
        { walletAddress: data.walletAddress },
        {
          $set: {
            'miningBreakdown.attendance.isCheckedToday': false,
            updatedAt: now
          }
        }
      );
      resetCount++;
    }

    console.log(`✅ 출석 상태 초기화 완료: ${resetCount}명`);

    res.json({
      success: true,
      message: '출석 상태가 초기화되었습니다.',
      count: resetCount
    });

  } catch (error) {
    console.error('❌ 출석 상태 초기화 오류:', error);
    res.status(500).json({
      success: false,
      message: '출석 상태 초기화 중 오류가 발생했습니다.',
      error: error.message
    });
  }
});
```

# 📋 **Phase 4: Backend API - 추천 보너스 시스템**

## 📌 API 4-1: 추천 코드로 가입 처리

```javascript
// ========================================
// API: POST /api/admin/bitwish/referral/register
// 설명: 추천 코드로 가입 처리 (1 BW 즉시 지급 + 2% 영구 보너스)
// ⚠️ 전역 변수/함수 일절 사용 안 함
// ⚠️ BitWish Network 전용
// ========================================

app.post('/api/admin/bitwish/referral/register', async (req, res) => {
  try {
    const { referrerCode, newWalletAddress } = req.body;

    // ========== 1. 입력 검증 ==========
    if (!referrerCode || !newWalletAddress) {
      return res.status(400).json({
        success: false,
        message: '추천 코드와 지갑 주소가 필요합니다.'
      });
    }

    if (!newWalletAddress.startsWith('BW')) {
      return res.status(400).json({
        success: false,
        message: '유효하지 않은 지갑 주소입니다.'
      });
    }

    console.log(`🎁 추천 가입 처리: ${referrerCode} → ${newWalletAddress}`);

    const db = await getMongoDBConnection();
    const miningCollection = db.collection('admin_user_mining_data');
    const referralCodesCollection = db.collection('admin_referral_codes');

    // ========== 2. 추천 코드 확인 ==========
    const referralCodeData = await referralCodesCollection.findOne({ referralCode: referrerCode });

    if (!referralCodeData) {
      return res.status(404).json({
        success: false,
        message: '존재하지 않는 추천 코드입니다.'
      });
    }

    const referrerWallet = referralCodeData.ownerWalletAddress;

    // 자기 자신 추천 방지
    if (referrerWallet === newWalletAddress) {
      return res.status(400).json({
        success: false,
        message: '자기 자신을 추천할 수 없습니다.'
      });
    }

    // ========== 3. 추천인 마이닝 데이터 조회 ==========
    const referrerData = await miningCollection.findOne({ walletAddress: referrerWallet });

    if (!referrerData) {
      return res.status(404).json({
        success: false,
        message: '추천인의 마이닝 데이터를 찾을 수 없습니다.'
      });
    }

    // ========== 4. 신규 가입자 마이닝 데이터 조회 ==========
    const newUserData = await miningCollection.findOne({ walletAddress: newWalletAddress });

    if (!newUserData) {
      return res.status(404).json({
        success: false,
        message: '신규 가입자의 마이닝 데이터를 찾을 수 없습니다. 먼저 마이닝을 시작하세요.'
      });
    }

    // ========== 5. 중복 추천 확인 ==========
    const existingReferee = referrerData.referralBonusVault.referees[newWalletAddress];
    if (existingReferee) {
      return res.status(400).json({
        success: false,
        message: '이미 추천 등록된 회원입니다.'
      });
    }

    // ========== 6. 추천 보너스 데이터 생성 ==========
    const now = new Date();
    const nowString = now.toISOString().replace('T', ' ').substring(0, 19);

    const refereeData = {
      refereeAddress: newWalletAddress,
      oneTimeBonus: "1.00000000",          // 즉시 1 BW
      permanentBonus: "0.00000000",        // 초기 0 (이후 누적)
      totalFromThisReferee: "1.00000000",
      
      kycStatus: newUserData.kyc.status || 'none',
      kycApprovedDate: newUserData.kyc.approvedDate || null,
      canMigrate: false,
      migrationDate: null,
      
      refereeJoinedAt: nowString,
      refereeStartedMiningAt: newUserData.miningStartTime
    };

    // ========== 7. 추천인 데이터 업데이트 ==========
    const currentReferralOneTimeTotal = parseFloat(referrerData.miningBreakdown.referralOneTime.total || 0);
    const newReferralOneTimeTotal = currentReferralOneTimeTotal + 1.0;
    
    const currentReferralCount = referrerData.miningBreakdown.referralOneTime.count || 0;
    const newReferralCount = currentReferralCount + 1;
    
    const currentActiveReferrals = referrerData.miningBreakdown.referralPermanent.activeReferrals || 0;
    const newActiveReferrals = currentActiveReferrals + 1;
    
    const currentTotalVault = parseFloat(referrerData.referralBonusVault.totalVault || 0);
    const newTotalVault = currentTotalVault + 1.0;

    await miningCollection.updateOne(
      { walletAddress: referrerWallet },
      {
        $set: {
          [`referralBonusVault.referees.${newWalletAddress}`]: refereeData,
          'referralBonusVault.totalVault': newTotalVault.toFixed(8),
          'miningBreakdown.referralOneTime.count': newReferralCount,
          'miningBreakdown.referralOneTime.total': newReferralOneTimeTotal.toFixed(8),
          'miningBreakdown.referralPermanent.activeReferrals': newActiveReferrals,
          updatedAt: now
        }
      }
    );

    // ========== 8. 추천 코드 사용 기록 업데이트 ==========
    await referralCodesCollection.updateOne(
      { referralCode: referrerCode },
      {
        $inc: {
          'usage.totalUses': 1,
          'usage.activeReferrals': 1
        },
        $push: {
          'usage.usedBy': {
            walletAddress: newWalletAddress,
            usedAt: nowString,
            status: 'active'
          }
        },
        $set: {
          updatedAt: now
        }
      }
    );

    console.log(`✅ 추천 등록 완료: ${referrerWallet} → ${newWalletAddress}`);

    // ========== 9. 응답 ==========
    res.json({
      success: true,
      message: '추천 등록이 완료되었습니다.',
      data: {
        referrerWallet: referrerWallet,
        refereeWallet: newWalletAddress,
        oneTimeBonus: "1.00000000",
        permanentBonusRate: "0.02",
        totalReferrals: newReferralCount,
        activeReferrals: newActiveReferrals
      }
    });

  } catch (error) {
    console.error('❌ 추천 등록 오류:', error);
    res.status(500).json({
      success: false,
      message: '추천 등록 중 오류가 발생했습니다.',
      error: error.message
    });
  }
});
```

## 📌 API 4-2: 추천 영구 보너스 업데이트 (2%)

```javascript
// ========================================
// API: POST /api/admin/bitwish/referral/update-permanent
// 설명: 추천받은 사람의 채굴량에 따른 2% 영구 보너스 업데이트
// ⚠️ 전역 변수/함수 일절 사용 안 함
// ========================================

app.post('/api/admin/bitwish/referral/update-permanent', async (req, res) => {
  try {
    const { referrerWallet } = req.body;

    if (!referrerWallet || !referrerWallet.startsWith('BW')) {
      return res.status(400).json({
        success: false,
        message: '유효하지 않은 지갑 주소입니다.'
      });
    }

    console.log(`🔄 추천 영구 보너스 업데이트: ${referrerWallet}`);

    const db = await getMongoDBConnection();
    const miningCollection = db.collection('admin_user_mining_data');

    // ========== 1. 추천인 데이터 조회 ==========
    const referrerData = await miningCollection.findOne({ walletAddress: referrerWallet });

    if (!referrerData) {
      return res.status(404).json({
        success: false,
        message: '추천인 데이터를 찾을 수 없습니다.'
      });
    }

    const referees = referrerData.referralBonusVault.referees || {};
    const refereeAddresses = Object.keys(referees);

    if (refereeAddresses.length === 0) {
      return res.json({
        success: true,
        message: '추천받은 사람이 없습니다.',
        totalPermanentBonus: "0.00000000"
      });
    }

    // ========== 2. 각 추천받은 사람의 채굴량 조회 ==========
    let totalPermanentBonus = 0;
    const now = new Date();

    for (const refereeAddress of refereeAddresses) {
      const refereeData = await miningCollection.findOne({ walletAddress: refereeAddress });

      if (!refereeData || !refereeData.miningStatus.isActive) {
        continue;
      }

      // 추천받은 사람의 기본 채굴량
      const startTime = new Date(refereeData.miningStartTime);
      const elapsedMs = now - startTime;
      const elapsedHours = elapsedMs / (1000 * 60 * 60);
      const refereeBasicMined = elapsedHours * 0.25;

      // 2% 영구 보너스 계산
      const permanentBonus = refereeBasicMined * 0.02;
      totalPermanentBonus += permanentBonus;

      // 개별 추천받은 사람의 보너스 업데이트
      referees[refereeAddress].permanentBonus = permanentBonus.toFixed(8);
      referees[refereeAddress].totalFromThisReferee = (
        parseFloat(referees[refereeAddress].oneTimeBonus) + permanentBonus
      ).toFixed(8);
    }

    // ========== 3. 추천인 데이터 업데이트 ==========
    const currentTotalVault = parseFloat(referrerData.referralBonusVault.totalVault || 0);
    const oneTimeBonusTotal = parseFloat(referrerData.miningBreakdown.referralOneTime.total || 0);
    const newTotalVault = oneTimeBonusTotal + totalPermanentBonus;

    await miningCollection.updateOne(
      { walletAddress: referrerWallet },
      {
        $set: {
          'referralBonusVault.referees': referees,
          'referralBonusVault.totalVault': newTotalVault.toFixed(8),
          'miningBreakdown.referralPermanent.total': totalPermanentBonus.toFixed(8),
          updatedAt: now
        }
      }
    );

    console.log(`✅ 추천 영구 보너스 업데이트 완료: ${referrerWallet}`);

    // ========== 4. 응답 ==========
    res.json({
      success: true,
      message: '추천 영구 보너스가 업데이트되었습니다.',
      data: {
        referrerWallet: referrerWallet,
        totalPermanentBonus: totalPermanentBonus.toFixed(8),
        totalVault: newTotalVault.toFixed(8),
        activeReferrals: refereeAddresses.length
      }
    });

  } catch (error) {
    console.error('❌ 추천 영구 보너스 업데이트 오류:', error);
    res.status(500).json({
      success: false,
      message: '추천 영구 보너스 업데이트 중 오류가 발생했습니다.',
      error: error.message
    });
  }
});
```

```javascript
    if (miningData.kyc.status !== 'pending') {

      });

    const migrationStartDate = new Date(now);
    migrationStartDate.setDate(migrationStartDate.getDate() + 15);
    const migrationStartDateString = migrationStartDate.toISOString().substring(0, 10);

    // 15일 카운트다운 (초 단위)
    const fifteenDaysInSeconds = 15 * 24 * 60 * 60;
    const countdownDays = 15;
    const countdownHours = 0;
    const countdownMinutes = 0;
    const countdownSeconds = 0;

      { walletAddress },
      {
        $set: {
          'kyc.status': 'approved',
          'kyc.approvedDate': nowString,
          'kyc.migrationStartDate': migrationStartDateString,
          'kyc.adminNote': adminNote || '',
          'kyc.countdown.days': countdownDays,
          'kyc.countdown.hours': countdownHours,
          'kyc.countdown.minutes': countdownMinutes,
          'kyc.countdown.seconds': countdownSeconds,

          updatedAt: now
        }

    const allMiningData = await miningCollection.find({}).toArray();
```

## 📌 API 7-1: 월별 마이그레이션 자동 실행

```javascript
// ========================================
// API: POST /api/admin/bitwish/migration/execute-monthly
// 설명: 월별 마이그레이션 자동 실행 (70% 지갑에서 회원 지갑으로 이동)
// ⚠️ 전역 변수/함수 일절 사용 안 함
// ⚠️ BitWish Network 전용 자체 검증
// ⚠️ 스텔라 관련 코드 절대 사용 안 함
// ========================================

app.post('/api/admin/bitwish/migration/execute-monthly', async (req, res) => {
  try {
    const { walletAddress, targetMonth } = req.body;

    // ========== 1. 입력 검증 (자체 검증) ==========
    if (!walletAddress || !walletAddress.startsWith('BW')) {
      return res.status(400).json({
        success: false,
        message: '유효하지 않은 지갑 주소입니다.'
      });
    }

    // targetMonth 형식: "2025-01"
    const monthRegex = /^\d{4}-\d{2}$/;
    if (!targetMonth || !monthRegex.test(targetMonth)) {
      return res.status(400).json({
        success: false,
        message: '올바른 월 형식이 아닙니다. (예: 2025-01)'
      });
    }

    console.log(`💰 마이그레이션 실행: ${walletAddress} (${targetMonth})`);

    const db = await getMongoDBConnection();
    const miningCollection = db.collection('admin_user_mining_data');
    const systemWalletsCollection = db.collection('admin_system_wallets');

    // ========== 2. 마이닝 데이터 조회 ==========
    const miningData = await miningCollection.findOne({ walletAddress });

    if (!miningData) {
      return res.status(404).json({
        success: false,
        message: '마이닝 데이터를 찾을 수 없습니다.'
      });
    }

    // ========== 3. KYC 승인 및 15일 경과 확인 ==========
    if (miningData.kyc.status !== 'approved') {
      return res.status(400).json({
        success: false,
        message: 'KYC가 승인되지 않았습니다.'
      });
    }

    const now = new Date();
    const migrationStartDate = new Date(miningData.kyc.migrationStartDate);

    if (now < migrationStartDate) {
      return res.status(400).json({
        success: false,
        message: '마이그레이션 시작일이 아직 도래하지 않았습니다.',
        migrationStartDate: miningData.kyc.migrationStartDate
      });
    }

    // ========== 4. 월별 채굴량 확인 ==========
    const monthlyMining = miningData.monthlyMining[targetMonth];

    if (!monthlyMining || !monthlyMining.saved) {
      return res.status(404).json({
        success: false,
        message: '해당 월의 채굴 데이터를 찾을 수 없습니다.'
      });
    }

    // ========== 5. 중복 마이그레이션 확인 ==========
    const existingMigration = miningData.migrationHistory.find(
      m => m.month === targetMonth
    );

    if (existingMigration) {
      return res.status(400).json({
        success: false,
        message: '이미 마이그레이션이 완료된 월입니다.',
        existingMigration: existingMigration
      });
    }

    // ========== 6. 70% 지갑 조회 ==========
    const communityWallet = await systemWalletsCollection.findOne({
      walletType: 'COMMUNITY_70PERCENT'
    });

    if (!communityWallet) {
      return res.status(500).json({
        success: false,
        message: '커뮤니티 지갑을 찾을 수 없습니다.'
      });
    }

    // ========== 7. 잔액 확인 ==========
    const migrationAmount = parseFloat(monthlyMining.amount);
    const currentBalance = parseFloat(communityWallet.balance.current);

    if (currentBalance < migrationAmount) {
      return res.status(400).json({
        success: false,
        message: '커뮤니티 지갑 잔액이 부족합니다.',
        required: migrationAmount.toFixed(8),
        available: currentBalance.toFixed(8)
      });
    }

    // ========== 8. 마이그레이션 실행 ==========
    const nowString = now.toISOString().replace('T', ' ').substring(0, 19);
    const transactionHash = `0x${Date.now().toString(16)}${Math.random().toString(16).substring(2, 66).padEnd(48, '0')}`;
    const migrationId = `MIG-${targetMonth}-${Date.now()}`;

    // 마이그레이션 기록 생성
    const migrationRecord = {
      migrationId: migrationId,
      month: targetMonth,
      amount: monthlyMining.amount,
      
      sourceWallet: 'COMMUNITY_70PERCENT_WALLET',
      sourceWalletAddress: communityWallet.walletInfo.address,
      sourceBalance: communityWallet.balance.current,
      
      migratedAt: nowString,
      transactionHash: transactionHash,
      status: 'completed',
      
      breakdown: monthlyMining.breakdown
    };

    // ========== 9. 회원 데이터 업데이트 ==========
    const currentAvailableAmount = parseFloat(miningData.walletBalance.availableAmount.amount || 0);
    const newAvailableAmount = currentAvailableAmount + migrationAmount;

    await miningCollection.updateOne(
      { walletAddress },
      {
        $push: {
          migrationHistory: migrationRecord
        },
        $set: {
          'walletBalance.availableAmount.amount': newAvailableAmount.toFixed(8),
          updatedAt: now
        }
      }
    );

    // ========== 10. 커뮤니티 지갑 업데이트 ==========
    const newBalance = currentBalance - migrationAmount;

    await systemWalletsCollection.updateOne(
      { walletType: 'COMMUNITY_70PERCENT' },
      {
        $set: {
          'balance.current': newBalance.toFixed(8)
        },
        $inc: {
          'balance.distributed': migrationAmount
        },
        $push: {
          transactions: {
            transactionId: `TX-MIG-${Date.now()}`,
            type: 'distribution',
            amount: monthlyMining.amount,
            recipient: walletAddress,
            recipientName: miningData.kyc.personalInfo.name || '회원',
            purpose: `${targetMonth}월 마이그레이션`,
            timestamp: nowString,
            status: 'completed'
          }
        },
        $set: {
          updatedAt: now
        }
      }
    );

    console.log(`✅ 마이그레이션 완료: ${walletAddress} (${targetMonth}) - ${monthlyMining.amount} BW`);

    // ========== 11. 응답 ==========
    res.json({
      success: true,
      message: '마이그레이션이 완료되었습니다.',
      data: {
        walletAddress: walletAddress,
        month: targetMonth,
        amount: monthlyMining.amount,
        transactionHash: transactionHash,
        migratedAt: nowString,
        newAvailableAmount: newAvailableAmount.toFixed(8),
        communityWalletBalance: newBalance.toFixed(8)
      }
    });

  } catch (error) {
    console.error('❌ 마이그레이션 실행 오류:', error);
    res.status(500).json({
      success: false,
      message: '마이그레이션 실행 중 오류가 발생했습니다.',
      error: error.message
    });
  }
});
```

## 📌 API 7-2: 마이그레이션 예약 자동 실행 (크론 작업용)

```javascript
// ========================================
// API: POST /api/admin/bitwish/migration/auto-execute
// 설명: 매월 23일 자동 마이그레이션 실행 (크론 작업)

app.post('/api/admin/bitwish/migration/auto-execute', async (req, res) => {

    // ========== 1. 마이그레이션 대상 조회 ==========
    const now = new Date();
    const migrationStartDateString = now.toISOString().substring(0, 10);

    const eligibleUsers = await miningCollection.find({
      'kyc.status': 'approved',
      'kyc.migrationStartDate': { $lte: migrationStartDateString }
    }).toArray();

    if (eligibleUsers.length === 0) {
      return res.json({
        success: true,
        message: '마이그레이션 대상자가 없습니다.',
        count: 0
      });
    }

    console.log(`📊 마이그레이션 대상자: ${eligibleUsers.length}명`);

    // ========== 2. 커뮤니티 지갑 조회 ==========
    const communityWallet = await systemWalletsCollection.findOne({
      walletType: 'COMMUNITY_70PERCENT'
    });

    if (!communityWallet) {
      return res.status(500).json({
        success: false,
        message: '커뮤니티 지갑을 찾을 수 없습니다.'
      });
    }

    // ========== 3. 각 회원별 마이그레이션 실행 ==========
    let successCount = 0;
    let failCount = 0;
    const results = [];

    for (const userData of eligibleUsers) {
      try {
        // 마이그레이션할 월 찾기
        const monthlyMining = userData.monthlyMining || {};
        const monthKeys = Object.keys(monthlyMining).sort();

        // 아직 마이그레이션 안 된 월 찾기
        const migratedMonths = userData.migrationHistory.map(m => m.month);
        const pendingMonths = monthKeys.filter(month => !migratedMonths.includes(month));

        if (pendingMonths.length === 0) {
          continue; // 마이그레이션할 월이 없음
        }

        // 가장 오래된 월부터 마이그레이션
        const targetMonth = pendingMonths[0];
        const monthData = monthlyMining[targetMonth];

        if (!monthData || !monthData.saved) {
          continue;
        }

        const migrationAmount = parseFloat(monthData.amount);
        const currentBalance = parseFloat(communityWallet.balance.current);

        if (currentBalance < migrationAmount) {
          failCount++;
          results.push({
            walletAddress: userData.walletAddress,
            month: targetMonth,
            status: 'failed',
            reason: '커뮤니티 지갑 잔액 부족'
          });
          continue;
        }

        // 마이그레이션 실행
        const nowString = now.toISOString().replace('T', ' ').substring(0, 19);
        const transactionHash = `0x${Date.now().toString(16)}${Math.random().toString(16).substring(2, 66).padEnd(48, '0')}`;
        const migrationId = `MIG-${targetMonth}-${Date.now()}`;

        const migrationRecord = {
          migrationId: migrationId,
          month: targetMonth,
          amount: monthData.amount,
          sourceWallet: 'COMMUNITY_70PERCENT_WALLET',
          sourceWalletAddress: communityWallet.walletInfo.address,
          sourceBalance: communityWallet.balance.current,
          migratedAt: nowString,
          transactionHash: transactionHash,
          status: 'completed',
          breakdown: monthData.breakdown
        };

        // 회원 데이터 업데이트
        const currentAvailableAmount = parseFloat(userData.walletBalance.availableAmount.amount || 0);
        const newAvailableAmount = currentAvailableAmount + migrationAmount;

        await miningCollection.updateOne(
          { walletAddress: userData.walletAddress },
          {
            $push: { migrationHistory: migrationRecord },
            $set: {
              'walletBalance.availableAmount.amount': newAvailableAmount.toFixed(8),
              updatedAt: now
            }
          }
        );

        // 커뮤니티 지갑 업데이트
        const newBalance = currentBalance - migrationAmount;

        await systemWalletsCollection.updateOne(
          { walletType: 'COMMUNITY_70PERCENT' },
          {
            $set: { 'balance.current': newBalance.toFixed(8) },
            $inc: { 'balance.distributed': migrationAmount },
            $push: {
              transactions: {
                transactionId: `TX-MIG-${Date.now()}`,
                type: 'distribution',
                amount: monthData.amount,
                recipient: userData.walletAddress,
                recipientName: userData.kyc.personalInfo.name || '회원',
                purpose: `${targetMonth}월 자동 마이그레이션`,
                timestamp: nowString,
                status: 'completed'
              }
            },
            $set: { updatedAt: now }
          }
        );

        // 커뮤니티 지갑 정보 갱신
        communityWallet.balance.current = newBalance.toFixed(8);

        successCount++;
        results.push({
          walletAddress: userData.walletAddress,
          month: targetMonth,
          amount: monthData.amount,
          status: 'success',
          transactionHash: transactionHash
        });

        console.log(`✅ 자동 마이그레이션 성공: ${userData.walletAddress} (${targetMonth})`);

      } catch (error) {
        failCount++;
        results.push({
          walletAddress: userData.walletAddress,
          status: 'failed',
          reason: error.message
        });
        console.error(`❌ 자동 마이그레이션 실패: ${userData.walletAddress}`, error);
      }
    }

    console.log(`✅ 자동 마이그레이션 완료: ${successCount}명 성공, ${failCount}명 실패`);

    // ========== 4. 응답 ==========
    res.json({
      success: true,
      message: '자동 마이그레이션이 완료되었습니다.',

        success: successCount,

      message: '자동 마이그레이션 중 오류가 발생했습니다.',
```

## 📌 API 7-3: 마이그레이션 내역 조회

```javascript
    const allMonths = Object.keys(monthlyMining).sort();

        migrationStartDate: miningData.kyc.migrationStartDate,
        totalMigrated: migrationHistory.length,
        totalPending: pendingMonths.length,
        completedMigrations: migrationHistory.sort((a, b) => 
          new Date(b.migratedAt) - new Date(a.migratedAt)
```

## 📌 API 7-4: 마이그레이션 가능 여부 확인

```javascript
// ========================================
// API: GET /api/admin/bitwish/migration/eligibility/:walletAddress
// 설명: 마이그레이션 가능 여부 확인
// ⚠️ 전역 변수/함수 일절 사용 안 함
// ⚠️ BitWish Network 전용 자체 검증
// ========================================

app.get('/api/admin/bitwish/migration/eligibility/:walletAddress', async (req, res) => {
  try {
    const { walletAddress } = req.params;

    // ========== 1. 입력 검증 (자체 검증) ==========
    if (!walletAddress || !walletAddress.startsWith('BW')) {
      return res.status(400).json({
        success: false,
        message: '유효하지 않은 지갑 주소입니다.'
      });
    }

    const db = await getMongoDBConnection();
    const miningCollection = db.collection('admin_user_mining_data');

    // ========== 2. 마이닝 데이터 조회 ==========
    const miningData = await miningCollection.findOne({ walletAddress });

    if (!miningData) {
      return res.status(404).json({
        success: false,
        message: '마이닝 데이터를 찾을 수 없습니다.'
      });
    }

    // ========== 3. 마이그레이션 가능 여부 확인 ==========
    const eligibilityChecks = {
      kycApproved: false,
      fifteenDaysPassed: false,
      hasPendingMonths: false,
      canMigrate: false
    };

    const reasons = [];

    // KYC 승인 확인
    if (miningData.kyc.status === 'approved') {
      eligibilityChecks.kycApproved = true;
    } else {
      reasons.push(`KYC 상태: ${miningData.kyc.status}`);
    }

    // 15일 경과 확인
    const now = new Date();
    const migrationStartDate = miningData.kyc.migrationStartDate ? 
      new Date(miningData.kyc.migrationStartDate) : null;

    if (migrationStartDate && now >= migrationStartDate) {
      eligibilityChecks.fifteenDaysPassed = true;
    } else if (migrationStartDate) {
      const remainingDays = Math.ceil((migrationStartDate - now) / (1000 * 60 * 60 * 24));
      reasons.push(`마이그레이션 시작까지 ${remainingDays}일 남음`);
    }

    // 대기 중인 월 확인
    const monthlyMining = miningData.monthlyMining || {};
    const migrationHistory = miningData.migrationHistory || [];
    const migratedMonths = migrationHistory.map(m => m.month);
    const allMonths = Object.keys(monthlyMining).sort();
    const pendingMonths = allMonths.filter(month => !migratedMonths.includes(month));

    if (pendingMonths.length > 0) {
      eligibilityChecks.hasPendingMonths = true;
    } else {
      reasons.push('마이그레이션할 월별 데이터 없음');
    }

    // 최종 판단
    eligibilityChecks.canMigrate = 
      eligibilityChecks.kycApproved && 
      eligibilityChecks.fifteenDaysPassed && 
      eligibilityChecks.hasPendingMonths;

    // ========== 4. 응답 ==========
    res.json({
      success: true,
      data: {
        walletAddress: walletAddress,
        eligibility: eligibilityChecks,
        reasons: reasons,
        nextMigrationMonth: pendingMonths.length > 0 ? pendingMonths[0] : null,
        migrationStartDate: miningData.kyc.migrationStartDate,
        countdown: miningData.kyc.countdown
      }
    });

  } catch (error) {
    console.error('❌ 마이그레이션 가능 여부 확인 오류:', error);
    res.status(500).json({
      success: false,
      message: '마이그레이션 가능 여부 확인 중 오류가 발생했습니다.',
      error: error.message
    });
  }
});
```

## 📌 API 7-5: 전체 시스템 통계 조회

```javascript
// ========================================
// API: GET /api/admin/bitwish/system/stats
// 설명: 전체 시스템 통계 조회 (홈페이지 메인 화면용)
// ⚠️ 전역 변수/함수 일절 사용 안 함
// ⚠️ BitWish Network 전용 자체 검증
// ========================================

app.get('/api/admin/bitwish/system/stats', async (req, res) => {
  try {
    console.log(`📊 시스템 통계 조회`);

    const db = await getMongoDBConnection();
    const miningCollection = db.collection('admin_user_mining_data');
    const blockCollection = db.collection('admin_block_records');
    const systemWalletsCollection = db.collection('admin_system_wallets');

    // ========== 1. 총 발행량 (고정) ==========
    const totalSupply = "21000000000.00000000"; // 210억 BW

    // ========== 2. 현재 발행량 (모든 회원 채굴량 합계) ==========
    const allMiningData = await miningCollection.find({
      'miningStatus.isActive': true
    }).toArray();

    let totalMined = 0;
    for (const data of allMiningData) {
      totalMined += parseFloat(data.totalMined.amount || 0);
    }

    const currentSupply = totalMined.toFixed(8);

    // ========== 3. 잔여 발행량 ==========
    const remainingSupply = (parseFloat(totalSupply) - totalMined).toFixed(8);

    // ========== 4. 발행률 ==========
    const issuanceRate = ((totalMined / parseFloat(totalSupply)) * 100).toFixed(2);

    // ========== 5. 총 블록 수 ==========
    const totalBlocks = await blockCollection.countDocuments();

    // ========== 6. 지갑 생성 수 ==========
    const totalWallets = await miningCollection.countDocuments();

    // ========== 7. 시스템 지갑 현황 ==========
    const systemWallets = await systemWalletsCollection.find({}).toArray();
    const walletBalances = {};

    for (const wallet of systemWallets) {
      walletBalances[wallet.walletType] = {
        name: wallet.walletInfo.name,
        initial: wallet.balance.initial,
        current: wallet.balance.current,
        distributed: wallet.balance.distributed,
        percentage: wallet.walletInfo.percentage
      };
    }

    // ========== 8. KYC 통계 ==========
    const kycStats = {
      none: await miningCollection.countDocuments({ 'kyc.status': 'none' }),
      pending: await miningCollection.countDocuments({ 'kyc.status': 'pending' }),
      approved: await miningCollection.countDocuments({ 'kyc.status': 'approved' }),
      rejected: await miningCollection.countDocuments({ 'kyc.status': 'rejected' })
    };

    // ========== 9. 마이그레이션 통계 ==========
    const allUsers = await miningCollection.find({}).toArray();
    let totalMigrations = 0;
    let totalMigratedAmount = 0;

    for (const user of allUsers) {
      const history = user.migrationHistory || [];
      totalMigrations += history.length;
      for (const migration of history) {
        totalMigratedAmount += parseFloat(migration.amount);
      }
    }

    // ========== 10. 응답 ==========
    res.json({
      success: true,
      data: {
        supply: {
          total: totalSupply,
          current: currentSupply,
          remaining: remainingSupply,
          issuanceRate: issuanceRate + '%'
        },
        blockchain: {
          totalBlocks: totalBlocks,
          totalWallets: totalWallets
        },
        systemWallets: walletBalances,
        kyc: kycStats,
        migration: {
          totalMigrations: totalMigrations,
          totalMigratedAmount: totalMigratedAmount.toFixed(8)
        }
      }
    });

  } catch (error) {
    console.error('❌ 시스템 통계 조회 오류:', error);
    res.status(500).json({
      success: false,
      message: '시스템 통계 조회 중 오류가 발생했습니다.',
      error: error.message
    });
  }
});
```

```typescript
const AdminPersonalMiningPage: React.FC = () => {
  // ========== 언어 상태 (자체 관리) ==========
  const [currentLanguage, setCurrentLanguage] = useState<'ko' | 'en' | 'zh' | 'ja'>('ko');
  
  // 번역 함수 (자체 구현)
  const translate = useCallback((key: string): string => {
    return ADMIN_MINING_TRANSLATIONS[currentLanguage][key] || key;
  }, [currentLanguage]);

  // ========== 상태 관리 (자체 관리, 전역 상태 사용 안 함) ==========
  const [miningData, setMiningData] = useState<AdminMiningData | null>(null);
  const [isLoading, setIsLoading] = useState<boolean>(false);
  const [errorMessage, setErrorMessage] = useState<string>('');
  const [successMessage, setSuccessMessage] = useState<string>('');
  const [adminWalletAddress, setAdminWalletAddress] = useState<string>('');

  // 자동 업데이트 타이머 (자체 관리)
  const updateIntervalRef = useRef<NodeJS.Timeout | null>(null);
  const countdownIntervalRef = useRef<NodeJS.Timeout | null>(null);

  // ========================================
  // 🔐 자체 API 호출 함수 (전역 함수 사용 안 함)
  // ========================================
  
  // API Base URL (자체 정의)
  const API_BASE_URL = 'http://localhost:3001/api/admin/bitwish';

  // 자체 fetch 함수 (자체 구현)
  const fetchWithAuth = useCallback(async (url: string, options: RequestInit = {}) => {
    try {
      const response = await fetch(url, {
        ...options,
        headers: {
          'Content-Type': 'application/json',
          ...options.headers
        }
      });

      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
      }

      return await response.json();
    } catch (error) {
      console.error('API 호출 오류:', error);
      throw error;
    }
  }, []);

  // ========================================
  // 📊 마이닝 데이터 로드
  // ========================================
  const loadMiningData = useCallback(async () => {
    if (!adminWalletAddress) return;

    try {
      setIsLoading(true);
      setErrorMessage('');

      const response = await fetchWithAuth(
        `${API_BASE_URL}/mining/data/${adminWalletAddress}`
      );

      if (response.success) {
        setMiningData(response.data);
      } else {
        setErrorMessage(translate('errorLoadingData'));
      }
    } catch (error) {
      console.error('마이닝 데이터 로드 오류:', error);
      setErrorMessage(translate('errorLoadingData'));
    } finally {
      setIsLoading(false);
    }
  }, [adminWalletAddress, fetchWithAuth, translate]);

  // ========================================
  // 🚀 마이닝 시작
  // ========================================
  const handleStartMining = useCallback(async () => {
    if (!adminWalletAddress) {
      setErrorMessage('지갑 주소를 입력해주세요.');
      return;
    }

    try {
      setIsLoading(true);
      setErrorMessage('');

      const response = await fetchWithAuth(
        `${API_BASE_URL}/mining/start`,
        {
          method: 'POST',
          body: JSON.stringify({ walletAddress: adminWalletAddress })
        }
      );

      if (response.success) {
        setSuccessMessage(translate('successStartMining'));
        await loadMiningData();
      } else {
        setErrorMessage(translate('errorStartingMining'));
      }
    } catch (error) {
      console.error('마이닝 시작 오류:', error);
      setErrorMessage(translate('errorStartingMining'));
    } finally {
      setIsLoading(false);
    }

        `${API_BASE_URL}/mining/update`,

          method: 'POST',
          body: JSON.stringify({ walletAddress: adminWalletAddress })
        }

      }

    }

  // ========================================
  useEffect(() => {

      countdownIntervalRef.current = setInterval(() => {
        loadMiningData();
      }, 1000);

        if (countdownIntervalRef.current) {
          clearInterval(countdownIntervalRef.current);
        }
      };
    }
  }, [miningData, loadMiningData]);

  // ========================================
  // 🎨 렌더링
  // ========================================
  return (
    <div className="admin-mining-page" style={{
```

```typescript
// 📦 유저용 타입 정의 (독립적으로 정의)

interface UserMiningDataType {
  walletAddress: string;
  referralCode: string;
  miningStartTime: string;
  miningStatus: {
    isActive: boolean;
    lastUpdateTime: string;
  };
  totalMined: {
    amount: string;
    currency: string;
  };
  miningBreakdown: {
    basic: {
      rate: string;
      total: string;
    };
    attendance: {
      total: string;
      lastCheckIn: string | null;
      isCheckedToday: boolean;
      monthlyCheckIns: {
        [month: string]: {
          days: number[];
          totalDays: number;
        };
      };
    };
    referralOneTime: {
      count: number;
      total: string;
    };
    referralPermanent: {
      total: string;
      activeReferrals: number;
    };
  };
  walletBalance: {
    myBW: {
      amount: string;
      type: string;
    };
    availableAmount: {
      amount: string;
      type: string;
    };
    referralVaultDisplay: {
      amount: string;
      type: string;
    };
  };
  kyc: {
    status: string;
    approvedDate: string | null;
    migrationStartDate: string | null;
    countdown: {
      days: number;
      hours: number;
      minutes: number;
      seconds: number;
      isActive: boolean;
    };
  };
}

// ========================================
// 🎨 유저 마이닝 페이지 메인 컴포넌트
// ========================================
const UserPersonalMiningPage: React.FC = () => {
  // ========== 언어 상태 (독립 관리) ==========
  const [userLanguage, setUserLanguage] = useState<'ko' | 'en' | 'zh' | 'ja'>('ko');
  
  // 번역 함수 (독립 구현)
  const getText = useCallback((key: string): string => {
    return USER_MINING_LANGUAGE_DATA[userLanguage][key] || key;
  }, [userLanguage]);

  // ========== 상태 관리 (독립 관리) ==========
  const [userMiningData, setUserMiningData] = useState<UserMiningDataType | null>(null);
  const [isUserLoading, setIsUserLoading] = useState<boolean>(false);
  const [userErrorMsg, setUserErrorMsg] = useState<string>('');
  const [userSuccessMsg, setUserSuccessMsg] = useState<string>('');
  const [userWalletAddr, setUserWalletAddr] = useState<string>('');
  const [showCopyNotification, setShowCopyNotification] = useState<boolean>(false);

  // 자동 업데이트 타이머 (독립 관리)
  const userUpdateTimer = useRef<NodeJS.Timeout | null>(null);
  const userCountdownTimer = useRef<NodeJS.Timeout | null>(null);

  // ========================================
  // 🔐 유저용 API 호출 함수 (독립 구현)
  // ========================================
  
  const USER_API_URL = 'http://localhost:3001/api/admin/bitwish';

  // 독립 fetch 함수
  const userFetchAPI = useCallback(async (endpoint: string, options: RequestInit = {}) => {
    try {
      const response = await fetch(`${USER_API_URL}${endpoint}`, {
        ...options,
        headers: {
          'Content-Type': 'application/json',
          ...options.headers
        }
      });

      if (!response.ok) {
        throw new Error(`API Error: ${response.status}`);
      }

      return await response.json();
    } catch (err) {
      console.error('User API Error:', err);
      throw err;
    }
  }, []);

  // ========================================
  // 📊 유저 마이닝 데이터 로드
  // ========================================
  const loadUserMiningData = useCallback(async () => {
    if (!userWalletAddr) return;

    try {
      setIsUserLoading(true);
      setUserErrorMsg('');

      const result = await userFetchAPI(`/mining/data/${userWalletAddr}`);

      if (result.success) {
        setUserMiningData(result.data);
      } else {
        setUserErrorMsg(getText('errorLoading'));
      }
    } catch (err) {
      console.error('Load Error:', err);
      setUserErrorMsg(getText('errorLoading'));
    } finally {
      setIsUserLoading(false);
    }
  }, [userWalletAddr, userFetchAPI, getText]);

  // ========================================
  // 🚀 유저 마이닝 시작
  // ========================================
  const startUserMining = useCallback(async () => {
    if (!userWalletAddr) {
      setUserErrorMsg('지갑 주소를 입력하세요.');
      return;
    }

    try {
      setIsUserLoading(true);
      setUserErrorMsg('');

      const result = await userFetchAPI('/mining/start', {
        method: 'POST',
        body: JSON.stringify({ walletAddress: userWalletAddr })
      });

      if (result.success) {
        setUserSuccessMsg('마이닝이 시작되었습니다!');
        await loadUserMiningData();
      } else {
        setUserErrorMsg(getText('errorStarting'));
      }
    } catch (err) {
      console.error('Start Mining Error:', err);
      setUserErrorMsg(getText('errorStarting'));
    } finally {
      setIsUserLoading(false);
    }
  }, [userWalletAddr, userFetchAPI, getText, loadUserMiningData]);

  // ========================================
  // 🔄 유저 채굴량 업데이트
  // ========================================
  const updateUserMining = useCallback(async () => {
    if (!userWalletAddr) return;

    try {
      await userFetchAPI('/mining/update', {
        method: 'POST',
        body: JSON.stringify({ walletAddress: userWalletAddr })
      });

      await loadUserMiningData();
    } catch (err) {
      console.error('Update Error:', err);
    }
  }, [userWalletAddr, userFetchAPI, loadUserMiningData]);

  // ========================================
  // 📅 유저 출석 체크
  // ========================================
  const checkInUser = useCallback(async () => {
    if (!userWalletAddr) return;

    try {
      setIsUserLoading(true);
      setUserErrorMsg('');

      const result = await userFetchAPI('/attendance/checkin', {
        method: 'POST',
        body: JSON.stringify({ walletAddress: userWalletAddr })
      });

      if (result.success) {
        setUserSuccessMsg('출석 체크 완료!');
        await loadUserMiningData();
      } else {
        setUserErrorMsg(getText('errorCheckIn'));
      }
    } catch (err) {
      console.error('Check-in Error:', err);
      setUserErrorMsg(getText('errorCheckIn'));
    } finally {
      setIsUserLoading(false);
    }
  }, [userWalletAddr, userFetchAPI, getText, loadUserMiningData]);

  // ========================================
  // 📋 추천 코드 복사
  // ========================================
  const copyReferralCode = useCallback(() => {
    if (!userMiningData) return;

    navigator.clipboard.writeText(userMiningData.referralCode).then(() => {
      setShowCopyNotification(true);
      setTimeout(() => setShowCopyNotification(false), 2000);
    });
  }, [userMiningData]);

  // ========================================
  // ⏰ 자동 업데이트
  // ========================================
  useEffect(() => {
    if (userMiningData && userMiningData.miningStatus.isActive) {
      userUpdateTimer.current = setInterval(() => {
        updateUserMining();
      }, 30000);

      return () => {
        if (userUpdateTimer.current) {
          clearInterval(userUpdateTimer.current);
        }
      };
    }
  }, [userMiningData, updateUserMining]);

  // ========================================
  // ⏱️ 카운트다운 자동 업데이트
  // ========================================
  useEffect(() => {
    if (userMiningData && userMiningData.kyc.countdown.isActive) {
      userCountdownTimer.current = setInterval(() => {
        loadUserMiningData();
      }, 1000);

      return () => {
        if (userCountdownTimer.current) {
          clearInterval(userCountdownTimer.current);
        }
      };
    }
  }, [userMiningData, loadUserMiningData]);

  // ========================================
  // 🎨 렌더링
  // ========================================
  return (
    <div style={{
      minHeight: '100vh',
      background: 'linear-gradient(135deg, #667eea 0%, #764ba2 100%)',
      padding: '20px'
    }}>
      <div style={{
        maxWidth: '1200px',
        margin: '0 auto'
      }}>
        {/* 헤더 */}
        <div style={{
          background: 'rgba(255, 255, 255, 0.95)',
          borderRadius: '20px',
          padding: '30px',
          marginBottom: '20px',
          boxShadow: '0 10px 30px rgba(0,0,0,0.2)'
        }}>
          <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', flexWrap: 'wrap', gap: '15px' }}>
            <div>
              <h1 style={{ margin: '0 0 5px 0', fontSize: '32px', color: '#667eea' }}>
                {getText('pageTitle')}
              </h1>
              <p style={{ margin: 0, color: '#666', fontSize: '16px' }}>
                {getText('pageSubtitle')}
              </p>
            </div>

            {/* 언어 선택 */}
            <select 
              value={userLanguage}
              onChange={(e) => setUserLanguage(e.target.value as 'ko' | 'en' | 'zh' | 'ja')}
              style={{
                padding: '10px 15px',
                borderRadius: '10px',
                border: '2px solid #667eea',
                fontSize: '14px',
                cursor: 'pointer',
                background: 'white'
              }}
            >
              <option value="ko">🇰🇷 한국어</option>
              <option value="en">🇺🇸 English</option>
              <option value="zh">🇨🇳 中文</option>
              <option value="ja">🇯🇵 日本語</option>
            </select>
          </div>
        </div>

        {/* 지갑 입력 */}
        <div style={{
          background: 'rgba(255, 255, 255, 0.95)',
          borderRadius: '20px',
          padding: '25px',
          marginBottom: '20px',
          boxShadow: '0 10px 30px rgba(0,0,0,0.2)'
        }}>
          <input
            type="text"
            value={userWalletAddr}
            onChange={(e) => setUserWalletAddr(e.target.value)}
            placeholder="BW..."
            style={{
              width: '100%',
              padding: '15px 20px',
              borderRadius: '12px',
              border: '2px solid #667eea',
              fontSize: '16px',
              marginBottom: '15px',
              boxSizing: 'border-box'
            }}
          />
          
          <div style={{ display: 'flex', gap: '10px', flexWrap: 'wrap' }}>
            <button
              onClick={loadUserMiningData}
              disabled={isUserLoading || !userWalletAddr}
              style={{
                flex: 1,
                minWidth: '150px',
                padding: '15px 20px',
                borderRadius: '12px',
                border: 'none',
                background: '#667eea',
                color: 'white',
                fontSize: '16px',
                fontWeight: 'bold',
                cursor: isUserLoading ? 'not-allowed' : 'pointer',
                opacity: (isUserLoading || !userWalletAddr) ? 0.6 : 1
              }}
            >
              {isUserLoading ? getText('loading') : getText('refresh')}
            </button>
          </div>
        </div>

        {/* 알림 메시지 */}
        {userErrorMsg && (
          <div style={{
            background: '#fee',
            border: '2px solid #fcc',
            borderRadius: '12px',
            padding: '15px 20px',
            marginBottom: '20px',
            color: '#c00',
            fontWeight: 'bold'
          }}>
            ⚠️ {userErrorMsg}
          </div>
        )}

        {userSuccessMsg && (
          <div style={{
            background: '#efe',
            border: '2px solid #cfc',
            borderRadius: '12px',
            padding: '15px 20px',
            marginBottom: '20px',
            color: '#060',
            fontWeight: 'bold'
          }}>
            ✅ {userSuccessMsg}
          </div>
        )}

        {/* 마이닝 데이터 표시 */}
        {userMiningData ? (
          <>
            {/* 마이닝 상태 카드 */}
            <div style={{
              background: 'rgba(255, 255, 255, 0.95)',
              borderRadius: '20px',
              padding: '30px',
              marginBottom: '20px',
              boxShadow: '0 10px 30px rgba(0,0,0,0.2)'
            }}>
              <h2 style={{ marginTop: 0, color: '#667eea' }}>{getText('miningOverview')}</h2>
              
              {userMiningData.miningStatus.isActive ? (
                <div style={{
                  background: 'linear-gradient(135deg, #667eea 0%, #764ba2 100%)',
                  borderRadius: '15px',
                  padding: '20px',
                  color: 'white',
                  marginBottom: '20px'
                }}>
                  <div style={{ fontSize: '16px', marginBottom: '10px' }}>
                    ✅ {getText('miningActiveMessage')}
                  </div>
                  <div style={{ fontSize: '48px', fontWeight: 'bold' }}>
                    {parseFloat(userMiningData.totalMined.amount).toFixed(4)} BW
                  </div>
                  <div style={{ fontSize: '14px', opacity: 0.9, marginTop: '10px' }}>
                    {getText('totalMined')}
                  </div>
                </div>
              ) : (
                <div style={{
                  background: '#f5f5f5',
                  borderRadius: '15px',
                  padding: '30px',
                  textAlign: 'center',
                  marginBottom: '20px'
                }}>
```
