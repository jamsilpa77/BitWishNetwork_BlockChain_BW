## **📌 Part 1: Backend API 8-1 - 월별 마이그레이션 실행 (simple-server.cjs에 추가)**

```javascript
// ========================================
// API 8-1: POST /api/admin/bitwish/migration/execute
// 설명: 월별 마이그레이션 실행 (50자리 정밀도)
// ========================================

app.post('/api/admin/bitwish/migration/execute', async (req, res) => {
  try {
    const { walletAddress, targetMonth } = req.body;

    // ========== 1. 입력 검증 ==========
    if (!walletAddress || !walletAddress.startsWith('BW')) {
      return res.status(400).json({
        success: false,
        message: '유효하지 않은 지갑 주소입니다.'
      });
    }

    if (!targetMonth || !/^\d{4}-\d{2}$/.test(targetMonth)) {
      return res.status(400).json({
        success: false,
        message: '유효하지 않은 월 형식입니다. (예: 2025-01)'
      });
    }

    const db = bitWishMongoDB.db;
    if (!db) {
      return res.status(500).json({
        success: false,
        message: 'MongoDB 연결이 필요합니다.'
      });
    }

    const miningCollection = db.collection('admin_user_mining_data');
    const systemWalletsCollection = db.collection('admin_system_wallets');

    // ========== 2. 유저 데이터 조회 (userType='user' 필터) ==========
    const userData = await miningCollection.findOne({ 
      walletAddress,
      userType: 'user'
    });

    if (!userData) {
      return res.status(404).json({
        success: false,
        message: '유저 데이터를 찾을 수 없습니다.'
      });
    }

    // ========== 3. KYC 승인 확인 ==========
    if (!userData.kyc || userData.kyc.status !== 'approved') {
      return res.status(400).json({
        success: false,
        message: 'KYC가 승인되지 않았습니다.'
      });
    }

    // ========== 4. 카운트다운 만료 확인 ==========
    if (!userData.kyc.countdown || !userData.kyc.countdown.isExpired) {
      return res.status(400).json({
        success: false,
        message: '15일 카운트다운이 아직 완료되지 않았습니다.'
      });
    }

    // ========== 5. 월별 채굴 데이터 확인 ==========
    const monthlyMining = userData.monthlyMining || {};
    const targetMonthData = monthlyMining[targetMonth];

    if (!targetMonthData) {
      return res.status(404).json({
        success: false,
        message: `${targetMonth}월의 채굴 데이터가 없습니다.`
      });
    }

    if (!targetMonthData.saved) {
      return res.status(400).json({
        success: false,
        message: `${targetMonth}월의 채굴 데이터가 저장되지 않았습니다.`
      });
    }

    // ========== 6. 이미 마이그레이션 되었는지 확인 ==========
    const migrationHistory = userData.migrationHistory || [];
    const alreadyMigrated = migrationHistory.find(m => m.month === targetMonth);

    if (alreadyMigrated) {
      return res.status(400).json({
        success: false,
        message: `${targetMonth}월은 이미 마이그레이션되었습니다.`,
        data: alreadyMigrated
      });
    }

    // ========== 7. 마이그레이션 금액 계산 (50자리 정밀도) ==========
    const migrationAmount = targetMonthData.amount;
    const migrationAmountFloat = parseFloat(migrationAmount);

    if (migrationAmountFloat <= 0) {
      return res.status(400).json({
        success: false,
        message: '마이그레이션할 금액이 0입니다.'
      });
    }

    // ========== 8. COMMUNITY_70PERCENT 지갑 조회 ==========
    const communityWallet = await systemWalletsCollection.findOne({ 
      walletType: 'COMMUNITY_70PERCENT'
    });

    if (!communityWallet) {
      return res.status(500).json({
        success: false,
        message: 'COMMUNITY_70PERCENT 지갑을 찾을 수 없습니다.'
      });
    }

    // ========== 9. 지갑 잔액 확인 (50자리 정밀도) ==========
    const communityCurrentBalance = parseFloat(communityWallet.balance.current);

    if (communityCurrentBalance < migrationAmountFloat) {
      return res.status(400).json({
        success: false,
        message: 'COMMUNITY_70PERCENT 지갑 잔액이 부족합니다.',
        data: {
          required: migrationAmount,
          available: communityCurrentBalance.toFixed(50)
        }
      });
    }

    const now = new Date();
    const nowString = now.toISOString().replace('T', ' ').substring(0, 19);

    // ========== 10. Transaction Hash 생성 (BitWish 방식) ==========
    const crypto = require('crypto');
    const txData = `${walletAddress}-${targetMonth}-${migrationAmount}-${nowString}`;
    const transactionHash = '0x' + crypto.createHash('sha256').update(txData).digest('hex');

    // ========== 11. 마이그레이션 ID 생성 ==========
    const migrationId = `MIG-${targetMonth.replace('-', '')}-${Date.now().toString().slice(-6)}`;

    // ========== 12. COMMUNITY_70PERCENT 지갑 잔액 차감 (50자리 정밀도) ==========
    const newCommunityBalance = (communityCurrentBalance - migrationAmountFloat).toFixed(50);
    const newCommunityDistributed = (parseFloat(communityWallet.balance.distributed) + migrationAmountFloat).toFixed(50);

    await systemWalletsCollection.updateOne(
      { walletType: 'COMMUNITY_70PERCENT' },
      {
        $set: {
          'balance.current': newCommunityBalance,
          'balance.distributed': newCommunityDistributed,
          updatedAt: now
        },
        $push: {
          transactions: {
            transactionId: migrationId,
            type: 'distribution',
            amount: migrationAmount,
            recipient: walletAddress,
            recipientName: userData.kyc?.personalInfo?.name || 'Unknown',
            purpose: `${targetMonth}월 마이그레이션`,
            timestamp: nowString,
            transactionHash: transactionHash,
            status: 'completed'
          }
        }
      }
    );

    // ========== 13. 마이그레이션 내역 생성 (50자리 정밀도) ==========
    const migrationRecord = {
      migrationId: migrationId,
      month: targetMonth,
      amount: migrationAmount,
      sourceWallet: 'COMMUNITY_70PERCENT',
      sourceWalletAddress: communityWallet.walletInfo.address,
      sourceBalance: newCommunityBalance,
      migratedAt: nowString,
      transactionHash: transactionHash,
      status: 'completed',
      breakdown: {
        basic: targetMonthData.breakdown.basic,
        attendance: targetMonthData.breakdown.attendance,
        referral: targetMonthData.breakdown.referral
      }
    };

    // ========== 14. 유저 데이터 업데이트 (50자리 정밀도, userType='user' 필터) ==========
    // 14-1. 사용 가능 금액 증가 (가상 BW → 실제 BW 변환)
    const currentAvailableAmount = parseFloat(userData.walletBalance?.availableAmount?.amount || 0);
    const newAvailableAmount = (currentAvailableAmount + migrationAmountFloat).toFixed(50);

    await miningCollection.updateOne(
      { 
        walletAddress,
        userType: 'user'
      },
      {
        $set: {
          'walletBalance.availableAmount.amount': newAvailableAmount,
          'walletBalance.availableAmount.type': 'real',
          'walletBalance.availableAmount.sourceWallet': 'COMMUNITY_70PERCENT',
          'walletBalance.availableAmount.description': '마이그레이션 완료된 실제 BW',
          updatedAt: now
        },
        $push: {
          migrationHistory: migrationRecord
        }
      }
    );

    // ========== 15. 응답 ==========
    res.json({
      success: true,
      message: `${targetMonth}월 마이그레이션이 완료되었습니다.`,
      data: {
        migrationId: migrationId,
        month: targetMonth,
        amount: migrationAmount,
        transactionHash: transactionHash,
        migratedAt: nowString,
        walletAddress: walletAddress,
        newAvailableAmount: newAvailableAmount,
        breakdown: migrationRecord.breakdown
      }
    });

  } catch (error) {
    console.error('❌ 마이그레이션 실행 오류:', error);
    res.status(500).json({
      success: false,
      message: '마이그레이션 실행 중 오류가 발생했습니다.',
      error: error.message
    });
  }
});
```

## **📌 Part 2: Backend API 8-2 - 자동 월별 마이그레이션 시스템 (simple-server.cjs에 추가)**

```javascript
// ========================================
// 자동 월별 마이그레이션 시스템 (매일 자정 실행)
// 설명: KYC 승인 후 15일이 지난 유저의 월별 채굴량 자동 마이그레이션
// ========================================

setInterval(async () => {
  try {
    const now = new Date();
    const hours = now.getHours();
    const minutes = now.getMinutes();
    const seconds = now.getSeconds();

    // 매일 자정(00:00:00)에만 실행
    if (hours === 0 && minutes === 0 && seconds < 10) {
      console.log('🔄 자동 월별 마이그레이션 시작...');

      const db = bitWishMongoDB.db;
      if (!db) {
        console.error('❌ MongoDB 연결 없음');
        return;
      }

      const miningCollection = db.collection('admin_user_mining_data');
      const systemWalletsCollection = db.collection('admin_system_wallets');

      // ========== 1. KYC 승인되고 카운트다운 만료된 유저 조회 (userType='user'만) ==========
      const eligibleUsers = await miningCollection.find({ 
        userType: 'user',
        'kyc.status': 'approved',
        'kyc.countdown.isExpired': true
      }).toArray();

      if (eligibleUsers.length === 0) {
        console.log('✅ 마이그레이션 대상 유저 없음');
        return;
      }

      console.log(`📊 마이그레이션 대상 유저: ${eligibleUsers.length}명`);

      // ========== 2. COMMUNITY_70PERCENT 지갑 조회 ==========
      const communityWallet = await systemWalletsCollection.findOne({ 
        walletType: 'COMMUNITY_70PERCENT'
      });

      if (!communityWallet) {
        console.error('❌ COMMUNITY_70PERCENT 지갑을 찾을 수 없습니다.');
        return;
      }

      let communityCurrentBalance = parseFloat(communityWallet.balance.current);
      let communityDistributed = parseFloat(communityWallet.balance.distributed);
      let totalMigrated = 0;
      let successCount = 0;

      const crypto = require('crypto');

      for (const user of eligibleUsers) {
        try {
          const monthlyMining = user.monthlyMining || {};
          const migrationHistory = user.migrationHistory || [];

          // ========== 3. 마이그레이션 대상 월 찾기 ==========
          for (const [month, data] of Object.entries(monthlyMining)) {
            // 이미 마이그레이션된 월은 스킵
            const alreadyMigrated = migrationHistory.find(m => m.month === month);
            if (alreadyMigrated || !data.saved) continue;

            const migrationAmount = parseFloat(data.amount);
            if (migrationAmount <= 0) continue;

            // 잔액 확인
            if (communityCurrentBalance < migrationAmount) {
              console.error(`❌ 지갑 잔액 부족: ${user.walletAddress} - ${month}`);
              continue;
            }

            const nowString = now.toISOString().replace('T', ' ').substring(0, 19);
            const txData = `${user.walletAddress}-${month}-${migrationAmount.toFixed(50)}-${nowString}`;
            const transactionHash = '0x' + crypto.createHash('sha256').update(txData).digest('hex');
            const migrationId = `MIG-${month.replace('-', '')}-${Date.now().toString().slice(-6)}`;

            // ========== 4. 마이그레이션 실행 (50자리 정밀도) ==========
            communityCurrentBalance -= migrationAmount;
            communityDistributed += migrationAmount;

            const migrationRecord = {
              migrationId: migrationId,
              month: month,
              amount: migrationAmount.toFixed(50),
              sourceWallet: 'COMMUNITY_70PERCENT',
              sourceWalletAddress: communityWallet.walletInfo.address,
              sourceBalance: communityCurrentBalance.toFixed(50),
              migratedAt: nowString,
              transactionHash: transactionHash,
              status: 'completed',
              breakdown: {
                basic: data.breakdown.basic,
                attendance: data.breakdown.attendance,
                referral: data.breakdown.referral
              }
            };

            // 사용 가능 금액 증가
            const currentAvailable = parseFloat(user.walletBalance?.availableAmount?.amount || 0);
            const newAvailable = (currentAvailable + migrationAmount).toFixed(50);

            await miningCollection.updateOne(
              { 
                walletAddress: user.walletAddress,
                userType: 'user'
              },
              {
                $set: {
                  'walletBalance.availableAmount.amount': newAvailable,
                  'walletBalance.availableAmount.type': 'real',
                  updatedAt: now
                },
                $push: {
                  migrationHistory: migrationRecord
                }
              }
            );

            totalMigrated += migrationAmount;
            successCount++;

            console.log(`✅ 마이그레이션 완료: ${user.walletAddress} - ${month} - ${migrationAmount.toFixed(8)} BW`);
          }

        } catch (error) {
          console.error(`❌ 유저 마이그레이션 오류 (${user.walletAddress}):`, error);
        }
      }

      // ========== 5. COMMUNITY_70PERCENT 지갑 업데이트 (50자리 정밀도) ==========
      if (successCount > 0) {
        await systemWalletsCollection.updateOne(
          { walletType: 'COMMUNITY_70PERCENT' },
          {
            $set: {
              'balance.current': communityCurrentBalance.toFixed(50),
              'balance.distributed': communityDistributed.toFixed(50),
              updatedAt: now
            }
          }
        );

        console.log(`✅ 자동 마이그레이션 완료: ${successCount}건, 총 ${totalMigrated.toFixed(8)} BW`);
      } else {
        console.log('✅ 마이그레이션 대상 없음');
      }

    }
  } catch (error) {
    console.error('❌ 자동 마이그레이션 오류:', error);
  }
}, 5000); // 5초마다 체크 (자정 감지용)

console.log('✅ 자동 월별 마이그레이션 시스템 시작 (매일 자정 실행)');
```

## **📌 Part 4: Backend API 8-4 - 마이그레이션 가능 월 조회 (simple-server.cjs에 추가)**

```javascript
// ========================================
// API 8-4: GET /api/admin/bitwish/migration/available-months
// 설명: 마이그레이션 가능한 월 목록 조회
// ========================================

app.get('/api/admin/bitwish/migration/available-months', async (req, res) => {
  try {
    const { walletAddress } = req.query;

    if (!walletAddress || !walletAddress.startsWith('BW')) {
      return res.status(400).json({
        success: false,
        message: '유효하지 않은 지갑 주소입니다.'
      });
    }

    const db = bitWishMongoDB.db;
    if (!db) {
      return res.status(500).json({
        success: false,
        message: 'MongoDB 연결이 필요합니다.'
      });
    }

    const miningCollection = db.collection('admin_user_mining_data');

    // ========== 데이터 조회 (userType='user' 필터) ==========
    const userData = await miningCollection.findOne({ 
      walletAddress,
      userType: 'user'
    });

    if (!userData) {
      return res.status(404).json({
        success: false,
        message: '유저 데이터를 찾을 수 없습니다.'
      });
    }

    // KYC 및 카운트다운 확인
    const canMigrate = userData.kyc?.status === 'approved' && userData.kyc?.countdown?.isExpired;

    const monthlyMining = userData.monthlyMining || {};
    const migrationHistory = userData.migrationHistory || [];
    const availableMonths = [];

    for (const [month, data] of Object.entries(monthlyMining)) {
      const alreadyMigrated = migrationHistory.find(m => m.month === month);
      
      if (!alreadyMigrated && data.saved && parseFloat(data.amount) > 0) {
        availableMonths.push({
          month: month,
          amount: data.amount,
          breakdown: data.breakdown,
          startDate: data.startDate,
          endDate: data.endDate,
          canMigrate: canMigrate
        });
      }
    }

    // 월별로 정렬 (오래된 것부터)
    availableMonths.sort((a, b) => a.month.localeCompare(b.month));

    res.json({
      success: true,
      data: {
        availableMonths: availableMonths,
        count: availableMonths.length,
        canMigrate: canMigrate,
        kycStatus: userData.kyc?.status || 'none',
        countdownExpired: userData.kyc?.countdown?.isExpired || false
      }
    });

  } catch (error) {
    console.error('❌ 마이그레이션 가능 월 조회 오류:', error);
    res.status(500).json({
      success: false,
      message: '마이그레이션 가능 월 조회 중 오류가 발생했습니다.',
      error: error.message
    });
  }
});
```

```typescript
// ========== 마이그레이션 내역 조회 함수 ==========
const handleLoadMigrationHistory = async () => {
  if (!walletAddress) return;

  try {
    const response = await fetch(`http://localhost:4001/api/admin/bitwish/migration/history?walletAddress=${walletAddress}`, {
      method: 'GET',
      headers: { 'Content-Type': 'application/json' }
    });

    const data = await response.json();

    if (data.success) {
      setMigrationHistory(data.data.history);
    }
  } catch (err: any) {
    console.error('마이그레이션 내역 조회 실패:', err);
  }
};

// ========== 마이그레이션 가능 월 조회 함수 ==========
const handleLoadAvailableMonths = async () => {
  if (!walletAddress) return;

  try {
    const response = await fetch(`http://localhost:4001/api/admin/bitwish/migration/available-months?walletAddress=${walletAddress}`, {
      method: 'GET',
      headers: { 'Content-Type': 'application/json' }
    });

    const data = await response.json();

    if (data.success) {
      setAvailableMonths(data.data.availableMonths);
    }
  } catch (err: any) {
    console.error('마이그레이션 가능 월 조회 실패:', err);
  }
};

// ========== 마이그레이션 실행 함수 ==========
const handleExecuteMigration = async () => {
  if (!walletAddress || !selectedMonth) {
    setError('월을 선택하세요');
    return;
  }

  setLoading(true);
  setError('');

  try {
    const response = await fetch('http://localhost:4001/api/admin/bitwish/migration/execute', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ walletAddress, targetMonth: selectedMonth })
    });

    const data = await response.json();

    if (data.success) {
      alert(`${t.migrationSuccess}\n${data.data.amount} BW`);
      setSelectedMonth('');
      await handleLoadMigrationHistory();
      await handleLoadAvailableMonths();
      await handleUpdateMining();
    } else {
      setError(data.message || '마이그레이션 실패');
    }
  } catch (err: any) {
    setError(err.message || 'Network error');
  } finally {
    setLoading(false);
  }
};

// ========== useEffect에 추가 (마이그레이션 데이터 자동 로드) ==========
useEffect(() => {
  if (walletAddress) {
    handleLoadMigrationHistory();
    handleLoadAvailableMonths();
  }
}, [walletAddress]);
```
