      setError('지갑 주소는 BW로 시작해야 합니다');
      return;
    }

    setLoading(true);
    setError('');

    try {
      const response = await fetch('http://localhost:4001/api/admin/bitwish/mining/start', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ walletAddress })
      });

      const data = await response.json();

      if (data.success) {
        await handleUpdateMining();
      } else {
        setError(data.message || 'Mining start failed');
      }
    } catch (err: any) {
      setError(err.message || 'Network error');
    } finally {
      setLoading(false);
    }
  };

  const renderMonthlyVaultSection = () => {

  // ========== 마이그레이션 섹션 렌더링 ==========
  const renderMigrationSection = () => {

        border: '2px solid #4CAF50', 
        padding: '15px',

        background: 'white',

        </h3>

        <div style={{
          background: 'linear-gradient(135deg, #667eea 0%, #764ba2 100%)',
          padding: '20px',
          borderRadius: '10px',
          marginBottom: '15px',
          color: 'white',
          textAlign: 'center'
        }}>
          <div style={{ fontSize: '14px', opacity: 0.9, marginBottom: '10px' }}>
            💰 {t.availableAmount}
          </div>
          <div style={{ fontSize: '32px', fontWeight: 'bold' }}>
            {formatNumber(miningData.walletBalance?.availableAmount?.amount || '0')} BW
          </div>
          <div style={{ fontSize: '12px', opacity: 0.8, marginTop: '10px' }}>
            {t.virtualBW}: {formatNumber(miningData.totalMined)} BW
          </div>
        </div>

        {availableMonths.length > 0 && (
          <div style={{ marginBottom: '15px' }}>

        <input
          type="text"
          placeholder={t.walletPlaceholder}
          value={walletAddress}

          onClick={handleStartMining} 
          disabled={loading}
          style={{ 
            flex: 1,
            padding: '12px 20px',
            fontSize: '16px',
            background: '#4CAF50',
            color: 'white',
            border: 'none',
            borderRadius: '5px',

          onClick={handleUpdateMining} 
          disabled={loading}
          style={{ 
            flex: 1,
            padding: '12px 20px',
            fontSize: '16px',
            background: '#2196F3',
            color: 'white',
            border: 'none',
            borderRadius: '5px',
            cursor: loading ? 'not-allowed' : 'pointer'
          }}
        >
          {loading ? t.loading : t.updateMining}
        </button>
        <button 
          onClick={handleCheckIn} 
          disabled={loading}
          style={{ 
            flex: 1,
            padding: '12px 20px',
            fontSize: '16px',
            background: '#FF9800',
            color: 'white',
            border: 'none',
            borderRadius: '5px',
            cursor: loading ? 'not-allowed' : 'pointer'
          }}
        >
          {loading ? t.loading : `📅 ${t.checkIn}`}
        </button>
      </div>

      {/* 오류 메시지 */}
      {error && (
        <div style={{ 
          color: 'white', 
          background: '#f44336',
          padding: '15px',
          borderRadius: '5px',
          marginBottom: '20px'

      )}

      {/* 채굴 데이터 표시 */}
      {miningData && (

          border: '2px solid #2196F3', 
          padding: '20px', 
          borderRadius: '8px',
          background: 'white',
          marginBottom: '20px'

            💰 {t.totalMined}: {formatNumber(miningData.totalMined)} BW
          </h2>
          <p style={{ fontSize: '18px', color: '#333' }}>
            📅 {t.currentMonth}: {formatNumber(miningData.currentMonthMining)} BW
          </p>
          
          <h3 style={{ borderBottom: '1px solid #ddd', paddingBottom: '5px' }}>상세 내역:</h3>
          <ul style={{ lineHeight: '1.8' }}>
            <li>⛏️ {t.basicMining}: <strong>{formatNumber(miningData.breakdown.basic)}</strong> BW</li>
            <li>📅 {t.attendanceBonus}: <strong>{formatNumber(miningData.breakdown.attendance)}</strong> BW</li>
            <li>👥 {t.referralBonus}: <strong>
              {formatNumber(
                (parseFloat(miningData.breakdown.referralPermanent) + 
                 parseFloat(miningData.breakdown.referralOneTime)).toFixed(50)
              )}
            </strong> BW</li>
          </ul>

          <div style={{ 
            background: '#f5f5f5', 
            padding: '10px', 
            borderRadius: '5px',
            marginTop: '15px'
          }}>
            <p style={{ fontSize: '12px', color: '#666', margin: 0 }}>
              🕐 {t.lastUpdate}: {miningData.lastUpdate}
            </p>
            <p style={{ fontSize: '11px', color: '#999', margin: '5px 0 0 0' }}>
              ⭐ BitWish 50자리 부동소수점 정밀 계산 적용
            </p>
          </div>
        </div>
      )}

      {/* 출석 달력 */}
      {miningData && renderAttendanceCalendar()}

      {/* 추천 보너스 시스템 */}
      {miningData && renderReferralSection()}

      {/* KYC 시스템 */}
      {miningData && renderKYCSection()}

      {/* 알림 */}
      {miningData && renderNotificationsSection()}

      {/* 15일 카운트다운 */}
      {miningData && renderCountdownSection()}

      {/* 월별 채굴 보관함 (신규!) */}
      {miningData && renderMonthlyVaultSection()}

      {/* 마이그레이션 시스템 */}
      {miningData && renderMigrationSection()}
    </div>
  );
};

export default UserMiningPage;
