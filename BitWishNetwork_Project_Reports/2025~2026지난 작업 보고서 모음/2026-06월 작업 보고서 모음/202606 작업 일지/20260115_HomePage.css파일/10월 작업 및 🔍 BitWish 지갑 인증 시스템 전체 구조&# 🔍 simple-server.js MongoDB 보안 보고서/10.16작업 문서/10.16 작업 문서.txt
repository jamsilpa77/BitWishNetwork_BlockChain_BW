// 관리자 전용 독립 블록체인 코어 클래스
class BitWishAdminBlockchainCore {
  constructor() {
    this.isAdminMode = true;
    this.virtualBlockchain = new Map();
    this.adminWallets = new Map();
    this.adminTransactions = new Map();
    this.adminBlocks = new Map();
    this.adminMiningSessions = new Map();
    
    // 50자리 부동소수점 정밀도 설정
    const Decimal = require('decimal.js');
    Decimal.set({ precision: 50 });
    this.Decimal = Decimal;
    
    // 관리자 전용 설정
    this.adminConfig = {
      networkId: 'BitWish-Admin-Test-Network',
      blockTime: 10000,
      totalSupply: '21000000000.000000000000000000000000000000000000000000000000000',
      miningRate: '0.250000000000000000000000000000000000000000000000000'
    };
  }

  // 관리자 전용 가상 블록체인 초기화
  initializeAdminBlockchain() {
    try {
      // 제네시스 블록 생성
      const genesisBlock = {
        index: 0,
        timestamp: new Date().toISOString(),
        previousHash: '0',
        hash: this.calculateAdminBlockHash(0, new Date().toISOString(), '0', []),
        transactions: [],
        nonce: 0,
        adminBlock: true
      };
      
      this.adminBlocks.set(0, genesisBlock);
      this.currentBlockHeight = 0;
      
      console.log('✅ 관리자 전용 가상 블록체인 초기화 완료');
      return { success: true, message: '관리자 블록체인 초기화 성공' };
    } catch (error) {
      console.error('❌ 관리자 블록체인 초기화 오류:', error);
      return { success: false, error: error.message };
    }
  }

  // 관리자 전용 블록 해시 계산
  calculateAdminBlockHash(index, timestamp, previousHash, transactions) {
    const crypto = require('crypto');
    const transactionString = transactions.map(tx => tx.hash).join('');
    const data = `${index}${timestamp}${previousHash}${transactionString}`;
    return crypto.createHash('sha256').update(data).digest('hex');
  }

  // 관리자 지갑 생성 (실제 블록체인과 완전 분리)
  createAdminWallet(address, seedPhrase) {
    try {
      const adminWallet = {
        address: address,
        balance: new this.Decimal('0.000000000000000000000000000000000000000000000000000'),
        availableBalance: new this.Decimal('0.000000000000000000000000000000000000000000000000000'),
        lockedBalance: new this.Decimal('0.000000000000000000000000000000000000000000000000000'),
        networkType: 'BITWISH_ADMIN_TESTNET',
        isAdminWallet: true,
        createdAt: new Date(),
        lastActivity: new Date(),
        adminTransactions: [],
        adminMiningData: {
          totalMined: new this.Decimal('0.000000000000000000000000000000000000000000000000000'),
          miningSessions: [],
          isMining: false
        }
      };

      this.adminWallets.set(address, adminWallet);
      
      console.log(`✅ 관리자 지갑 생성 완료: ${address}`);
      return { success: true, wallet: adminWallet };
    } catch (error) {
      console.error('❌ 관리자 지갑 생성 오류:', error);
      return { success: false, error: error.message };
    }
  }

  // 관리자 전용 마이닝 세션 시작
  startAdminMiningSession(address) {
    try {
      const wallet = this.adminWallets.get(address);
      if (!wallet) {
        return { success: false, error: '관리자 지갑을 찾을 수 없습니다' };
      }

      if (wallet.adminMiningData.isMining) {
        return { success: false, error: '이미 마이닝이 진행 중입니다' };
      }

      const sessionId = require('crypto').randomUUID();
      const miningSession = {
        sessionId: sessionId,
        address: address,
        startTime: new Date(),
        lastUpdate: new Date(),
        isActive: true,
        totalEarned: new this.Decimal('0.000000000000000000000000000000000000000000000000000'),
        currentRate: new this.Decimal(this.adminConfig.miningRate),
        blocksMined: 0,
        efficiency: 1.0,
        adminSession: true
      };

      wallet.adminMiningData.isMining = true;
      wallet.adminMiningData.miningSessions.push(miningSession);
      this.adminMiningSessions.set(sessionId, miningSession);

      console.log(`✅ 관리자 마이닝 세션 시작: ${address}`);
      return { success: true, sessionId: sessionId };
    } catch (error) {
      console.error('❌ 관리자 마이닝 세션 시작 오류:', error);
      return { success: false, error: error.message };
    }
  }

  // 관리자 전용 마이닝 보상 계산 (50자리 정밀도)
  calculateAdminMiningReward(sessionId, timeElapsed) {
    try {
      const session = this.adminMiningSessions.get(sessionId);
      if (!session || !session.isActive) {
        return { success: false, error: '활성화된 마이닝 세션을 찾을 수 없습니다' };
      }

      const hours = timeElapsed / (1000 * 60 * 60);
      const reward = new this.Decimal(session.currentRate).mul(hours);
      
      session.totalEarned = session.totalEarned.plus(reward);
      session.lastUpdate = new Date();
      session.blocksMined++;

      const wallet = this.adminWallets.get(session.address);
      if (wallet) {
        wallet.balance = wallet.balance.plus(reward);
        wallet.adminMiningData.totalMined = wallet.adminMiningData.totalMined.plus(reward);
        wallet.lastActivity = new Date();
      }

      console.log(`✅ 관리자 마이닝 보상 계산: ${reward.toString()} BW`);
      return { 
        success: true, 
        reward: reward.toString(),
        totalEarned: session.totalEarned.toString()
      };
    } catch (error) {
      console.error('❌ 관리자 마이닝 보상 계산 오류:', error);
      return { success: false, error: error.message };
    }
  }
}

// ====================================================================================
// 실시간 마이닝-지갑 연동 시스템 - BitWish Network 전용
// ====================================================================================

// 실시간 마이닝 동기화 클래스
class BitWishRealTimeMiningSync {
  constructor() {
    this.wsConnections = new Map();
    this.miningData = new Map();
    this.walletData = new Map();
    this.syncInterval = null;
    
    // 50자리 부동소수점 정밀도 설정
    const Decimal = require('decimal.js');
    Decimal.set({ precision: 50 });
    this.Decimal = Decimal;
    
    // 실시간 동기화 설정
    this.syncConfig = {
      updateInterval: 15000, // 15초마다 동기화
      maxConnections: 10000,
      enableWebSocket: true
    };
  }

  // WebSocket 연결 설정
  setupWebSocketConnection(address, ws) {
    try {
      this.wsConnections.set(address, {
        ws: ws,
        lastPing: new Date(),
        isActive: true,
        syncData: {
          miningBalance: new this.Decimal('0.000000000000000000000000000000000000000000000000000'),
          walletBalance: new this.Decimal('0.000000000000000000000000000000000000000000000000000'),
          lastSync: new Date()
        }
      });

      ws.on('message', (message) => {
        this.handleWebSocketMessage(address, message);
      });

      ws.on('close', () => {
        this.wsConnections.delete(address);
      });

      console.log(`✅ WebSocket 연결 설정 완료: ${address}`);
      return { success: true };
    } catch (error) {
      console.error('❌ WebSocket 연결 설정 오류:', error);
      return { success: false, error: error.message };
    }
  }

  // 실시간 마이닝 데이터 동기화
  syncMiningToWallet(address, miningData) {
    try {
      const connection = this.wsConnections.get(address);
      if (!connection) {
        return { success: false, error: 'WebSocket 연결을 찾을 수 없습니다' };
      }

      // 50자리 부동소수점 계산으로 마이닝 보상 계산
      const miningReward = new this.Decimal(miningData.reward || '0');
      const currentBalance = connection.syncData.miningBalance;
      const newBalance = currentBalance.plus(miningReward);

      // 실시간 지갑 잔액 업데이트
      connection.syncData.miningBalance = newBalance;
      connection.syncData.walletBalance = newBalance;
      connection.syncData.lastSync = new Date();

      // WebSocket으로 실시간 데이터 전송
      const syncData = {
        type: 'mining_sync',
        address: address,
        miningBalance: newBalance.toString(),
        walletBalance: newBalance.toString(),
        reward: miningReward.toString(),
        timestamp: new Date().toISOString()
      };

      connection.ws.send(JSON.stringify(syncData));

      console.log(`✅ 실시간 마이닝-지갑 동기화: ${address} - ${miningReward.toString()} BW`);
      return { success: true, syncData: syncData };
    } catch (error) {
      console.error('❌ 실시간 마이닝-지갑 동기화 오류:', error);
      return { success: false, error: error.message };
    }
  }

  // 마이닝 시작 시 실시간 동기화 시작
  startRealTimeMiningSync(address, sessionId) {
    try {
      const miningData = {
        address: address,
        sessionId: sessionId,
        startTime: new Date(),
        totalEarned: new this.Decimal('0.000000000000000000000000000000000000000000000000000'),
        isActive: true,
        syncEnabled: true
      };

      this.miningData.set(address, miningData);

      // 실시간 동기화 시작
      this.startSyncInterval(address);

      console.log(`✅ 실시간 마이닝 동기화 시작: ${address}`);
      return { success: true, sessionId: sessionId };
    } catch (error) {
      console.error('❌ 실시간 마이닝 동기화 시작 오류:', error);
      return { success: false, error: error.message };
    }
  }

  // 실시간 동기화 간격 설정
  startSyncInterval(address) {
    const interval = setInterval(() => {
      this.performRealTimeSync(address);
    }, this.syncConfig.updateInterval);

    // 간격 ID 저장
    const miningData = this.miningData.get(address);
    if (miningData) {
      miningData.syncInterval = interval;
    }
  }

  // 실시간 동기화 수행
  performRealTimeSync(address) {
    try {
      const miningData = this.miningData.get(address);
      if (!miningData || !miningData.isActive) {
        return;
      }

      // 마이닝 보상 계산
      const timeElapsed = new Date() - miningData.startTime;
      const hours = timeElapsed / (1000 * 60 * 60);
      const hourlyRate = new this.Decimal('0.250000000000000000000000000000000000000000000000000');
      const reward = hourlyRate.mul(hours);

      // 실시간 동기화
      const syncResult = this.syncMiningToWallet(address, { reward: reward.toString() });
      
      if (syncResult.success) {
        miningData.totalEarned = miningData.totalEarned.plus(reward);
        miningData.lastUpdate = new Date();
      }

    } catch (error) {
      console.error(`❌ 실시간 동기화 오류 (${address}):`, error);
    }
  }

  // WebSocket 메시지 처리
  handleWebSocketMessage(address, message) {
    try {
      const data = JSON.parse(message);
      
      switch (data.type) {
        case 'ping':
          this.handlePing(address);
          break;
        case 'mining_status':
          this.handleMiningStatus(address, data);
          break;
        case 'wallet_sync':
          this.handleWalletSync(address, data);
          break;
        default:
          console.log(`알 수 없는 메시지 타입: ${data.type}`);
      }
    } catch (error) {
      console.error('❌ WebSocket 메시지 처리 오류:', error);
    }
  }

  // 핑 처리
  handlePing(address) {
    const connection = this.wsConnections.get(address);
    if (connection) {
      connection.lastPing = new Date();
      connection.ws.send(JSON.stringify({ type: 'pong', timestamp: new Date().toISOString() }));
    }
  }

  // 마이닝 상태 처리
  handleMiningStatus(address, data) {
    const miningData = this.miningData.get(address);
    if (miningData) {
      miningData.isActive = data.isActive;
      miningData.lastUpdate = new Date();
    }
  }

  // 지갑 동기화 처리
  handleWalletSync(address, data) {
    const connection = this.wsConnections.get(address);
    if (connection) {
      connection.syncData.walletBalance = new this.Decimal(data.balance);
      connection.syncData.lastSync = new Date();
    }
  }
}

// 실시간 마이닝-지갑 연동 API 엔드포인트
app.post('/api/bitwish/mining/realtime/sync', async (req, res) => {
  try {
    const { address, sessionId } = req.body;

    if (!syncResult.success) {
      return res.status(500).json({
        success: false,
        error: syncResult.error
      });
    }

    console.log(`✅ 실시간 마이닝-지갑 연동 시작: ${address}`);
    res.json({
      success: true,
      message: '실시간 마이닝-지갑 동기화가 시작되었습니다',
      sessionId: sessionId,
      syncEnabled: true
    });

  } catch (error) {
    console.error('❌ 실시간 마이닝-지갑 연동 오류:', error);
    res.status(500).json({
      success: false,
      error: error.message
    });
  }
});

// ====================================================================================
// KYC 마이그레이션 월별 채굴 보관 시스템 - BitWish Network 전용
// ====================================================================================

// KYC 마이그레이션 시스템 클래스
class BitWishKycMigrationSystem {
  constructor() {
    this.monthlyStorage = new Map();
    this.migrationQueue = new Map();
    this.kycApplications = new Map();
    this.migrationHistory = new Map();
    
    // 50자리 부동소수점 정밀도 설정
    const Decimal = require('decimal.js');
    Decimal.set({ precision: 50 });
    this.Decimal = Decimal;
    
    // KYC 마이그레이션 설정
    this.migrationConfig = {
      kycApprovalDelay: 15 * 24 * 60 * 60 * 1000, // 15일
      monthlyStorageEnabled: true,
      autoMigrationEnabled: true,
      maxMonthlyStorage: 12 // 최대 12개월 보관
    };
  }

  // 월별 채굴량 자동 보관
  storeMonthlyMining(address, year, month, totalMined) {
    try {
      const storageKey = `${address}_${year}_${month}`;
      const monthlyData = {
        address: address,
        year: year,
        month: month,
        totalMined: new this.Decimal(totalMined),
        startDate: new Date(year, month - 1, 1),
        endDate: new Date(year, month, 0, 23, 59, 59),
        storedAt: new Date(),
        migrationStatus: 'pending',
        migrationDate: null
      };

      this.monthlyStorage.set(storageKey, monthlyData);
      
      console.log(`✅ 월별 채굴량 보관: ${address} - ${year}년 ${month}월 - ${totalMined} BW`);
      return { success: true, storageKey: storageKey };
    } catch (error) {
      console.error('❌ 월별 채굴량 보관 오류:', error);
      return { success: false, error: error.message };
    }
  }

  // 월별 채굴량 자동 보관 (월말 자동 실행)
  autoStoreMonthlyMining(address, currentDate) {
    try {
      const currentYear = currentDate.getFullYear();
      const currentMonth = currentDate.getMonth() + 1;
      const lastMonth = currentMonth === 1 ? 12 : currentMonth - 1;
      const lastYear = currentMonth === 1 ? currentYear - 1 : currentYear;

      // 지난 달 채굴량 조회
      const lastMonthMining = this.getMonthlyMiningTotal(address, lastYear, lastMonth);
      
      if (lastMonthMining.greaterThan(0)) {
        const storeResult = this.storeMonthlyMining(address, lastYear, lastMonth, lastMonthMining.toString());
        
        if (storeResult.success) {
          console.log(`✅ 자동 월별 채굴량 보관: ${address} - ${lastYear}년 ${lastMonth}월`);
          return { success: true, stored: true };
        }
      }

      return { success: true, stored: false };
    } catch (error) {
      console.error('❌ 자동 월별 채굴량 보관 오류:', error);
      return { success: false, error: error.message };
    }
  }

  // 월별 채굴량 총합 조회
  getMonthlyMiningTotal(address, year, month) {
    try {
      const startDate = new Date(year, month - 1, 1);
      const endDate = new Date(year, month, 0, 23, 59, 59);
      
      // 실제 구현에서는 MongoDB에서 조회
      // 여기서는 예시로 0 반환
      return new this.Decimal('0.000000000000000000000000000000000000000000000000000');
    } catch (error) {
      console.error('❌ 월별 채굴량 총합 조회 오류:', error);
      return new this.Decimal('0.000000000000000000000000000000000000000000000000000');
    }
  }

  // KYC 신청 처리
  processKycApplication(address, kycData) {
    try {
      const applicationId = require('crypto').randomUUID();
      const kycApplication = {
        applicationId: applicationId,
        address: address,
        kycData: kycData,
        status: 'pending',
        submittedAt: new Date(),
        reviewedAt: null,
        approvedAt: null,
        migrationDate: null,
        migrationStatus: 'pending'
      };

      this.kycApplications.set(applicationId, kycApplication);
      
      console.log(`✅ KYC 신청 처리: ${address} - ${applicationId}`);
      return { success: true, applicationId: applicationId };
    } catch (error) {
      console.error('❌ KYC 신청 처리 오류:', error);
      return { success: false, error: error.message };
    }
  }

  // KYC 승인 처리
  approveKycApplication(applicationId, approvedBy) {
    try {
      const application = this.kycApplications.get(applicationId);
      if (!application) {
        return { success: false, error: 'KYC 신청을 찾을 수 없습니다' };
      }

      application.status = 'approved';
      application.approvedAt = new Date();
      application.reviewedBy = approvedBy;
      
      // 15일 후 마이그레이션 예약
      const migrationDate = new Date(application.approvedAt.getTime() + this.migrationConfig.kycApprovalDelay);
      application.migrationDate = migrationDate;

      // 마이그레이션 대기열에 추가
      this.migrationQueue.set(application.address, {
        applicationId: applicationId,
        address: application.address,
        migrationDate: migrationDate,
        status: 'scheduled',
        createdAt: new Date()
      });

      console.log(`✅ KYC 승인: ${application.address} - 마이그레이션 예정일: ${migrationDate.toISOString()}`);
      return { success: true, migrationDate: migrationDate };
    } catch (error) {
      console.error('❌ KYC 승인 처리 오류:', error);
      return { success: false, error: error.message };
    }
  }

  // KYC 후 15일 마이그레이션 실행
  executeKycMigration(address) {
    try {
      const migrationQueue = this.migrationQueue.get(address);
      if (!migrationQueue) {
        return { success: false, error: '마이그레이션 대기열을 찾을 수 없습니다' };
      }

      // 월별 보관된 채굴량 조회
      const monthlyStorages = this.getMonthlyStoragesForMigration(address);
      
      if (monthlyStorages.length === 0) {
        return { success: false, error: '마이그레이션할 월별 채굴량이 없습니다' };
      }

      // 첫 번째 월별 채굴량 마이그레이션
      const firstMonth = monthlyStorages[0];
      const migrationResult = this.migrateMonthlyMining(address, firstMonth);
      
      if (migrationResult.success) {
        // 마이그레이션 완료 처리
        firstMonth.migrationStatus = 'completed';
        firstMonth.migrationDate = new Date();
        
        // 다음 달 마이그레이션 예약
        this.scheduleNextMonthMigration(address, firstMonth);
        
        console.log(`✅ KYC 마이그레이션 실행: ${address} - ${firstMonth.year}년 ${firstMonth.month}월`);
        return { success: true, migratedAmount: migrationResult.amount };
      }

      return { success: false, error: migrationResult.error };
    } catch (error) {
      console.error('❌ KYC 마이그레이션 실행 오류:', error);
      return { success: false, error: error.message };
    }
  }

  // 월별 마이그레이션 실행
  migrateMonthlyMining(address, monthlyStorage) {
    try {
      const migrationAmount = monthlyStorage.totalMined;
      
      // 실제 BW 토큰 전송 (P2P 지갑)
      const transferResult = this.transferRealBitWishToken(address, migrationAmount.toString());
      
      if (transferResult.success) {
        // 마이그레이션 기록 저장
        const migrationRecord = {
          address: address,
          year: monthlyStorage.year,
          month: monthlyStorage.month,
          amount: migrationAmount.toString(),
          migratedAt: new Date(),
          transactionHash: transferResult.transactionHash
        };

        this.migrationHistory.set(`${address}_${monthlyStorage.year}_${monthlyStorage.month}`, migrationRecord);
        
        console.log(`✅ 월별 마이그레이션 완료: ${address} - ${migrationAmount.toString()} BW`);
        return { success: true, amount: migrationAmount.toString() };
      }

      return { success: false, error: transferResult.error };
    } catch (error) {
      console.error('❌ 월별 마이그레이션 오류:', error);
      return { success: false, error: error.message };
    }
  }

  // 실제 BW 토큰 전송
  transferRealBitWishToken(address, amount) {
    try {
      // 실제 구현에서는 P2P 지갑으로 전송
      const transactionHash = require('crypto').randomUUID();
      
      console.log(`✅ 실제 BW 토큰 전송: ${address} - ${amount} BW`);
      return { success: true, transactionHash: transactionHash };
    } catch (error) {
      console.error('❌ 실제 BW 토큰 전송 오류:', error);
      return { success: false, error: error.message };
    }
  }

  // 다음 달 마이그레이션 예약
  scheduleNextMonthMigration(address, currentMonth) {
    try {
      const nextMonth = currentMonth.month === 12 ? 1 : currentMonth.month + 1;
      const nextYear = currentMonth.month === 12 ? currentMonth.year + 1 : currentMonth.year;
      
      const nextMigrationDate = new Date();
      nextMigrationDate.setMonth(nextMigrationDate.getMonth() + 1);
      
      this.migrationQueue.set(address, {
        address: address,
        migrationDate: nextMigrationDate,
        status: 'scheduled',
        nextMonth: nextMonth,
        nextYear: nextYear,
        createdAt: new Date()
      });

      console.log(`✅ 다음 달 마이그레이션 예약: ${address} - ${nextYear}년 ${nextMonth}월`);
    } catch (error) {
      console.error('❌ 다음 달 마이그레이션 예약 오류:', error);
    }
  }

  // 월별 보관 데이터 조회
  getMonthlyStoragesForMigration(address) {
    try {
      const storages = [];
      
      for (const [key, storage] of this.monthlyStorage) {
        if (storage.address === address && storage.migrationStatus === 'pending') {
          storages.push(storage);
        }
      }
      
      // 년월순 정렬
      storages.sort((a, b) => {
        if (a.year !== b.year) return a.year - b.year;
        return a.month - b.month;
      });
      
      return storages;
    } catch (error) {
      console.error('❌ 월별 보관 데이터 조회 오류:', error);
      return [];
    }
  }
}

// KYC 마이그레이션 API 엔드포인트
app.post('/api/bitwish/kyc/migration/execute', async (req, res) => {
  try {
    const { address, adminKey } = req.body;
    
    if (adminKey !== 'admin_bitwish_2024') {
      return res.status(401).json({
        success: false,
        error: '관리자 권한이 필요합니다'
      });
    }

    if (!address) {
      return res.status(400).json({
        success: false,
        error: '지갑 주소가 필요합니다'
      });
    }

    // KYC 마이그레이션 실행
    const kycMigration = new BitWishKycMigrationSystem();
    const migrationResult = kycMigration.executeKycMigration(address);
    
    if (!migrationResult.success) {
      return res.status(500).json({
        success: false,
        error: migrationResult.error
      });
    }

    console.log(`✅ KYC 마이그레이션 실행 완료: ${address}`);
    res.json({
      success: true,
      message: 'KYC 마이그레이션이 성공적으로 실행되었습니다',
      migratedAmount: migrationResult.migratedAmount
    });

  } catch (error) {
    console.error('❌ KYC 마이그레이션 실행 오류:', error);
    res.status(500).json({
      success: false,
      error: error.message
    });
  }
});

// ====================================================================================
// 추천 보너스 시스템 - BitWish Network 전용

// 추천 보너스 시스템 클래스
class BitWishReferralBonusSystem {
  constructor() {
    this.referralCodes = new Map();
    this.referralRelations = new Map();
    this.bonusCalculations = new Map();
    this.referralRewards = new Map();
    
    // 50자리 부동소수점 정밀도 설정
    const Decimal = require('decimal.js');
    Decimal.set({ precision: 50 });
    this.Decimal = Decimal;
    
    // 추천 보너스 설정
    this.bonusConfig = {
      immediateReward: '1.000000000000000000000000000000000000000000000000000', // 1BW
      permanentBonusRate: 0.02, // 2%
      maxReferrals: 1000,
      bonusEnabled: true
    };
  }

  // 추천 코드 자동 발급 (지갑 생성 시)
  generateReferralCode(address) {
    try {
      // BitWish Network 전용 추천 코드 생성
      const prefix = 'BW';
      const randomPart = require('crypto').randomBytes(8).toString('hex').toUpperCase();
      const referralCode = `${prefix}${randomPart}`;

      const referralData = {
        code: referralCode,
        address: address,
        generatedAt: new Date(),
        isActive: true,
        totalReferrals: 0,
        totalRewards: new this.Decimal('0.000000000000000000000000000000000000000000000000000'),
        immediateRewards: new this.Decimal('0.000000000000000000000000000000000000000000000000000'),
        permanentRewards: new this.Decimal('0.000000000000000000000000000000000000000000000000000'),
        referrals: [],
        bonusHistory: []
      };

      this.referralCodes.set(referralCode, referralData);
      
      console.log(`✅ 추천 코드 자동 발급: ${address} - ${referralCode}`);
      return { success: true, referralCode: referralCode };
    } catch (error) {
      console.error('❌ 추천 코드 자동 발급 오류:', error);
      return { success: false, error: error.message };
    }
  }

  // 추천 관계 등록
  registerReferralRelation(referralCode, refereeAddress) {
    try {
      const referralData = this.referralCodes.get(referralCode);
      if (!referralData) {
        return { success: false, error: '유효하지 않은 추천 코드입니다' };
      }

      // 중복 추천 확인
      const existingRelation = this.referralRelations.get(refereeAddress);
      if (existingRelation) {
        return { success: false, error: '이미 추천인이 등록되어 있습니다' };
      }

      // 추천 관계 등록
      const relationData = {
        referrerCode: referralCode,
        referrerAddress: referralData.address,
        refereeAddress: refereeAddress,
        registeredAt: new Date(),
        isActive: true,
        totalMiningReward: new this.Decimal('0.000000000000000000000000000000000000000000000000000'),
        immediateRewardPaid: false,
        permanentRewardsPaid: new this.Decimal('0.000000000000000000000000000000000000000000000000000')
      };

      this.referralRelations.set(refereeAddress, relationData);
      
      // 추천인 데이터 업데이트
      referralData.totalReferrals++;
      referralData.referrals.push({
        refereeAddress: refereeAddress,
        registeredAt: new Date(),
        status: 'active'
      });

      console.log(`✅ 추천 관계 등록: ${referralCode} → ${refereeAddress}`);
      return { success: true, relationData: relationData };
    } catch (error) {
      console.error('❌ 추천 관계 등록 오류:', error);
      return { success: false, error: error.message };
    }
  }

  // 추천인 보상 계산 (1BW + 2% 영구보너스)
  calculateReferralBonus(refereeAddress, miningAmount) {
    try {
      const relationData = this.referralRelations.get(refereeAddress);
      if (!relationData) {
        return { success: false, error: '추천 관계를 찾을 수 없습니다' };
      }

      const referralData = this.referralCodes.get(relationData.referrerCode);
      if (!referralData) {
        return { success: false, error: '추천인 데이터를 찾을 수 없습니다' };
      }

      // 1. 즉시 지급 보상 (1BW)
      let immediateReward = new this.Decimal('0.000000000000000000000000000000000000000000000000000');
      if (!relationData.immediateRewardPaid) {
        immediateReward = new this.Decimal(this.bonusConfig.immediateReward);
        relationData.immediateRewardPaid = true;
      }

      // 2. 영구 보너스 (2% - 추천받은 사람의 마이닝 보상의 2%)
      const miningAmountDecimal = new this.Decimal(miningAmount);
      const permanentBonus = miningAmountDecimal.mul(this.bonusConfig.permanentBonusRate);

      // 총 보상 계산
      const totalReward = immediateReward.plus(permanentBonus);

      // 추천인 계정 업데이트
      referralData.totalRewards = referralData.totalRewards.plus(totalReward);
      referralData.immediateRewards = referralData.immediateRewards.plus(immediateReward);
      referralData.permanentRewards = referralData.permanentRewards.plus(permanentBonus);

      // 보너스 기록 저장
      const bonusRecord = {
        refereeAddress: refereeAddress,
        immediateReward: immediateReward.toString(),
        permanentBonus: permanentBonus.toString(),
        totalReward: totalReward.toString(),
        miningAmount: miningAmount,
        calculatedAt: new Date()
      };

      referralData.bonusHistory.push(bonusRecord);

      console.log(`✅ 추천인 보상 계산: ${relationData.referrerAddress} - ${totalReward.toString()} BW`);
      return {
        success: true,
        immediateReward: immediateReward.toString(),
        permanentBonus: permanentBonus.toString(),
        totalReward: totalReward.toString(),
        bonusRecord: bonusRecord
      };
    } catch (error) {
      console.error('❌ 추천인 보상 계산 오류:', error);
      return { success: false, error: error.message };
    }
  }

  // 추천인 누적 보상 조회
  getReferralCumulativeReward(address) {
    try {
      const referralData = this.referralCodes.get(address);
      if (!referralData) {
        return { success: false, error: '추천인 데이터를 찾을 수 없습니다' };
      }

      const cumulativeData = {
        totalReferrals: referralData.totalReferrals,
        totalRewards: referralData.totalRewards.toString(),
        immediateRewards: referralData.immediateRewards.toString(),
        permanentRewards: referralData.permanentRewards.toString(),
        pendingMigration: referralData.totalRewards.toString(),
        bonusHistory: referralData.bonusHistory
      };

      console.log(`✅ 추천인 누적 보상 조회: ${address} - ${referralData.totalRewards.toString()} BW`);
      return { success: true, cumulativeData: cumulativeData };
    } catch (error) {
      console.error('❌ 추천인 누적 보상 조회 오류:', error);
      return { success: false, error: error.message };
    }
  }

  // KYC 후 추천 보너스 마이그레이션
  migrateReferralBonus(address, kycApprovalDate) {
    try {
      const referralData = this.referralCodes.get(address);
      if (!referralData) {
        return { success: false, error: '추천인 데이터를 찾을 수 없습니다' };
      }

      // 15일 후 마이그레이션 날짜 계산
      const migrationDate = new Date(kycApprovalDate.getTime() + (15 * 24 * 60 * 60 * 1000));
      
      // 마이그레이션 예약
      const migrationRecord = {
        address: address,
        totalRewards: referralData.totalRewards.toString(),
        migrationDate: migrationDate,
        status: 'scheduled',
        createdAt: new Date()
      };

      this.referralRewards.set(address, migrationRecord);

      console.log(`✅ 추천 보너스 마이그레이션 예약: ${address} - ${migrationDate.toISOString()}`);
      return { success: true, migrationDate: migrationDate };
    } catch (error) {
      console.error('❌ 추천 보너스 마이그레이션 오류:', error);
      return { success: false, error: error.message };
    }
  }

  // 추천 보너스 실제 지급
  executeReferralBonusMigration(address) {
    try {
      const migrationRecord = this.referralRewards.get(address);
      if (!migrationRecord) {
        return { success: false, error: '마이그레이션 기록을 찾을 수 없습니다' };
      }

      const referralData = this.referralCodes.get(address);
      if (!referralData) {
        return { success: false, error: '추천인 데이터를 찾을 수 없습니다' };
      }

      // 실제 BW 토큰 전송
      const transferResult = this.transferReferralBonus(address, referralData.totalRewards.toString());
      
      if (transferResult.success) {
        migrationRecord.status = 'completed';
        migrationRecord.completedAt = new Date();
        migrationRecord.transactionHash = transferResult.transactionHash;

        console.log(`✅ 추천 보너스 실제 지급: ${address} - ${referralData.totalRewards.toString()} BW`);
        return { success: true, amount: referralData.totalRewards.toString() };
      }

      return { success: false, error: transferResult.error };
    } catch (error) {
      console.error('❌ 추천 보너스 실제 지급 오류:', error);
      return { success: false, error: error.message };
    }
  }

  // 추천 보너스 실제 전송
  transferReferralBonus(address, amount) {
    try {
      // 실제 구현에서는 P2P 지갑으로 전송
      const transactionHash = require('crypto').randomUUID();
      
      console.log(`✅ 추천 보너스 실제 전송: ${address} - ${amount} BW`);
      return { success: true, transactionHash: transactionHash };
    } catch (error) {
      console.error('❌ 추천 보너스 실제 전송 오류:', error);
      return { success: false, error: error.message };
    }
  }
}

        return { success: false, error: '개인 저장소를 찾을 수 없습니다' };
      }

      const miningRecord = {
        recordId: require('crypto').randomUUID(),
        address: address,
        startTime: startTime,
        year: startTime.getFullYear(),
        month: startTime.getMonth() + 1,
        day: startTime.getDate(),
        hour: startTime.getHours(),
        minute: startTime.getMinutes(),
        second: startTime.getSeconds(),
        timestamp: startTime.toISOString(),
        status: 'active',
        createdAt: new Date()
      };

      storage.miningData.miningHistory.push(miningRecord);
      storage.miningData.totalMiningSessions++;
      storage.miningData.currentSession = miningRecord;
      storage.totalRecords++;
      storage.lastUpdated = new Date();

      this.miningRecords.set(miningRecord.recordId, miningRecord);
      
      console.log(`✅ 마이닝 시작 시간 정밀 저장: ${address} - ${startTime.toISOString()}`);
      return { success: true, recordId: miningRecord.recordId };
    } catch (error) {
      console.error('❌ 마이닝 시작 시간 저장 오류:', error);
      return { success: false, error: error.message };
    }
  }

  // 개인별 마이닝 데이터 실시간 저장
  storeIndividualMiningData(address, miningData) {
    try {
      const storage = this.individualStorages.get(address);
      if (!storage) {
        return { success: false, error: '개인 저장소를 찾을 수 없습니다' };
      }

      const miningAmount = new this.Decimal(miningData.amount);
      
      // 마이닝 데이터 업데이트
      storage.miningData.totalMined = storage.miningData.totalMined.plus(miningAmount);
      
      if (storage.miningData.currentSession) {
        storage.miningData.currentSession.totalEarned = storage.miningData.currentSession.totalEarned || new this.Decimal('0');
        storage.miningData.currentSession.totalEarned = storage.miningData.currentSession.totalEarned.plus(miningAmount);
        storage.miningData.currentSession.lastUpdate = new Date();
      }

      // 지갑 데이터 업데이트
      storage.walletData.totalBalance = storage.walletData.totalBalance.plus(miningAmount);
      storage.walletData.lastActivity = new Date();

      storage.lastUpdated = new Date();

      console.log(`✅ 개인별 마이닝 데이터 저장: ${address} - ${miningAmount.toString()} BW`);
      return { success: true };
    } catch (error) {
      console.error('❌ 개인별 마이닝 데이터 저장 오류:', error);
      return { success: false, error: error.message };
    }
  }

      storage.walletData.lastActivity = new Date();
      storage.totalRecords++;
      storage.lastUpdated = new Date();

      this.walletRecords.set(transactionRecord.transactionId, transactionRecord);
      
      console.log(`✅ 개인별 지갑 데이터 저장: ${address} - ${transactionRecord.type}`);

  try {
    const { address } = req.body;
    
    if (!address) {
      return res.status(400).json({
        success: false,
        error: '지갑 주소가 필요합니다'
      });
    }

    // 개인별 독립 저장소 생성
    const individualStorage = new BitWishIndividualStorageSystem();
    const storageResult = individualStorage.createIndividualStorage(address);
    
    if (!storageResult.success) {
      return res.status(500).json({
        success: false,
        error: storageResult.error
      });
    }

    console.log(`✅ 개인별 독립 저장소 생성 완료: ${address}`);
    res.json({
      success: true,
      message: '개인별 독립 저장소가 생성되었습니다',
      storageId: storageResult.storageId
    });

  } catch (error) {
    console.error('❌ 개인별 독립 저장소 생성 오류:', error);
    res.status(500).json({

    });
  }
