#### **1.3 인증 처리 로직 (라인 127-208)**
```typescript
const authenticateBitWishWallet = async () => {
  try {
    setIsAuthenticating(true);
    
    // 1. 입력 검증
    if (!walletAddress.trim() || !walletPassword.trim()) {
      setError('지갑 주소와 비밀번호를 입력하세요');
      return;
    }
    
    // 2. BitWish 주소 형식 검증
    const bitwishAddressRegex = /^BW[A-F0-9]{40}$/;
    if (!bitwishAddressRegex.test(walletAddress)) {
      setError('올바른 BitWish 지갑 주소 형식이 아닙니다');
      return;
    }
    
    // 3. 비밀번호 해싱 (PBKDF2 + 솔팅)
    const hashedPassword = await hashPassword(walletPassword);
    
    // 4. 백엔드 API 호출 (검증만 수행)
    const response = await fetch('http://localhost:4001/api/bitwish/integrated/wallet/authenticate', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        walletAddress: walletAddress,
        password: hashedPassword  // ✅ 해시된 비밀번호만 전송
      })
    });
    
    const result = await response.json();
    
    // 5. 인증 성공 시 콜백 호출
    if (result.success) {
      onWalletAuthenticated(walletAddress);
    }
  } catch (error) {
    setError('지갑 인증 중 오류가 발생했습니다');
  } finally {
    setIsAuthenticating(false);
  }
};
```

**보안 특징:**
- ✅ 평문 비밀번호 절대 전송 안 함
- ✅ PBKDF2 해싱 (100,000회 반복)
- ✅ 솔트 기반 암호화
- ✅ 클라이언트에서 해싱 후 전송

#### **2.2 2차 비밀번호 설정 로직 (BitWishSecondPasswordModal.tsx 라인 159-249)**
```typescript
const handleSetPassword = async () => {
  try {
    setIsLoading(true);
    setError('');
    setSuccess('');
    
    // 1. 입력 검증
    const trimmedWalletAddress = walletAddress.trim().toUpperCase();
    const trimmedPassword = password.trim();
    const trimmedConfirmPassword = confirmPassword.trim();
    
    if (!trimmedWalletAddress || !trimmedPassword || !trimmedConfirmPassword) {
      setError('모든 필드를 입력해주세요.');
      return;
    }
    
    // 2. 비밀번호 일치 확인
    if (trimmedPassword !== trimmedConfirmPassword) {
      setError('비밀번호가 일치하지 않습니다.');
      return;
    }
    
    // 3. 비밀번호 강도 검증 (8자 이상, 특수문자 포함 등)
    const strengthCheck = checkPasswordStrength(trimmedPassword);
    if (!strengthCheck.isValid) {
      setError(strengthCheck.message);
      return;
    }
    
    // 4. BitWish 주소 형식 검증
    const bitwishAddressRegex = /^BW[A-F0-9]{40}$/;
    if (!bitwishAddressRegex.test(trimmedWalletAddress)) {
      setError('올바른 BitWish 지갑 주소 형식이 아닙니다.');
      return;
    }
    
    // 5. 지갑 주소 존재 여부 확인
    await validateWalletAddress(trimmedWalletAddress);
    
    // 6. 비밀번호 해싱 (PBKDF2 + 솔팅)
    const hashedPassword = await hashPassword(trimmedPassword);
    
    // 7. 백엔드 API 호출 - 2차 비밀번호 설정
    const response = await fetch('http://localhost:4001/api/bitwish/integrated/wallet/setup-second-password', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        walletAddress: trimmedWalletAddress,
        password: hashedPassword  // ✅ 해시된 비밀번호만 전송
      })
    });
    
    const result = await response.json();
    
    if (result.success) {
      setSuccess('2차 비밀번호가 성공적으로 설정되었습니다.');
      
      // 8. 로컬 스토리지에 지갑 주소만 저장 (비밀번호는 절대 저장 안 함)
      localStorage.setItem('bitwish_wallet_address', trimmedWalletAddress);
      
      // 9. 1초 후 모달 닫기
      setTimeout(() => {
        handleSuccess();
      }, 1000);
    }
  } catch (error) {
    setError(error instanceof Error ? error.message : '비밀번호 설정 중 오류가 발생했습니다.');
  } finally {
    setIsLoading(false);
  }
};
```

**보안 특징:**
- ✅ 비밀번호 평문 절대 저장 안 함
- ✅ PBKDF2 해싱 (100,000회 반복)
- ✅ 클라이언트 측 검증 + 서버 측 검증
- ✅ 로컬 스토리지에는 지갑 주소만 저장

#### **3.2 비밀번호 변경 로직 (라인 231-320)**
```typescript
const changeBitWishPassword = async () => {
  try {
    setIsChangingPassword(true);
    setError('');
    setSuccess('');
    
    // 1. 입력 검증
    if (!walletAddress.trim()) {
      setError('지갑 주소가 필요합니다.');
      return;
    }
    
    if (!currentPassword.trim()) {
      setError('기존 비밀번호가 필요합니다.');
      return;
    }
    
    if (!newPassword.trim()) {
      setError('새 비밀번호가 필요합니다.');
      return;
    }
    
    if (!confirmNewPassword.trim()) {
      setError('새 비밀번호 확인이 필요합니다.');
      return;
    }
    
    // 2. 새 비밀번호 일치 확인
    if (newPassword !== confirmNewPassword) {
      setError('새 비밀번호와 확인 비밀번호가 일치하지 않습니다.');
      return;
    }
    
    // 3. 비밀번호 강도 검증
    if (newPassword.length < 8) {
      setError('새 비밀번호는 최소 8자 이상이어야 합니다.');
      return;
    }
    
    // 4. BitWish 주소 형식 검증
    const bitwishAddressRegex = /^BW[A-F0-9]{40}$/;
    if (!bitwishAddressRegex.test(walletAddress)) {
      setError('올바른 BitWish 지갑 주소 형식이 아닙니다.');
      return;
    }
    
    // 5. 비밀번호 해싱 (기존 + 새 비밀번호)
    const hashedCurrentPassword = await hashPassword(currentPassword);
    const hashedNewPassword = await hashPassword(newPassword);
    
    // 6. 백엔드 API 호출 - 비밀번호 변경
    const response = await fetch('http://localhost:4001/api/bitwish/integrated/wallet/change-password', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        walletAddress: walletAddress,
        currentPassword: hashedCurrentPassword,  // ✅ 해시된 비밀번호
        newPassword: hashedNewPassword           // ✅ 해시된 비밀번호
      })
    });
    
    const result = await response.json();
    
    if (result.success) {
      setSuccess('비밀번호가 성공적으로 변경되었습니다.');
      
      // 7. 입력 필드 초기화
      setCurrentPassword('');
      setNewPassword('');
      setConfirmNewPassword('');
      
      // 8. 2초 후 모달 닫기
      setTimeout(() => {
        handleClosePasswordChangeModal();
      }, 2000);
    }
  } catch (error) {
    setError('비밀번호 변경 중 오류가 발생했습니다.');
  } finally {
    setIsChangingPassword(false);
  }
};
```

**보안 특징:**
- ✅ 기존 비밀번호 검증 필수
- ✅ 평문 비밀번호 절대 전송 안 함
- ✅ 양방향 해싱 (기존 + 새 비밀번호)
- ✅ 서버 측 기존 비밀번호 재검증
