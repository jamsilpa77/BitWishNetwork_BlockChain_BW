## 🚨 **4. 누락된 구현 항목**

### **4-1. 홈페이지 메인화면 통계 API (❌ 미구현)**

**.md 요구사항 (6번, 6-3, 6-4, 6-5번)**:
- 총 발행량: 210억 BW
- 현재 발행량: 실시간 합계
- 잔여 발행량: 210억 - 현재 발행량
- 발행률: (현재 발행량 / 210억) × 100%
- 15~30초마다 자동 새로고침

**메모장 상태**: ❌ 완전 누락

**보완 필요**:
```javascript
// 추가 필요: GET /api/admin/bitwish/stats/homepage
app.get('/api/admin/bitwish/stats/homepage', async (req, res) => {
  const miningCollection = db.collection('admin_user_mining_data');
  
  // 모든 회원의 totalMined 합계
  const allUsers = await miningCollection.find({}).toArray();
  const currentSupply = allUsers.reduce((sum, user) => 
    sum + parseFloat(user.totalMined.amount), 0
  );
  
  const TOTAL_SUPPLY = 21000000000; // 210억
  const remainingSupply = TOTAL_SUPPLY - currentSupply;
  const issueRate = (currentSupply / TOTAL_SUPPLY * 100).toFixed(2);
  
  res.json({
    totalSupply: TOTAL_SUPPLY,
    currentSupply: currentSupply.toFixed(8),
    remainingSupply: remainingSupply.toFixed(8),
    issueRate: `${issueRate}%`,
    totalBlocks: await db.collection('admin_block_records').countDocuments(),
    totalWallets: await db.collection('bitwish_wallets').countDocuments()
  });
});
```

## **1-1. Backend API (simple-server.cjs 추가)**

```javascript
// ========================================
// 📊 홈페이지 메인화면 통계 API (최종 완벽 버전)
// ========================================
app.get('/api/bitwish/stats/homepage', async (req, res) => {
  try {
    const db = await getMongoDBConnection();
    
    // 1. 총 발행량 (고정)
    const TOTAL_SUPPLY = 21000000000; // 210억 BW
    const SYSTEM_WALLETS_COUNT = 5; // 초기 시스템 지갑
    
    // 2. ✅ 현재 발행량 (모든 유저의 totalMined 합계 - isActive 조건 제거)
    const miningCollection = db.collection('admin_user_mining_data');
    const allUsers = await miningCollection.find({}).toArray(); // ✅ 활성 여부 무관
    
    let currentSupply = 0;
    for (const user of allUsers) {
      const amount = parseFloat(user.totalMined?.amount || 0);
      if (!isNaN(amount) && isFinite(amount)) {
        currentSupply += amount;
      } else {
        console.warn(`⚠️ 잘못된 채굴량 데이터: ${user.walletAddress}`);
      }
    }
    
    // 3. 잔여 발행량
    const remainingSupply = TOTAL_SUPPLY - currentSupply;
    
    // 4. 발행률 (%)
    const issueRate = currentSupply > 0 
      ? (currentSupply / TOTAL_SUPPLY * 100).toFixed(2) 
      : "0.00";
    
    // 5. 총 블록 수
    const blockCollection = db.collection('admin_block_records');
    const totalBlocks = await blockCollection.countDocuments();
    
    // 6. ✅ 지갑 생성 수 (시스템 지갑 포함)
    const walletCollection = db.collection('bitwish_wallets');
    const userWalletsCount = await walletCollection.countDocuments({
      address: { $regex: /^BW[A-F0-9]{40}$/ } // BW + 40자리 16진수 (사용자 지갑만)
    });
    const totalWallets = SYSTEM_WALLETS_COUNT + userWalletsCount;
    
    res.json({
      success: true,
      data: {
        totalSupply: {
          value: TOTAL_SUPPLY,
          formatted: "21,000,000,000 BW",
          display: "210억 BW"
        },
        currentSupply: {
          value: currentSupply.toFixed(8),
          formatted: `${currentSupply.toLocaleString('ko-KR', { 
            minimumFractionDigits: 2, 
            maximumFractionDigits: 2 
          })} BW`,
          raw: currentSupply
        },
        remainingSupply: {
          value: remainingSupply.toFixed(8),
          formatted: `${remainingSupply.toLocaleString('ko-KR', { 
            minimumFractionDigits: 2, 
            maximumFractionDigits: 2 
          })} BW`,
          raw: remainingSupply
        },
        issueRate: {
          value: issueRate,
          formatted: `${issueRate}%`,
          percentage: parseFloat(issueRate)
        },
        totalBlocks: {
          value: totalBlocks,
          formatted: `${totalBlocks.toLocaleString('ko-KR')}개`,
          raw: totalBlocks
        },
        totalWallets: {
          value: totalWallets,
          formatted: `${totalWallets.toLocaleString('ko-KR')}개`,
          raw: totalWallets,
          breakdown: {
            system: SYSTEM_WALLETS_COUNT,
            user: userWalletsCount
          }
        },
        lastUpdated: new Date().toISOString()
      }
    });
    
  } catch (error) {
    console.error('❌ 홈페이지 통계 조회 오류:', error);
    res.status(500).json({
      success: false,
      message: '통계 조회 중 오류가 발생했습니다.',
      error: error.message
    });
  }
});
```

## **1-2. Frontend Component (src/pages/HomePage.tsx 생성)**

```typescript
// ========================================
// 📊 홈페이지 메인화면 통계 표시 (최종 완벽 버전)
// ========================================
import React, { useState, useEffect } from 'react';

interface HomepageStats {
  totalSupply: { value: number; formatted: string; display: string };
  currentSupply: { value: string; formatted: string; raw: number };
  remainingSupply: { value: string; formatted: string; raw: number };
  issueRate: { value: string; formatted: string; percentage: number };
  totalBlocks: { value: number; formatted: string; raw: number };
  totalWallets: { 
    value: number; 
    formatted: string; 
    raw: number; 
    breakdown: { system: number; user: number } 
  };
  lastUpdated: string;
}

const HomePage: React.FC = () => {
  const [stats, setStats] = useState<HomepageStats | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  // 통계 데이터 로드
  const loadStats = async () => {
    try {
      setError(null);
      const response = await fetch('http://localhost:4001/api/bitwish/stats/homepage');
      const result = await response.json();
      
      if (result.success) {
        setStats(result.data);
        console.log('✅ 통계 업데이트:', new Date().toLocaleTimeString());
      } else {
        setError(result.message || '통계를 불러올 수 없습니다.');
      }
    } catch (err) {
      console.error('❌ 통계 로드 오류:', err);
      setError('서버와 연결할 수 없습니다.');
    } finally {
      setLoading(false);
    }
  };

  // 초기 로드
  useEffect(() => {
    loadStats();
  }, []);

  // ✅ 15~30초마다 자동 새로고침 (20초)
  useEffect(() => {
    const interval = setInterval(() => {
      loadStats();
    }, 20000); // 20초
    
    return () => clearInterval(interval);
  }, []);

  if (loading) {
    return (
      <div className="flex items-center justify-center min-h-screen">
        <div className="text-center">
          <div className="animate-spin rounded-full h-16 w-16 border-b-2 border-blue-500 mx-auto mb-4"></div>
          <p className="text-gray-600">통계 로딩 중...</p>
        </div>
      </div>
    );
  }

  if (error) {
    return (
      <div className="flex items-center justify-center min-h-screen">
        <div className="bg-red-50 border border-red-200 rounded-lg p-6 max-w-md">
          <h3 className="text-red-800 font-bold mb-2">⚠️ 오류 발생</h3>
          <p className="text-red-600">{error}</p>
          <button 
            onClick={loadStats}
            className="mt-4 bg-red-600 text-white px-4 py-2 rounded hover:bg-red-700"
          >
            다시 시도
          </button>
        </div>
      </div>
    );
  }

  if (!stats) {
    return (
      <div className="flex items-center justify-center min-h-screen">
        <p className="text-gray-600">통계 데이터가 없습니다.</p>
      </div>
    );
  }

  return (
    <div className="min-h-screen bg-gradient-to-br from-gray-50 to-gray-100 py-12 px-4">
      <div className="container mx-auto max-w-7xl">
        
        {/* 헤더 */}
        <div className="text-center mb-12">
          <h1 className="text-4xl font-bold text-gray-800 mb-2">
            🚀 BitWish Network
          </h1>
          <p className="text-gray-600">
            실시간 블록체인 통계 (20초마다 자동 업데이트)
          </p>
        </div>

        {/* 메인 통계 그리드 */}
        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6 mb-8">
          
          {/* 총 발행량 */}
          <div className="bg-gradient-to-br from-blue-500 to-blue-700 text-white rounded-xl p-6 shadow-xl hover:shadow-2xl transition-shadow">
            <div className="flex items-center justify-between mb-4">
              <div className="text-sm font-medium opacity-90">총 발행량</div>
              <div className="text-3xl">💎</div>
            </div>
            <div className="text-3xl font-bold mb-1">{stats.totalSupply.display}</div>
            <div className="text-xs opacity-75">Fixed Total Supply</div>
          </div>

          {/* 현재 발행량 */}
          <div className="bg-gradient-to-br from-green-500 to-green-700 text-white rounded-xl p-6 shadow-xl hover:shadow-2xl transition-shadow">
            <div className="flex items-center justify-between mb-4">
              <div className="text-sm font-medium opacity-90">현재 발행량</div>
              <div className="text-3xl">⛏️</div>
            </div>
            <div className="text-3xl font-bold mb-1">
              {stats.currentSupply.formatted}
            </div>
            <div className="text-xs opacity-75">
              실시간 채굴량 (활성/비활성 포함)
            </div>
          </div>

          {/* 잔여 발행량 */}
          <div className="bg-gradient-to-br from-purple-500 to-purple-700 text-white rounded-xl p-6 shadow-xl hover:shadow-2xl transition-shadow">
            <div className="flex items-center justify-between mb-4">
              <div className="text-sm font-medium opacity-90">잔여 발행량</div>
              <div className="text-3xl">🎁</div>
            </div>
            <div className="text-3xl font-bold mb-1">
              {stats.remainingSupply.formatted}
            </div>
            <div className="text-xs opacity-75">Remaining Supply</div>
          </div>

          {/* 발행률 */}
          <div className="bg-gradient-to-br from-orange-500 to-orange-700 text-white rounded-xl p-6 shadow-xl hover:shadow-2xl transition-shadow">
            <div className="flex items-center justify-between mb-4">
              <div className="text-sm font-medium opacity-90">발행률</div>
              <div className="text-3xl">📊</div>
            </div>
            <div className="text-3xl font-bold mb-1">{stats.issueRate.formatted}</div>
            <div className="text-xs opacity-75">
              진행률: {stats.issueRate.percentage.toFixed(4)}%
            </div>
          </div>

          {/* 총 블록 */}
          <div className="bg-gradient-to-br from-red-500 to-red-700 text-white rounded-xl p-6 shadow-xl hover:shadow-2xl transition-shadow">
            <div className="flex items-center justify-between mb-4">

              <div className="text-3xl">🧱</div>

export default HomePage;
```

## **2-1. Custom Hook (src/hooks/useAutoRefresh.ts 생성)**

```typescript
// ========================================
// 🔄 자동 새로고침 커스텀 훅 (최종 완벽 버전)
// ========================================
import { useEffect, useRef } from 'react';

/**
 * 자동 새로고침 커스텀 훅
 * @param callback - 실행할 함수
 * @param intervalMs - 새로고침 간격 (밀리초)
 * @param enabled - 활성화 여부
 */
export const useAutoRefresh = (
  callback: () => void | Promise<void>,
  intervalMs: number = 20000, // 기본 20초
  enabled: boolean = true
) => {
  const savedCallback = useRef<() => void | Promise<void>>();

  // ✅ 콜백 최신화
  useEffect(() => {
    savedCallback.current = callback;
  }, [callback]);

  // ✅ 자동 새로고침 설정
  useEffect(() => {
    if (!enabled) {
      console.log('🛑 자동 새로고침 비활성화됨');
      return;
    }
    
    const tick = async () => {
      if (savedCallback.current) {
        try {
          await savedCallback.current();
        } catch (error) {
          console.error('❌ 자동 새로고침 오류:', error);
        }
      }
    };

    const id = setInterval(tick, intervalMs);
    
    console.log(`✅ 자동 새로고침 시작: ${intervalMs / 1000}초마다`);
    
    return () => {
      clearInterval(id);
      console.log('🛑 자동 새로고침 중지');
    };
  }, [intervalMs, enabled]);
};

/**
 * 사용 예시:
 * 
 * import { useAutoRefresh } from '@/hooks/useAutoRefresh';
 * 
 * const MyComponent = () => {
 *   const [data, setData] = useState(null);
 *   
 *   const fetchData = async () => {
 *     const response = await fetch('/api/data');
 *     const result = await response.json();
 *     setData(result);
 *   };
 *   
 *   // 초기 로드
 *   useEffect(() => {
 *     fetchData();
 *   }, []);
 *   
 *   // 20초마다 자동 새로고침
 *   useAutoRefresh(fetchData, 20000);
 *   
 *   return <div>{JSON.stringify(data)}</div>;
 * };
 */
```

# 🔥 **Phase 3: P2P 거래 수수료 시스템 (최종 완벽 버전)**

## **3-1. Backend API (simple-server.cjs 추가)**

```javascript
// ========================================
// 💸 P2P 송금 API (수수료 0.01 BW 포함 - 최종 완벽 버전)
// ========================================
app.post('/api/bitwish/wallet/send-with-fee', async (req, res) => {
  try {
    const { senderAddress, recipientAddress, amount, otpCode } = req.body;
    
    // 1. 입력 검증
    if (!senderAddress || !recipientAddress || !amount) {
      return res.status(400).json({
        success: false,
        message: '필수 정보가 누락되었습니다.'
      });
    }
    
    if (senderAddress === recipientAddress) {
      return res.status(400).json({
        success: false,
        message: '자기 자신에게 송금할 수 없습니다.'
      });
    }
    
    const sendAmount = parseFloat(amount);
    if (sendAmount <= 0) {
      return res.status(400).json({
        success: false,
        message: '송금액은 0보다 커야 합니다.'
      });
    }
    
    const FEE_AMOUNT = 0.01; // 0.01 BW 고정 수수료
    const totalAmount = sendAmount + FEE_AMOUNT;
    
    const db = await getMongoDBConnection();
    const miningCollection = db.collection('admin_user_mining_data');
    const feeCollection = db.collection('admin_fee_collection');
    const systemWalletsCollection = db.collection('admin_system_wallets');
    const transactionCollection = db.collection('bitwish_transactions');
    
    // 2. ✅ admin_user_mining_data에서 보내는 사람 확인
    const senderData = await miningCollection.findOne({ walletAddress: senderAddress });
    if (!senderData) {
      return res.status(404).json({
        success: false,
        message: '보내는 지갑을 찾을 수 없습니다.'
      });
    }
    
    // 3. ✅ 실제 BW 잔액 확인 (마이그레이션 완료된 금액만)
    const availableBalance = parseFloat(senderData.walletBalance?.availableAmount?.amount || 0);
    
    if (availableBalance === 0) {
      return res.status(400).json({
        success: false,
        message: 'KYC 승인 후 마이그레이션 완료된 실제 BW만 송금 가능합니다.',
        hint: '현재 가상 BW는 송금할 수 없습니다.'
      });
    }
    
    if (availableBalance < totalAmount) {
      return res.status(400).json({
        success: false,
        message: `사용 가능한 잔액이 부족합니다.`,
        required: totalAmount.toFixed(8),
        available: availableBalance.toFixed(8),
        breakdown: {
          sendAmount: sendAmount.toFixed(8),
          fee: FEE_AMOUNT.toFixed(8)
        }
      });
    }
    
    // 4. ✅ 받는 사람 확인
    const recipientData = await miningCollection.findOne({ walletAddress: recipientAddress });
    if (!recipientData) {
      return res.status(404).json({
        success: false,
        message: '받는 지갑을 찾을 수 없습니다.'
      });
    }
    
    // 5. OTP 검증 (선택사항)
    // TODO: OTP 검증 로직 추가
    
    const now = new Date();
    const nowString = now.toISOString().replace('T', ' ').substring(0, 19);
    const transactionId = `TX-P2P-${Date.now()}`;
    
    // 6. 트랜잭션 실행 (MongoDB Transaction 사용)
    const session = db.client.startSession();
    
    try {
      await session.withTransaction(async () => {
        // 6-1. ✅ 보내는 사람 availableAmount 차감
        await miningCollection.updateOne(
          { walletAddress: senderAddress },
          { 
            $inc: { 
              'walletBalance.availableAmount.amount': -totalAmount 
            },
            $set: { updatedAt: now }
          },
          { session }
        );
        
        // 6-2. ✅ 받는 사람 availableAmount 증가
        await miningCollection.updateOne(
          { walletAddress: recipientAddress },
          { 
            $inc: { 
              'walletBalance.availableAmount.amount': sendAmount 
            },
            $set: { updatedAt: now }
          },
          { session }
        );
        
        // 6-3. 수수료 분배 (70% + 30%)
        const ecosystemFee = FEE_AMOUNT * 0.7; // 0.007 BW
        const devFee = FEE_AMOUNT * 0.3;       // 0.003 BW
        
        // 생태계 지갑에 수수료 입금
        await systemWalletsCollection.updateOne(
          { walletType: 'ECOSYSTEM_FEE_70' },
          { 
            $inc: { 'balance.current': ecosystemFee },
            $push: {
              transactions: {
                transactionId: transactionId,
                type: 'fee_collection',
                amount: ecosystemFee.toFixed(8),
                source: `P2P-${senderAddress.substring(0, 10)}`,
                purpose: 'P2P 거래 수수료 (70%)',
                timestamp: nowString,
                status: 'completed'
              }
            },
            $set: { updatedAt: now }
          },
          { session }
        );
        
        // 개발팀 지갑에 수수료 입금
        await systemWalletsCollection.updateOne(
          { walletType: 'DEV_FEE_30' },
          { 
            $inc: { 'balance.current': devFee },
            $push: {
              transactions: {
                transactionId: `${transactionId}-DEV`,
                type: 'fee_collection',
                amount: devFee.toFixed(8),
                source: `P2P-${senderAddress.substring(0, 10)}`,
                purpose: 'P2P 거래 수수료 (30%)',
                timestamp: nowString,
                status: 'completed'
              }
            },
            $set: { updatedAt: now }
          },
          { session }
        );
        
        // 6-4. 수수료 기록 저장
        await feeCollection.insertOne({
          feeType: 'P2P_TRANSACTION',
          feeInfo: {
            amount: FEE_AMOUNT.toFixed(8),
            sourceType: 'transaction',
            sourceId: transactionId,
            sender: senderAddress,
            recipient: recipientAddress
          },
          distribution: {
            ecosystemShare: {
              percentage: 70,
              amount: ecosystemFee.toFixed(8),
              status: 'completed'
            },
            devTeamShare: {
              percentage: 30,
              amount: devFee.toFixed(8),
              status: 'completed'
            }
          },
          collectedAt: now,
          distributedAt: now
        }, { session });
        
        // 6-5. ✅ 거래 내역 저장
        await transactionCollection.insertOne({
          transactionId: transactionId,
          type: 'P2P_TRANSFER',
          sender: senderAddress,
          recipient: recipientAddress,
          amount: sendAmount.toFixed(8),
          fee: FEE_AMOUNT.toFixed(8),
          totalDeducted: totalAmount.toFixed(8),
          status: 'completed',
          timestamp: nowString,
          createdAt: now
        }, { session });
        
        console.log(`✅ P2P 거래 완료: ${senderAddress.substring(0, 10)}... → ${recipientAddress.substring(0, 10)}... (${sendAmount} BW + 수수료 ${FEE_AMOUNT} BW)`);
      });
      
      await session.endSession();
      
      res.json({
        success: true,
        message: 'P2P 송금이 완료되었습니다.',
        data: {
          transactionId: transactionId,
          sender: senderAddress,
          recipient: recipientAddress,
          amount: sendAmount.toFixed(8),
          fee: FEE_AMOUNT.toFixed(8),
          totalDeducted: totalAmount.toFixed(8),
          timestamp: nowString
        }
      });
      
    } catch (txError) {
      await session.endSession();
      throw txError;
    }
    
  } catch (error) {
    console.error('❌ P2P 송금 오류:', error);
    res.status(500).json({
      success: false,
      message: 'P2P 송금 중 오류가 발생했습니다.',
      error: error.message
    });
  }
});
```

## **4-2. Backend API - 가맹점 승인 (simple-server.cjs 추가)**

```javascript
// ========================================
// ✅ 가맹점 승인 API - 즉시 지급 (최종 완벽 버전)
// ========================================
app.post('/api/admin/bitwish/partner/approve', async (req, res) => {
  try {
    const { walletAddress, approvalAmount } = req.body;
    
    if (!walletAddress) {
      return res.status(400).json({
        success: false,
        message: '지갑 주소가 필요합니다.'
      });
    }
    
    const db = await getMongoDBConnection();
    const partnerCollection = db.collection('admin_partner_stores');
    const systemWalletsCollection = db.collection('admin_system_wallets');
    const miningCollection = db.collection('admin_user_mining_data');
    
    // 1. 가맹점 정보 확인
    const partner = await partnerCollection.findOne({ walletAddress });
    if (!partner) {
      return res.status(404).json({
        success: false,
        message: '가맹점을 찾을 수 없습니다.'
      });
    }
    
    if (partner.status === 'approved') {
      return res.status(400).json({
        success: false,
        message: '이미 승인된 가맹점입니다.'
      });
    }
    
    const now = new Date();
    const nowString = now.toISOString().replace('T', ' ').substring(0, 19);
    
    // 2. ✅ PARTNER_10PERCENT 지갑 확인
    const partnerSystemWallet = await systemWalletsCollection.findOne({ 
      walletType: 'PARTNER_10PERCENT' 
    });
    
    if (!partnerSystemWallet) {
      return res.status(500).json({
        success: false,
        message: 'PARTNER_10PERCENT 시스템 지갑을 찾을 수 없습니다.'
      });
    }
    
    const approveAmount = parseFloat(approvalAmount || 1000); // 기본 1000 BW
    const currentBalance = parseFloat(partnerSystemWallet.balance?.current || 0);
    
    if (currentBalance < approveAmount) {
      return res.status(400).json({
        success: false,
        message: `PARTNER_10PERCENT 지갑 잔액이 부족합니다. (필요: ${approveAmount} BW, 현재: ${currentBalance} BW)`
      });
    }
    
    // 3. ✅ 트랜잭션 실행 (MongoDB Transaction 사용)
    const session = db.client.startSession();
    
    try {
      await session.withTransaction(async () => {
        // 3-1. 가맹점 상태 업데이트
        await partnerCollection.updateOne(
          { walletAddress },
          {
            $set: {
              status: 'approved',
              approvedAt: nowString,
              'partnerWallet.initial': approveAmount.toFixed(8),
              'partnerWallet.current': approveAmount.toFixed(8),
              updatedAt: now
            },
            $push: {
              transactions: {
                transactionId: `TX-PARTNER-INIT-${Date.now()}`,
                type: 'initial_distribution',
                amount: approveAmount.toFixed(8),
                source: 'PARTNER_10PERCENT_WALLET',
                purpose: '가맹점 승인 즉시 지급',
                timestamp: nowString,
                status: 'completed'
              }
            }
          },
          { session }
        );
        
        // 3-2. ✅ 회원의 admin_user_mining_data에도 파트너 보너스 추가
        await miningCollection.updateOne(
          { walletAddress },
          {
            $inc: { 
              'walletBalance.availableAmount.amount': approveAmount 
            },
            $set: { 
              isPartner: true,
              partnerApprovedAt: nowString,
              updatedAt: now 
            }
          },
          { session }
        );
        
        // 3-3. PARTNER_10PERCENT 지갑 잔액 차감
        await systemWalletsCollection.updateOne(
          { walletType: 'PARTNER_10PERCENT' },
          {
            $inc: { 'balance.current': -approveAmount },
            $push: {
              transactions: {
                transactionId: `TX-PARTNER-OUT-${Date.now()}`,
                type: 'distribution',
                amount: approveAmount.toFixed(8),
                recipient: walletAddress,
                recipientName: partner.businessInfo.businessName,
                purpose: '가맹점 승인 지급',
                timestamp: nowString,
                status: 'completed'
              }
            },
            $set: { updatedAt: now }
          },
          { session }
        );
        
        console.log(`✅ 가맹점 승인 완료: ${partner.businessInfo.businessName} (${approveAmount} BW 즉시 지급)`);
      });
      
      await session.endSession();
      
      res.json({
        success: true,
        message: '가맹점 승인이 완료되었습니다.',
        data: {
          walletAddress: walletAddress,
          businessName: partner.businessInfo.businessName,
          approvedAmount: approveAmount.toFixed(8),
          approvedAt: nowString
        }
      });
      
    } catch (txError) {
      await session.endSession();
      throw txError;
    }
    
  } catch (error) {
    console.error('❌ 가맹점 승인 오류:', error);
    res.status(500).json({
      success: false,
      message: '가맹점 승인 중 오류가 발생했습니다.',
      error: error.message
    });
  }
});
```

```typescript
// ========================================
import React, { useEffect, useState } from 'react';
import { BrowserRouter as Router, Routes, Route, Navigate } from 'react-router-dom';

// 사용자 페이지
import UserPersonalMiningPage from './pages/UserPersonalMiningPage';
import MyWalletPage from './pages/MyWalletPage';
import HomePage from './pages/HomePage';

// 관리자 페이지
import AdminPersonalMiningPage from './pages/AdminPersonalMiningPage';
import AdminSettingsPage from './pages/AdminSettingsPage';

function App() {
  const [isAdmin, setIsAdmin] = useState(false);
  
  // ✅ URL 파라미터 확인 (?admin=true)
  useEffect(() => {
    const checkAdminMode = () => {
      const urlParams = new URLSearchParams(window.location.search);
      const adminParam = urlParams.get('admin');
      
      if (adminParam === 'true') {
        setIsAdmin(true);
        console.log('🔐 관리자 모드 활성화');
      } else {
        setIsAdmin(false);
        console.log('👤 일반 사용자 모드');
      }
    };
    
    checkAdminMode();
    
    // URL 변경 감지
    window.addEventListener('popstate', checkAdminMode);
    
    return () => {
      window.removeEventListener('popstate', checkAdminMode);
    };
  }, []);

  return (
    <Router>
      <Routes>
        {/* 홈페이지 (공통) */}
        <Route path="/" element={<HomePage />} />
        
        {/* ✅ 조건부 라우팅: 관리자 vs 사용자 */}
        {isAdmin ? (
          <>
            {/* 관리자 전용 라우팅 */}
            <Route path="/mining" element={<AdminPersonalMiningPage />} />
            <Route path="/settings" element={<AdminSettingsPage />} />
            <Route path="/wallet" element={<Navigate to="/mining" replace />} />
          </>
        ) : (
          <>
            {/* 사용자 전용 라우팅 */}
            <Route path="/mining" element={<UserPersonalMiningPage />} />
            <Route path="/wallet" element={<MyWalletPage />} />
            <Route path="/settings" element={<Navigate to="/wallet" replace />} />
          </>
        )}
        
        {/* 404 페이지 */}
        <Route path="*" element={<Navigate to="/" replace />} />
      </Routes>
    </Router>
  );
}

export default App;
```

### **8-1. Backend API - 마이닝 시작 시 블록 자동 생성 (simple-server.cjs 수정)**

```javascript
// ⛏️ 마이닝 시작 API - 블록 생성 연동 (완벽 버전)
// ========================================
app.post('/api/admin/bitwish/mining/start', async (req, res) => {
  try {
    const { walletAddress } = req.body;

      return res.status(400).json({

              isActive: true,
              startedAt: nowString,

      success: false,
      message: '마이닝 시작 중 오류가 발생했습니다.',
      error: error.message
    });
  }
});

// ✅ 블록 해시 생성 함수 (간단 버전)
function generateBlockHash(blockHeight, timestamp) {
  const crypto = require('crypto');
  const data = `${blockHeight}-${timestamp}-${Date.now()}`;
  return crypto.createHash('sha256').update(data).digest('hex');
}
```

## 🔥 **Phase 9: 월별 채굴량 보관 시스템**

### **9-1. Backend API - 월별 채굴량 자동 저장 (simple-server.cjs 추가)**

```javascript
// ========================================
// 📦 월별 채굴량 보관 시스템
// ========================================

// ✅ 마이닝 업데이트 시 월별 저장 로직 추가
app.post('/api/admin/bitwish/mining/update', async (req, res) => {
  try {
    const { walletAddress } = req.body;
    
    if (!walletAddress) {
      return res.status(400).json({
        success: false,
        message: '지갑 주소가 필요합니다.'
      });
    }
    
    const db = await getMongoDBConnection();
    const miningCollection = db.collection('admin_user_mining_data');
    
    const userData = await miningCollection.findOne({ walletAddress });
    
    if (!userData || !userData.miningStatus?.isActive) {
      return res.status(400).json({
        success: false,
        message: '마이닝이 활성화되지 않았습니다.'
      });
    }
    
    const now = new Date();
    const startedAt = new Date(userData.miningStatus.startedAt);
    const lastUpdated = userData.totalMined.lastUpdated 
      ? new Date(userData.totalMined.lastUpdated) 
      : startedAt;
    
    // 경과 시간 계산 (시간 단위)
    const elapsedMs = now - lastUpdated;
    const elapsedHours = elapsedMs / (1000 * 60 * 60);
    
    // 채굴량 계산
    const baseRate = parseFloat(userData.miningRate.baseRate || 0.25);
    const minedAmount = elapsedHours * baseRate;
    
    const currentTotal = parseFloat(userData.totalMined.amount || 0);
    const newTotal = currentTotal + minedAmount;
    
    // ✅ 월별 저장 로직
    const currentYear = now.getFullYear();
    const currentMonth = now.getMonth() + 1; // 1-12
    const monthKey = `${currentYear}-${String(currentMonth).padStart(2, '0')}`; // "2025-01"
    
    // 월말 날짜 계산
    const monthEnd = new Date(currentYear, currentMonth, 0, 23, 59, 59); // 해당 월 마지막 날
    
    // 기존 월별 저장소 확인
    const monthlyStorage = userData.monthlyStorage || [];
    let currentMonthData = monthlyStorage.find(m => m.month === monthKey);
    
    if (!currentMonthData) {
      // 새 월 데이터 생성
      currentMonthData = {
        month: monthKey,
        year: currentYear,
        monthNumber: currentMonth,
        startDate: new Date(currentYear, currentMonth - 1, 1).toISOString().replace('T', ' ').substring(0, 19),
        endDate: monthEnd.toISOString().replace('T', ' ').substring(0, 19),
        minedAmount: "0.00000000",
        isClosed: false,
        closedAt: null,
        migratedAmount: "0.00000000",
        isMigrated: false,
        migratedAt: null
      };
      monthlyStorage.push(currentMonthData);
    }
    
    // 현재 월 채굴량 업데이트
    const monthMinedAmount = parseFloat(currentMonthData.minedAmount || 0) + minedAmount;
    currentMonthData.minedAmount = monthMinedAmount.toFixed(8);
    
    // ✅ 월말 체크 (자동 마감)
    if (now > monthEnd && !currentMonthData.isClosed) {
      currentMonthData.isClosed = true;
      currentMonthData.closedAt = monthEnd.toISOString().replace('T', ' ').substring(0, 19);
      console.log(`📦 월별 보관 마감: ${walletAddress.substring(0, 10)}... → ${monthKey} (${monthMinedAmount.toFixed(8)} BW)`);
    }
    
    // DB 업데이트
    await miningCollection.updateOne(
      { walletAddress },
      {
        $set: {
          'totalMined.amount': newTotal.toFixed(8),
          'totalMined.lastUpdated': now.toISOString().replace('T', ' ').substring(0, 19),
          monthlyStorage: monthlyStorage,
          updatedAt: now
        }
      }
    );
    
    res.json({
      success: true,
      message: '채굴량이 업데이트되었습니다.',
      data: {
        walletAddress: walletAddress,
        minedAmount: minedAmount.toFixed(8),
        totalMined: newTotal.toFixed(8),
        currentMonth: {
          month: monthKey,
          monthlyMined: monthMinedAmount.toFixed(8),
          isClosed: currentMonthData.isClosed
        }
      }
    });
    
  } catch (error) {
    console.error('❌ 마이닝 업데이트 오류:', error);
    res.status(500).json({
      success: false,
      message: '마이닝 업데이트 중 오류가 발생했습니다.',
      error: error.message
    });
  }
});
```

```javascript
    const { walletAddress } = req.body;
    
    if (!walletAddress) {
      return res.status(400).json({
        success: false,
        message: '지갑 주소가 필요합니다.'

    if (!userData) {

        message: '사용자를 찾을 수 없습니다.'
      });

        success: false,
        message: 'KYC 승인이 필요합니다.'
      });
    }
    
    const kycApprovedDate = new Date(kycData.approvedAt);
    const now = new Date();
    
    // 3. ✅ 15일 경과 확인
    const daysSinceApproval = (now - kycApprovedDate) / (1000 * 60 * 60 * 24);
    
    if (daysSinceApproval < 15) {
      return res.status(400).json({
        success: false,
        message: `KYC 승인 후 15일이 지나야 마이그레이션이 가능합니다. (현재: ${Math.floor(daysSinceApproval)}일 경과)`
      });
    }
    
    // 4. ✅ 마이그레이션할 월 찾기 (마감되었지만 미마이그레이션된 가장 오래된 월)
    const monthlyStorage = userData.monthlyStorage || [];
    const eligibleMonths = monthlyStorage.filter(m => 
      m.isClosed && !m.isMigrated && parseFloat(m.minedAmount) > 0
    );
    
    if (eligibleMonths.length === 0) {
      return res.status(400).json({
        success: false,
        message: '마이그레이션 가능한 월별 채굴량이 없습니다.'
      });
    }
    
    // 가장 오래된 월 선택 (순차 마이그레이션)
    eligibleMonths.sort((a, b) => a.month.localeCompare(b.month));
    const targetMonth = eligibleMonths[0];
    const migrateAmount = parseFloat(targetMonth.minedAmount);
    
    // 5. ✅ 마지막 마이그레이션 날짜 확인 (매월 동일한 날짜에 마이그레이션)
    const lastMigration = monthlyStorage.find(m => m.isMigrated);
    let nextMigrationDate;
    
    if (lastMigration) {
      const lastMigrationDate = new Date(lastMigration.migratedAt);
      const dayOfMonth = kycApprovedDate.getDate(); // KYC 승인 날짜의 일

      });

    const transactionId = `TX-MIGRATION-${targetMonth.month}-${Date.now()}`;
    
    const session = db.client.startSession();

          { walletAddress },
          {
            $inc: { 
              'walletBalance.availableAmount.amount': migrateAmount 
            },
            $set: {
              [`monthlyStorage.$[elem].isMigrated`]: true,
              [`monthlyStorage.$[elem].migratedAmount`]: migrateAmount.toFixed(8),
              [`monthlyStorage.$[elem].migratedAt`]: nowString,
              updatedAt: now
            }

          { 
            arrayFilters: [{ 'elem.month': targetMonth.month }],
            session

        await systemWalletsCollection.updateOne(

                timestamp: nowString,
                status: 'completed'
```

```javascript
              'kycStatus.isEligibleForMigration': false, // 15일 후 true로 변경
              updatedAt: now
            }
          },
          { session }
        );
        
        console.log(`✅ KYC 승인 완료: ${walletAddress.substring(0, 10)}... (마이그레이션 예정: ${migrationDateString})`);
      });
      
      await session.endSession();
      
      res.json({
        success: true,
        message: 'KYC가 승인되었습니다.',

          walletAddress: walletAddress,

          migrationEligibleDate: migrationDateString,
          daysUntilMigration: 15

    } catch (txError) {
      await session.endSession();
      throw txError;

      success: false,
```

## 🔥 **Phase 12: 추천 코드 자동 생성 및 마이닝 시작 시간 완벽 저장**

### **12-1. Backend API - 지갑 생성 시 추천 코드 자동 생성 (simple-server.cjs 수정)**

```javascript
// ========================================
// 👛 지갑 생성 API - 추천 코드 자동 생성 포함
// ========================================
app.post('/api/bitwish/wallet/create', async (req, res) => {
  try {
    const { username, password, referralCode } = req.body;
    
    // 입력 검증... (기존 로직)
    
    const db = await getMongoDBConnection();
    const walletCollection = db.collection('bitwish_wallets');
    const referralCollection = db.collection('admin_referral_codes');
    const miningCollection = db.collection('admin_user_mining_data');
    
    // 지갑 생성... (기존 로직)
    const newWalletAddress = generateWalletAddress(); // BW + 40자리
    
    // ✅ 추천 코드 자동 생성
    const myReferralCode = generateReferralCode(newWalletAddress);
    
    const now = new Date();
    const session = db.client.startSession();
    
    try {
      await session.withTransaction(async () => {
        // 1. 지갑 생성 (기존 로직)
        await walletCollection.insertOne({
          address: newWalletAddress,
          username: username,
          // ... 기존 필드들
          referralInfo: {
            myReferralCode: myReferralCode, // ✅ 자동 생성된 추천 코드
            usedReferralCode: referralCode || null,
            totalReferrals: 0,
            referrals: []
          },
          createdAt: now
        }, { session });
        
        // 2. ✅ 추천 코드 등록
        await referralCollection.insertOne({
          referralCode: myReferralCode,
          ownerWallet: newWalletAddress,
          isActive: true,
          totalUses: 0,
          users: [],
          createdAt: now
        }, { session });
        
        // 3. ✅ 추천인 보너스 처리 (추천 코드 사용 시)
        if (referralCode) {
          const referrerData = await referralCollection.findOne({ referralCode });
          
          if (referrerData) {
            // 추천인에게 1 BW 보너스
            await miningCollection.updateOne(
              { walletAddress: referrerData.ownerWallet },
              {
                $inc: {
                  'bonuses.referral.oneTimeBonus': 1.0,
                  'bonuses.referral.totalBonus': 1.0
                },
                $push: {
                  'referralInfo.referrals': {
                    referredWallet: newWalletAddress,
                    registeredAt: now.toISOString().replace('T', ' ').substring(0, 19),
                    bonusEarned: { amount: "1.00000000", type: 'one_time' }
                  }
                }
              },
              { session }
            );
            
            // 추천 코드 사용 횟수 증가
            await referralCollection.updateOne(
              { referralCode },
              {
                $inc: { totalUses: 1 },
                $push: {
                  users: {
                    walletAddress: newWalletAddress,
                    registeredAt: now.toISOString().replace('T', ' ').substring(0, 19)
                  }
                }
              },
              { session }
            );
            
            console.log(`✅ 추천 보너스 지급: ${referrerData.ownerWallet.substring(0, 10)}... → 1 BW`);
          }
        }
        
        console.log(`✅ 지갑 생성 완료: ${newWalletAddress} (추천 코드: ${myReferralCode})`);
      });
      
      await session.endSession();
      
      res.json({
        success: true,
        message: '지갑이 생성되었습니다.',
        data: {
          walletAddress: newWalletAddress,
          referralCode: myReferralCode
        }
      });
      
    } catch (txError) {
      await session.endSession();
      throw txError;
    }
    
  } catch (error) {
    console.error('❌ 지갑 생성 오류:', error);
    res.status(500).json({
      success: false,
      message: '지갑 생성 중 오류가 발생했습니다.',
      error: error.message
    });
  }
});

// ✅ 추천 코드 생성 함수
function generateReferralCode(walletAddress) {
  const crypto = require('crypto');
  const hash = crypto.createHash('sha256').update(walletAddress + Date.now()).digest('hex');
  return `REF-${hash.substring(0, 8).toUpperCase()}`; // REF-A1B2C3D4
}

// ✅ 지갑 주소 생성 함수
function generateWalletAddress() {
  const crypto = require('crypto');
  const randomBytes = crypto.randomBytes(20); // 20 bytes = 40 hex chars
  return `BW${randomBytes.toString('hex').toUpperCase()}`; // BW + 40자리
}
```
