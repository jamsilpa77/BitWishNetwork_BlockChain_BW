```typescript

const completeVerification = async () => {
  // 시드문구 해시만 생성
  const seedHash = await crypto.subtle.digest('SHA-256', 
    new TextEncoder().encode(seedPhrase.trim())
  );
  const seedHashHex = Array.from(new Uint8Array(seedHash))
    .map(b => b.toString(16).padStart(2, '0')).join('');
  
  const completeWalletData = {
    address: generatedWalletAddress,
    seedHash: seedHashHex, // 🚨 핵심: 시드문구 해시만 저장
    // seedPhrase는 저장하지 않음!
  };
  
  // 백엔드는 검증만 하고 시드문구 저장 안함

    body: JSON.stringify({
      seedPhrase: seedPhrase,  // 서버에서 주소 검증용만 사용
      userId: null
    })
  });
}
```

## 2️⃣ 2차 비밀번호 시스템 ⚠️ **정상이지만 오해 가능성 있음**

### 현재 상태:
- ✅ **2차 비밀번호 원본은 절대 저장되지 않음**

- ⚠️ 사용자가 "비밀번호가 저장되지 않는다"고 오해할 수 있으나, 실제로는 **해시가 저장되는 것이 정상**입니다

**코드 근거:**
```javascript
// simple-server.cjs (613-628줄)
async setPassword(address, password) {
  const hashedPassword = await this.passwordManager.hashPassword(password);
  
  wallet.hasPassword = true;
  wallet.passwordHash = hashedPassword; // 해시만 저장!
  wallet.passwordSetAt = new Date();
  
  // 원본 비밀번호는 저장 안됨
}
```

**4. API 호출 로직 수정 (HomePage.tsx):**

```typescript
// HomePage.tsx - fetchBlockchainStatus 함수 수정
const fetchBlockchainStatus = async () => {
  try {
    setIsLoadingStatus(true);
    
    const blockchainResponse = await fetch('http://localhost:4001/api/bitwish/blockchain/status');
    const economyResponse = await fetch('http://localhost:4001/tokenomics/total');
    
    if (blockchainResponse.ok && economyResponse.ok) {
      const blockchainData = await blockchainResponse.json();
      const economyData = await economyResponse.json();
      
      const totalSupply = 21000000000;
      const circulatingSupply = economyData?.circulatingSupply || 0;
      const remainingSupply = totalSupply - circulatingSupply;
      const issuancePercentage = circulatingSupply > 0 ? (circulatingSupply / totalSupply) * 100 : 0;
      
      setBlockchainStatus({
        totalBlocks: blockchainData?.data?.totalBlocks || 0,
        totalTokens: circulatingSupply.toString(),
        totalSupply: totalSupply,
        remainingSupply: remainingSupply,
        issuancePercentage: issuancePercentage,
        networkStatus: t('blockchain.connected'),
        lastUpdated: new Date().toISOString(),
        recentBlocks: [],
        lastUpdate: new Date(),
        totalWallets: blockchainData?.data?.totalWallets || 0  // 🔥 추가
      });
    }
  } catch (error) {
    console.error('블록체인 상태 조회 오류:', error);
  } finally {
    setIsLoadingStatus(false);
  }
};
```
