### **📋 요구사항 이해 및 분석**

#### **1. 관리자 마이닝 페이지 = 유저 마이닝 페이지 (기능 동일)**

// ====================================================================================
// 관리자 전용 테스트 API 엔드포인트들 - BitWish Network 완전 분리

// 관리자 전용 테스트 마이닝 시작
app.post('/api/admin/bitwish/test/mining/start', async (req, res) => {
  try {
    const { walletAddress } = req.body;
    
    // 관리자 전용 테스트 환경 확인
    if (!walletAddress || !walletAddress.startsWith('BW')) {
      return res.status(400).json({ 
        success: false, 
        error: '관리자 테스트용 지갑 주소가 필요합니다' 
      });
    }

    // 관리자 전용 독립 데이터베이스에 저장
    const db = bitWishMongoDB.db;
    const adminMiningCollection = db.collection('admin_test_mining_sessions');
    
    // 50자리 정밀도로 테스트 마이닝 세션 생성
    const Decimal = require('decimal.js');
    Decimal.set({ precision: 50 });
    
    const sessionData = {
      sessionId: require('crypto').randomUUID(),
      walletAddress: walletAddress,
      userType: 'admin_test',
      startTime: new Date(),
      isActive: true,
      isTestMode: true,
      totalEarned: new Decimal('0.000000000000000000000000000000000000000000000000000'),
      hourlyRate: new Decimal('0.250000000000000000000000000000000000000000000000000'),
      createdAt: new Date(),
      networkId: 'BitWish-Admin-Test-Network' // 관리자 전용 테스트 네트워크
    };

    await adminMiningCollection.insertOne(sessionData);
    
    console.log(`✅ 관리자 테스트 마이닝 시작: ${walletAddress}`);
    res.json({ 
      success: true, 
      message: '관리자 테스트 마이닝이 시작되었습니다',
      sessionId: sessionData.sessionId,
      isTestMode: true,
      networkId: 'BitWish-Admin-Test-Network'
    });
    
  } catch (error) {
    console.error('❌ 관리자 테스트 마이닝 시작 오류:', error);
    res.status(500).json({ success: false, error: error.message });
  }
});

// 관리자 전용 테스트 마이닝 업데이트
app.post('/api/admin/bitwish/test/mining/update', async (req, res) => {
  try {
    const { walletAddress } = req.body;
    
    const db = bitWishMongoDB.db;
    const adminMiningCollection = db.collection('admin_test_mining_sessions');
    
    const Decimal = require('decimal.js');
    Decimal.set({ precision: 50 });
    
    // 관리자 테스트 마이닝 세션 조회
    const miningSession = await adminMiningCollection.findOne({
      walletAddress: walletAddress,
      isActive: true,
      isTestMode: true
    });
    
    if (!miningSession) {
      return res.status(404).json({ 
        success: false, 
        error: '관리자 테스트 마이닝 세션을 찾을 수 없습니다' 
      });
    }
    
    // 테스트용 마이닝 보상 계산 (50자리 정밀도)
    const timeElapsed = new Date() - miningSession.startTime;
    const hours = timeElapsed / (1000 * 60 * 60);
    const hourlyRate = new Decimal('0.250000000000000000000000000000000000000000000000000');
    const reward = hourlyRate.mul(hours);
    
    // 테스트 마이닝 데이터 업데이트
    const newTotalEarned = new Decimal(miningSession.totalEarned).plus(reward);
    
    await adminMiningCollection.updateOne(
      { sessionId: miningSession.sessionId },
      { 
        $set: { 
          totalEarned: newTotalEarned.toString(),
          lastUpdate: new Date()
        }
      }
    );
    
    console.log(`✅ 관리자 테스트 마이닝 업데이트: ${walletAddress} - ${reward.toString()} BW`);
    res.json({ 
      success: true, 
      totalEarned: newTotalEarned.toString(),
      currentReward: reward.toString(),
      isTestMode: true,
      networkId: 'BitWish-Admin-Test-Network'
    });
    
  } catch (error) {
    console.error('❌ 관리자 테스트 마이닝 업데이트 오류:', error);
    res.status(500).json({ success: false, error: error.message });
  }
});

app.get('/api/admin/bitwish/test/attendance/stats', async (req, res) => {

    const todayUsers = await adminAttendanceCollection.countDocuments({

      attendanceBonusRate: attendanceBonusRate.toString(),
      totalBonusPaid: totalBonusPaid.toString(),
      isTestMode: true,
