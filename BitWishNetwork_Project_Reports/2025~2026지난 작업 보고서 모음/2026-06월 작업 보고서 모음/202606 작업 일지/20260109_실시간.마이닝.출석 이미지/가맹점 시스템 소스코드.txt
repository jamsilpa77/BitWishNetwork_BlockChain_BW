## **3. 가맹점 보너스 계산 함수 (App.tsx 라인 3316-3344)**

```typescript
// 위치: Node_HomePage/src/App.tsx (라인 3316-3344)
//    가맹점 보너스 계산 함수 (BigDecimal 기반 완전 재구현)
const calculateMerchantBonus = (isMerchant: boolean): {
  bonusRate: number;
  enhancedRate: number;
  description: string;
  decimalBonusRate: Decimal;
  decimalEnhancedRate: Decimal;
} => {
  if (isMerchant) {
    const decimalBonusRate = new Decimal(1.25); // 125% (백서 정책)
    const decimalEnhancedRate = new Decimal(0.25).times(new Decimal(1).plus(decimalBonusRate));
    return {
      bonusRate: decimalBonusRate.toDecimalPlaces(8).toNumber(),
      enhancedRate: decimalEnhancedRate.toDecimalPlaces(8).toNumber(),
      description: '🏪 가맹점 보너스 활성화 (125%)',
      decimalBonusRate,
      decimalEnhancedRate
    };
  } else {
    const decimalBonusRate = new Decimal(0); // 0%
    const decimalEnhancedRate = new Decimal(0.25); // 기본값
    return {
      bonusRate: decimalBonusRate.toDecimalPlaces(8).toNumber(),
      enhancedRate: decimalEnhancedRate.toDecimalPlaces(8).toNumber(),
      description: '❌ 가맹점 보너스 미활성화',
      decimalBonusRate,
      decimalEnhancedRate
    };
  }
};
```

## **9. 최종 보너스 계산 예시**

```typescript
// 가맹점 등록 시 보너스 계산
const decimalBonusRate = new Decimal(1.25); // 125% 보너스
const decimalEnhancedRate = new Decimal(0.25).times(new Decimal(1).plus(decimalBonusRate));
// = 0.25 × (1 + 1.25) = 0.25 × 2.25 = 0.5625 BW/시간

// 최종 채굴률 계산 예시
// 예시: 가맹점 등록 + 30일 출석 + 50명 추천인 + 180일 락업
기본_채굴률 = 0.25 BW/시간
출석_보너스 = 5.0% = 0.05
추천인_보너스 = 46.2% = 0.462
락업_보너스 = 3.0% = 0.03
가맹점_보너스 = 125% = 1.25

최종_채굴률 = 0.25 × (1 + 0.05 + 0.462 + 0.03 + 1.25)
최종_채굴률 = 0.25 × 2.792
최종_채굴률 = 0.698 BW/시간

일일_보상 = 0.698 × 24 = 16.752 BW/일
월간_보상 = 16.752 × 30 = 502.56 BW/월
연간_보상 = 502.56 × 12 = 6,030.72 BW/년
```
